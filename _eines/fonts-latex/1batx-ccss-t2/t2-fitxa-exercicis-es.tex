\seccio{1}{Polinomios. Operaciones}

\exf{1} Dados los polinomios $P(x)=2x^3-x+3$ y $Q(x)=x^2-2x+1$, calcula:\begin{tasks}(4)
  \task $P(x)+Q(x)$
  \task $P(x)-2Q(x)$
  \task $P(x)\cdot Q(x)$
  \task $\big(Q(x)\big)^2$
\end{tasks}

\ex{1} Simplifica las expresiones polinómicas siguientes:\begin{tasks}(1)
  \task $2(x-2)^2-3(3x+2)-2(3x-2)(3x+2)$
  \task $(3x+2)^2+2(2x-3)^2-(x-5)(2x-5)$
  \task $(2x^2-2x-13)(3x^2-2x)-3x$
  \task $(x^2+4)(x^2-4)(x-2)+2x(x^3-10)x$
  \task $\left(\frac23x-\frac35\right)\left(\frac32x^2+\frac25\right)+\frac{6}{25}$
\end{tasks}\recorda{$(a\pm b)^2=a^2\pm2ab+b^2$, \ $(a+b)(a-b)=a^2-b^2$. Desarrolla cada producto y reduce los términos semejantes.}

\ex{2} Calcula $(4x^2-3x+2)^2$. (Puedes usar que $(a+b+c)^2=a^2+b^2+c^2+2ab+2ac+2bc$.)

\ex{2} Calcula los valores de $a$ y $b$ para que se verifiquen las igualdades siguientes:\begin{tasks}(1)
  \task $(3x^2-5x+a)(2x^2+bx)=6x^4-19x^3+19x^2-6x$
  \task $2(x+3)^2-3(x+a)=2x^2+bx+3$
\end{tasks}

\ex{1} Efectúa las divisiones siguientes e indica el cociente y el resto:\begin{tasks}(1)
  \task $(3x^3+2x^2+x-5):(3x^2+2)$
  \task $(3x^4-2x^2-x+4):(x^2+x+2)$
  \task $(x^3-3x^2-x+6):(2x^3-2x+3)$
  \task $(6x^4-x^2+3x):(2x^2+1)$
\end{tasks}\recorda{Compruébalo con la prueba de la división: $D(x)=d(x)\cdot c(x)+r(x)$, con el grado de $r(x)$ menor que el de $d(x)$.}

\seccio{2}{Regla de Ruffini. Teorema del resto}

\exf{1} Efectúa las divisiones siguientes aplicando la regla de Ruffini:\begin{tasks}(2)
  \task $(3x^3+2x^2+x-5):(x+2)$
  \task $(3x^4-2x^2-x+4):(x+2)$
  \task $(x^3-3x^2-x+6):(2x+3)$
  \task $(4x^4-3x^2+5):(-x+5)$
  \task $(2x^5-12x^3-6x^2-117):(x-3)$
\end{tasks}\recorda{Si el divisor es $ax+b$, haz Ruffini con $-\frac ba$ y divide el cociente entre $a$; el resto no cambia.}

\ex{1} Calcula el valor numérico del polinomio $P(x)=x^2-2x-3$ para $x=-1$ y $x=3$. ¿Podrías encontrar otro valor de $x$ para el que el valor del polinomio sea cero? Razona la respuesta.

\ex{1} Comprueba si los números siguientes son raíces del polinomio $P(x)=x^4+3x^3-2x^2+6x-8$:\begin{tasks}(5)
  \task $x=1$
  \task $x=2$
  \task $x=-1$
  \task $x=-4$
  \task $x=-3$
\end{tasks}

\ex{1} Comprueba si $M(x)=2x^3-5x^2+4x-4$ es divisible por $x-2$ y, si lo es, encuentra un polinomio $N(x)$ que permita escribir $M(x)=(x-2)\cdot N(x)$.

\ex{1} Calcula, sin hacer la división, el resto de las divisiones siguientes:\begin{tasks}(2)
  \task $(x^3-2x+5):(x+2)$
  \task $(x^{100}-2x^{51}+1):(x+1)$
  \task $(2x^4-3x^2+x-1):(x-1)$
  \task $(x^5-32):(x-2)$
\end{tasks}\recorda{Teorema del resto: el resto de dividir $P(x)$ entre $x-a$ es $P(a)$.}

\ex{2} Calcula el valor de $k$ para que el polinomio:\begin{tasks}(1)
  \task $P(x)=x^3+x^2-2x+k$ sea divisible por $x-2$
  \task $P(x)=x^3-2x^2+kx+4$ sea divisible por $x+2$
  \task $P(x)=(k-3)x^3-15x+k$ sea divisible por $x+4$
  \task $P(x)=x^3-(k-3)x^2+(k-5)x-3$ sea divisible por $x-3$
\end{tasks}

\ex{2} Utiliza Ruffini (o el teorema del resto) para calcular el valor de $a$ y $b$ que hace que las divisiones siguientes sean exactas:\begin{tasks}(1)
  \task $(2x^3+5x^2-2x+a):(x+3)$
  \task $(2x^4+3x^3-2x^2-ax+3):(2x+3)$
  \task $(-6x^4+11x^3+ax^2+5x+4):(-3x+4)$
  \task $(4x^4+6x^3-13x^2+x+b):(-x+5)$
\end{tasks}

\ex{2} Calcula el valor de $k$ para que la división de $x^3-(3k+2)x^2-3$ entre $(x+3)$:\begin{tasks}(2)
  \task Sea exacta.
  \task Tenga resto 15.
\end{tasks}

\ex{2} Halla el valor de $m$ para que el polinomio $2x^4+9x^3+2x^2-6x+3m$ tenga resto 12 al dividirlo por $x+2$, sin hacer la división.

\seccio{3}{Factorización de polinomios}

\exf{1} Determina las raíces de los polinomios siguientes:\begin{tasks}(2)
  \task $(x-3)(x+5)(x-2)$
  \task $x(x-2)^3(2x+1)$
  \task $(2x-1)(3x+2)(x+1)^3$
  \task $x^3-3x^2-6x+8$
  \task $x^3+8x^2+17x+10$
  \task $2x^5-11x^4+21x^3-16x^2+4x$
\end{tasks}

\ex{2} Determina, si existen, las raíces enteras de cada uno de los polinomios siguientes y factorízalos:\begin{tasks}(2)
  \task $P(x)=x^3-2x^2-5x+6$
  \task $P(x)=x^3+x^2-5x+3$
  \task $P(x)=2x^4-x^3-5x^2+x+3$
  \task $P(x)=4x^3-4x^2-11x+6$
  \task $P(x)=x^4-4x^2$
  \task $P(x)=6x^5+11x^4+3x^3-3x^2-x$
  \task $P(x)=x^6-4x^4-x^2+4$
  \task $P(x)=3x^2-12x+12$
  \task $P(x)=\frac{x^4}{4}+\frac34x^2+\frac{9}{16}$ \ (comprueba que no tiene raíces reales)
\end{tasks}\recorda{Primero, saca factor común y busca identidades notables. Las raíces enteras son divisores del término independiente: pruébalas con Ruffini hasta que quede un polinomio de grado 2.}

\ex{2} De un polinomio de segundo grado, $P(x)$, sabemos que $P(1)=-6$, $P(0)=-3$ y que una de sus raíces es 3. Determínalo y factorízalo.

\ex{2} Determina $a$ y $b$ de manera que el polinomio $x^3+ax^2+bx-6$ sea divisible por $x-2$ y por $x+3$.

\ex{2} Halla el polinomio de grado 3 que tiene las raíces $1$, $-2$ y $3$ y que cumple $P(0)=12$.

\seccio{4}{Fracciones algebraicas}

\exf{2} Simplifica las fracciones algebraicas siguientes:\begin{tasks}(2)
  \task $\dfrac{2x^2-8}{x^3+x^2-2x}$
  \task $\dfrac{x^3-x}{x^3+3x^2+2x}$
  \task $\dfrac{x^4-2x^3-3x^2}{x^4-9x^2}$
  \task $\dfrac{x^3-3x^2+3x-1}{x^3-2x^2+x}$
  \task $\dfrac{2x^4-5x^3+x^2+5x-3}{2x^3-x^2-7x+6}$
  \task $\dfrac{x^3-27}{x^2-9}$
  \task $\dfrac{x^3-2x^2-9x+18}{x^3-7x^2+16x-12}$
  \task $\dfrac{x^4-5x^3+9x^2-7x+2}{x^4-6x^3+13x^2-12x+4}$
\end{tasks}\recorda{Factoriza el numerador y el denominador y simplifica solo los \emph{factores} comunes: $\dfrac{x+3}{x}\neq3$.}

\ex{2} Efectúa las sumas y restas siguientes reduciendo previamente a común denominador:\begin{tasks}(2)
  \task $\dfrac{1}{x+1}+\dfrac{3}{x^2-1}$
  \task $\dfrac{1}{x^2+x}+\dfrac{3}{x+1}$
  \task $\dfrac{1}{2x}-\dfrac{3}{4x^2-4x}$
  \task $\dfrac{1}{x^2+x}+\dfrac{3}{x^2-x}$
  \task $\dfrac{2x}{x-2}+\dfrac{3x-1}{x+2}-\dfrac{1}{x^2-4}$
  \task $\dfrac{1}{(x-1)^2}+\dfrac{2}{x-1}+\dfrac{1}{x^2-1}$
\end{tasks}\recorda{Factoriza los denominadores: el común denominador es el m.c.m. (factores comunes y no comunes con el mayor exponente).}

\ex{2} Efectúa los productos y cocientes siguientes y simplifica el resultado:\begin{tasks}(2)
  \task $\dfrac{3x+3}{12x-12}:\dfrac{(x+1)^2}{x^2-1}$
  \task $\dfrac{x^2+2x-3}{(x-2)^3}\cdot\dfrac{(x-2)^2}{x^2-1}$
  \task $\dfrac{6x+4}{x-2}\cdot\dfrac{4x+1}{2x+4}$
  \task $\dfrac{-6x^2}{x^2+2x+1}\cdot\dfrac{x+1}{3x^3}\cdot\dfrac{x+1}{3x}$
  \task $\dfrac{2x^2-7x+6}{x^2+2x+1}\cdot\dfrac{x^2+6x+5}{2x-3}$
  \task $\dfrac{x^2-4}{x^2+x}:\dfrac{x-2}{x+1}$
\end{tasks}\recorda{Factoriza antes de multiplicar: así podrás simplificar sin desarrollar los productos.}

\ex{3} Calcula y simplifica el resultado de las operaciones combinadas siguientes:\begin{tasks}(2)
  \task $\left(\dfrac3x-\dfrac{2x}{x+1}\right)\cdot\dfrac{x^2+x}{x-1}$
  \task $\dfrac{(x-1)^2}{2}\cdot\dfrac{1}{x^2-1}-\dfrac{3x}{(x+1)^2}$
  \task $\left(\dfrac{2x-1}{x+1}-\dfrac{3x}{x-1}\right)\cdot\dfrac{x^3-x}{-x^2-6x+1}$
  \task $\left(\dfrac{x}{x-1}-\dfrac1x\right):\dfrac{x^2-x+1}{x^2-x}$
  \task $\dfrac{1+\frac{2x+1}{x-1}}{\frac{1}{x-1}+\frac{x}{x+1}}$
  \task $\dfrac{6}{2+x}-\dfrac{4}{2-x}+\dfrac{16}{x^2-4}$
  \task $\dfrac{1-\frac1x}{1-\frac1{x^2}}$
  \task $1+\cfrac{1}{1+\cfrac{1}{1+\frac1x}}$
\end{tasks}\recorda{Primero los paréntesis (y los numeradores y denominadores de las fracciones compuestas), después productos y cocientes y, al final, sumas y restas.}

\seccio{5}{Ecuaciones polinómicas}

\exf{1} Resuelve las ecuaciones de primer grado siguientes:\begin{tasks}(2)
  \task $3(2x-1)-2(x+4)=5-(x-2)$
  \task $2(x+3)-3(x-1)=4(2-x)+1$
  \task $5x-2(3-x)=3(x+2)-4$
  \task $4(x-1)-(2x+3)=2(x-3)+1$
  \task $3(x+2)-(x-4)=2(x+5)$
  \task $x-\big[2x-(3-x)\big]=5$
\end{tasks}\recorda{Un signo $-$ delante de un paréntesis cambia el signo de todos los términos de dentro: \mbox{$-(x-2)=-x+2$}.}

\ex{2} Resuelve las ecuaciones de primer grado con denominadores siguientes:\begin{tasks}(2)
  \task $\dfrac{2x-3}{4}+\dfrac x2-\dfrac{3x-1}{5}=2x-1$
  \task $\dfrac{x+3}{4}-\dfrac{2x-1}{6}=\dfrac x3-1$
  \task $\dfrac{x-1}{2}-\dfrac{x+2}{3}=1$
  \task $\dfrac x2-\dfrac{x-1}{3}=\dfrac{x+1}{6}+1$
  \task $\dfrac{2(x+1)}{3}-\dfrac{x-3}{4}=1+\dfrac{x}{12}$
  \task $\dfrac{3x-1}{2}-\dfrac{2(x+1)}{5}=x-\dfrac{x-2}{10}$
\end{tasks}\recorda{Multiplica todos los términos por el m.c.m. de los denominadores. Si delante de una fracción hay un $-$, pon el numerador entre paréntesis.}

\ex{1} Resuelve las ecuaciones de segundo grado siguientes:\begin{tasks}(2)
  \task $x^2-5x+6=0$
  \task $2x^2+3x-2=0$
  \task $3x^2-12=0$
  \task $x^2+4x+5=0$
  \task $4x^2-4x+1=0$
  \task $5x^2-10x=0$
  \task $2(3x-2)+x(x-1)=-4$
  \task $\dfrac{x^2+1}{2}-\dfrac{2x-3}{4}+\dfrac{x^2}{6}=\dfrac{59}{12}$
  \task $(x+1)^3-(x-1)^3=7$
  \task $(2x-1)^2-3(x+1)=x(x-2)$
\end{tasks}\recorda{$ax^2+bx+c=0\Rightarrow x=\dfrac{-b\pm\sqrt{b^2-4ac}}{2a}$. Las incompletas se resuelven más rápido: sacando factor común ($c=0$) o despejando $x^2$ ($b=0$).}

\ex{2} Resuelve las ecuaciones bicuadradas siguientes (o que se resuelven con un cambio de variable):\begin{tasks}(2)
  \task $x^4-17x^2+16=0$
  \task $x^6-9x^3+8=0$
  \task $x^4-8x^2-9=0$
  \task $\dfrac{9}{x^2}=10-x^2$
  \task $2(x+1)^4-8x^3-8(x+3)+8=0$
  \task $4x^4-17x^2+4=0$
\end{tasks}

\ex{2} Resuelve las ecuaciones polinómicas de grado superior a 2 siguientes:\begin{tasks}(2)
  \task $6x^4+13x^3-8x^2-17x+6=0$
  \task $2x^4-x^3-3x-18=0$
  \task $6x^4+5x^3=2x^2+x$
  \task $x^3-7x+6=0$
  \task $x^3-3x^2-9x+27=0$
  \task $x^4-x^3-5x^2-x-6=0$
  \task $x^4-1=0$
  \task $x^4+x^3-9x^2-9x=0$
\end{tasks}\recorda{Pásalo todo a un miembro y factoriza (factor común, Ruffini…): un producto vale cero si algún factor vale cero.}

\ex{1} Resuelve las ecuaciones factorizadas siguientes:\begin{tasks}(2)
  \task $(x^2-3)(x^2-4)=0$
  \task $(3x+2)(x^2-4x)(x^2-9)=0$
  \task $(x^2-4x+4)(x^2+6x+9)=0$
  \task $(x-2)^2(x+1)^3(2x-7)=0$
\end{tasks}

\seccio{7}{Ecuaciones racionales}

\exf{2} Resuelve las ecuaciones racionales siguientes:\begin{tasks}(2)
  \task $\dfrac1x+\dfrac1{x^2}+\dfrac1{x^3}=\dfrac78$
  \task $\dfrac{2x}{3}+\dfrac{2x+3}{x-1}=\dfrac{11}{3x-3}$
  \task $\dfrac{2x}{x-2}+\dfrac{3x}{x+2}=\dfrac{6x^2}{x^2-4}$
  \task $\dfrac{3}{x^2-6x+8}+\dfrac{x-5}{x^2-5x+4}=\dfrac{1}{x-4}$
  \task $\dfrac{x}{\sqrt2}+\dfrac{\sqrt2}{x}=\sqrt2\,x$
  \task $\dfrac{x}{x-2}+\dfrac{2x}{2-x}=-x$
  \task $\dfrac{x}{x+1}+\dfrac{x+1}{x}=\dfrac52$
  \task $\dfrac{1+\frac{x+1}{x-1}}{2-\frac{x-1}{x+1}}=2$
  \task $\dfrac{1}{1-\frac{1}{x+1}}=\dfrac{x+1}{x}$
