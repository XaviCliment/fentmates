  \task $\res{x=2,\ y=1}$
  \task $\res{x=1,\ y=3}$
  \task $\res{x=1,\ y=-1}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $\left\{\begin{array}{l}2x-3y=5\\x-y=3\end{array}\right.$ \ $\res{x=4,\ y=1}$
  \task $\left\{\begin{array}{l}5x-2y=13\\2x-3y=3\end{array}\right.$ \ $\res{x=3,\ y=1}$
  \task $\left\{\begin{array}{l}x+7y=17\\2x-5y=-4\end{array}\right.$ \ $\res{x=3,\ y=2}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task Por 6 y por 2: $\left\{\begin{array}{l}2x+3y=18\\x-y=4\end{array}\right.$ \ $\res{x=6,\ y=2}$
  \task Por 6 y por 3: $\left\{\begin{array}{l}3x-2y=7\\x+y=9\end{array}\right.$ \ $\res{x=5,\ y=4}$
  \task Por 10 y por 15: $\left\{\begin{array}{l}5x-2y=10\\5x+3y=35\end{array}\right.$ \ $\res{x=4,\ y=5}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task Las dos quedan $x-2y=1$: rectas coincidentes. \rest{SCI}: \ $\res{(x,y)=(1+2\lambda,\ \lambda)}$
  \task $2x-2y=3$ y $2x-2y=2$: rectas paralelas. \rest{SI}: \ $\res{\text{no tiene solución}}$
  \task Rectas que se cortan en un punto. \rest{SCD}: \ $\res{(x,y)=(2,\,1)}$
\end{tasks}

\ex{2} $\left\{\begin{array}{l}x+y=30\\2x+4y=80\end{array}\right.\Rightarrow 2y=20$: \ $\res{20\text{ gallinas y }10\text{ conejos}}$

\ex{2} $\left\{\begin{array}{l}x+y=120\\7x+5y=720\end{array}\right.\Rightarrow 7x+5(120-x)=720\Rightarrow 2x=120$: \ $\res{60\text{ normales y }60\text{ reducidas}}$

\seccio{2}{Método de Gauss}

\exf{1} \begin{tasks}(2)
  \task $z=3,\ y=2,\ x=1$: \ $\res{(1,\,2,\,3)}$
  \task $z=2,\ y=1,\ x=1$: \ $\res{(1,\,1,\,2)}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task \rest{SCD}: \ $\res{(x,y,z)=(2,\ 1,\ {-}1)}$
  \task \rest{SCD}: \ $\res{(x,y,z)=(1,\ 2,\ 3)}$
  \task \rest{SI}: incompatible (aparece una ecuación $0=k\neq0$)
  \task \rest{SCI}: \ $\res{(x,y,z)=(- \lambda + 2,\ \lambda + 1,\ \lambda),\ \lambda\in\mathbb R}$
\end{tasks}

\ex{2} $\left\{\begin{array}{l}2x+y+z=9\\x+2y+z=10\\x+y+2z=13\end{array}\right.$ (sumándolas, $x+y+z=8$): \ $\res{\text{bolígrafo 1 €,\, libreta 2 €,\, carpeta 5 €}}$

\ex{3} $\left\{\begin{array}{l}x+y+z=30\\2x-y=36\\x=2(y+z)\end{array}\right.\Rightarrow\left(\begin{array}{rrr|r}1&1&1&30\\ 2&-1&0&36\\ 1&-2&-2&0\end{array}\right)$: \ $\res{20\text{ aciertos,\, }4\text{ errores y }6\text{ en blanco}}$

\seccio{3}{Sistemas de ecuaciones no lineales}

\exf{2} \begin{tasks}(1)
  \task $y=7-x\Rightarrow x^2-7x+12=0$: \ $\res{(3,\,4),\ (4,\,3)}$
  \task $x^2-4=x+2\Rightarrow x^2-x-6=0$: \ $\res{(-2,\,0),\ (3,\,5)}$
  \task $y=2x-1\Rightarrow 5x^2-4x-1=0$: \ $\res{(- \frac{1}{5},\,{-} \frac{7}{5}),\ (1,\,1)}$
  \task $y=2-x\Rightarrow x^2-2x+5=0,\ \Delta\lt0$: \ $\res{\text{no tiene solución}}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task Sumando: $2x^2=18$, $y^2=4$: \ $\res{(-3,\,{-}2),\ (-3,\,2),\ (3,\,{-}2),\ (3,\,2)}$
  \task La 1.ª más el doble de la 2.ª: $5x^2=80$, $y^2=1$: \ $\res{(-4,\,{-}1),\ (-4,\,1),\ (4,\,{-}1),\ (4,\,1)}$
\end{tasks}

\ex{2} $\left\{\begin{array}{l}x+y=17\\x^2+y^2=169\end{array}\right.\Rightarrow x^2-17x+60=0$: \ $\res{5\text{ cm y }12\text{ cm}}$

\ex{2} $\left\{\begin{array}{l}x-y=3\\xy=40\end{array}\right.\Rightarrow y^2+3y-40=0\Rightarrow y=5,\ y=-8$: \ $\res{8\text{ y }5,\ \text{o bien }-5\text{ y }-8}$
