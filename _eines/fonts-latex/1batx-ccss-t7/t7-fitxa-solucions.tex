\seccio{2}{Derivada en un punt i funció derivada}

\exf{1} \begin{tasks}(1)
  \task $\dfrac{f(-2+h)-f(-2)}{h}=\dfrac{\frac{1}{-2+h}+\frac12}{h}=\dfrac{1}{2(-2+h)}\to\res{f'(-2)=-\frac14}$
  \task $\dfrac{-2(a+h)+4-(-2a+4)}{h}=\dfrac{-2h}{h}=-2$: \ $\res{f'(a)=-2}$ en tots els punts, perquè la gràfica és una recta de pendent $-2$.
  \task $\dfrac{f(a+h)-f(a)}{h}=6a-5+3h\to6a-5$: \ $f'(-2)=\res{-17}$, \ $f'(1)=\res{1}$, \ $f'(0)=\res{-5}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $\dfrac{(x+h)^2+3(x+h)-x^2-3x}{h}=\dfrac{2xh+h^2+3h}{h}=2x+h+3\to\res{2x+3}$
  \task $\dfrac{\frac{2}{x+h+1}-\frac{2}{x+1}}{h}=\dfrac{-2h}{h(x+h+1)(x+1)}\to\res{\dfrac{-2}{(x+1)^2}}$
  \task Pel conjugat: $\dfrac{\sqrt{x+h+1}-\sqrt{x+1}}{h}=\dfrac{1}{\sqrt{x+h+1}+\sqrt{x+1}}\to\res{\dfrac{1}{2\sqrt{x+1}}}$
\end{tasks}

\ex{1} $\dfrac{(3+h)^2-9}{h}=6+h\to f'(3)=6$. \ Tangent per $(3,9)$: $y=9+6(x-3)\Rightarrow\res{y=6x-9}$

\seccio{6}{Monotonia i extrems}

\exf{1} \begin{tasks}(1)
  \task $f'(x)=-2x+5=0\Rightarrow x=\frac52$. \ Creixent en $\res{\left(-\infty,\frac52\right)}$ i decreixent en $\res{\left(\frac52,+\infty\right)}$. Màxim relatiu (i absolut) en $\res{\left(\frac52,\frac{21}{4}\right)}$.
  \task $f'(x)=6x^2-6x-12=6(x+1)(x-2)$. \ Creixent en $\res{(-\infty,\,-1)\cup(2,+\infty)}$, decreixent en $\res{(-1,2)}$. Màxim relatiu $\res{(-1,12)}$, mínim relatiu $\res{(2,\,-15)}$.
\end{tasks}

\ex{2} \label{ex:racional}\begin{tasks}(1)
  \task $D=\mathbb R-\{1\}$, \ $f'(x)=\frac{x^2-2x}{(x-1)^2}=0\Rightarrow x=0,\ x=2$. \ Creixent en $\res{(-\infty,0)\cup(2,+\infty)}$, decreixent en $\res{(0,1)\cup(1,2)}$. Màxim $\res{(0,0)}$, mínim $\res{(2,4)}$.
  \task $g'(x)=e^x(x+1)=0\Rightarrow x=-1$. \ Decreixent en $\res{(-\infty,\,-1)}$, creixent en $\res{(-1,+\infty)}$. Mínim $\res{\left(-1,\,-\frac1e\right)}$.
  \task $h'(x)=\frac{2x}{x^2+1}=0\Rightarrow x=0$. \ Decreixent en $\res{(-\infty,0)}$, creixent en $\res{(0,+\infty)}$. Mínim $\res{(0,0)}$.
\end{tasks}

\ex{2} $f'(x)=4x^3-16x=4x(x^2-4)=0\Rightarrow x=0,\ x=\pm2$. \ $f''(x)=12x^2-16$: \ $f''(0)=-16<0$: màxim $\res{(0,3)}$; \ $f''(\pm2)=32>0$: mínims $\res{(-2,\,-13)}$ i $\res{(2,\,-13)}$.

\ex{2} $f'(x)=3x^2-3=0\Rightarrow x=\pm1$ (tots dos dins). \ $f(-2)=0$, $f(-1)=4$, $f(1)=0$, $f(3)=20$. \ Màxim absolut $\res{(3,20)}$; mínim absolut $\res{0}$, en $x=-2$ i en $x=1$.

\ex{3} $f'(2)=12+4a=0\Rightarrow\res{a=-3}$. \ $f(2)=8-12+b=-3\Rightarrow\res{b=1}$. \ $f''(x)=6x-6$, $f''(2)=6>0$: \rest{mínim relatiu}.

\seccio{8}{Problemes d'optimització}

\exf{1} $B'(x)=-2x+40=0\Rightarrow x=20$. \ Creixent en $\res{(0,20)}$ i decreixent en $\res{(20,+\infty)}$: màxim en $x=20$. \ Ha de vendre $\res{2\,000\text{ unitats}}$ i el benefici màxim és $B(20)=\res{100\,000\text{ €}}$. \ $B(x)=-(x-10)(x-30)<0$ si $x<10$ o $x>30$: pèrdues amb \rest{menys de 1\,000 o més de 3\,000 unitats}.

\ex{2} $M(x)=x+100+\frac{3\,600}{x}$, $x>0$. \ $M'(x)=1-\frac{3\,600}{x^2}=0\Rightarrow x=60$. \ Decreixent en $\res{(0,60)}$ i creixent en $\res{(60,+\infty)}$: mínim. \ $\res{60\text{ unitats}}$, amb un cost mitjà de $M(60)=\res{220\text{ €}}$.

\ex{2} $N'(t)=-6t^2+42t-36=-6(t-1)(t-6)=0\Rightarrow t=1,\ t=6$. \ Decreix en $\res{(0,1)\cup(6,7)}$ i creix en $\res{(1,6)}$. \ $N(0)=40$, $N(1)=23$, $N(6)=148$, $N(7)=131$. \ Màxim a les $\res{23\text{ h}}$ (148\,000 persones) i mínim a les $\res{18\text{ h}}$ (23\,000 persones).

\ex{2} $V'(t)=\frac{40(t^2+16)-40t\cdot2t}{(t^2+16)^2}=\frac{40(16-t^2)}{(t^2+16)^2}=0\Rightarrow t=4$ ($t=-4$ no té sentit). \ Creixent en $\res{(0,4)}$ i decreixent en $\res{(4,+\infty)}$. \ Màxim al $\res{\text{quart mes}}$: $V(4)=\frac{160}{32}=\res{5\,000\text{ unitats}}$. \ $\lim_{t\to+\infty}V(t)=0$: les vendes \rest{tendeixen a desaparéixer}.

\ex{3} $B'(x)=10e^{-0{,}2x}+10x\cdot(-0{,}2)e^{-0{,}2x}=(10-2x)\,e^{-0{,}2x}=0\Rightarrow x=5$ (l'exponencial no s'anul·la mai). \ Creixent en $\res{(0,5)}$ i decreixent en $\res{(5,+\infty)}$. \ Ha d'invertir $\res{5\,000\text{ €}}$ i el benefici màxim és $B(5)=\frac{50}{e}\approx\res{18\,394\text{ €}}$.

\ex{2} $B(x)=120x-(2x^2+40x+300)=-2x^2+80x-300$. \ $B'(x)=-4x+80=0\Rightarrow x=20$; $B''=-4<0$: màxim. \ $\res{20\text{ unitats}}$ al dia, amb un benefici de $B(20)=\res{500\text{ €}}$.

\ex{2} $x$ = euros que puja el preu: s'ocupen $80-x$ habitacions i cadascuna deixa $50+x-10=40+x$ euros. \ $B(x)=(80-x)(40+x)=3\,200+40x-x^2$, \ $0\leq x\leq80$. \ $B'(x)=40-2x=0\Rightarrow x=20$; $B''=-2<0$: màxim. \ Preu $\res{70\text{ €}}$, $\res{60\text{ habitacions}}$ ocupades i un benefici de $60\cdot60=\res{3\,600\text{ €}}$ per nit.

\ex{2} $x$ = euros de rebaixa: preu $20-x$, espectadors $800+50x$. \ $R(x)=(20-x)(800+50x)=16\,000+200x-50x^2$. \ $R'(x)=200-100x=0\Rightarrow x=2$; $R''=-100<0$: màxim. \ Preu $\res{18\text{ €}}$, $\res{900\text{ persones}}$ i una recaptació de $\res{16\,200\text{ €}}$.

\ex{3} $B(p)=\underbrace{p\,(600-20p)}_{\text{ingressos}}-\underbrace{\big(6(600-20p)+1\,000\big)}_{\text{costos}}=(p-6)(600-20p)-1\,000=-20p^2+720p-4\,600$, \ $6\leq p\leq30$. \ $B'(p)=-40p+720=0\Rightarrow p=18$; $B''<0$: màxim. \ Preu $\res{18\text{ €}}$, $q=\res{240\text{ unitats}}$ i benefici $B(18)=\res{1\,880\text{ €}}$.

\ex{3} Ingressos = quilos $\cdot$ preu: \ $I(t)=(10\,000-10t^2)(2+0{,}05t)$, $t\in[0,20]$. \ $I'(t)=-1{,}5t^2-40t+500=0\Rightarrow t\approx9{,}27$. \ Amb dies enters: $I(9)=22\,515{,}5$, $I(10)=22\,500$; \ $I(0)=20\,000$, $I(20)=18\,000$. \ Li convé vendre el \rest{9é dia}: $\res{22\,515{,}50\text{ €}}$.
