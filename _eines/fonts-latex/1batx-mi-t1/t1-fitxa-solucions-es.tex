\ex{1} \begin{tasks}(1)
  \task \rest{Verdadera}: todo entero $n$ se puede escribir como $\frac n1$.
  \task \rest{Falsa}: solo los exactos y los periódicos; $\pi=3,1415\ldots$ no.
  \task \rest{Verdadera}: por ejemplo, $5=\frac{10}{2}$.
  \task \rest{Verdadera}: $\frac ab+\frac cd=\frac{ad+bc}{bd}$, que es una fracción.
  \task \rest{Falsa}: $\frac32\cdot\frac23=1$, o bien $2,5\cdot0,4=1$.
\end{tasks}

\ex{2} \begin{tasks}(2)
  \task $\frac{25}{9}+\frac{31}{9}-\frac{10}{9}=\res{\frac{46}{9}}$
  \task $\frac{17}{9}+\frac19=\res{2}$
  \task $\frac{37}{30}+\frac{1143}{250}-\frac{1271}{495}=\res{\frac{40066}{12375}}$
  \task $\frac{38}{11}\cdot\frac{1111}{900}-\frac52=\res{\frac{397}{225}}$
\end{tasks}(En la ficha original el apartado b) ponía $1\widehat8$; lo hemos leído como $1,\widehat8$.)
