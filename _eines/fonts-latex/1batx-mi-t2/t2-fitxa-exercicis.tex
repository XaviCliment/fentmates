\seccio{1}{Polinomis. Operacions}

\exf{1} Simplifica les següents expressions polinòmiques.\begin{tasks}(1)
  \task $2(x-2)^2-3(3x+2)-2(3x-2)(3x+2)$
  \task $(3x+2)^2+2(2x-3)^2-(x-5)(2x-5)$
  \task $(2x^2-2x-13)(3x^2-2x)-3x$
  \task $(x^2+4)(x^2-4)(x-2)+2x(x^3-10)x$
  \task $\left(\frac23x-\frac35\right)\left(\frac32x^2+\frac25\right)+\frac{6}{25}$
\end{tasks}\recorda{$(a\pm b)^2=a^2\pm2ab+b^2$, \ $(a+b)(a-b)=a^2-b^2$. Desenvolupa cada producte i redueix els termes semblants.}

\ex{2} Demostra la igualtat algebraica: \[(x+y+z)^2=x^2+y^2+z^2+2xy+2xz+2yz\]

\ex{1} Calcula $(4x^2-3x+2)^2$.

\ex{2} Calcula els valors de $a$ i $b$ perquè es verifiquen les següents igualtats.\begin{tasks}(1)
  \task $(3x^2-5x+a)(2x^2+bx)=6x^4-19x^3+19x^2-6x$
  \task $2(x+3)^2-3(x+a)=2x^2+bx+3$
\end{tasks}

\ex{1} Efectua les següents divisions:\begin{tasks}(1)
  \task $(3x^3+2x^2+x-5):(3x^2+2)$
  \task $(3x^4-2x^2-x+4):(x^2+x+2)$
  \task $(x^3-3x^2-x+6):(2x^3-2x+3)$
\end{tasks}

\ex{1} Efectua les següents divisions aplicant la regla de Ruffini:\begin{tasks}(2)
  \task $(3x^3+2x^2+x-5):(x+2)$
  \task $(3x^4-2x^2-x+4):(x+2)$
  \task $(x^3-3x^2-x+6):(2x+3)$
  \task $(4x^4-3x^2+5):(-x+5)$
  \task $(2x^5-12x^3-6x^2-117):(x-3)$
\end{tasks}\recorda{Si el divisor és $ax+b$, fes Ruffini amb $-\frac ba$ i divideix el quocient entre $a$; el residu no canvia.}

\ex{2} Utilitza Ruffini per a calcular el valor de $a$ i $b$ que fa que les següents divisions siguen exactes:\begin{tasks}(1)

  \task $(4x^4+6x^3-13x^2+x+b):(-x+5)$
\end{tasks}

\ex{1} Comprova si $M(x)=2x^3-5x^2+4x-4$ és divisible per $x-2$ i, si ho és, troba un polinomi $N(x)$ que permeta escriure $M(x)=(x-2)\cdot N(x)$.

  \task $\dfrac{2x^2-8}{x^3+x^2-2x}$
  \task $\dfrac{x^3-x}{x^3+3x^2+2x}$
  \task $\dfrac{x^4-2x^3-3x^2}{x^4-9x^2}$
  \task $\dfrac{x^3-3x^2+3x-1}{x^3-2x^2+x}$
  \task $\dfrac{2x^4-5x^3+x^2+5x-3}{2x^3-x^2-7x+6}$
  \task $\dfrac{2x^4+x^3-11x^2+11x-3}{2x^3+3x^2-8x+3}$
  \task $\dfrac{x^3-2x^2-9x+18}{x^3-7x^2+16x-12}$
  \task $\dfrac{x^4-5x^3+9x^2-7x+2}{x^4-6x^3+13x^2-12x+4}$
\end{tasks}\recorda{Factoritza el numerador i el denominador i simplifica només els \emph{factors} comuns.}

\ex{2} Efectua les següents sumes i restes reduint prèviament a comú denominador:\begin{tasks}(2)
  \task $\dfrac{1}{x+1}+\dfrac{3}{x^2-1}$

  \task $\dfrac{1}{2x}-\dfrac{3}{4x^2-4x}$
  \task $\dfrac{1}{x^2+x}+\dfrac{3}{x^2-x}$
  \task $\dfrac{2x}{x-2}+\dfrac{3x-1}{x+2}-\dfrac{1}{x^2-4}$
  \task $\dfrac{1}{(x-1)^2}+\dfrac{2}{x-1}+\dfrac{1}{x^2-1}$
\end{tasks}

\ex{2} Opera i simplifica les fraccions algebraiques:\begin{tasks}(2)
  \task $\dfrac{3x+3}{12x-12}:\dfrac{(x+1)^2}{x^2-1}$

  \task $\dfrac{(x-1)^2}{2}\cdot\dfrac{1}{x^2-1}-\dfrac{3x}{(x+1)^2}$
  \task $\left(\dfrac{2x-1}{x+1}-\dfrac{3x}{x-1}\right)\cdot\dfrac{x^3-x}{-x^2-6x+1}$
\end{tasks}

\ex{3} Calcula i simplifica el resultat de les operacions següents:\begin{tasks}(2)
