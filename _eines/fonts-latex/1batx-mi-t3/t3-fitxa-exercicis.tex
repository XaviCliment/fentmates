\ex{3} Demostra les identitats següents:\begin{tasks}(2)
  \task $\dfrac{\sin\alpha-\cos\alpha}{\text{tg}\,\alpha-1}=\cos\alpha$
  \task $\text{tg}^2\alpha-\sin^2\alpha=\text{tg}^2\alpha\cdot\sin^2\alpha$
  \task $\dfrac{1+\text{cotg}\,\alpha}{\sin\alpha+\cos\alpha}=\text{cosec}\,\alpha$
  \task $\sec^2\alpha-1=\text{tg}^2\alpha$
  \task $\dfrac{1+\sin2\alpha}{1+\cos2\alpha}=\tfrac12(1+\text{tg}\,\alpha)^2$
  \task $2\,\text{tg}\,\alpha\sin^2\tfrac{\alpha}{2}+\sin\alpha=\text{tg}\,\alpha$
  \task $\cos3\alpha=4\cos^3\alpha-3\cos\alpha$
  \task $\dfrac{\text{tg}\,\alpha}{\cos2\alpha}=\text{tg}\,2\alpha-\text{tg}\,\alpha$
  \task $\text{tg}\,\alpha+\text{cotg}\,\alpha=\sec\alpha\cdot\text{cosec}\,\alpha$
  \task $\dfrac{\sin2\alpha}{1+\cos2\alpha}=\text{tg}\,\alpha$
