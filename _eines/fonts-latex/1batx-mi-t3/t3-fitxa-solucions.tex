\seccio{2}{Sistemes d'equacions no lineals}

\exf{2} \begin{tasks}(1)
  \task $x=\frac{-4-3y}{2}$ i substituïm: \ $\res{(- \frac{15}{7},\,\frac{2}{21}),\ (1,\,{-}2)}$
  \task $\frac{y+2x}{xy}=4\Rightarrow y+2x=\frac{18}{5}$: \ $\res{(\frac{3}{10},\,3),\ (\frac{3}{2},\,\frac{3}{5})}$
  \task Reducció: $7x^2=28$: \ $\res{(-2,\,{-}1),\ (-2,\,1),\ (2,\,{-}1),\ (2,\,1)}$
  \task Restant: $3y^2-y-14=0$: \ $\res{(\frac{17}{6},\,\frac{7}{3}),\ (5,\,{-}2)}$
  \task $x=\frac{-8-5y}{2}$ i substituïm: \ $\res{(-2,\,{-} \frac{4}{5}),\ (1,\,{-}2)}$
  \task $-x^2=1$: \ $\res{\text{no té solució real}}$
  \task $y=2x-3\Rightarrow x^2-4x+4=0$: \ $\res{(2,\,1)}$
  \task $y+1=12-x\Rightarrow x^2-7x+12=0$: \ $\res{(3,\,8),\ (4,\,7)}$
  \task $y=-\frac3x\Rightarrow x^4-19x^2+18=0$: \ $\res{(- 3 \sqrt{2},\,\frac{\sqrt{2}}{2}),\ (-1,\,3),\ (1,\,{-}3),\ (3 \sqrt{2},\,{-} \frac{\sqrt{2}}{2})}$
  \task Reducció: $17y^2=0$: \ $\res{(-4,\,0),\ (4,\,0)}$
  \task $(\sqrt x-\sqrt y)(\sqrt x+\sqrt y)=105\Rightarrow\sqrt x-\sqrt y=7$: \ $\res{(121,\,16)}$
  \task $(x-y)^2=16\Rightarrow x-y=\pm4$: \ $\res{(-11,\,{-}7),\ (11,\,7)}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $u+v=9,\ 4u+5v=41\Rightarrow u=4,\ v=5$: \ $\res{x=2,\ y=1}$
  \task $u+v=5,\ 2u-v=7\Rightarrow u=4,\ v=1$: \ $\res{x=2,\ y=0}$
  \task $2^x=128\cdot2^y\Rightarrow127\cdot2^y=1016\Rightarrow2^y=8$: \ $\res{x=10,\ y=3}$
  \task $u+v=16,\ \frac u3-49v=-340\Rightarrow u=9,\ v=7$: \ $\res{x=2,\ y=1}$
  \task $y=5-x$ i $t=2^x$: $\frac1t+\frac{3t}{32}=\frac58\Rightarrow3t^2-20t+32=0\Rightarrow t=4,\ t=\frac83$:\par $\res{(2,\,3)}$ \ i \ $\res{\left(\log_2\tfrac83,\ 5-\log_2\tfrac83\right)\approx(1,415;\ 3,585)}$
\end{tasks}

\ex{2} \begin{tasks}(1)

  \task $x=10y\Rightarrow32y=64$: \ $\res{x=20,\ y=2}$
  \task $2u-2v=6,\ 2u+2v=2\Rightarrow u=2,\ v=-1$: \ $\res{x=100,\ y=\frac1{10}}$
\end{tasks}
