\ex{2} Calcula, para cada función, $\displaystyle\lim_{x\to-\infty}f(x)$ y $\displaystyle\lim_{x\to+\infty}f(x)$. Fíjate bien en el dominio de las raíces.\begin{tasks}(2)
  \task $f(x)=-2x^3+7x^2$
  \task $f(x)=3x^4-7x$
  \task $f(x)=10^x$
  \task $f(x)=\sqrt{5x-8}$
  \task $f(x)=\sqrt{-2x^2+1}$
  \task $f(x)=-5^x$
  \task $f(x)=\sqrt{3-x}$
  \task $f(x)=\dfrac{x^2+3}{-x^3}$
  \task $f(x)=\dfrac{-x^3}{x^2+3}$
  \task $f(x)=\dfrac{5x^3-10}{3x^3+10x^2}$
\end{tasks}\recorda{Potencias, exponenciales de base mayor que 1, raíces y logaritmos tienden a $+\infty$ cuando $x\to+\infty$. Un polinomio se comporta como su término de mayor grado. Si la función no está definida hacia un infinito, el límite no existe.}

\ex{1} Calcula los límites siguientes comparando los grados del numerador y del denominador:\begin{tasks}(3)
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{x^2-5x+3}{3x-5}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{x^2+3}{x^3}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{3x^2-5x+1}{2x^2-6}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{4x^3-x}{2-x^2}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{1-6x}{3x+2}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{2x^2-1}{5-x}$
\end{tasks}\recorda{$\displaystyle\lim_{x\to\pm\infty}\frac{P(x)}{Q(x)}=\lim_{x\to\pm\infty}\frac{a_nx^n}{b_mx^m}$: \ si $n>m$, $\pm\infty$ (hay que mirar el signo); \ si $n=m$, $\frac{a_n}{b_m}$; \ si $n<m$, $0$.}

\ex{2} Calcula los límites siguientes con exponenciales y raíces:\begin{tasks}(3)
  \task $\displaystyle\lim_{x\to +\infty}\left(\tfrac13\right)^x$
  \task $\displaystyle\lim_{x\to -\infty}\left(3^x+1\right)$

  \task $\displaystyle\lim_{x\to +\infty}\dfrac{\sqrt{4x^2+1}}{x+3}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{\sqrt{x^2+1}}{2x}$
  \task $\displaystyle\lim_{x\to +\infty}\left(e^{-x}+2\right)$
\end{tasks}

\seccio{3}{Indeterminaciones en el infinito}

\exf{2} Indeterminación $\frac{\infty}{\infty}$ con raíces:\begin{tasks}(2)
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{\sqrt{x^2+1}+2x}{3x-1}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{\sqrt{4x^4+x}-x}{x^2+5}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{3x+\sqrt x}{x-\sqrt x}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{\sqrt{4x^3+x-2}}{8x^2-7}$
\end{tasks}\recorda{Nos quedamos con el término de mayor grado de cada parte: \ $\sqrt{x^2+1}\approx x$ (si $x>0$), \ $\sqrt{4x^3}=2x^{3/2}$.}

\ex{2} Indeterminación $\infty-\infty$ con fracciones:\begin{tasks}(2)
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{2x^2-5x}{x+3}-2x\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x^2}{x-2}-\dfrac{x^2}{x+2}\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x^3}{x^2-1}-x\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x^2+1}{x}+\dfrac{5x-x^2}{3x-1}\right)$
\end{tasks}\recorda{Si la indeterminación $\infty-\infty$ viene de una suma o resta de fracciones, \textbf{hacemos la operación} con denominador común y calculamos el límite de la fracción que resulta.}

\ex{2} Indeterminación $\infty-\infty$ con raíces:\begin{tasks}(3)
  \task $\displaystyle\lim_{x\to +\infty}\left(x-\sqrt{x^2-1}\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\sqrt{x+2}-\sqrt{x-2}\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\sqrt{x^2+6x}-x\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\sqrt{x^2-2}-\sqrt{x^2+x}\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(2x-\sqrt{4x^2-3x}\right)$
\end{tasks}\recorda{Multiplicamos y dividimos por el \textbf{conjugado}: $(A-B)(A+B)=A^2-B^2$ hace desaparecer las raíces del numerador.}

\ex{2} Indeterminación $1^{\infty}$:\begin{tasks}(3)
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x}{x+2}\right)^{2x+1}$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{2x-2}{2x+1}\right)^{3x^2}$
  \task $\displaystyle\lim_{x\to +\infty}\left(1+\dfrac3x\right)^{2x}$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x+1}{x-2}\right)^{x}$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x^2+1}{x^2-3}\right)^{x^2}$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{3x+1}{x+2}\right)^{x}$
\end{tasks}\recorda{Si $\lim f(x)=1$ y $\lim g(x)=\pm\infty$: \ $\lim f(x)^{g(x)}=e^{\lim\,(f(x)-1)\cdot g(x)}$. Si la base no tiende a 1, no hay indeterminación.}

\ex{2} Problemas. Calcula el límite que responde a cada pregunta:\begin{tasks}(1)
  \task El coste medio (en euros) de fabricar $x$ unidades de un producto es $C(x)=\dfrac{12x+3\,000}{x}$. ¿A qué valor se acerca el coste por unidad si se fabrican muchísimas?
  \task La concentración de un medicamento en sangre $t$ horas después de tomarlo es $c(t)=\dfrac{8t}{t^2+4}$ (mg/l). ¿Qué pasa con la concentración a muy largo plazo?
  \task Una población de conejos evoluciona según $P(t)=\dfrac{500t+300}{t+1}$, con $t$ en meses. ¿Cuál es la población inicial? ¿Tiene algún límite el número de conejos?
\end{tasks}
