\seccio{2}{Derivada en un punto y función derivada}

\exf{1} \begin{tasks}(1)
  \task $\dfrac{f(-2+h)-f(-2)}{h}=\dfrac{\frac{1}{-2+h}+\frac12}{h}=\dfrac{1}{2(-2+h)}\to\res{f'(-2)=-\frac14}$
  \task $\dfrac{-2(a+h)+4-(-2a+4)}{h}=\dfrac{-2h}{h}=-2$: \ $\res{f'(a)=-2}$ en todos los puntos, porque la gráfica es una recta de pendiente $-2$.
  \task $\dfrac{f(a+h)-f(a)}{h}=6a-5+3h\to6a-5$: \ $f'(-2)=\res{-17}$, \ $f'(1)=\res{1}$, \ $f'(0)=\res{-5}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $\dfrac{(x+h)^2+3(x+h)-x^2-3x}{h}=\dfrac{2xh+h^2+3h}{h}=2x+h+3\to\res{2x+3}$
  \task $\dfrac{\frac{2}{x+h+1}-\frac{2}{x+1}}{h}=\dfrac{-2h}{h(x+h+1)(x+1)}\to\res{\dfrac{-2}{(x+1)^2}}$
  \task Por el conjugado: $\dfrac{\sqrt{x+h+1}-\sqrt{x+1}}{h}=\dfrac{1}{\sqrt{x+h+1}+\sqrt{x+1}}\to\res{\dfrac{1}{2\sqrt{x+1}}}$
\end{tasks}

\ex{1} $\dfrac{(3+h)^2-9}{h}=6+h\to f'(3)=6$. \ Tangente por $(3,9)$: $y=9+6(x-3)\Rightarrow\res{y=6x-9}$

\seccio{6}{Monotonía y extremos}

\exf{1} \begin{tasks}(1)
  \task $f'(x)=-2x+5=0\Rightarrow x=\frac52$. \ Creciente en $\res{\left(-\infty,\frac52\right)}$ y decreciente en $\res{\left(\frac52,+\infty\right)}$. Máximo relativo (y absoluto) en $\res{\left(\frac52,\frac{21}{4}\right)}$.
  \task $f'(x)=6x^2-6x-12=6(x+1)(x-2)$. \ Creciente en $\res{(-\infty,\,-1)\cup(2,+\infty)}$, decreciente en $\res{(-1,2)}$. Máximo relativo $\res{(-1,12)}$, mínimo relativo $\res{(2,\,-15)}$.
\end{tasks}

\ex{2} \label{ex:racional}\begin{tasks}(1)
  \task $D=\mathbb R-\{1\}$, \ $f'(x)=\frac{x^2-2x}{(x-1)^2}=0\Rightarrow x=0,\ x=2$. \ Creciente en $\res{(-\infty,0)\cup(2,+\infty)}$, decreciente en $\res{(0,1)\cup(1,2)}$. Máximo $\res{(0,0)}$, mínimo $\res{(2,4)}$.
  \task $g'(x)=e^x(x+1)=0\Rightarrow x=-1$. \ Decreciente en $\res{(-\infty,\,-1)}$, creciente en $\res{(-1,+\infty)}$. Mínimo $\res{\left(-1,\,-\frac1e\right)}$.
  \task $h'(x)=\frac{2x}{x^2+1}=0\Rightarrow x=0$. \ Decreciente en $\res{(-\infty,0)}$, creciente en $\res{(0,+\infty)}$. Mínimo $\res{(0,0)}$.
\end{tasks}

\ex{2} $f'(x)=4x^3-16x=4x(x^2-4)=0\Rightarrow x=0,\ x=\pm2$. \ $f''(x)=12x^2-16$: \ $f''(0)=-16<0$: máximo $\res{(0,3)}$; \ $f''(\pm2)=32>0$: mínimos $\res{(-2,\,-13)}$ y $\res{(2,\,-13)}$.

\ex{2} $f'(x)=3x^2-3=0\Rightarrow x=\pm1$ (los dos dentro). \ $f(-2)=0$, $f(-1)=4$, $f(1)=0$, $f(3)=20$. \ Máximo absoluto $\res{(3,20)}$; mínimo absoluto $\res{0}$, en $x=-2$ y en $x=1$.

\ex{3} $f'(2)=12+4a=0\Rightarrow\res{a=-3}$. \ $f(2)=8-12+b=-3\Rightarrow\res{b=1}$. \ $f''(x)=6x-6$, $f''(2)=6>0$: \rest{mínimo relativo}.

\seccio{8}{Problemas de optimización}

\exf{1} $y=20-x$, \ $P(x)=x(20-x)=20x-x^2$. \ $P'(x)=20-2x=0\Rightarrow x=10$; $P''=-2<0$: máximo. \ Los números son $\res{10\text{ y }10}$ y el producto, $\res{100}$.

\ex{2} Lados perpendiculares al río $x$ y lado paralelo $y=\frac{800}{x}$: \ $L(x)=2x+\frac{800}{x}$, $x>0$. \ $L'(x)=2-\frac{800}{x^2}=0\Rightarrow x=20$ ($L''>0$). \ $\res{20\text{ m}\times40\text{ m}}$, con $\res{80\text{ m}}$ de valla.

\ex{2} $V(x)=x(12-2x)^2$, $0<x<6$. \ $V'(x)=(12-2x)(12-6x)=0\Rightarrow x=2$ ($x=6$ no es válido). \ $\res{x=2\text{ cm}}$ y $V(2)=2\cdot8^2=\res{128\text{ cm}^3}$.

\ex{2} Si $x$ es el lado de moldura y $z$ el otro: $C=19x+x+2z=20x+2z$ y $xz=1\,000$. \ $C(x)=20x+\frac{2\,000}{x}$, \ $C'(x)=20-\frac{2\,000}{x^2}=0\Rightarrow x=10$. \ $\res{10\text{ cm}\times100\text{ cm}}$, con un coste de $\res{400\text{ €}}$.

\ex{3} $C(t)=(10\,000-10t^2)(2+0{,}05t)$, $t\in[0,20]$. \ $C'(t)=-1{,}5t^2-40t+500=0\Rightarrow t\approx9{,}27$. \ Con días enteros: $C(9)=22\,515{,}5$, $C(10)=22\,500$; \ $C(0)=20\,000$, $C(20)=18\,000$. \ Le conviene vender el \rest{noveno día}: $\res{22\,515{,}50\text{ €}}$.

\ex{2} $M(x)=\frac{C(x)}{x}=x+100+\frac{3\,600}{x}$. \ $M'(x)=1-\frac{3\,600}{x^2}=0\Rightarrow x=60$. \ $\res{60\text{ unidades}}$, coste medio $\res{220\text{ €}}$.

\ex{3} $d^2(x)=x^2+(x^2-2)^2$ (minimizar $d^2$ es lo mismo que minimizar $d$). \ $(d^2)'=2x+4x(x^2-2)=2x(2x^2-3)=0\Rightarrow x=0,\ x=\pm\sqrt{\tfrac32}$. \ $d^2(0)=4$, $d^2\!\left(\pm\sqrt{\tfrac32}\right)=\frac74$. \ Puntos $\res{\left(\pm\frac{\sqrt6}{2},\frac32\right)}$, a distancia $\frac{\sqrt7}{2}$.
