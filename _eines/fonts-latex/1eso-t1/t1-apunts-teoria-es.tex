Para calcular de cabeza: \ $7\cdot99=7\cdot(100-1)=700-7=693$\\[3pt]

\begin{idea}[La división entera]
Si no es exacta, da un \textbf{cociente} $q$ y un \textbf{resto} $r$, siempre \textbf{menor} que el divisor:\\[4pt]
{\centering\Large$D=d\cdot q+r$ \qquad \normalsize$247=12\cdot20+7$\par}\smallskip
Es la \textbf{prueba de la división}: si no se cumple, hay un error.
\end{idea}
\end{minipage}

\vfill
\begin{exemple}[Ejemplos]
\begin{minipage}[t]{0.48\linewidth}
250 caramelos en bolsas de 12: \ $250=12\cdot20+10$\\ \hspace*{1em}\boxed{20\text{ bolsas y sobran }10}\\[4pt]
$25\cdot101=25\cdot100+25=\boxed{2\,525}$
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
¿Qué dividendo tiene divisor 15, cociente 8 y resto 6?\\ \hspace*{1em}$D=15\cdot8+6=\boxed{126}$\\[4pt]
$36\cdot7+36\cdot3=36\cdot10=\boxed{360}$
\end{minipage}
\end{exemple}

\vfill
\begin{compte}[El resto siempre es menor que el divisor]
$100=7\cdot13+9$ \mal{} \ no es una división entera: $9>7$. \ La correcta: \ $100=7\cdot14+2$ \be{} \qquad Y \textbf{no se puede dividir entre 0}.
\end{compte}

\seccio{sec2}{Aproximación: truncamiento y redondeo}{Un número próximo, más fácil de leer y de recordar.}

\vfill
\begin{idea}[Aproximar]
Cambiar un número por otro próximo y más fácil. En un concierto hay 44\,863 personas: el periódico dirá «unas 45\,000». Se aproxima a un \textbf{orden}: decenas, centenas, millares\dots
\end{idea}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmA!8,colframe=fmA,coltitle=white,title={\faIcon{cut}\ \ Truncar}]
Cambiamos por \textbf{ceros} todas las cifras a la derecha del orden: «cortamos».\par\smallskip
{\centering$44\,8\underline{63}\to44\,800$ \ \nota{a las centenas}\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmB!8,colframe=fmB,coltitle=white,title={\faIcon{circle}\ \ Redondear}]
Miramos la cifra \textbf{siguiente}: si es 0, 1, 2, 3 o 4, la del orden se queda igual; si es 5, 6, 7, 8 o 9, le \textbf{sumamos 1}. Después, ceros.\par\smallskip
{\centering$44\,8\fbox{6}3\to44\,900$ \qquad $44\,8\fbox{2}3\to44\,800$\par}
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{center}
\begin{tikzpicture}[x=0.11cm]
  \draw[fmnavy,line width=1.1pt] (0,0) -- (100,0);
  \foreach \k in {0,10,...,100} \draw[fmnavy] (\k,-0.12) -- (\k,0.12);
  \node[below,font=\small] at (0,-0.1) {44\,800}; \node[below,font=\small] at (50,-0.1) {44\,850}; \node[below,font=\small] at (100,-0.1) {44\,900};
  \fill[fmR] (63,0) circle (3pt) node[above=2pt,font=\small\heavy,text=fmR] {44\,863};
  \draw[-{Latex[length=2mm]},fmK,line width=1.2pt] (63,0.5) to[bend left=20] node[above,font=\scriptsize] {más cerca} (100,0.5);
\end{tikzpicture}\\
{\comic\small\color{fmmuted}44\,863 está más cerca de 44\,900 que de 44\,800}
\end{center}

\vfill
\begin{exemple}[Ejemplo: \ 44\,863 aproximado de las dos maneras]
\centering\small
\taularod[fmex]{>{\heavy}l*{4}{>{\hspace{8pt}}c<{\hspace{8pt}}}}{
\rowcolor{fmex}\color{white}44\,863 & \color{white}Truncamiento & \color{white}Error & \color{white}Redondeo & \color{white}Error\\
a las decenas & 44\,860 & 3 & 44\,860 & 3\\
\rowcolor{fmexbg}a las centenas & 44\,800 & 63 & 44\,900 & \cK{37}\\
a los millares & 44\,000 & 863 & 45\,000 & \cK{137}
}\par\smallskip\raggedright
El \textbf{error} es la diferencia entre el exacto y el aproximado: el redondeo siempre da el \textbf{más próximo}.
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
\begin{minipage}[t]{0.48\linewidth}
Redondea 7\,459 a las centenas: \ $\boxed{7\,500}$\\[4pt]
Redondea 128\,350 a los millares: \ $\boxed{128\,000}$
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
Trunca 9\,987 a las decenas: \ $\boxed{9\,980}$\\[4pt]
Redondea 9\,987 a las decenas: \ $\boxed{9\,990}$
\end{minipage}
\end{exemple}

\vfill
\begin{compte}[Cuando se arrastra]
2\,996 a las centenas: la cifra siguiente es un 9, así que sumamos 1 a las 9 centenas: se hacen 10 y nos llevamos una al millar: \ $\boxed{3\,000}$.
\end{compte}

\seccio{sec3}{Operaciones con números naturales}{Las propiedades nos ayudan a calcular más rápido.}

\vfill
\begin{tcbraster}[raster columns=3,raster equal height,raster column skip=8pt]
\begin{tcolorbox}[fmbase,colback=fmA!8,colframe=fmA,coltitle=white,title={Conmutativa},before upper=\centering]
El orden no importa:\par\smallskip
$7+5=5+7$\par $4\cdot9=9\cdot4$
\end{tcolorbox}

\end{tcolorbox}

$7\cdot99=700-7=693$
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{compte}[La resta y la división no son conmutativas]
$9-4\neq4-9$ \qquad $12:3\neq3:12$. \ Y recuerda que \textbf{no se puede dividir entre 0}.
\end{compte}

$=30-24+4$ \ \nota{\pas{2}}\\[3pt]

\end{exemple}

\passosproblema{\textbf{Piensa}\\un plan}{\textbf{Elige}\\las operaciones}

\vfill
\begin{minipage}[c]{0.55\linewidth}\centering\small
\taularod{>{\hspace{3pt}}l<{\hspace{3pt}}>{\hspace{3pt}}c<{\hspace{3pt}}}{
\rowcolor{fmblue}\color{white}\heavy Si el problema dice\dots & \color{white}\heavy Operación\\
juntar, añadir, en total, entre todos & $+$\\
\rowcolor{fmsurface}quitar, gastar, cuánto queda, cuánto falta & $-$\\

}

\begin{minipage}[c]{0.42\linewidth}

\end{compte}
\end{minipage}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt,raster row skip=8pt]
\begin{exemple}[Ejemplo: \ autobuses]
150 alumnos van de excursión en autobuses de 55 plazas. ¿Cuántos autobuses hacen falta?\\[3pt]
$150=55\cdot2+40$: sobran 40 alumnos, hace falta un autobús más: \ \boxed{3\text{ autobuses}}
\end{exemple}
\begin{exemple}[Ejemplo: \ cajas llenas]
Tenemos 150 huevos y los ponemos en cajas de 12. ¿Cuántas cajas llenas?\\[3pt]
$150=12\cdot12+6$: \ \boxed{12\text{ cajas}} \ y sobran 6.
\end{exemple}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplo: \ una familia compra una tele de 549 € y paga 3 mensualidades de 125 € y el resto al final. ¿Cuánto paga al final?]
Mensualidades: \ $3\cdot125=375$ € \qquad Resto: \ $549-375=\boxed{174\text{ €}}$ \qquad \nota{comprueba: $375+174=549$}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ en una biblioteca hay 12 estanterías de 5 baldas con 35 libros cada una. Llegan 300 libros más]
Libros: \ $12\cdot5\cdot35=2\,100$ \qquad Ahora: \ $2\,100+300=\boxed{2\,400}$ \qquad Por estantería: \ $2\,400:12=\boxed{200}$
\end{exemple}

\vfill
\begin{truc}[Escribe qué calculas]
En cada paso, una frase corta: «Mensualidades: 375 €». Al final, responde con una \textbf{frase} y comprueba que tiene sentido.
\end{truc}

\begin{minipage}[c]{0.47\linewidth}
\begin{idea}[Potencia]
La \cA{\textbf{base}} es el factor que se repite; el \cB{\textbf{exponente}}, cuántas veces.\\[3pt]
{\centering$a^n=\underbrace{a\cdot a\cdots a}_{n\text{ veces}}$\par}
\end{idea}
\end{minipage}

\vfill
\begin{compte}[El exponente no multiplica]
\be{2^5=2\cdot2\cdot2\cdot2\cdot2=32} \qquad \mal{2^5=2\cdot5=10}
\end{compte}

\vfill

\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Cómo se leen},before upper=\centering]
$a^2$: «al cuadrado»\par $a^3$: «al cubo»\par $7^4$: «siete elevado a cuatro»
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Casos especiales},before upper=\centering]
$a^1=a$\par\smallskip $a^0=1$ \ (si $a\neq0$)\par\smallskip $1^n=1$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Cuadrados y cubos},before upper=\centering]
\begin{tikzpicture}[scale=0.28]

\end{tikzpicture}
\end{tcolorbox}
\end{tcbraster}

\vfill

\begin{minipage}[t]{0.48\linewidth}

$2^6=\boxed{64}$ \qquad $10^4=\boxed{10\,000}$\\[4pt]

\end{minipage}\hfill

\end{minipage}
\end{exemple}

\end{exemple}

\seccio{sec7}{Propiedades de las potencias}{Con la misma base, jugamos con los exponentes.}

\providecommand{\prop}{}\renewcommand{\prop}[3]{\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={#1},before upper=\centering]
{\Large$#2$}\par\smallskip{\small$#3$}
\end{tcolorbox}}
\begin{tcbraster}[raster columns=3,raster equal height=rows,raster column skip=8pt]
\prop{\pas{1}\ Producto}{a^{m}\cdot a^{n}=a^{\cK{m+n}}}{2^{5}\cdot2^{3}=2^{8}=256}
\prop{\pas{2}\ Cociente}{a^{m}:a^{n}=a^{\cK{m-n}}}{7^{9}:7^{7}=7^{2}=49}
\prop{\pas{3}\ Potencia de potencia}{\left(a^{m}\right)^{n}=a^{\cK{m\cdot n}}}{\left(5^{2}\right)^{3}=5^{6}=15\,625}
\end{tcbraster}

\vfill
\begin{minipage}[c]{0.5\linewidth}
\begin{idea}[Exponente 1 y exponente 0]
$a^{1}=a$: \ $7^{1}=7$\\[4pt]
$a^{0}=1$ \ (si $a\neq0$): \ $5^{3}:5^{3}=5^{0}=1$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
\begin{truc}[¿Por qué se suman?]
$2^{3}\cdot2^{2}=(2\cdot2\cdot2)\cdot(2\cdot2)$: \ en total, $3+2=5$ factores iguales: \ $2^{5}$.
\end{truc}
\end{minipage}

\vfill
\begin{compte}[Solo con la misma base, y para productos y cocientes]
\mal{2^{2}+2^{3}=2^{5}}\ \ {\small$4+8=12$, \ pero $2^5=32$}\qquad\quad
$2^{3}\cdot3^{2}$: bases distintas, calculamos cada una: \ $8\cdot9=72$
\end{compte}

\vfill
\begin{exemple}[Ejemplos de cada propiedad]
\begin{minipage}[t]{0.32\linewidth}
\pas{1} $4^{2}\cdot4^{5}=4^{7}$\\[4pt]
\hspace*{1.3em}$10\cdot10^{4}=10^{5}$\\[4pt]
\hspace*{1.3em}$2^{3}\cdot2^{3}\cdot2^{3}=2^{9}$
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}
\pas{2} $3^{9}:3^{4}=3^{5}$\\[4pt]
\hspace*{1.3em}$10^{6}:10^{2}=10^{4}$\\[4pt]
\hspace*{1.3em}$8^{5}:8^{4}=8^{1}=8$
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}
\pas{3} $\left(3^{2}\right)^{4}=3^{8}$\\[4pt]
\hspace*{1.3em}$\left(10^{3}\right)^{2}=10^{6}$\\[4pt]
\hspace*{1.3em}$\left(6^{5}\right)^{1}=6^{5}$
\end{minipage}
\end{exemple}

\subsection{Aplicamos las propiedades}
\begin{exemple}[Ejemplos: \ escribe como una sola potencia]
\begin{minipage}[t]{0.48\linewidth}
$3^{4}\cdot3^{2}\cdot3=3^{4+2+1}=\boxed{3^{7}}$ \ \nota{$3=3^1$}\\[5pt]
$5^{8}:5^{3}=5^{8-3}=\boxed{5^{5}}$\\[5pt]
$\left(2^{3}\right)^{4}=2^{3\cdot4}=\boxed{2^{12}}$
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
$10^{2}\cdot10^{3}\cdot10^{4}=\boxed{10^{9}}$\\[5pt]
$6^{10}:6^{9}=6^{1}=\boxed{6}$\\[5pt]
$\left(4^{2}\right)^{0}=4^{0}=\boxed{1}$
\end{minipage}
\end{exemple}

\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{exemple}[Ejemplo: \ $\left(2^{5}\cdot2^{3}\right):2^{6}$]
$=2^{8}:2^{6}$ \ \nota{\pas{1} primero el paréntesis}\\[3pt]
$=2^{2}=\boxed{4}$ \ \nota{\pas{2}}
\end{exemple}
\begin{exemple}[Ejemplo: \ $\left(7^{2}\right)^{3}:7^{4}$]
$=7^{6}:7^{4}$ \ \nota{\pas{3}}\\[3pt]
$=7^{2}=\boxed{49}$ \ \nota{\pas{2}}
\end{exemple}
\begin{exemple}[{Ejemplo: \ $\left[\left(3^{2}\right)^{3}\cdot3^{4}\right]:3^{8}$}]
$=\left[3^{6}\cdot3^{4}\right]:3^{8}=3^{10}:3^{8}$\\[3pt]
$=3^{2}=\boxed{9}$
\end{exemple}
\begin{exemple}[Ejemplo: \ $\left(5^{4}\cdot5^{2}\right):\left(5^{3}\right)^{2}$]
$=5^{6}:5^{6}$\\[3pt]
$=5^{0}=\boxed{1}$ \ \nota{dividendo y divisor iguales}
\end{exemple}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplos: \ completa el exponente que falta]
$4^{\square}\cdot4^{3}=4^{7}$: \ $\square=\boxed{4}$ \qquad $9^{8}:9^{\square}=9^{5}$: \ $\square=\boxed{3}$ \qquad $\left(2^{\square}\right)^{3}=2^{12}$: \ $\square=\boxed{4}$ \qquad $\dfrac{10^{7}\cdot10^{2}}{10^{6}}=10^{\square}$: \ $\square=\boxed{3}$
\end{exemple}

$3^{2}\cdot3^{4}=3^{6}$, \ \mal{9^{6}}: la base no se multiplica\\[3pt] $\left(2^{3}\right)^{2}=2^{6}$, \ \mal{2^{5}}: los exponentes se multiplican\\[3pt] $5^{6}:5^{2}=5^{4}$, \ \mal{5^{3}}: se restan, no se dividen
\end{compte}

\seccio{sec8}{Raíces cuadradas}{La operación inversa de elevar al cuadrado.}

\vfill
\begin{minipage}[c]{0.55\linewidth}
\begin{idea}[Raíz cuadrada]
{\centering\Large$\sqrt a=b\iff b^2=a$\par}\smallskip
$\sqrt{49}=7$ porque $7^2=49$. \ El signo $\sqrt{\ }$ es el \textbf{radical} y el número de dentro, el \textbf{radicando}.\\[3pt]
Elevar al cuadrado y hacer la raíz se \textbf{deshacen} la una a la otra.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.42\linewidth}\centering
\begin{tikzpicture}[scale=0.32]
  \draw[fmA,fill=fmA!20] (0,0) grid (7,7); \draw[fmA,line width=1.3pt] (0,0) rectangle (7,7);
  \node[below,font=\small] at (3.5,0) {7 cm}; \node[font=\small\heavy] at (3.5,3.5) {49 cm$^2$};
\end{tikzpicture}\\
{\comic\small\color{fmmuted}un cuadrado de 49 cm$^2$ mide 7 cm de lado}
\end{minipage}

\vfill
\begin{center}\small
\taularod{>{\heavy}l*{15}{>{\hspace{1pt}}c<{\hspace{1pt}}}}{
\rowcolor{fmblue}\color{white}Raíz & \color{white}1 & \color{white}2 & \color{white}3 & \color{white}4 & \color{white}5 & \color{white}6 & \color{white}7 & \color{white}8 & \color{white}9 & \color{white}10 & \color{white}11 & \color{white}12 & \color{white}13 & \color{white}14 & \color{white}15\\
Cuadrado & 1 & 4 & 9 & 16 & 25 & 36 & 49 & 64 & 81 & 100 & 121 & 144 & 169 & 196 & 225
}\\[3pt]
{\comic\color{fmmuted}los \textbf{cuadrados perfectos}: su raíz es exacta}
\end{center}

\vfill
\begin{minipage}[c]{0.5\linewidth}
\begin{idea}[Raíz entera y resto]
Si no es un cuadrado perfecto, la \textbf{raíz entera} es el natural más grande cuyo cuadrado no pasa del radicando:\\[3pt]
$\sqrt{40}$: \ $6^2=36\le40<49=7^2$ \ $\to$ \ raíz entera 6\\[2pt]
Resto: \ $40-36=4$ \qquad $40=6^2+4$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
\begin{truc}[Los ceros]
Si acaba en un número par de ceros, la raíz tiene la mitad:\\[3pt]
$\sqrt{400}=20$ \qquad $\sqrt{2\,500}=50$\\[2pt]
$\sqrt{10\,000}=100$
\end{truc}
\end{minipage}

\vfill
\begin{exemple}[Ejemplos]
\begin{minipage}[t]{0.48\linewidth}
$\sqrt{81}=\boxed{9}$ \qquad $\sqrt{144}=\boxed{12}$ \qquad $\sqrt{900}=\boxed{30}$\\[4pt]
$\sqrt{60}$: \ $7^2=49$, $8^2=64$: \ raíz entera \boxed{7}, resto 11
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
$\sqrt{150}$: \ $12^2=144$: \ raíz entera \boxed{12}, resto 6\\[4pt]
Con 30 baldosas, el cuadrado más grande: \ $5\times5$ (25) y sobran 5
\end{minipage}
\end{exemple}

\seccio{sec9}{Problemas con potencias y raíces}{La jerarquía completa, y problemas con cuadrados y cubos.}

\vfill
\begin{minipage}[c]{0.42\linewidth}\centering
\begin{tikzpicture}[st/.style={draw=#1,fill=#1!12,line width=1.2pt,rounded corners=2mm,minimum height=0.8cm,anchor=south west,font=\large,inner xsep=6pt}]
  \node[st=fmV] at (0,2.7) {\pas{1}\ \ Paréntesis $(\ )\ [\ ]$};

  \node[st=fmK] at (1.0,0.9) {\pas{3}\ \ $\cdot$ \ y \ $:$};
  \node[st=fmB] at (1.5,0) {\pas{4}\ \ $+$ \ y \ $-$};
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.55\linewidth}
\begin{exemple}[Ejemplo: \ el paréntesis cambia el resultado]
$2+3\cdot4^2=2+3\cdot16=2+48=\boxed{50}$\\[4pt]
$(2+3)\cdot4^2=5\cdot16=\boxed{80}$
\end{exemple}
\begin{truc}[El radical hace de paréntesis]
Primero calculamos lo que hay dentro: \ $\sqrt{16+9}=\sqrt{25}=5$.
\end{truc}
\end{minipage}

\vfill
\begin{compte}[No se reparten en las sumas]
$\sqrt{9+16}=5$, pero $\sqrt9+\sqrt{16}=7$. \qquad $(2+3)^2=25$, pero $2^2+3^2=13$.
\end{compte}

\vfill
\begin{exemple}[Ejemplos de operaciones]
\begin{minipage}[t]{0.48\linewidth}
$20-3^2\cdot2+\sqrt{16}=20-18+4=\boxed{6}$\\[4pt]
$\sqrt{49}+2\cdot(8-3)^2=7+50=\boxed{57}$
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
$50-[18:(3^2-6)]\cdot4=50-24=\boxed{26}$\\[4pt]
$5\cdot\sqrt{36}-2^4:8=30-2=\boxed{28}$
\end{minipage}
\end{exemple}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt,raster row skip=8pt]
\begin{exemple}[Ejemplo: \ un patio cuadrado tiene 225 m$^2$. ¿Cuánto mide el lado? ¿Y la valla?]
Lado: \ $\sqrt{225}=\boxed{15\text{ m}}$\\[3pt]
Valla: \ $4\cdot15=\boxed{60\text{ m}}$
\end{exemple}
\begin{exemple}[Ejemplo: \ un cubo de 4 cm de arista, hecho de cubitos de 1 cm. ¿Cuántos cubitos tiene?]
$4^3=4\cdot4\cdot4=\boxed{64}$ cubitos
\end{exemple}
\begin{exemple}[Ejemplo: \ un mensaje lo reciben 3 personas; cada una lo envía a 3 más, y así 5 veces]
En el 5.º paso lo reciben \ $3^5=\boxed{243}$ personas
\end{exemple}
\begin{exemple}[Ejemplo: \ queremos hacer el cuadrado más grande posible con 90 baldosas]
$9^2=81\le90<100$: \ un cuadrado de $\boxed{9\times9}$ y sobran 9
\end{exemple}
\end{tcbraster}
