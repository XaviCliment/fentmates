\seccio{1}{La relación de divisibilidad}

\ex{1} \begin{tasks}(2)
  \task $84=7\cdot12$: \res{\text{sí}}
  \task $95=8\cdot11+7$: \res{\text{no}}
  \task $132=12\cdot11$: \res{\text{sí}}
  \task $200=15\cdot13+5$: \res{\text{no}}
\end{tasks}

\ex{1} \begin{tasks}(4)
  \task múltiplo
  \task divisor
  \task múltiplo
  \task divisor
\end{tasks}

\ex{1} $72$ es múltiplo de $8$; \ $72$ es múltiplo de $9$; \ $8$ es divisor de $72$; \ $9$ es divisor de $72$.

\ex{2} \begin{tasks}(1)
  \task \res{\text{V}}: $48=6\cdot8$.
  \task \res{\text{V}}: $77=7\cdot11$.
  \task \res{\text{F}}: $120=9\cdot13+3$.
  \task \res{\text{V}}: $0=15\cdot0$.
  \task \res{\text{F}}: $1:3$ no es exacta (es $3$ el que es múltiplo de $1$).
\end{tasks}

\ex{2} $100=9\cdot11+1$. \ Sumar: $\res{8}$ ($108=9\cdot12$). \ Restar: $\res{1}$ ($99=9\cdot11$).

\ex{3} Exacta: $13\cdot17=\res{221}$. \ El resto como mucho es $12$: $221+12=\res{233}$.

\ex{3} De los múltiplos de 7 de dos cifras ($14, 21, \ldots, 98$), acaban en 4: $\res{14}$ y $\res{84}$.

\seccio{2}{Múltiplos de un número}

\ex{1} \begin{tasks}(1)
  \task $7,\ 14,\ 21,\ 28,\ 35,\ 42,\ 49,\ 56,\ 63,\ 70$
  \task $11,\ 22,\ 33,\ 44,\ 55,\ 66,\ 77,\ 88,\ 99,\ 110$
  \task $15,\ 30,\ 45,\ 60,\ 75,\ 90,\ 105,\ 120,\ 135,\ 150$
\end{tasks}

\ex{1} $\res{18,\ 36,\ 54,\ 60,\ 96}$ \ (los demás no dan división exacta entre 6).

\ex{2} $100=13\cdot7+9$: el primero es $13\cdot8=104$. \ $\res{104,\ 117,\ 130,\ 143,\ 156,\ 169}$.

\ex{2} $200:9\to q=22$: \ hay $\res{22}$. \ Hasta $99$ hay $11$, así que entre 100 y 200: $22-11=\res{11}$ (del $108$ al $198$).

\ex{2} $\res{18,\ 36,\ 54,\ 72,\ 90}$. El más pequeño es $18$ (su m.c.m.).

\ex{3} $300=17\cdot17+11$: los múltiplos más próximos son $289$ (a 11) y $306$ (a 6). \ El más próximo es $\res{306}$.

\ex{3} $999=12\cdot83+3$: \ $12\cdot83=\res{996}$ \ y \ $12\cdot84=\res{1008}$.

\seccio{4}{Criterios de divisibilidad}

\ex{1} \par\smallskip{\renewcommand{\arraystretch}{1.5}\centering\begin{tabular}{|C{2.4cm}|C{1.5cm}|C{1.5cm}|C{1.5cm}|C{1.5cm}|C{1.5cm}|}\hline\rowcolor{soft}\textbf{Número} & $\boldsymbol{\div2}$ & $\boldsymbol{\div3}$ & $\boldsymbol{\div5}$ & $\boldsymbol{\div9}$ & $\boldsymbol{\div11}$\\\hline $3080$ & \checkmark & $\times$ & \checkmark & $\times$ & \checkmark\\\hline $7425$ & $\times$ & \checkmark & \checkmark & \checkmark & \checkmark\\\hline $1188$ & \checkmark & \checkmark & $\times$ & \checkmark & \checkmark\\\hline $2543$ & $\times$ & $\times$ & $\times$ & $\times$ & $\times$\\\hline $6735$ & $\times$ & \checkmark & \checkmark & $\times$ & $\times$\\\hline \end{tabular}\par}\par El $2543$ no es divisible por ninguno: es impar, las cifras suman 14 y la diferencia para el 11 es $8-6=2$.

\ex{1} Por ejemplo, $\res{102,\ 111,\ 123}$: las cifras suman 3, 3 y 6 (múltiplos de 3, pero no de 9).

\ex{2} $5+4=9$. \ Por 3: $\square=0,3,6,9$ \ ($\res{504,\ 534,\ 564,\ 594}$). \ Por 9: $\square=0$ o $9$ \ ($\res{504,\ 594}$).

\ex{2} Por 3: $11+\square$ múltiplo de 3, \ $\square=1, 4, 7$. Por 2 tiene que ser par: $\square=4$, \ $\res{384}$.

\ex{2} \begin{tasks}(2)
  \task $(9+8)-(4+2)=11$: \res{\text{sí}}
  \task $(7+9)-(5+3)=8$: \res{\text{no}}
  \task $(2+1+5)-(8+0)=0$: \res{\text{sí}}
  \task $(5+5+1)-(7+3)=1$: \res{\text{no}}
\end{tasks}

\ex{3} Tiene que acabar en 0 (por 2 y por 5) y las cifras tienen que sumar un múltiplo de 3: $\res{990}$ ($9+9+0=18$).

\ex{3} Por 5: $b=0$ o $5$. \ Si $b=0$: $9+a$ múltiplo de 9, $a=0$ o $9$. \ Si $b=5$: $14+a$, $a=4$. \ \ $\res{7020,\ 7920,\ 7425}$.

\ex{2} Si acaba en 5, es divisible por 5: tiene al menos los divisores 1, 5 y él mismo. Si es par, es divisible por 2: tiene al menos los divisores 1, 2 y él mismo.

\ex{3} $2027$ no es divisible por ningún primo hasta el $43$, y $47\cdot47=2209>2027$: es \res{\text{primo}}. \ $2028$ es par: \res{\text{compuesto}} ($2028=2^{2}\cdot3\cdot13^{2}$).

\ex{3} Por ejemplo, $5+7+11=\res{23}$ \ o \ $7+11+13=\res{31}$.
