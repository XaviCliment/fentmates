\seccio{1}{La relació de divisibilitat}

\ex{1} \begin{tasks}(2)
  \task $84=7\cdot12$: \res{\text{sí}}
  \task $95=8\cdot11+7$: \res{\text{no}}
  \task $132=12\cdot11$: \res{\text{sí}}
  \task $200=15\cdot13+5$: \res{\text{no}}
\end{tasks}

\ex{1} \begin{tasks}(4)
  \task múltiple
  \task divisor
  \task múltiple
  \task divisor
\end{tasks}

\ex{1} $72$ és múltiple de $8$; \ $72$ és múltiple de $9$; \ $8$ és divisor de $72$; \ $9$ és divisor de $72$.

\ex{2} \begin{tasks}(1)
  \task \res{\text{V}}: $48=6\cdot8$.
  \task \res{\text{V}}: $77=7\cdot11$.
  \task \res{\text{F}}: $120=9\cdot13+3$.
  \task \res{\text{V}}: $0=15\cdot0$.
  \task \res{\text{F}}: $1:3$ no és exacta (és $3$ el que és múltiple d'$1$).
\end{tasks}

\ex{2} $100=9\cdot11+1$. \ Sumar: $\res{8}$ ($108=9\cdot12$). \ Restar: $\res{1}$ ($99=9\cdot11$).

\ex{3} Exacta: $13\cdot17=\res{221}$. \ El residu com a molt és $12$: $221+12=\res{233}$.

\ex{3} Dels múltiples de 7 de dues xifres ($14, 21, \ldots, 98$), acaben en 4: $\res{14}$ i $\res{84}$.

\seccio{2}{Múltiples d'un nombre}

\ex{1} \begin{tasks}(1)
  \task $7,\ 14,\ 21,\ 28,\ 35,\ 42,\ 49,\ 56,\ 63,\ 70$
  \task $11,\ 22,\ 33,\ 44,\ 55,\ 66,\ 77,\ 88,\ 99,\ 110$
  \task $15,\ 30,\ 45,\ 60,\ 75,\ 90,\ 105,\ 120,\ 135,\ 150$
\end{tasks}

\ex{1} $\res{18,\ 36,\ 54,\ 60,\ 96}$ \ (la resta no donen divisió exacta entre 6).

\ex{2} $100=13\cdot7+9$: el primer és $13\cdot8=104$. \ $\res{104,\ 117,\ 130,\ 143,\ 156,\ 169}$.

\ex{2} $200:9\to q=22$: \ n'hi ha $\res{22}$. \ Fins a $99$ n'hi ha $11$, així que entre 100 i 200: $22-11=\res{11}$ (del $108$ al $198$).

\ex{2} $\res{18,\ 36,\ 54,\ 72,\ 90}$. El més xicotet és $18$ (el seu mcm).

\ex{3} $300=17\cdot17+11$: els múltiples més pròxims són $289$ (a 11) i $306$ (a 6). \ El més pròxim és $\res{306}$.

\ex{3} $999=12\cdot83+3$: \ $12\cdot83=\res{996}$ \ i \ $12\cdot84=\res{1008}$.

  \task $\{1,\ 2,\ 4,\ 8,\ 16,\ 32\}$

\ex{1} $\res{4,\ 12,\ 15,\ 30}$ \ ($60=4\cdot15=12\cdot5=15\cdot4=30\cdot2$).

\ex{2} \begin{tasks}(1)
  \task $\{1,2,3,4,6,8,12,16,24,48\}$
  \task $\{1,2,3,4,6,8,9,12,18,24,36,72\}$
  \task $\{1,2,3,5,6,9,10,15,18,30,45,90\}$

\ex{2} Tantes com divisors té 36: $1\times36,\ 2\times18,\ 3\times12,\ 4\times9,\ 6\times6,\ 9\times4,\ 12\times3,\ 18\times2,\ 36\times1$: \ $\res{9\text{ maneres}}$.

\ex{3} El $\res{48}$, amb 10 divisors: $1, 2, 3, 4, 6, 8, 12, 16, 24, 48$.

\ex{3} Per exemple, $\res{6}$: $\{1,2,3,6\}$ \ i \ $\res{16}$: $\{1,2,4,8,16\}$.

\seccio{4}{Criteris de divisibilitat}

\ex{1} \par\smallskip{\renewcommand{\arraystretch}{1.5}\centering\begin{tabular}{|C{2.4cm}|C{1.5cm}|C{1.5cm}|C{1.5cm}|C{1.5cm}|C{1.5cm}|}\hline\rowcolor{soft}\textbf{Nombre} & $\boldsymbol{\div2}$ & $\boldsymbol{\div3}$ & $\boldsymbol{\div5}$ & $\boldsymbol{\div9}$ & $\boldsymbol{\div11}$\\\hline $3080$ & \checkmark & $\times$ & \checkmark & $\times$ & \checkmark\\\hline $7425$ & $\times$ & \checkmark & \checkmark & \checkmark & \checkmark\\\hline $1188$ & \checkmark & \checkmark & $\times$ & \checkmark & \checkmark\\\hline $2543$ & $\times$ & $\times$ & $\times$ & $\times$ & $\times$\\\hline $6735$ & $\times$ & \checkmark & \checkmark & $\times$ & $\times$\\\hline \end{tabular}\par}\par El $2543$ no és divisible per cap: és senar, les xifres sumen 14 i la diferència per a l'11 és $8-6=2$.

\ex{1} Per exemple, $\res{102,\ 111,\ 123}$: les xifres sumen 3, 3 i 6 (múltiples de 3, però no de 9).

\ex{2} $5+4=9$. \ Per 3: $\square=0,3,6,9$ \ ($\res{504,\ 534,\ 564,\ 594}$). \ Per 9: $\square=0$ o $9$ \ ($\res{504,\ 594}$).

\ex{2} Per 3: $11+\square$ múltiple de 3, \ $\square=1, 4, 7$. Per 2 ha de ser parella: $\square=4$, \ $\res{384}$.

\ex{2} \begin{tasks}(2)
  \task $(9+8)-(4+2)=11$: \res{\text{sí}}
  \task $(7+9)-(5+3)=8$: \res{\text{no}}
  \task $(2+1+5)-(8+0)=0$: \res{\text{sí}}
  \task $(5+5+1)-(7+3)=1$: \res{\text{no}}
\end{tasks}

\ex{3} Ha d'acabar en 0 (per 2 i per 5) i les xifres han de sumar un múltiple de 3: $\res{990}$ ($9+9+0=18$).

\ex{3} Per 5: $b=0$ o $5$. \ Si $b=0$: $9+a$ múltiple de 9, $a=0$ o $9$. \ Si $b=5$: $14+a$, $a=4$. \ \ $\res{7020,\ 7920,\ 7425}$.

\ex{1} $\res{61,\ 67,\ 71,\ 73,\ 79,\ 83,\ 89,\ 97}$.

\ex{2} \begin{tasks}(2)
  \task $87=3\cdot29$: \res{\text{compost}}
  \task $107:11\to q=9<11$: \res{\text{primer}}
  \task $133=7\cdot19$: \res{\text{compost}}
  \task $149:13\to q=11<13$: \res{\text{primer}}

\ex{2} \begin{tasks}(2)
  \task $3+13=5+11$
  \task $5+19=7+17=11+13$
  \task $5+31=7+29=13+23=17+19$

\end{tasks}\par (N'hi ha prou amb una manera.)

\ex{2} Si acaba en 5, és divisible per 5: té almenys els divisors 1, 5 i ell mateix. Si és parell, és divisible per 2: té almenys els divisors 1, 2 i ell mateix.

\ex{3} $2027$ no és divisible per cap primer fins al $43$, i $47\cdot47=2209>2027$: és \res{\text{primer}}. \ $2028$ és parell: \res{\text{compost}} ($2028=2^{2}\cdot3\cdot13^{2}$).

\ex{3} Per exemple, $5+7+11=\res{23}$ \ o \ $7+11+13=\res{31}$.
