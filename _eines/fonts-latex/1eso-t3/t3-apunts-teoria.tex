\seccio{sec1}{Els nombres enters}{Nombres amb signe: per damunt o per davall del zero.}

\vfill
\begin{idea}[Per què els necessitem?]
Amb els naturals no podem expressar una temperatura \textbf{baix zero}, un \textbf{soterrani}, un \textbf{deute}, una \textbf{profunditat} o un any \textbf{abans de Crist}, ni fer restes com $3-5$. \ El signe $+$ diu que estem per damunt d'una referència (el zero) i el signe $-$, per davall.
\end{idea}

\vfill
\begin{minipage}[c]{0.55\linewidth}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=2mm,bottom=2mm]
{\Large$\mathbb Z=\{\ldots,\ -3,\ -2,\ -1,\ 0,\ +1,\ +2,\ +3,\ \ldots\}$}
\end{tcolorbox}
\chip{fmK}{positius} $+1,\ +2,\ +3\ldots$ \ el $+$ es pot ometre: $+7=7$\\[4pt]
\chip{fmR}{negatius} $-1,\ -2,\ -3\ldots$ \ el $-$ sempre s'escriu\\[4pt]
\chip{fmmuted}{zero} \ el 0 no és ni positiu ni negatiu
\end{minipage}\hfill

\seccio{sec5}{Parèntesis i claudàtors}{Llevem els parèntesis de dins cap a fora.}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={Mateix signe: \textbf{sumem}}]
Sumem els valors absoluts i posem el \textbf{signe comú}.\par\smallskip
{\centering$(+4)+(+3)=+7\qquad (-4)+(-3)=-7$\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmB!8,colframe=fmB,coltitle=white,title={Signes diferents: \textbf{restem}}]
El major menys el menor, amb el signe del que té \textbf{major} valor absolut.\par\smallskip
{\centering$(+4)+(-9)=-5\qquad (-4)+(+9)=+5$\par}
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{minipage}[c]{0.46\linewidth}
\begin{idea}[Llevar parèntesis]
{\centering\renewcommand{\arraystretch}{1.4}\Large
\begin{tabular}{cccc}
$+(+a)$ & $+(-a)$ & $-(+a)$ & $-(-a)$\\
$\cK{+a}$ & $\cR{-a}$ & $\cR{-a}$ & $\cK{+a}$
\end{tabular}\par}\smallskip
Un $-$ davant del parèntesi canvia \textbf{tots} els signes: \ $-(3-5+2)=-3+5-2$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.51\linewidth}
\begin{exemple}[Exemple: \ $7-(-3)+(-5)-(+2)$]
$=7+3-5-2$ \ \nota{llevem parèntesis}\\[3pt]
$=10-7=\boxed{3}$ \ \nota{positius i negatius}
\end{exemple}
\begin{truc}[Agrupa]
Suma a part els positius i els negatius:\\[2pt]
$-4+9-12+3-7=(9+3)-(4+12+7)=12-23=-11$
\end{truc}
\end{minipage}

\vfill
\begin{minipage}[t]{0.48\linewidth}
\begin{exemple}[Exemple: \ $-8+(4-9)-(-6+2)$]
$=-8+(-5)-(-4)$ \ \nota{dins dels parèntesis}\\[3pt]
$=-8-5+4$ \ \nota{llevem parèntesis}\\[3pt]
$=-13+4=\boxed{-9}$
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
\begin{exemple}[{Exemple: \ $15-\left[6-(2-9)\right]$}]
$=15-\left[6-(-7)\right]$ \ \nota{el de dins primer}\\[3pt]
$=15-\left[6+7\right]$\\[3pt]
$=15-13=\boxed{2}$
\end{exemple}
\end{minipage}

\vfill
\begin{compte}[El signe menys davant del parèntesi]
\be{5-(-3)=5+3=8}\qquad\quad \be{-2-6=-2+(-6)=-8}\qquad\quad \mal{-(4-9)=-4-9}
\end{compte}

\seccio{sec6}{Multiplicació i divisió d'enters}{Signes iguals, positiu. Signes diferents, negatiu.}

\vfill
\begin{minipage}[c]{0.36\linewidth}\centering
\renewcommand{\arraystretch}{1.6}\taulaclip{fmblue,line width=1.2pt}{\fontsize{20}{24}\selectfont%
\begin{tabular}{>{\columncolor{fmblue}\color{white}}w{c}{1.6cm}w{c}{1.4cm}w{c}{1.4cm}}
\rowcolor{fmblue}{\large$\cdot$ \ $:$} & \color{white}$+$ & \color{white}$-$\\
$+$ & \cellcolor{fmK!18}$\cK{+}$ & \cellcolor{fmR!14}$\cR{-}$\\
$-$ & \cellcolor{fmR!14}$\cR{-}$ & \cellcolor{fmK!18}$\cK{+}$
\end{tabular}}
\end{minipage}\hfill
\begin{minipage}[c]{0.6\linewidth}
\begin{idea}[La regla dels signes]
{\large Multipliquem (o dividim) els valors absoluts i posem el signe de la taula.}\\[8pt]
$(-6)\cdot(+4)=\cR{-}24\qquad (-3)\cdot(-5)=\cK{+}15$\\[4pt]
$(-36):(-4)=\cK{+}9\qquad 42:(-7)=\cR{-}6$
\end{idea}
\end{minipage}

\vfill
\begin{minipage}[t]{0.5\linewidth}
\begin{idea}[Molts factors? Compta els negatius]
\textbf{Parell} de negatius $\to$ \cK{positiu}. \ \textbf{Senar} $\to$ \cR{negatiu}.\\[4pt]
$(-2)\cdot3\cdot(-1)\cdot(-5)=-30$ \ \nota{3 negatius}\\[3pt]
$(-1)^{15}=-1\qquad (-1)^{20}=+1$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{truc}[I si no és exacta?]
La divisió només dona un enter si és \textbf{exacta}. Si no, dona una \textbf{fracció}:\\[3pt]
$(-7):2=-\dfrac72$ \ \ \nota{per això calen les fraccions!}
\end{truc}
\end{minipage}

\vfill
\begin{exemple}[Exemples]
\begin{minipage}[t]{0.48\linewidth}
$(-1)\cdot(-2)\cdot(-3)=\cR{-6}$ \ \nota{3 negatius}\\[4pt]
$(-1)\cdot(-2)\cdot(-3)\cdot(-4)=\cK{+24}$ \ \nota{4 negatius}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
$(-60):(+5):(-3)=$ \ \nota{d'esquerra a dreta}\\[4pt]
$=(-12):(-3)=\boxed{4}$
\end{minipage}
\end{exemple}

\vfill
\begin{compte}[Divisions seguides: d'esquerra a dreta]
\be{(-100):(-4):(-5)=25:(-5)=-5}\qquad\quad \mal{(-100):\big((-4):(-5)\big)}
\end{compte}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faSuperscript\ \ Potències de base negativa}]
Exponent \textbf{parell} $\to$ \cK{positiu}. \ \textbf{Senar} $\to$ \cR{negatiu}.\par\smallskip
{\centering$(-2)^4=\cK{+16}\qquad (-2)^3=\cR{-8}$\par}
\end{tcolorbox}
\begin{compte}[Amb parèntesi o sense?]
\be{(-2)^4=(-2)\cdot(-2)\cdot(-2)\cdot(-2)=16}\\[3pt]
\be{-2^4=-(2\cdot2\cdot2\cdot2)=-16}\\[2pt]
{\small Sense parèntesi, el $-$ no s'eleva.}
\end{compte}
\end{tcbraster}

$=3-2\cdot(-3)-(-6):2$ \ \nota{\pas{1}}\\[3pt]
$=3-(-6)-(-3)$ \ \nota{\pas{3}}\\[3pt]
$=3+6+3=\boxed{12}$ \ \nota{\pas{4}}
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
\begin{exemple}[{Exemple: \ $2\cdot\left\{3-\left[4-(1-6)\right]\right\}$}]
$=2\cdot\left\{3-\left[4-(-5)\right]\right\}$ \ \nota{de dins cap a fora}\\[3pt]
$=2\cdot\left\{3-9\right\}$\\[3pt]
$=2\cdot(-6)=\boxed{-12}$
\end{exemple}
\end{minipage}

\clearpage
\seccio{sec8}{Problemes amb enters}{Primer, on és el zero.}

\passosproblema{\textbf{Busquem}\\el zero}{\textbf{Cada dada}\\amb el seu signe}

\vfill
\begin{minipage}[c]{0.52\linewidth}
\begin{idea}[Pujar i baixar]
puja, guanya, ingressa, sobre zero $\to$ \cK{$+$}\\[2pt]
baixa, perd, gasta, deu, baix zero $\to$ \cR{$-$}\\[4pt]
\textbf{Canvi}: valor final $-$ valor inicial. Positiu, ha pujat; negatiu, ha baixat.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.45\linewidth}
\begin{truc}[La distància entre dos valors]
El major menys el menor. De $-3$ °C a $5$ °C: \ $5-(-3)=\boxed{8}$ graus.
\end{truc}
\end{minipage}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt,raster row skip=8pt]
\begin{exemple}[Exemple: \ un ascensor]
Està a la 3a planta, baixa 5 plantes i després puja 2. On és?\\[3pt]
$3-5+2=\boxed{0}$: a la planta baixa.
\end{exemple}
\begin{exemple}[Exemple: \ un compte]
Tinc 45 €, gaste 60 € i ingresse 20 €.\\[3pt]
$45-60+20=\boxed{5}$ €: em queden 5 €.
\end{exemple}
\begin{exemple}[Exemple: \ temperatures]
A les 7 h feia $-4$ °C i a les 14 h, 9 °C. Quant ha pujat?\\[3pt]
$9-(-4)=9+4=\boxed{13}$ °C
\end{exemple}
\begin{exemple}[Exemple: \ història]
Arquimedes va nàixer el 287 a. C. i va morir el 212 a. C. Quants anys va viure?\\[3pt]
$-212-(-287)=\boxed{75}$ anys
\end{exemple}
\end{tcbraster}

\vfill
\begin{exemple}[Exemple: \ un submarí a 120 m de profunditat puja 45 m i després baixa 30 m]
$-120+45-30=\boxed{-105}$: \ està a 105 m de profunditat. \qquad Si ara puja fins a la superfície: \ $0-(-105)=105$ m.
\end{exemple}

\vfill
\begin{compte}[Interpreta el resultat]
Respon amb una frase: un $-5$ en un compte vol dir «dec 5 €»; un $-2$ en un ascensor, «el segon soterrani».
\end{compte}
