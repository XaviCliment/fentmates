\ex{1} Expresa en lenguaje algebraico (usa la letra $x$):\begin{tasks}(2)
  \task El doble de un número
  \task Un número más 7
  \task El triple de un número menos 2
  \task La mitad de un número
  \task El cuadrado de un número
  \task Un número menos su cuarta parte
  \task El consecutivo de un número
  \task El doble de un número más su triple
\end{tasks}\recorda{Entre un número y una letra no escribimos el punto de multiplicar: $3\cdot x=3x$. «El doble de un número más 3» es $2x+3$, pero «el doble de la suma de un número y 3» es $2(x+3)$.}

\seccio{2}{Expresiones algebraicas. Valor numérico}

\ex{1} Calcula el valor numérico de $4x-3$ para: \quad a) $x=2$ \quad b) $x=0$ \quad c) $x=-1$ \quad d) $x=5$\recorda{Para calcular el valor numérico, sustituimos la letra por el número y hacemos las operaciones respetando la jerarquía. Los números negativos, siempre entre paréntesis: si $x=-3$, \ $x^2=(-3)^2=9$.}

\ex{1} Calcula el valor numérico de cada expresión para los valores indicados:\begin{tasks}(2)

\end{tasks}

\ex{2} Copia y completa la tabla de valores:\par\smallskip{\centering\renewcommand{\arraystretch}{1.35}\begin{tabular}{|c|C{0.9cm}|C{0.9cm}|C{0.9cm}|C{0.9cm}|C{0.9cm}|C{0.9cm}||}\hline $x$ & $-2$ & $-1$ & $0$ & $1$ & $2$ & $3$\\\hline $2x+1$ &  &  &  &  &  & \\\hline $x^2-x$ &  &  &  &  &  & \\\hline\end{tabular}\par}

\seccio{3}{Monomios y polinomios}

\ex{1} Copia y completa la tabla:\par\smallskip{\centering\renewcommand{\arraystretch}{1.35}\begin{tabular}{|c|C{2.2cm}|C{2.2cm}|C{1.4cm}|}\hline Monomio & Coeficiente & Parte literal & Grado\\\hline $6x^3$ & & & \\\hline $-a^2b$ & & & \\\hline $xy$ & & & \\\hline $-7$ & & & \\\hline $\tfrac{3}{4}x^2y^2$ & & & \\\hline $-z$ & & & \\\hline\end{tabular}\par}\recorda{En el monomio $-5x^2y$: el coeficiente es $-5$, la parte literal es $x^2y$ y el grado es la suma de los exponentes, $2+1=3$. Si no hay número, el coeficiente es 1 (o $-1$).}

\ex{1} ¿Cuáles de estas expresiones son monomios?\begin{tasks}(4)
  \task $5x^2$
  \task $x+1$
  \task $-3ab$

  \task $8$
  \task $x^2y^3$
  \task $2^x$
  \task $\dfrac{x}{5}$
\end{tasks}

\ex{2} Agrupa los monomios semejantes: \quad $4x,\ \ -2x^2,\ \ 3xy,\ \ x,\ \ 5x^2,\ \ -xy,\ \ 7,\ \ -3x$

\ex{2} Escribe: \ a) dos monomios semejantes a $-2x^3$; \ b) un monomio de grado 5 con coeficiente 4; \ c) un monomio de grado 0; \ d) un monomio de grado 2 con dos letras diferentes.

\ex{2} Indica el número de términos, el grado y el término independiente de cada polinomio:\begin{tasks}(2)
  \task $3x^2-5x+1$
  \task $7-x^3$
  \task $x^4+2x^2-x$
  \task $5x-8$
\end{tasks}

\ex{3} Escribe un monomio semejante a $6x^2y$ que: \ a) sumado con él dé 0; \ b) sumado con él dé $10x^2y$; \ c) restado de él dé $x^2y$.

\ex{3} Calcula el valor numérico: \ a) del monomio $-3x^2y$ para $x=-2$, $y=5$; \ b) del polinomio $x^3-2x+1$ para $x=-1$; \ c) del polinomio $2x^2-3x$ para $x=\frac{1}{2}$.

\seccio{4}{Operaciones con monomios. Reducir expresiones}

\ex{1} Calcula:\begin{tasks}(4)
  \task $5x+3x$
  \task $9a-4a$
  \task $x+x+x$
  \task $2x-7x$
  \task $4x^{2}+x^{2}$
  \task $-3ab+8ab$
  \task $6y-y$
  \task $-x-x$
\end{tasks}\recorda{Solo podemos sumar o restar monomios semejantes: sumamos los coeficientes y dejamos la parte literal. $3x+5x=8x$, pero $3x+2$ no se puede reducir.}

\ex{1} Reduce:\begin{tasks}(2)
  \task $3x+2+5x-1$
  \task $7-2x+4-x$
  \task $x^{2}+3x-2x^{2}+x$
  \task $5a+2b-3a+b$
\end{tasks}

\ex{1} Multiplica:\begin{tasks}(3)
  \task $3\cdot 5x$
  \task $2x\cdot 4x$
  \task $(-3a)\cdot 2a$
  \task $x\cdot x^{2}$
  \task $5x\cdot (-2y)$
  \task $(-4x^{2})\cdot (-x)$
\end{tasks}

\ex{2} Aplica la propiedad distributiva:\begin{tasks}(3)
  \task $4(x+3)$
  \task $2(5x-1)$
  \task $-3(x+2)$
  \task $-(4x-7)$
  \task $x(x+5)$
  \task $2x(3x-1)$
\end{tasks}\recorda{$a(b+c)=ab+ac$: multiplicamos por \emph{cada} término del paréntesis. Un signo menos delante cambia todos los signos: $-(x-5)=-x+5$.}

\ex{2} Quita paréntesis y reduce:\begin{tasks}(2)
  \task $2(x+4)+3x$
  \task $5x-(2x+3)$
  \task $3(x-2)+2(x+5)$
  \task $4(2x+1)-(x-6)$
\end{tasks}

\ex{2} Halla el error en cada igualdad y corrígela:\begin{tasks}(2)
  \task $3x+4=7x$
  \task $2(x+5)=2x+5$
  \task $-(x-3)=-x-3$
  \task $x\cdot x=2x$
\end{tasks}

\ex{3} Reduce:\begin{tasks}(1)
  \task $3(2x-1)-2(x-4)+x$
  \task $5-2(3x+1)+4(x-2)$
  \task $x(x+2)-3(x-1)$
\end{tasks}

\ex{3} Escribe de forma reducida el perímetro de: \ a) un rectángulo de base $2x+1$ y altura $x-1$; \ b) un triángulo de lados $x$, $x+3$ y $2x-1$. Calcúlalos para $x=4$.

\ex{3} Adivinanza: piensa un número, súmale 5, multiplica el resultado por 2, réstale 10 y divide entre 2. ¿Qué obtienes? Explica por qué pasa siempre usando el álgebra.

\seccio{6}{Ecuaciones equivalentes. Resolución}

\ex{1} Resuelve:\begin{tasks}(4)
  \task $x+8=15$
  \task $x-6=-2$

\end{tasks}\recorda{Transposición: lo que \emph{suma} pasa \emph{restando} y lo que \emph{resta} pasa \emph{sumando}; lo que \emph{multiplica} pasa \emph{dividiendo} (con su signo) y lo que \emph{divide} pasa \emph{multiplicando}.}

\seccio{7}{Ecuaciones con paréntesis}

\ex{1} Resuelve:\begin{tasks}(2)
  \task $3(x+2)=18$
  \task $5(x-1)=20$
  \task $2(x+7)=x+12$
  \task $-2(x-3)=10$
\end{tasks}\recorda{Primero quitamos paréntesis (propiedad distributiva) y después resolvemos como siempre. Cuidado con el signo menos: $-2(x-3)=-2x+6$.}

\ex{2} Resuelve:\begin{tasks}(2)
  \task $4(x-1)=2(x+3)$
  \task $3(2x+1)=5x+7$
  \task $6-(x+2)=x$
  \task $2x-(5-x)=10$
\end{tasks}

\ex{2} Resuelve:\begin{tasks}(2)
  \task $5(x+2)-3x=16$
  \task $7x-2(x+4)=12$
  \task $3(x-4)+2(x+1)=5$
  \task $4(2x-3)=5(x+3)$
\end{tasks}

\ex{3} Resuelve. ¿Qué pasa en la última ecuación?\begin{tasks}(1)
  \task $2(3x-1)-3(x+2)=x+4$
  \task $5-2(x-3)=3(x-8)+5$
  \task $4(x+1)-3(x-2)=2(x+5)$
  \task $3(x-1)-(2x+5)=x-8$

\seccio{9}{Resolución de problemas con ecuaciones}

\ex{1} Si a un número le sumamos 12, obtenemos 30. ¿Cuál es el número?\recorda{Pasos: 1) lee y entiende el enunciado; 2) elige la incógnita ($x=\ldots$); 3) plantea la ecuación; 4) resuélvela; 5) comprueba la solución y responde a la pregunta.}

\ex{1} El triple de un número menos 8 es 25. ¿Cuál es el número?

\ex{1} La suma de dos números consecutivos es 73. ¿Cuáles son?

\ex{2} La suma de tres números consecutivos es 159. ¿Cuáles son?

\ex{2} Laura tiene 6 años más que su hermano. Entre los dos suman 30 años. ¿Cuántos años tiene cada uno?

\ex{2} Un bolígrafo y una libreta cuestan juntos $4,50$\,€. La libreta cuesta el doble que el bolígrafo. ¿Cuánto cuesta cada cosa?

\ex{2} El perímetro de un rectángulo es 46\,cm y la base mide 5\,cm más que la altura. ¿Cuánto miden los lados?

\ex{2} Repartimos 90 cromos entre tres amigos. El segundo recibe el doble que el primero y el tercero, el triple que el primero. ¿Cuántos recibe cada uno?

\ex{3} Una madre tiene 42 años y su hijo, 10. ¿Dentro de cuántos años la madre tendrá el triple de edad que el hijo?

\ex{3} La mitad de un número más su cuarta parte es 21. ¿Cuál es el número?

\ex{3} En una clase hay 28 alumnos. Si hubiera 4 chicas más, habría tantas chicas como chicos. ¿Cuántas chicas y cuántos chicos hay?

\ex{3} En un aparcamiento hay coches y motos: en total, 30 vehículos y 96 ruedas. ¿Cuántos coches y cuántas motos hay?

\ex{3} Los ángulos de un triángulo miden $x$, $2x$ y $3x$. ¿Cuánto mide cada ángulo? (Recuerda que los tres ángulos de un triángulo suman $180^\circ$.)
