\seccio{2}{Expressions algebraiques. Valor numèric}

\seccio{4}{Operacions amb monomis. Reduir expressions}

\ex{1} Calcula:\begin{tasks}(4)
  \task $5x+3x$
  \task $9a-4a$
  \task $x+x+x$
  \task $2x-7x$
  \task $4x^{2}+x^{2}$
  \task $-3ab+8ab$
  \task $6y-y$
  \task $-x-x$
\end{tasks}\recorda{Només podem sumar o restar monomis semblants: sumem els coeficients i deixem la part literal. $3x+5x=8x$, però $3x+2$ no es pot reduir.}

\ex{1} Redueix:\begin{tasks}(2)
  \task $3x+2+5x-1$
  \task $7-2x+4-x$
  \task $x^{2}+3x-2x^{2}+x$
  \task $5a+2b-3a+b$
\end{tasks}

\ex{1} Multiplica:\begin{tasks}(3)
  \task $3\cdot 5x$
  \task $2x\cdot 4x$
  \task $(-3a)\cdot 2a$
  \task $x\cdot x^{2}$
  \task $5x\cdot (-2y)$
  \task $(-4x^{2})\cdot (-x)$
\end{tasks}

\ex{2} Aplica la propietat distributiva:\begin{tasks}(3)
  \task $4(x+3)$
  \task $2(5x-1)$
  \task $-3(x+2)$
  \task $-(4x-7)$
  \task $x(x+5)$
  \task $2x(3x-1)$
\end{tasks}\recorda{$a(b+c)=ab+ac$: multipliquem per \emph{cada} terme del parèntesi. Un signe menys davant canvia tots els signes: $-(x-5)=-x+5$.}

\ex{2} Trau parèntesis i redueix:\begin{tasks}(2)
  \task $2(x+4)+3x$
  \task $5x-(2x+3)$
  \task $3(x-2)+2(x+5)$
  \task $4(2x+1)-(x-6)$
\end{tasks}

\ex{2} Troba l'error en cada igualtat i corregeix-la:\begin{tasks}(2)
  \task $3x+4=7x$
  \task $2(x+5)=2x+5$
  \task $-(x-3)=-x-3$
  \task $x\cdot x=2x$
\end{tasks}

\ex{3} Redueix:\begin{tasks}(1)
  \task $3(2x-1)-2(x-4)+x$
  \task $5-2(3x+1)+4(x-2)$
  \task $x(x+2)-3(x-1)$
\end{tasks}

\ex{3} Escriu de manera reduïda el perímetre de: \ a) un rectangle de base $2x+1$ i altura $x-1$; \ b) un triangle de costats $x$, $x+3$ i $2x-1$. Calcula'ls per a $x=4$.

\seccio{6}{Equacions equivalents. Resolució}

\ex{1} Resol:\begin{tasks}(4)
  \task $x+8=15$
  \task $x-6=-2$
  \task $x+9=4$
  \task $-5+x=-11$
\end{tasks}\recorda{Transposició: el que \emph{suma} passa \emph{restant} i el que \emph{resta} passa \emph{sumant}; el que \emph{multiplica} passa \emph{dividint} (amb el seu signe) i el que \emph{divideix} passa \emph{multiplicant}.}

\ex{1} Resol:\begin{tasks}(3)
  \task $6x=42$
  \task $-4x=20$
  \task $9x=-27$
  \task $\dfrac{x}{5}=3$
  \task $\dfrac{x}{4}=-3$
  \task $-x=8$
\end{tasks}

\ex{1} Resol:\begin{tasks}(4)
  \task $2x+3=11$
  \task $5x-4=21$
  \task $3x+10=1$
  \task $7-2x=13$
\end{tasks}

\ex{2} Resol i comprova la solució:\begin{tasks}(2)
  \task $5x+2=3x+10$
  \task $8x-5=6x+1$
  \task $4x+7=x-8$
  \task $2x-9=5x+3$
\end{tasks}

\ex{2} Resol:\begin{tasks}(2)
  \task $3x+4-x=x+9$
  \task $6-2x+5=3x+1$
  \task $x+2x+3x=4x+10$
  \task $9x-3-4x=2x+12$
\end{tasks}

\ex{2} Resol (la solució pot ser una fracció: dona-la irreductible):\begin{tasks}(4)
  \task $4x=10$
  \task $6x+1=2x+3$
  \task $3x-2=5$
  \task $10x=-4$
\end{tasks}

\ex{3} Resol. Què passa en les dues últimes?\begin{tasks}(2)
  \task $2x+5-7x=3-x-6$

  \task $x+1=x+5$
  \task $3x+6-x=2x+6$
\end{tasks}

  \task $5(x-1)=20$
  \task $2(x+7)=x+12$
  \task $-2(x-3)=10$
\end{tasks}\recorda{Primer traiem parèntesis (propietat distributiva) i després resolem com sempre. Compte amb el signe menys: $-2(x-3)=-2x+6$.}

\ex{2} Resol:\begin{tasks}(2)
  \task $4(x-1)=2(x+3)$
  \task $3(2x+1)=5x+7$
  \task $6-(x+2)=x$
  \task $2x-(5-x)=10$
\end{tasks}

\ex{2} Resol:\begin{tasks}(2)

\end{tasks}

\ex{3} Resol. Què passa en l'última equació?\begin{tasks}(1)

\seccio{8}{Equacions amb denominadors}

\ex{1} Resol:\begin{tasks}(4)
  \task $\dfrac{x}{3}=5$
  \task $\dfrac{x}{4}+2=5$
  \task $\dfrac{2x}{3}=8$
  \task $\dfrac{x}{2}-1=4$
\end{tasks}\recorda{Multipliquem \emph{tots} els termes dels dos membres pel m.c.m. dels denominadors (també els que no tenen denominador). Si un numerador té més d'un terme, el posem entre parèntesis.}

\ex{2} Resol:\begin{tasks}(4)
  \task $\dfrac{x}{2}+\dfrac{x}{4}=6$
  \task $\dfrac{x}{3}-\dfrac{x}{6}=2$
  \task $\dfrac{x}{2}+\dfrac{x}{3}=10$
  \task $x-\dfrac{x}{5}=8$
\end{tasks}

\ex{2} Resol:\begin{tasks}(4)
  \task $\dfrac{x+3}{2}=5$
  \task $\dfrac{x-1}{3}=x-5$

\end{tasks}

\ex{2} Resol:\begin{tasks}(2)
