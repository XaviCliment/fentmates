\draw[thick,latex-latex] (2.800,-0.900) -- (6.200,2.600);

  \task $r$ y $t$
  \task $s$ y $t$
  \task $r$ y $u$
\end{tasks}\recorda{Dos rectas son \textbf{secantes} si se cortan en un punto, \textbf{paralelas} si no se cortan nunca y \textbf{perpendiculares} si se cortan formando cuatro ángulos rectos ($r\perp s$).}

\ex{1} Di si son paralelas, secantes o perpendiculares:\begin{tasks}(2)

\ex{1} Dibuja un segmento $\overline{AB}$ de 6\,cm y traza su mediatriz con regla y compás. ¿A qué distancia de $A$ corta el segmento?

\ex{2} Di si es cierto o falso:\begin{tasks}(1)
  \task Dos rectas perpendiculares son siempre secantes.
  \task Dos rectas secantes son siempre perpendiculares.
  \task Si $r\parallel s$ y $s\parallel t$, entonces $r\parallel t$.
  \task Si $r\perp s$ y $s\perp t$, entonces $r\perp t$.
\end{tasks}

\ex{2} El punto $P$ está sobre la mediatriz del segmento $\overline{AB}$, que mide 8\,cm, y la distancia de $P$ a $A$ es 5\,cm. ¿Cuánto mide $\overline{PB}$? ¿A qué distancia de $A$ está el punto medio $M$?

\ex{3} Dos casas, $A$ y $B$, están al mismo lado de un camino recto. ¿Dónde se tiene que poner una fuente, junto al camino, para que esté a la misma distancia de las dos casas? Explica cómo la encontrarías.
