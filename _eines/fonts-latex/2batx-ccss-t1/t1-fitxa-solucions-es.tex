  \task Todas las filas son proporcionales a $(1\ 2\ 3)$: \ $\res{\operatorname{rg}(A)=1}$
  \task \[A=\begin{pmatrix}1 & -1 & 2 & 0\\2 & 1 & 0 & 3\\3 & 0 & 2 & 3\end{pmatrix}\xrightarrow[F_3'=F_3-3F_1]{F_2'=F_2-2F_1}\begin{pmatrix}1 & -1 & 2 & 0\\0 & 3 & -4 & 3\\0 & 3 & -4 & 3\end{pmatrix}\]\[\xrightarrow{F_3''=F_3'-F_2'}\begin{pmatrix}1 & -1 & 2 & 0\\0 & 3 & -4 & 3\\0 & 0 & 0 & 0\end{pmatrix}\]Quedan dos filas no nulas: \ $\res{\operatorname{rg}(B)=2}$ \ (de hecho, $F_3=F_1+F_2$).
  \task \[C=\begin{pmatrix}1 & 2 & 1\\0 & 1 & 3\\2 & 1 & 0\\1 & 0 & 2\end{pmatrix}\xrightarrow[F_4'=F_4-F_1]{F_3'=F_3-2F_1}\begin{pmatrix}1 & 2 & 1\\0 & 1 & 3\\0 & -3 & -2\\0 & -2 & 1\end{pmatrix}\]\[\xrightarrow[F_4''=F_4'+2F_2]{F_3''=F_3'+3F_2}\begin{pmatrix}1 & 2 & 1\\0 & 1 & 3\\0 & 0 & 7\\0 & 0 & 7\end{pmatrix}\]\[\xrightarrow{F_4'''=F_4''-F_3''}\begin{pmatrix}1 & 2 & 1\\0 & 1 & 3\\0 & 0 & 7\\0 & 0 & 0\end{pmatrix}\]Quedan tres filas no nulas: \ $\res{\operatorname{rg}(C)=3}$

\ex{3} $|A|=k+k+k-k^3-1-1=-k^3+3k-2=-(k-1)^2(k+2)$, que se anula si $k=1$ o $k=-2$.\begin{tasks}(1)

\end{tasks}

\ex{3} \[B\xrightarrow[F_3'=F_3-F_1]{F_2'=F_2-2F_1}\begin{pmatrix}1 & 2 & -1 & 3\\ 0 & -3 & a+2 & -5\\ 0 & -3 & 4 & -5\end{pmatrix}\xrightarrow{F_3''=F_3'-F_2'}\begin{pmatrix}1 & 2 & -1 & 3\\ 0 & -3 & a+2 & -5\\ 0 & 0 & 2-a & 0\end{pmatrix}\]Si $\res{a=2}$, la 3.\textsuperscript{a} fila es nula: \ $\operatorname{rg}(B)=2$. \ Si $\res{a\neq2}$: \ $\operatorname{rg}(B)=3$.

\ex{2} \label{ex:inv3}\begin{tasks}(1)
  \task $|A|=1\neq0$, por tanto $A$ tiene inversa:\[\mathrm{Adj}(A)=\begin{pmatrix}3 & 0 & -2\\-6 & 1 & 4\\-1 & 0 & 1\end{pmatrix},\qquad A^{-1}=\frac{1}{1}\big(\mathrm{Adj}(A)\big)^t=\begin{pmatrix}3 & -6 & -1\\0 & 1 & 0\\-2 & 4 & 1\end{pmatrix}\]
  \task $|B|=5\neq0$, por tanto $B$ tiene inversa:\[\mathrm{Adj}(B)=\begin{pmatrix}1 & 3 & -1\\-1 & 2 & 1\\3 & -6 & 2\end{pmatrix},\qquad B^{-1}=\frac{1}{5}\big(\mathrm{Adj}(B)\big)^t=\frac{1}{5}\begin{pmatrix}1 & -1 & 3\\3 & 2 & -6\\-1 & 1 & 2\end{pmatrix}\]
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $|M|=-k^2-0+0-(-4k)-3-0=-k^2+4k-3=-(k-1)(k-3)$: \ tiene inversa si \ $\res{k\neq1\ \text{y}\ k\neq3}$
  \task $|M|=1\neq0$, por tanto $M$ tiene inversa:\[\mathrm{Adj}(M)=\begin{pmatrix}-7 & 12 & -8\\-1 & 2 & -1\\2 & -3 & 2\end{pmatrix},\qquad M^{-1}=\frac{1}{1}\big(\mathrm{Adj}(M)\big)^t=\begin{pmatrix}-7 & -1 & 2\\12 & 2 & -3\\-8 & -1 & 2\end{pmatrix}\]
\end{tasks}

\ex{3} \begin{tasks}(1)

  \task $A^2-3A=-2I\Rightarrow A(A-3I)=-2I\Rightarrow A\cdot\dfrac{3I-A}{2}=I$: \ $\res{A^{-1}=\dfrac12(3I-A)=\begin{pmatrix}\tfrac{1}{2} & -\tfrac{1}{2}\\0 & 1\end{pmatrix}}$
\end{tasks}

\seccio{7}{Ecuaciones y sistemas matriciales}

\ex{2} $|A|=1$, \ $A^{-1}=\begin{pmatrix}3 & -1\\-5 & 2\end{pmatrix}$.\begin{tasks}(1)
  \task $X=A^{-1}\cdot B=\begin{pmatrix}3 & -1\\-5 & 2\end{pmatrix}\begin{pmatrix}1 & 0\\2 & -1\end{pmatrix}=\res{\begin{pmatrix}1 & 1\\-1 & -2\end{pmatrix}}$
  \task $X=B\cdot A^{-1}=\begin{pmatrix}1 & 0\\2 & -1\end{pmatrix}\begin{pmatrix}3 & -1\\-5 & 2\end{pmatrix}=\res{\begin{pmatrix}3 & -1\\11 & -4\end{pmatrix}}$
