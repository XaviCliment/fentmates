\exf{1} Di si las variables siguientes son discretas o continuas:\begin{tasks}(2)
  \task El número de hermanos de un alumno.
  \task El tiempo que tarda un autobús en llegar.
  \task El peso de un bebé.
  \task El número de clientes que entran en una tienda en una hora.
  \task La nota de un examen con decimales.
  \task El número de caras en 10 lanzamientos.
\end{tasks}

\ex{2} Una variable aleatoria $X$ toma los valores 0, 1, 2 y 3 con probabilidades $0{,}1$, $k$, $0{,}3$ y $2k$, respectivamente.\begin{tasks}(1)
  \task Calcula $k$.
  \task Calcula la media y la desviación típica de $X$.
  \task Calcula $P(X\geq2)$.
\end{tasks}\recorda{$\sum p_i=1$; \ $\mu=\sum x_ip_i$; \ $\sigma^2=\sum x_i^2p_i-\mu^2$.}

\ex{2} En un juego se lanzan dos dados: si sale un doble, ganas 5 euros; si no, pierdes 1 euro. Sea $X$ la ganancia. Calcula la ganancia media y la desviación típica. ¿Es un juego justo?

\seccio{4}{La normal tipificada $N(0,1)$ y su tabla}

\exf{1} Si $Z\sim N(0,1)$, calcula con la tabla de la PAU:\begin{tasks}(4)
  \task $P(Z\leq1{,}53)$
  \task $P(Z>2{,}05)$
  \task $P(Z<-0{,}82)$
  \task $P(Z\geq-1{,}2)$
  \task $P(0{,}75\leq Z\leq1{,}5)$
  \task $P(-2< Z<1)$
  \task $P(|Z|\leq2{,}58)$
\end{tasks}\recorda{la tabla da $P(Z\leq z)$ para $z\geq0$. \ $P(Z\geq z)=1-P(Z\leq z)$, \ $P(Z\leq-z)=1-P(Z\leq z)$.}

\ex{2} Si $Z\sim N(0,1)$, halla el valor de $k$ en cada caso:\begin{tasks}(2)
  \task $P(Z\leq k)=0{,}8907$
  \task $P(Z\geq k)=0{,}0505$
  \task $P(Z\leq k)=0{,}25$
  \task $P(-k\leq Z\leq k)=0{,}9$
\end{tasks}\recorda{busca la probabilidad \emph{dentro} de la tabla; si es menor que 0,5, $k$ es negativo.}

\seccio{7}{Aproximación de la binomial por la normal}

\exf{1} Sea $X\sim B(100;\ 0{,}2)$.\begin{tasks}(1)
  \task Comprueba que se puede aproximar por una normal e indica cuál.
  \task Calcula aproximadamente $P(X\leq15)$, $P(X=20)$ y $P(X\geq25)$.
\end{tasks}\recorda{si $np\geq5$ y $nq\geq5$, $B(n,p)\approx N\left(np,\sqrt{npq}\right)$. No olvides la corrección de continuidad ($\pm0{,}5$).}

\ex{2} \pau{Canarias, EBAU ordinaria 2022} Una compañía de seguros tiene asegurados 2500 coches, 560 autobuses y 220 motos. Las probabilidades de tener un accidente a lo largo de un año son 0,1 para los coches, 0,08 para los autobuses y 0,16 para las motos.\begin{tasks}(1)
  \task ¿Cuántos vehículos en total se puede esperar que tengan un accidente a lo largo del próximo año?
  \task ¿Cuál es la probabilidad de que el año que viene tengan un accidente al menos 270 de los coches asegurados?
\end{tasks}

\ex{2} \pau{País Vasco, EAU extraordinaria 2023} El jugador $A$ lanza una moneda equilibrada y el jugador $B$, una moneda trucada en la que la probabilidad de cara es $0{,}4$. ¿Qué es más probable, que el jugador $A$ obtenga menos de 190 caras en 400 lanzamientos, o que el jugador $B$ obtenga menos de 110 cruces en 200 lanzamientos? Justifica la respuesta.

\ex{2} \pau{Andalucía, PAU ordinaria 2026 (adaptado)} Un suceso tiene probabilidad $0{,}4$ y el experimento se repite 1350 veces de manera independiente. Sea $X$ el número de veces que ocurre.\begin{tasks}(1)
  \task ¿Qué distribución sigue $X$?
  \task Calcula la probabilidad de que el suceso ocurra como mucho 580 veces, pero más de 499 veces.
\end{tasks}

\ex{2} Sea $X\sim B(50;\ 0{,}3)$. Calcula $P(10\leq X\leq20)$ con la aproximación normal y compárala con el valor exacto, $0{,}9120$.
