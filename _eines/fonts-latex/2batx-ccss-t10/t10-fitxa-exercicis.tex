\seccio{1}{Variables aleatòries discretes}

\ex{1} Digues si les variables següents són discretes o contínues:\begin{tasks}(2)
  \task El nombre de germans d'un alumne.
  \task El temps que tarda un autobús a arribar.
  \task El pes d'un nadó.
  \task El nombre de clients que entren en una botiga en una hora.
  \task La nota d'un examen amb decimals.
  \task El nombre de cares en 10 llançaments.
\end{tasks}

\ex{2} Una variable aleatòria $X$ pren els valors 0, 1, 2 i 3 amb probabilitats $0{,}1$, $k$, $0{,}3$ i $2k$, respectivament.\begin{tasks}(1)
  \task Calcula $k$.
  \task Calcula la mitjana i la desviació típica de $X$.
  \task Calcula $P(X\geq2)$.
\end{tasks}\recorda{$\sum p_i=1$; \ $\mu=\sum x_ip_i$; \ $\sigma^2=\sum x_i^2p_i-\mu^2$.}

\ex{2} En un joc es llancen dos daus: si ix un doble, guanyes 5 euros; si no, perds 1 euro. Siga $X$ el guany. Calcula el guany mitjà i la desviació típica. És un joc just?

  \task $P(X\geq4)$
  \task $P(X>0)$
\end{tasks}\recorda{la taula dona $P(X=x)$ per a $n\leq9$. Per a les acumulades, suma caselles de la mateixa columna o fes el contrari.}

\seccio{4}{La normal típica $N(0,1)$ i la seua taula}

\ex{1} Si $Z\sim N(0,1)$, calcula amb la taula de la PAU:\begin{tasks}(4)
  \task $P(Z\leq1{,}53)$
  \task $P(Z>2{,}05)$
  \task $P(Z<-0{,}82)$
  \task $P(Z\geq-1{,}2)$
  \task $P(0{,}75\leq Z\leq1{,}5)$
  \task $P(-2< Z<1)$
  \task $P(|Z|\leq2{,}58)$
\end{tasks}\recorda{la taula dona $P(Z\leq z)$ per a $z\geq0$. \ $P(Z\geq z)=1-P(Z\leq z)$, \ $P(Z\leq-z)=1-P(Z\leq z)$.}

\ex{2} Si $Z\sim N(0,1)$, troba el valor de $k$ en cada cas:\begin{tasks}(2)
  \task $P(Z\leq k)=0{,}8907$
  \task $P(Z\geq k)=0{,}0505$
  \task $P(Z\leq k)=0{,}25$
  \task $P(-k\leq Z\leq k)=0{,}9$
\end{tasks}\recorda{busca la probabilitat \emph{dins} de la taula; si és menor que 0,5, $k$ és negatiu.}

\seccio{5}{Tipificació: problemes amb la normal}

\ex{1} \pau{Illes Balears, PBAU extraordinària 2025} Quan un client entra en una botiga d'electrodomèstics, el temps (en minuts) que hi passa segueix una distribució $N(15,10)$.\begin{tasks}(1)

  \task Quina és la probabilitat que un client estiga dins de la botiga entre 5 i 25 minuts?
