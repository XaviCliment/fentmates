\ex{1} $\res{\left\{\begin{aligned}x+2y&=4\\-x+3z&=1\end{aligned}\right.}$ \ \ Tiene \rest{2 ecuaciones y 3 incógnitas} (el producto de una matriz $2\times3$ por una $3\times1$ es $2\times1$).\par No puede ser SCD: con 2 ecuaciones y 3 incógnitas, al escalonar siempre sobra al menos una incógnita. Las filas de $A$ no son proporcionales y, con $F_2'=F_2+F_1$, queda $2y+3z=5$: no aparece ninguna fila del tipo $0=d\neq0$. Por tanto, es \rest{SCI}, con un parámetro: \ $x=-1+3\lambda;\ y=\tfrac{5-3\lambda}{2};\ z=\lambda$.

\seccio{5}{Teorema de Rouché-Frobenius}

\ex{1} \label{ex:rf}\begin{tasks}(1)
  \task $\begin{vmatrix}1&2\\ 3&-1\end{vmatrix}=-7\neq0\Rightarrow\operatorname{rg}(A)=2$; \ $|A^*|=\begin{vmatrix}1 & 2 & 5\\3 & -1 & 1\\2 & 1 & 4\end{vmatrix}=0\Rightarrow\operatorname{rg}(A^*)=2=n$: \ \rest{SCD} (la solución es $x=1,\ y=2$).
  \task $\operatorname{rg}(A)=2$; \ $|A^*|=\begin{vmatrix}1 & 2 & 5\\3 & -1 & 1\\1 & 1 & 2\end{vmatrix}=7\neq0\Rightarrow\operatorname{rg}(A^*)=3$: \ \rest{SI}.
  \task $|A|=0$ y $\begin{vmatrix}1&1\\ 2&-1\end{vmatrix}=-3\neq0\Rightarrow\operatorname{rg}(A)=2$. Los menores de orden 3 de $A^*$ con la columna de términos independientes también valen 0 $\Rightarrow\operatorname{rg}(A^*)=2<3$: \ \rest{SCI} (la 3.\textsuperscript{a} ecuación es la 1.\textsuperscript{a} más el doble de la 2.\textsuperscript{a}).
\end{tasks}

\ex{2} \pau{Murcia, PAU ordinaria 2025}$|A|=\begin{vmatrix}3 & 2 & -1\\1 & -1 & 2\\2 & 3 & -1\end{vmatrix}=-10\neq0\Rightarrow\operatorname{rg}(A)=\operatorname{rg}(A^*)=3=n$: \ \rest{SCD}. Por Cramer o por Gauss: \ $\res{x=1;\ y=1;\ z=2}$

\ex{2} $\begin{vmatrix}1 & 1 & 1\\1 & -1 & 1\\2 & 1 & -1\end{vmatrix}=6\neq0\Rightarrow\operatorname{rg}(A)=3$; \ $|A^*|=0\Rightarrow\operatorname{rg}(A^*)=3=n$: \ \rest{SCD}. Suprimimos la 4.\textsuperscript{a} ecuación (no forma parte del menor) y resolvemos las tres primeras: \ $\res{x=1;\ y=1;\ z=1}$

\ex{2} \begin{tasks}(1)

  \task \rest{Verdadera}: $\operatorname{rg}(A)=3$ y $\operatorname{rg}(A^*)$ no puede pasar de 3 (tiene 3 filas): $\operatorname{rg}(A)=\operatorname{rg}(A^*)=3=n$.
  \task \rest{Falsa}: los rangos son distintos y el sistema es incompatible (ninguna solución).
  \task \rest{Falsa}: $\operatorname{rg}(A)\leq2<3$, así que, si es compatible, es indeterminado.
\end{tasks}

\seccio{6}{Discusión de sistemas con parámetros}

\ex{2} \pau{Asturias, PAU extraordinaria 2025}\begin{tasks}(1)
  \task $|A|=\begin{vmatrix}1&-1\\ 10&10+m\end{vmatrix}=20+m$. \ Tiene solución única (SCD) si y solo si \ $\res{m\neq-20}$. (Para $m=-20$: $x-y=20$ y $x-y=15$, SI.)
  \task Debe ser $|A|=0$: $\res{m=-20}$; entonces la 2.\textsuperscript{a} ecuación es $x-y=15$ y, para que coincida con la 1.\textsuperscript{a}, $\res{\lambda=-15}$. Una solución: $\res{x=15,\ y=0}$ (en general, $x=15+t,\ y=t$).
\end{tasks}

\ex{2} \pau{Asturias, EBAU extraordinaria 2024}\begin{tasks}(1)
  \task $A\cdot B=\begin{pmatrix}1&2m+1\\ -1-m&0\end{pmatrix}$, \ $A\cdot B-C=\begin{pmatrix}1&1\\ m-1&1\end{pmatrix}$: \ \ $\res{\left\{\begin{aligned}x+y&=1\\ (m-1)x+y&=1\end{aligned}\right.}$
