\seccio{5}{Teorema de Rouché-Frobenius}

\ex{1} \label{ex:rf}\begin{tasks}(1)
  \task $\begin{vmatrix}1&2\\ 3&-1\end{vmatrix}=-7\neq0\Rightarrow\operatorname{rg}(A)=2$; \ $|A^*|=\begin{vmatrix}1 & 2 & 5\\3 & -1 & 1\\2 & 1 & 4\end{vmatrix}=0\Rightarrow\operatorname{rg}(A^*)=2=n$: \ \rest{SCD} (la solució és $x=1,\ y=2$).
  \task $\operatorname{rg}(A)=2$; \ $|A^*|=\begin{vmatrix}1 & 2 & 5\\3 & -1 & 1\\1 & 1 & 2\end{vmatrix}=7\neq0\Rightarrow\operatorname{rg}(A^*)=3$: \ \rest{SI}.
  \task $|A|=0$ i $\begin{vmatrix}1&1\\ 2&-1\end{vmatrix}=-3\neq0\Rightarrow\operatorname{rg}(A)=2$. Els menors d'ordre 3 de $A^*$ amb la columna de termes independents també valen 0 $\Rightarrow\operatorname{rg}(A^*)=2<3$: \ \rest{SCI} (la 3a equació és la 1a més el doble de la 2a).
\end{tasks}

\ex{2} \pau{Múrcia, PAU ordinària 2025}$|A|=\begin{vmatrix}3 & 2 & -1\\1 & -1 & 2\\2 & 3 & -1\end{vmatrix}=-10\neq0\Rightarrow\operatorname{rg}(A)=\operatorname{rg}(A^*)=3=n$: \ \rest{SCD}. Per Cramer o per Gauss: \ $\res{x=1;\ y=1;\ z=2}$

\ex{2} $\begin{vmatrix}1 & 1 & 1\\1 & -1 & 1\\2 & 1 & -1\end{vmatrix}=6\neq0\Rightarrow\operatorname{rg}(A)=3$; \ $|A^*|=0\Rightarrow\operatorname{rg}(A^*)=3=n$: \ \rest{SCD}. Suprimim la 4a equació (no forma part del menor) i resolem les tres primeres: \ $\res{x=1;\ y=1;\ z=1}$

\ex{2} \begin{tasks}(1)
  \task \rest{Falsa}: si $\operatorname{rg}(A)=\operatorname{rg}(A^*)=2$ és SCD (com al primer apartat de l'exercici \ref{ex:rf}).
  \task \rest{Certa}: $\operatorname{rg}(A)=3$ i $\operatorname{rg}(A^*)$ no pot passar de 3 (té 3 files): $\operatorname{rg}(A)=\operatorname{rg}(A^*)=3=n$.
  \task \rest{Falsa}: els rangs són diferents i el sistema és incompatible (cap solució).
  \task \rest{Falsa}: $\operatorname{rg}(A)\leq2<3$, així que, si és compatible, és indeterminat.
\end{tasks}

  \task Cal $|A|=0$: $\res{m=-20}$; aleshores la 2a equació és $x-y=15$ i, perquè coincidisca amb la 1a, $\res{\lambda=-15}$. Una solució: $\res{x=15,\ y=0}$ (en general, $x=15+t,\ y=t$).
\end{tasks}

  \task $|A|=2-m$. \ Si $m\neq2$: \rest{SCD}, solució única. \ Si $m=2$: les dues equacions són $x+y=1$, \rest{SCI}. \ Té solució \rest{per a tot $m$}, però només és única si $m\neq2$. \ Per a $m=-1$: $x+y=1$, $-2x+y=1$ \ $\Rightarrow\res{x=0;\ y=1}$.
