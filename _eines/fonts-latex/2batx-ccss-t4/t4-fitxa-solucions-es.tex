\ex{1} $\lim_{x\to1^-}f=\lim_{x\to1^+}f=+\infty$: el límite es $+\infty$ (no es un número) y $\nexists f(1)$: \rest{discontinuidad asintótica}. \ En el ejercicio \ref{ex:g4}: $f(0)=2$, laterales $-2$ y $2$: \rest{salto finito} en $x=0$.

\ex{1} \begin{tasks}(2)

  \task $k(1)=2$, laterales $2$ y $4$: \ \rest{salto finito}

En $x=5$ ($\frac{1}{0}$) la discontinuidad es asintótica y no se puede evitar.

\seccio{6}{Continuidad de funciones a trozos}

\ex{1} \pau{Navarra, EvAU extraordinaria 2021}Trozos polinómicos: continuos. \ En $x=2$: $f(2)=0$, \ $\lim_{x\to2^-}(x^2-2x)=0$, \ $\lim_{x\to2^+}(-x^2+6x-8)=0$. \ \rest{Continua en todo $\mathbb R$}.

\ex{2} \pau{Castilla y León, EBAU septiembre 2017}El primer trozo no está definido en $x=\pm2$, que no son de $(0,2)$: continuo. \ En $x=2$: $f(2)=4$, \ $\lim_{x\to2^-}\frac{1}{4-x^2}=\frac{1}{0^+}=+\infty$, \ $\lim_{x\to2^+}2x=4$. \ Continua en $\res{(0,2)\cup(2,4)}$; en $x=2$, \rest{salto infinito}.

\ex{2} \pau{Madrid, EvAU junio 2018}$f(2)=\frac41=4$, \ $\lim_{x\to2^-}f=4$, \ $\lim_{x\to2^+}f=\frac{12-4}{4}=2$: \ \rest{no es continua en $x=2$} (salto finito). \ Además, el primer trozo no existe en $x=1$ ($\frac30$): \rest{asintótica en $x=1$}. ($x=-2$ no es de $x>2$.) \ Continua en $\mathbb R-\{1,2\}$.

\ex{2} \pau{Madrid, EvAU julio 2018}El denominador $3+x$ se anula en $-3$, que no es de $x\geq0$: dominio $\res{\mathbb R}$. \ En $x=0$: $f(0)=\frac23$, \ $\lim_{x\to0^-}(x^3+2e^x)=2$, \ $\lim_{x\to0^+}\frac{2}{3+x}=\frac23$: \ salto finito. \ Continua en $\res{\mathbb R-\{0\}}$.

\ex{2} \pau{Aragón, EvAU extraordinaria 2020}Trozos: el primero, continuo en $x<-1$; el segundo, polinómico; el tercero, $4x^2-7x=x(4x-7)>0$ si $x>4$: continuo. \ $x=-1$: $\lim_{x\to-1^-}\frac{3}{x+1}=-\infty$, $\lim_{x\to-1^+}=-1-4-2-10=-17=f(-1)$: \ salto infinito. \ $x=4$: $f(4)=64-64+8-10=-2$, $\lim_{x\to4^+}=\sqrt{36}-8=-2$: continua. \ \rest{Continua en $\mathbb R-\{-1\}$}.

\ex{3} \pau{Madrid, EvAU junio 2019}$x=0$: $f(0)=\lim_{x\to0^-}f=1+k$, \ $\lim_{x\to0^+}(1-x^2)=1$: continua solo si $k=0$. \ $x=3$: $f(3)=-8$, \ $\lim_{x\to3^+}\frac{1}{x-3}=+\infty$: siempre discontinua (salto infinito). \ \rest{Si $k=0$, continua en $\mathbb R-\{3\}$; si $k\neq0$, continua en $\mathbb R-\{0,3\}$}.

\ex{2} \begin{tasks}(1)
  \task $f$: \ $x=0$: $\nexists f(0)$ y los laterales valen $1$ y $1$: \rest{evitable}. \ $x=1$: $f(1)=-1$, $\lim_{x\to1^-}f=2$: \rest{salto finito}. Continua en $\mathbb R-\{0,1\}$.
  \task $g$: \ $x=3$: $g(3)=0$, $\lim_{x\to3^+}g=0$: continua. \ $x=6$: $\lim_{x\to6^-}g=3$, $g(6)=0$: \rest{salto finito}. Continua en $\mathbb R-\{6\}$.
\end{tasks}

\seccio{7}{Funciones a trozos con parámetros}

\ex{1} \pau{Madrid, EvAU septiembre 2017}$\lim_{x\to-1^-}(ax+1)=-a+1$, \ $\lim_{x\to-1^+}=f(-1)=1-1-2=-2$. \ $-a+1=-2\Rightarrow\res{a=3}$.

\ex{1} \pau{Andalucía, PEvAU junio 2019}El primer trozo solo falla en $x=1$, que no es de $x<0$. \ $\lim_{x\to0^-}\frac{1}{x-1}=-1$, \ $\lim_{x\to0^+}(x^2+a)=a=f(0)$ \ $\Rightarrow\res{a=-1}$.

\ex{2} \pau{Castilla-La Mancha, EvAU extraordinaria 2020}$f(0)=\lim_{x\to0^-}f=-t$, \ $\lim_{x\to0^+}f=t^2-5t+4$. \ $t^2-5t+4=-t\Rightarrow t^2-4t+4=0\Rightarrow(t-2)^2=0\Rightarrow\res{t=2}$.

\ex{2} \pau{Castilla-La Mancha, EvAU extraordinaria 2020}$f(c)=\lim_{x\to c^-}f=-4c+4$, \ $\lim_{x\to c^+}f=c^2-4c$. \ $c^2-4c=-4c+4\Rightarrow c^2=4\Rightarrow\res{c=-2\ \text{o}\ c=2}$.

\ex{2} \pau{Islas Baleares, PBAU extraordinaria 2021}$x=-2$: $8+a=4-5=-1\Rightarrow\res{a=-9}$. \ $x=1$: $\lim_{x\to1^-}(x^2-5)=-4$ y $f(1)=\lim_{x\to1^+}(-7x+3)=-4$: continua para cualquier $a$.
