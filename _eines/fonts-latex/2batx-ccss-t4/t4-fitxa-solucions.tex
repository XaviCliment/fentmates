\seccio{5}{Continuïtat en un punt. Funcions racionals}

\ex{1} $\lim_{x\to1^-}f=\lim_{x\to1^+}f=+\infty$: el límit és $+\infty$ (no és un nombre) i $\nexists f(1)$: \rest{discontinuïtat asimptòtica}. \ A l'exercici \ref{ex:g4}: $f(0)=2$, laterals $-2$ i $2$: \rest{salt finit} en $x=0$.

\ex{1} \begin{tasks}(2)
  \task $\nexists f(-2)$, $\lim(x-2)=-4$: \ \rest{evitable}
  \task Laterals $-\infty$ i $+\infty$: \ \rest{asimptòtica}
  \task $h(0)=1$, laterals $1$ i $1$: \ \rest{contínua}
  \task $k(1)=2$, laterals $2$ i $4$: \ \rest{salt finit}
\end{tasks}

\ex{2} $x^2-4x+3=0\Rightarrow x=1,\ x=3$. \ $f(x)=\frac{(x-1)(x+2)}{(x-1)(x-3)}$. \ $x=1$: $\lim\frac{x+2}{x-3}=-\frac32$ \ $\Rightarrow$ \rest{evitable}. \ $x=3$: $\frac50$ \ $\Rightarrow$ \rest{asimptòtica}. Contínua en $\mathbb R-\{1,3\}$.

\ex{2} \pau{Cantàbria, EBAU juliol 2019}$x^2-2x-15=(x-5)(x+3)$ i $x^2-x-12=(x-4)(x+3)$: discontínua en $\res{x=-3}$ i $\res{x=5}$. \ En $x=-3$: $\lim\frac{x-4}{x-5}=\frac{-7}{-8}=\frac78$ \ (evitable): definint $\res{f(-3)=\frac78}$ deixa de ser discontínua. En $x=5$ ($\frac{1}{0}$) la discontinuïtat és asimptòtica i no es pot evitar.

\ex{2} \pau{Cantàbria, EBAU ordinària 2020}\begin{tasks}(1)

\seccio{6}{Continuïtat de funcions a trossos}

\ex{1} \pau{Navarra, EvAU extraordinària 2021}Trossos polinòmics: continus. \ En $x=2$: $f(2)=0$, \ $\lim_{x\to2^-}(x^2-2x)=0$, \ $\lim_{x\to2^+}(-x^2+6x-8)=0$. \ \rest{Contínua en tot $\mathbb R$}.

\ex{2} \pau{Castella i Lleó, EBAU setembre 2017}El primer tros no està definit en $x=\pm2$, que no són de $(0,2)$: continu. \ En $x=2$: $f(2)=4$, \ $\lim_{x\to2^-}\frac{1}{4-x^2}=\frac{1}{0^+}=+\infty$, \ $\lim_{x\to2^+}2x=4$. \ Contínua en $\res{(0,2)\cup(2,4)}$; en $x=2$, \rest{salt infinit}.

\ex{2} \pau{Madrid, EvAU juny 2018}$f(2)=\frac41=4$, \ $\lim_{x\to2^-}f=4$, \ $\lim_{x\to2^+}f=\frac{12-4}{4}=2$: \ \rest{no és contínua en $x=2$} (salt finit). \ A més, el primer tros no existeix en $x=1$ ($\frac30$): \rest{asimptòtica en $x=1$}. ($x=-2$ no és de $x>2$.) \ Contínua en $\mathbb R-\{1,2\}$.

\ex{2} \pau{Madrid, EvAU juliol 2018}El denominador $3+x$ s'anul·la en $-3$, que no és de $x\geq0$: domini $\res{\mathbb R}$. \ En $x=0$: $f(0)=\frac23$, \ $\lim_{x\to0^-}(x^3+2e^x)=2$, \ $\lim_{x\to0^+}\frac{2}{3+x}=\frac23$: \ salt finit. \ Contínua en $\res{\mathbb R-\{0\}}$.

\ex{2} \pau{Aragó, EvAU extraordinària 2020}Trossos: el primer, continu en $x<-1$; el segon, polinòmic; el tercer, $4x^2-7x=x(4x-7)>0$ si $x>4$: continu. \ $x=-1$: $\lim_{x\to-1^-}\frac{3}{x+1}=-\infty$, $\lim_{x\to-1^+}=-1-4-2-10=-17=f(-1)$: \ salt infinit. \ $x=4$: $f(4)=64-64+8-10=-2$, $\lim_{x\to4^+}=\sqrt{36}-8=-2$: contínua. \ \rest{Contínua en $\mathbb R-\{-1\}$}.

\ex{3} \pau{Madrid, EvAU juny 2019}$x=0$: $f(0)=\lim_{x\to0^-}f=1+k$, \ $\lim_{x\to0^+}(1-x^2)=1$: contínua només si $k=0$. \ $x=3$: $f(3)=-8$, \ $\lim_{x\to3^+}\frac{1}{x-3}=+\infty$: sempre discontínua (salt infinit). \ \rest{Si $k=0$, contínua en $\mathbb R-\{3\}$; si $k\neq0$, contínua en $\mathbb R-\{0,3\}$}.

\ex{2} \begin{tasks}(1)
  \task $f$: \ $x=0$: $\nexists f(0)$ i els laterals valen $1$ i $1$: \rest{evitable}. \ $x=1$: $f(1)=-1$, $\lim_{x\to1^-}f=2$: \rest{salt finit}. Contínua en $\mathbb R-\{0,1\}$.
  \task $g$: \ $x=3$: $g(3)=0$, $\lim_{x\to3^+}g=0$: contínua. \ $x=6$: $\lim_{x\to6^-}g=3$, $g(6)=0$: \rest{salt finit}. Contínua en $\mathbb R-\{6\}$.
\end{tasks}

\seccio{7}{Funcions a trossos amb paràmetres}

\ex{1} \pau{Madrid, EvAU setembre 2017}$\lim_{x\to-1^-}(ax+1)=-a+1$, \ $\lim_{x\to-1^+}=f(-1)=1-1-2=-2$. \ $-a+1=-2\Rightarrow\res{a=3}$.

\ex{1} \pau{Andalusia, PEvAU juny 2019}El primer tros només falla en $x=1$, que no és de $x<0$. \ $\lim_{x\to0^-}\frac{1}{x-1}=-1$, \ $\lim_{x\to0^+}(x^2+a)=a=f(0)$ \ $\Rightarrow\res{a=-1}$.

\ex{2} \pau{Castella-la Manxa, EvAU extraordinària 2020}$f(0)=\lim_{x\to0^-}f=-t$, \ $\lim_{x\to0^+}f=t^2-5t+4$. \ $t^2-5t+4=-t\Rightarrow t^2-4t+4=0\Rightarrow(t-2)^2=0\Rightarrow\res{t=2}$.

\ex{2} \pau{Castella-la Manxa, EvAU extraordinària 2020}$f(c)=\lim_{x\to c^-}f=-4c+4$, \ $\lim_{x\to c^+}f=c^2-4c$. \ $c^2-4c=-4c+4\Rightarrow c^2=4\Rightarrow\res{c=-2\ \text{o}\ c=2}$.
