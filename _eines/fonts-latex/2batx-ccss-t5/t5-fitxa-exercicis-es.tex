\seccio{1}{Tasa de variación media e interpretación de la derivada}

\ex{1} Calcula la tasa de variación media de $f(x)=x^2-4x+1$ en los intervalos siguientes:\begin{tasks}(3)
  \task $[0,2]$
  \task $[1,4]$
  \task $[2,5]$
\end{tasks}\recorda{$\text{TVM}[a,b]=\dfrac{f(b)-f(a)}{b-a}$: es la pendiente de la recta secante que pasa por $(a,f(a))$ y $(b,f(b))$.}

\ex{1} Calcula la TVM de $g(x)=\sqrt x$ en $[1,4]$ y en $[4,9]$. ¿En cuál de los dos intervalos crece más rápidamente la función?

\ex{2} \label{ex:cost}El coste, en euros, de fabricar $x$ unidades de un producto es $C(x)=0{,}01x^2+5x+200$. Calcula la TVM del coste entre 100 y 200 unidades y entre 200 y 300 unidades. ¿Qué significan estos valores?

\ex{1} Responde a las preguntas siguientes:\begin{tasks}(1)
  \task Sabemos que $f(3)=2$ y $f'(3)=-4$. Escribe la ecuación de la recta tangente a la gráfica de $f$ en $x=3$.
  \task La recta tangente a $g$ en $x=1$ es $y=-2x+7$. ¿Cuánto valen $g(1)$ y $g'(1)$?
  \task Si $f'(a)=0$, ¿cómo es la recta tangente en $x=a$?
\end{tasks}\recorda{$f'(a)$ es la pendiente de la recta tangente en $x=a$. Recta tangente: $y-f(a)=f'(a)\,(x-a)$.}

\seccio{3}{Propiedades de la derivada: suma, producto y cociente}

\ex{1} Deriva estas funciones polinómicas:\begin{tasks}(2)
  \task $f(x)=5x^4-3x^3+2x-8$
  \task $f(x)=\dfrac{x^5}{5}-\dfrac{2x^3}{3}+x$
  \task $f(x)=\dfrac{3x^2-2x+1}{4}$
  \task $f(x)=-x^6+\dfrac x2-7$
\end{tasks}\recorda{$(k)'=0$, \ $(x^n)'=n\,x^{n-1}$, \ $(f\pm g)'=f'\pm g'$, \ $(k\cdot f)'=k\cdot f'$.}

\ex{1} Deriva aplicando la regla del producto y simplifica:\begin{tasks}(3)
  \task $f(x)=(2x-1)(x^2+3)$
  \task $f(x)=(x^3+x)(1-x^2)$
  \task $f(x)=x^2(x-1)(x+1)$
\end{tasks}\recorda{$(f\cdot g)'=f'\cdot g+f\cdot g'$. \ Cuidado: $(f\cdot g)'\neq f'\cdot g'$.}

\ex{2} Deriva aplicando la regla del cociente y simplifica:\begin{tasks}(2)
  \task $f(x)=\dfrac{x+3}{x-2}$
  \task $f(x)=\dfrac{2x^2}{x+1}$
  \task $f(x)=\dfrac{x^2-1}{x^2+1}$
  \task $f(x)=\dfrac{3}{x^2-4}$
\end{tasks}\recorda{$\left(\dfrac fg\right)'=\dfrac{f'\cdot g-f\cdot g'}{g^2}$. Simplifica el numerador, pero deja el denominador sin desarrollar.}

\ex{1} Dada $f(x)=x^4-2x^3+5x$, calcula $f'(x)$, $f''(x)$ y $f''(1)$.

\ex{2} El coste de fabricar $x$ unidades es $C(x)=0{,}01x^2+5x+200$ euros (ejercicio \ref{ex:cost}). Calcula el coste marginal $C'(x)$ y $C'(100)$. ¿Qué significa? Compáralo con la TVM del ejercicio \ref{ex:cost}.

\seccio{6}{Continuidad y derivabilidad. Derivadas laterales}

\ex{2} Calcula las derivadas laterales en el punto indicado y di si la función es derivable en él:\begin{tasks}(2)
  \task $f(x)=|x+1|$ en $x=-1$
  \task $f(x)=|x^2-9|$ en $x=3$
  \task $f(x)=x\,|x-1|$ en $x=1$
  \task $f(x)=\sqrt[3]{x^2}$ en $x=0$
\end{tasks}\recorda{$f'(a^-)=\displaystyle\lim_{h\to0^-}\frac{f(a+h)-f(a)}{h}$ y $f'(a^+)=\displaystyle\lim_{h\to0^+}\frac{f(a+h)-f(a)}{h}$. \ $f$ es derivable en $a$ si las dos existen, son finitas y coinciden. Un valor absoluto se escribe primero a trozos.}

\ex{1} Razona si son verdaderas o falsas las afirmaciones siguientes. Si son falsas, pon un contraejemplo:\begin{tasks}(1)
  \task Si $f$ es continua en $x=a$, entonces es derivable en $x=a$.
  \task Si $f$ no es continua en $x=a$, entonces no es derivable en $x=a$.
  \task Si $f$ no es derivable en $x=a$, entonces no es continua en $x=a$.
  \task Si en una función a trozos las derivadas de los dos trozos valen lo mismo en el punto de cambio $a$, la función es derivable en $a$.
\end{tasks}\recorda{Derivable $\Rightarrow$ continua. El recíproco no es cierto.}
