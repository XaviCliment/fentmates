\seccio{1}{Taxa de variació mitjana i interpretació de la derivada}

\ex{1} Calcula la taxa de variació mitjana de $f(x)=x^2-4x+1$ en els intervals següents:\begin{tasks}(3)
  \task $[0,2]$
  \task $[1,4]$
  \task $[2,5]$
\end{tasks}\recorda{$\text{TVM}[a,b]=\dfrac{f(b)-f(a)}{b-a}$: és el pendent de la recta secant que passa per $(a,f(a))$ i $(b,f(b))$.}

\ex{1} Calcula la TVM de $g(x)=\sqrt x$ en $[1,4]$ i en $[4,9]$. En quin dels dos intervals creix més ràpidament la funció?

\ex{2} \label{ex:cost}El cost, en euros, de fabricar $x$ unitats d'un producte és $C(x)=0{,}01x^2+5x+200$. Calcula la TVM del cost entre 100 i 200 unitats i entre 200 i 300 unitats. Què signifiquen aquests valors?

\ex{1} Respon les preguntes següents:\begin{tasks}(1)
  \task Sabem que $f(3)=2$ i $f'(3)=-4$. Escriu l'equació de la recta tangent a la gràfica de $f$ en $x=3$.
  \task La recta tangent a $g$ en $x=1$ és $y=-2x+7$. Quant valen $g(1)$ i $g'(1)$?
  \task Si $f'(a)=0$, com és la recta tangent en $x=a$?
\end{tasks}\recorda{$f'(a)$ és el pendent de la recta tangent en $x=a$. Recta tangent: $y-f(a)=f'(a)\,(x-a)$.}

\seccio{3}{Propietats de la derivada: suma, producte i quocient}

\ex{1} Deriva aquestes funcions polinòmiques:\begin{tasks}(2)
  \task $f(x)=5x^4-3x^3+2x-8$
  \task $f(x)=\dfrac{x^5}{5}-\dfrac{2x^3}{3}+x$
  \task $f(x)=\dfrac{3x^2-2x+1}{4}$
  \task $f(x)=-x^6+\dfrac x2-7$
\end{tasks}\recorda{$(k)'=0$, \ $(x^n)'=n\,x^{n-1}$, \ $(f\pm g)'=f'\pm g'$, \ $(k\cdot f)'=k\cdot f'$.}

\ex{1} Deriva aplicant la regla del producte i simplifica:\begin{tasks}(3)
  \task $f(x)=(2x-1)(x^2+3)$
  \task $f(x)=(x^3+x)(1-x^2)$
  \task $f(x)=x^2(x-1)(x+1)$
\end{tasks}\recorda{$(f\cdot g)'=f'\cdot g+f\cdot g'$. \ Compte: $(f\cdot g)'\neq f'\cdot g'$.}

\ex{2} Deriva aplicant la regla del quocient i simplifica:\begin{tasks}(2)
  \task $f(x)=\dfrac{x+3}{x-2}$
  \task $f(x)=\dfrac{2x^2}{x+1}$
  \task $f(x)=\dfrac{x^2-1}{x^2+1}$
  \task $f(x)=\dfrac{3}{x^2-4}$
\end{tasks}\recorda{$\left(\dfrac fg\right)'=\dfrac{f'\cdot g-f\cdot g'}{g^2}$. Simplifica el numerador, però deixa el denominador sense desenvolupar.}

\ex{1} Donada $f(x)=x^4-2x^3+5x$, calcula $f'(x)$, $f''(x)$ i $f''(1)$.

\ex{2} El cost de fabricar $x$ unitats és $C(x)=0{,}01x^2+5x+200$ euros (exercici \ref{ex:cost}). Calcula el cost marginal $C'(x)$ i $C'(100)$. Què significa? Compara-ho amb la TVM de l'exercici \ref{ex:cost}.

\seccio{6}{Continuïtat i derivabilitat. Derivades laterals}

\ex{2} Calcula les derivades laterals en el punt indicat i digues si la funció hi és derivable:\begin{tasks}(2)
  \task $f(x)=|x+1|$ en $x=-1$
  \task $f(x)=|x^2-9|$ en $x=3$
  \task $f(x)=x\,|x-1|$ en $x=1$
  \task $f(x)=\sqrt[3]{x^2}$ en $x=0$
\end{tasks}\recorda{$f'(a^-)=\displaystyle\lim_{h\to0^-}\frac{f(a+h)-f(a)}{h}$ i $f'(a^+)=\displaystyle\lim_{h\to0^+}\frac{f(a+h)-f(a)}{h}$. \ $f$ és derivable en $a$ si les dues existeixen, són finites i coincideixen. Un valor absolut s'escriu primer a trossos.}

\ex{1} Raona si són vertaderes o falses les afirmacions següents. Si són falses, posa'n un contraexemple:\begin{tasks}(1)
  \task Si $f$ és contínua en $x=a$, aleshores és derivable en $x=a$.
  \task Si $f$ no és contínua en $x=a$, aleshores no és derivable en $x=a$.
  \task Si $f$ no és derivable en $x=a$, aleshores no és contínua en $x=a$.
  \task Si en una funció a trossos les derivades dels dos trossos valen el mateix en el punt de canvi $a$, la funció és derivable en $a$.
\end{tasks}\recorda{Derivable $\Rightarrow$ contínua. El recíproc no és cert.}

\seccio{8}{Funcions a trossos amb paràmetres}
