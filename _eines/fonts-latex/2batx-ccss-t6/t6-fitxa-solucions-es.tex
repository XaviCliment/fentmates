\seccio{1}{Recta tangente y recta normal}

\ex{1} \begin{tasks}(1)
  \task $f(2)=1$, $f'(x)=3x^2-4x$, $f'(2)=4$: \ tangente $\res{y=4x-7}$; \ normal $y-1=-\frac14(x-2)\Rightarrow\res{y=-\frac14x+\frac32}$
  \task $f(0)=-1$, $f'(x)=\frac{3}{(x+1)^2}$, $f'(0)=3$: \ tangente $\res{y=3x-1}$; \ normal $\res{y=-\frac13x-1}$
  \task $f(1)=0$, $f'(x)=\ln x+1$, $f'(1)=1$: \ tangente $\res{y=x-1}$; \ normal $\res{y=-x+1}$
  \task $f(1)=1$, $f'(x)=2x\,e^{x^2-1}$, $f'(1)=2$: \ tangente $\res{y=2x-1}$; \ normal $\res{y=-\frac12x+\frac32}$
\end{tasks}

\ex{2} $f'(x)=3x^2-12x=-9\Rightarrow x^2-4x+3=0\Rightarrow x=1,\ x=3$. \ $f(1)=3$: $\res{y=-9x+12}$; \ $f(3)=-19$: $\res{y=-9x+8}$.

\ex{2} $-\frac{1}{f'(a)}=-3\Rightarrow f'(a)=\frac13$. \ $f'(x)=\frac{1}{\sqrt{2x+1}}=\frac13\Rightarrow2x+1=9\Rightarrow x=4$, punto $\res{(4,3)}$. \ Normal: $y-3=-3(x-4)\Rightarrow\res{y=-3x+15}$.
