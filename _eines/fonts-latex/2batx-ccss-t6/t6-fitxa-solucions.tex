\seccio{1}{Recta tangent i recta normal}

\ex{1} \begin{tasks}(1)
  \task $f(2)=1$, $f'(x)=3x^2-4x$, $f'(2)=4$: \ tangent $\res{y=4x-7}$; \ normal $y-1=-\frac14(x-2)\Rightarrow\res{y=-\frac14x+\frac32}$
  \task $f(0)=-1$, $f'(x)=\frac{3}{(x+1)^2}$, $f'(0)=3$: \ tangent $\res{y=3x-1}$; \ normal $\res{y=-\frac13x-1}$
  \task $f(1)=0$, $f'(x)=\ln x+1$, $f'(1)=1$: \ tangent $\res{y=x-1}$; \ normal $\res{y=-x+1}$
  \task $f(1)=1$, $f'(x)=2x\,e^{x^2-1}$, $f'(1)=2$: \ tangent $\res{y=2x-1}$; \ normal $\res{y=-\frac12x+\frac32}$
\end{tasks}

\ex{2} $f'(x)=3x^2-12x=-9\Rightarrow x^2-4x+3=0\Rightarrow x=1,\ x=3$. \ $f(1)=3$: $\res{y=-9x+12}$; \ $f(3)=-19$: $\res{y=-9x+8}$.

\ex{2} $-\frac{1}{f'(a)}=-3\Rightarrow f'(a)=\frac13$. \ $f'(x)=\frac{1}{\sqrt{2x+1}}=\frac13\Rightarrow2x+1=9\Rightarrow x=4$, punt $\res{(4,3)}$. \ Normal: $y-3=-3(x-4)\Rightarrow\res{y=-3x+15}$.

\ex{1} \pau{Madrid, EvAU juliol 2019}$f'(x)=6x^2-8=0\Rightarrow x=\pm\frac{2}{\sqrt3}=\pm\frac{2\sqrt3}{3}$. \ Punts $\res{\left(\frac{2\sqrt3}{3},-\frac{32\sqrt3}{9}\right)}$ i $\res{\left(-\frac{2\sqrt3}{3},\frac{32\sqrt3}{9}\right)}$.

\ex{2} \pau{Catalunya, PAU juny 2019}Tangent horitzontal: $f'(x)=a-\frac{8}{x^2}\Rightarrow f'(-2)=a-2=0\Rightarrow\res{a=2}$. \ $f(-2)=-4+b-4=-6\Rightarrow\res{b=2}$.

\ex{2} \pau{Andalusia, PEvAU juny 2018}\begin{tasks}(1)
  \task Continuïtat: $1+a=b+2$. \ Derivabilitat: $3+2a=b-2$. \ $\res{a=-6,\ b=-7}$.
  \task En $x=2$ fem servir $f(x)=3x+\frac2x$: $f(2)=7$, $f'(x)=3-\frac{2}{x^2}$, $f'(2)=\frac52$: \ $y-7=\frac52(x-2)\Rightarrow\res{5x-2y+4=0}$.
\end{tasks}

\ex{1} \pau{Múrcia, EBAU setembre 2019}$f(1)=2e^2$, $f'(x)=2e^{x+1}$, $f'(1)=2e^2$: \ $y-2e^2=2e^2(x-1)\Rightarrow\res{y=2e^2x}$.

\seccio{2}{Monotonia d'una funció}

\ex{1} \begin{tasks}(1)
  \task $f'(x)=3(x+1)(x-3)$: creixent en $\res{(-\infty,-1)\cup(3,+\infty)}$, decreixent en $\res{(-1,3)}$.
  \task $f'(x)=4x(x+2)(x-2)$: decreixent en $\res{(-\infty,-2)\cup(0,2)}$, creixent en $\res{(-2,0)\cup(2,+\infty)}$.
  \task $f'(x)=-3(x+1)(x-1)$: decreixent en $\res{(-\infty,-1)\cup(1,+\infty)}$, creixent en $\res{(-1,1)}$.
  \task $f'(x)=3x^2+3>0$ sempre: \rest{creixent en tot $\mathbb R$}.
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $f'(x)=\frac{-2}{(x-2)^2}<0$: decreixent en $\res{(-\infty,2)}$ i en $\res{(2,+\infty)}$.
  \task $f'(x)=\frac{x^2-1}{x^2}$: creixent en $\res{(-\infty,-1)\cup(1,+\infty)}$, decreixent en $\res{(-1,0)\cup(0,1)}$.
  \task $f'(x)=\frac{-2x}{(x^2+1)^2}$: creixent en $\res{(-\infty,0)}$, decreixent en $\res{(0,+\infty)}$.
  \task $f'(x)=\frac{-2x}{(x^2-1)^2}$: creixent en $\res{(-\infty,-1)\cup(-1,0)}$, decreixent en $\res{(0,1)\cup(1,+\infty)}$.
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $f'(x)=x(x+2)e^x$: creixent en $\res{(-\infty,-2)\cup(0,+\infty)}$, decreixent en $\res{(-2,0)}$.
  \task $f'(x)=\frac{e^x(x-1)}{x^2}$: decreixent en $\res{(-\infty,0)\cup(0,1)}$, creixent en $\res{(1,+\infty)}$.
  \task $f'(x)=\frac{2x}{x^2+4}$: decreixent en $\res{(-\infty,0)}$, creixent en $\res{(0,+\infty)}$.
  \task Domini $(0,+\infty)$. $f'(x)=2x-\frac2x=\frac{2(x^2-1)}{x}$: decreixent en $\res{(0,1)}$, creixent en $\res{(1,+\infty)}$.
\end{tasks}

\ex{1} \begin{tasks}(1)
  \task Creixent en $\res{(-\infty,-1)\cup(1,+\infty)}$ i decreixent en $\res{(-1,1)}$.
  \task Màxim relatiu $\res{(-1,2)}$ i mínim relatiu $\res{(1,-2)}$.
  \task $\res{f'(0)<0}$ (decreix), $\res{f'(2)>0}$ (creix) i $\res{f'(-1)=0}$ (tangent horitzontal).
\end{tasks}

\ex{2} \pau{Madrid, EvAU juny 2018}\begin{tasks}(1)
  \task $\res{\mathbb R-\{-1\}}$
  \task $f'(x)=\frac{3x^2(x+1)^2-2x^3(x+1)}{(x+1)^4}=\frac{x^2(x+3)}{(x+1)^3}$. Creixent en $\res{(-\infty,-3)\cup(-1,+\infty)}$ i decreixent en $\res{(-3,-1)}$. (En $x=0$, $f'=0$ però no canvia de signe.)
\end{tasks}

\ex{2} \pau{Balears, PBAU ordinària 2021}\begin{tasks}(1)
  \task $f'(1)=3a+2b+1=0$ i $f(3)=27a+9b+3=0$: \ $\res{a=\frac19,\ b=-\frac23}$.
  \task $f'(x)=9x^2+6x+1=(3x+1)^2\geq0$: \rest{creixent en tot $\mathbb R$} (en $x=-\frac13$ la tangent és horitzontal, però no hi ha extrem).
\end{tasks}

\ex{2} \pau{Canàries, EBAU juny 2017}\begin{tasks}(1)
  \task $f'(x)=-0{,}5<0$ si $0<x<6$ i $f'(x)=\frac{5(x+4)-5(x+14)}{(x+4)^2}=\frac{-50}{(x+4)^2}<0$ si $x>6$: \rest{sempre decreix}.
  \task $f(6)=10$ i és decreixent: després revela \rest{menys de 10}. $\lim_{x\to+\infty}\frac{5(x+14)}{x+4}=5$ i s'hi acosta per dalt: \rest{mai menys de 5}.
\end{tasks}

\seccio{3}{Màxims i mínims relatius}

\ex{1} \begin{tasks}(1)
  \task $f'(x)=6(x-1)(x-2)$, $f''(x)=12x-18$: \ màxim $\res{(1,1)}$, mínim $\res{(2,0)}$.
  \task $f'(x)=4x(x^2-1)$, $f''(x)=12x^2-4$: \ màxim $\res{(0,3)}$, mínims $\res{(-1,2)}$ i $\res{(1,2)}$.
  \task $f'(x)=1-\frac{1}{x^2}$, $f''(x)=\frac{2}{x^3}$: \ màxim $\res{(-1,-2)}$, mínim $\res{(1,2)}$.
  \task $f'(x)=e^{-x}(1-x)$, $f''(x)=e^{-x}(x-2)$, $f''(1)=-\frac1e<0$: \ màxim $\res{\left(1,\frac1e\right)}$.
\end{tasks}

\ex{2} $f'(x)=12x^3-12x^2=12x^2(x-1)=0\Rightarrow x=0,\ x=1$. \ $f''(x)=36x^2-24x$: $f''(1)=12>0$, mínim $\res{(1,-1)}$. \ $f''(0)=0$ no decideix: $f'<0$ a l'esquerra i a la dreta de 0 (no canvia de signe), així que en $x=0$ \rest{no hi ha extrem}.

\ex{2} \pau{Catalunya, PAU ordinària 2021}\begin{tasks}(1)

\ex{2} \pau{Múrcia, EBAU juliol 2020}\begin{tasks}(1)

\ex{2} \pau{País Basc, EAU juliol 2019}$f(0)=c=0$; \ $f(2)=8+4a+2b=-4$; \ $f'(2)=12+4a+b=0$. \ $\res{a=-3,\ b=0,\ c=0}$. \ $f'(x)=3x(x-2)$: màxim $\res{(0,0)}$ i mínim $\res{(2,-4)}$.

\ex{2} \pau{La Rioja, EBAU ordinària 2020}\begin{tasks}(1)

\seccio{4}{Extrems absoluts en un interval tancat}

\ex{1} \begin{tasks}(1)
  \task $f'(x)=3(x^2-1)\Rightarrow x=1$. \ $f(0)=0$, $f(1)=-2$, $f(2)=2$: màxim $\res{2}$ en $x=2$ i mínim $\res{-2}$ en $x=1$.
  \task $f'(x)=2x-6\Rightarrow x=3$. \ $f(-1)=12$, $f(3)=-4$, $f(4)=-3$: màxim $\res{12}$ en $x=-1$ i mínim $\res{-4}$ en $x=3$.
  \task $f'(x)=1-\frac{4}{x^2}\Rightarrow x=2$ ($-2$ no és de l'interval). \ $f(1)=5$, $f(2)=4$, $f(5)=5{,}8$: màxim $\res{5{,}8}$ en $x=5$ i mínim $\res{4}$ en $x=2$.
\end{tasks}

\ex{2} \pau{Andalusia, PEvAU setembre 2018}\begin{tasks}(1)

\end{tasks}

\ex{2} \pau{Extremadura, EBAU juliol 2019}$P'(t)=-3(t-2)(t-8)$. \ $P(0)=50$, $P(2)=6$, $P(8)=114$, $P(10)=70$. \ Mínim a les $\res{2\text{ h}}$ ($\res{6}$ kW) i màxim a les $\res{8\text{ h}}$ ($\res{114}$ kW).

\ex{2} \pau{Aragó, EvAU juny 2018}\begin{tasks}(1)

\end{tasks}

\ex{2} \pau{Galícia, ABAU juliol 2019}\begin{tasks}(1)
