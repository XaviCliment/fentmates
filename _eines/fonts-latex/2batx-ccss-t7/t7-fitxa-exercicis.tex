\seccio{1}{Domini i punts de tall amb els eixos}

\ex{1} Calcula el domini de les funcions següents:\begin{tasks}(3)
  \task $f(x)=x^3-5x+2$
  \task $f(x)=\dfrac{x+3}{x^2-9}$
  \task $f(x)=\dfrac{2x}{x^2+x-6}$
  \task $f(x)=\dfrac{5}{x^2+9}$
  \task $f(x)=e^{\frac{1}{x-2}}$
  \task $f(x)=\dfrac{x-1}{x^3-x}$
\end{tasks}\recorda{Polinomis: $\mathbb R$. Fraccions: traiem els valors que anul·len el denominador. Exponencials: el domini de l'exponent.}

\ex{2} Calcula el domini de les funcions logarítmiques següents:\begin{tasks}(3)
  \task $f(x)=\ln(2x-6)$
  \task $f(x)=\ln(x^2-9)$
  \task $f(x)=\ln(x^2+1)$
  \task $f(x)=\ln\dfrac{x+1}{x-3}$
  \task $f(x)=\dfrac{\ln x}{x-2}$
  \task $f(x)=\ln(4-x^2)$
\end{tasks}\recorda{L'argument del logaritme ha de ser estrictament positiu. Si cal, fes una taula de signes.}

\ex{1} Troba els punts de tall amb els eixos de coordenades:\begin{tasks}(3)
  \task $f(x)=x^3-x^2-6x$
  \task $f(x)=x^4-5x^2+4$
  \task $f(x)=\dfrac{x^2-4}{x+1}$
  \task $f(x)=\dfrac{x+2}{x^2+1}$
  \task $f(x)=e^{x}-e$
  \task $f(x)=\ln(2x-1)$
\end{tasks}\recorda{Eix OX: resolem $f(x)=0$. Eix OY: calculem $f(0)$, si $0$ és del domini.}

\ex{2} Calcula el domini de les funcions radicals següents:\begin{tasks}(3)
  \task $f(x)=\sqrt{3x-6}$
  \task $f(x)=\sqrt{10-2x}$
  \task $f(x)=\sqrt{x^2-16}$
  \task $f(x)=\sqrt{x^2-4x+3}$
  \task $f(x)=\dfrac{x}{\sqrt{x+5}}$
  \task $f(x)=\sqrt[3]{x^2-7}$
\end{tasks}\recorda{Arrel d'índex parell: el radicand ha de ser $\geq0$ (els extrems on val 0 sí que hi entren); si l'arrel és al denominador, ha de ser $>0$. Arrel d'índex senar: el domini del radicand.}

\ex{2} Troba els punts de tall amb els eixos de les funcions radicals següents. Recorda calcular primer el domini:\begin{tasks}(2)
  \task $f(x)=\sqrt{x^2-9}$
  \task $f(x)=\sqrt{x+5}-3$
  \task $f(x)=\sqrt{16-x^2}$
  \task $f(x)=\sqrt{x+1}-x+1$
\end{tasks}\recorda{$\sqrt{g(x)}=0\iff g(x)=0$. Si hi ha més termes, aïllem l'arrel i elevem al quadrat; després \textbf{comprovem} les solucions, perquè en elevar al quadrat poden aparéixer solucions falses.}

\ex{2} Calcula $a$ perquè la gràfica de $f(x)=x^2+ax+4$ talle l'eix OX en $x=1$. Quin és l'altre punt de tall amb OX? I el tall amb OY?

\seccio{2}{Simetries}

\ex{1} Estudia si les funcions següents són parelles, senars o cap de les dues coses:\begin{tasks}(4)
  \task $f(x)=x^4-x^2$
  \task $f(x)=x^3+2x$
  \task $f(x)=x^3+x^2$
  \task $f(x)=\dfrac{x^2}{x^2+1}$
  \task $f(x)=\dfrac{x}{x^2-4}$
  \task $f(x)=e^{-x^2}+1$
  \task $f(x)=\ln(x^2+1)$
  \task $f(x)=\dfrac{x+1}{x}$
\end{tasks}\recorda{Parella: $f(-x)=f(x)$ (simètrica respecte de l'eix OY). Senar: $f(-x)=-f(x)$ (simètrica respecte de l'origen).}

\ex{2} Estudia la simetria de les funcions radicals següents. Abans, comprova si el domini és simètric:\begin{tasks}(4)
  \task $f(x)=\sqrt{x^2+9}$
  \task $f(x)=x\sqrt{4-x^2}$
  \task $f(x)=\sqrt{x+4}$
  \task $f(x)=\dfrac{x^3}{\sqrt{x^2+1}}$
\end{tasks}

  \task si $f$ és parella;

\seccio{3}{Asímptotes}

\ex{1} Calcula les asímptotes verticals i horitzontals d'aquestes funcions racionals i estudia els límits laterals en cada asímptota vertical:\begin{tasks}(3)
  \task $f(x)=\dfrac{3x-1}{x+2}$
  \task $f(x)=\dfrac{x^2+1}{x^2-4}$
  \task $f(x)=\dfrac{2x}{x^2+1}$
  \task $f(x)=\dfrac{x-3}{x^2-9}$
  \task $f(x)=\dfrac{5}{(x-1)^2}$
\end{tasks}\recorda{AV: on s'anul·la el denominador i el numerador no (límit $\frac{k}{0}$). AH: grau $P<$ grau $Q\Rightarrow y=0$; graus iguals $\Rightarrow y=$ quocient dels coeficients principals.}

\ex{2} Calcula les asímptotes obliqües pel mètode del quocient (fes la divisió $P(x):Q(x)$) i també les verticals. En els apartats a) i b), estudia la posició de la gràfica respecte de l'asímptota obliqua.\begin{tasks}(2)
  \task $f(x)=\dfrac{x^2+1}{x-2}$
  \task $f(x)=\dfrac{2x^2-3x}{x+1}$
  \task $f(x)=\dfrac{x^3-1}{x^2}$
  \task $f(x)=\dfrac{x^2-x+2}{x-1}$
\end{tasks}\recorda{Si grau $P=$ grau $Q+1$: $P(x)=Q(x)\cdot C(x)+R(x)\Rightarrow f(x)=C(x)+\dfrac{R(x)}{Q(x)}$ i l'asímptota obliqua és $y=C(x)$. El signe de $\frac{R(x)}{Q(x)}$ diu si la gràfica queda per damunt o per davall.}

\ex{2} Calcula les asímptotes de les funcions exponencials i logarítmiques següents:\begin{tasks}(3)
  \task $f(x)=3-e^{x}$
  \task $f(x)=2e^{-x}+1$
  \task $f(x)=\ln(x-4)$
  \task $f(x)=\ln(x^2+1)$
  \task $f(x)=e^{1/x}$
  \task $f(x)=x\,e^{-x}$
\end{tasks}\recorda{$\lim_{x\to-\infty}e^x=0$, $\lim_{x\to+\infty}e^{-x}=0$ i $\lim_{x\to0^+}\ln x=-\infty$. L'exponencial creix més ràpid que qualsevol potència.}

\ex{2} Asímptotes amb paràmetres:\begin{tasks}(1)
  \task Calcula $a$ i $b$ perquè la funció $f(x)=\dfrac{ax^2+b}{x-1}$ tinga l'asímptota obliqua $y=2x+2$ i passe pel punt $(2,11)$.
  \task Per a quin valor de $k$ la funció $g(x)=\dfrac{x^2+k}{x-2}$ no té asímptota vertical? Com és la gràfica en aquest cas?
\end{tasks}
