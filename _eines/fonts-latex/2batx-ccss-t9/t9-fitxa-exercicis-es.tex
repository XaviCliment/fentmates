\seccio{1}{Espacio muestral y operaciones con sucesos}

\exf{1} Escribe el espacio muestral de los experimentos siguientes y di cuántos elementos tiene:\begin{tasks}(1)
  \task Lanzar una moneda y un dado.
  \task Anotar, por orden de nacimiento, si cada uno de los tres hijos de una familia es chico ($X$) o chica ($Y$).
  \task Sacar, una detrás de otra y sin reemplazamiento, dos bolas de una urna que tiene una roja ($R$), una azul ($B$) y una verde ($V$).
  \task Lanzar un dado hasta que salga un 6 y anotar el número de tiradas que han hecho falta.
\end{tasks}

\ex{1} Lanzamos un dado y consideramos los sucesos $A=$«salir un número menor que 4», $B=$«salir par» y $C=$«salir 5 o 6». Calcula:\begin{tasks}(4)
  \task $A\cup B$
  \task $A\cap B$
  \task $\co A$
  \task $\co B$
  \task $A-B$
  \task $B\cap C$
  \task $\co A\cap\co B$
  \task $\overline{A\cup C}$
\end{tasks}¿Qué pares de estos sucesos son incompatibles?

\ex{1} Sean $M=$«un alumno aprueba Matemáticas» y $N=$«aprueba Inglés». Expresa con operaciones entre $M$ y $N$ los sucesos siguientes:\begin{tasks}(2)
  \task Aprueba las dos.
  \task No aprueba ninguna.
  \task Aprueba solo Matemáticas.
  \task Aprueba como mucho una.
  \task Aprueba al menos una.
  \task Aprueba exactamente una.
  \task Suspende alguna de las dos.
\end{tasks}\recorda{«al menos uno» es la unión; «ambos», la intersección; «ninguno» es $\co M\cap\co N=\overline{M\cup N}$ (De Morgan).}

\ex{1} En el espacio muestral $E=\{1,2,3,4,5,6,7,8\}$ consideramos $A=\{1,2,3,4\}$ y $B=\{2,4,6,8\}$. Comprueba las dos leyes de De Morgan.

\ex{2} Sean $A$, $B$ y $C$ tres sucesos de un mismo experimento. Expresa con operaciones:\begin{tasks}(2)
  \task Ocurre solo $A$.
  \task Ocurre al menos uno de los tres.
  \task No ocurre ninguno.
  \task Ocurre exactamente uno.
  \task Ocurren los tres.
  \task Ocurren al menos dos.
\end{tasks}

\seccio{3}{Fórmulas de la probabilidad}

\exf{1} Sean $A$ y $B$ dos sucesos con $P(A)=0{,}45$, $P(B)=0{,}3$ y $P(A\cup B)=0{,}6$. Calcula:\begin{tasks}(3)
  \task $P(A\cap B)$
  \task $P(\co A\cap\co B)$
  \task $P(A\cap\co B)$
  \task $P(\co A\cup\co B)$
  \task $P(\text{solo uno de los dos})$
\end{tasks}\recorda{$P(A\cup B)=P(A)+P(B)-P(A\cap B)$; \ $P(\co A)=1-P(A)$; \ $P(A\cap\co B)=P(A)-P(A\cap B)$; \ De Morgan: $\co A\cap\co B=\overline{A\cup B}$ y $\co A\cup\co B=\overline{A\cap B}$.}

\ex{1} \pau{Madrid, EvAU julio 2018} Sean $A$ y $B$ dos sucesos de un experimento aleatorio tales que $P(A)=0{,}4$, $P(B)=0{,}6$ y $P(A\cup B)=0{,}8$. Calcula $P(\co A\cap B)$ y $P(\co A\cup\co B)$.

\ex{1} De dos sucesos se sabe que $P(\co A)=0{,}7$, $P(\co B)=0{,}4$ y $P(\co A\cup\co B)=0{,}9$. Calcula $P(A)$, $P(B)$, $P(A\cap B)$, $P(A\cup B)$ y $P(\co A\cap\co B)$.

\ex{1} Los sucesos $A$ y $B$ son incompatibles, con $P(A)=0{,}35$ y $P(B)=0{,}4$. Calcula $P(A\cup B)$, $P(\co A\cap\co B)$, $P(A\cap\co B)$ y $P(\co A\cup\co B)$.

\ex{2} Sean $A$ y $B$ dos sucesos con $P(A)=0{,}6$ y $P(B)=0{,}7$.\begin{tasks}(1)
  \task ¿Pueden ser incompatibles?
  \task ¿Entre qué valores puede estar $P(A\cap B)$?
  \task ¿Y $P(A\cup B)$?
\end{tasks}

\ex{2} Tres sucesos $A$, $B$ y $C$ tienen probabilidad $0{,}4$ cada uno; las intersecciones de dos en dos tienen probabilidad $0{,}15$ y $P(A\cap B\cap C)=0{,}05$. Calcula la probabilidad de que ocurra al menos uno y de que no ocurra ninguno.

\seccio{5}{Probabilidad condicionada}

\exf{1} Sean $A$ y $B$ dos sucesos con $P(A)=0{,}5$, $P(B)=0{,}4$ y $P(A\cap B)=0{,}1$. Calcula:\begin{tasks}(3)
  \task $P(A\cond B)$
  \task $P(B\cond A)$
  \task $P(A\cond\co B)$
  \task $P(\co A\cond\co B)$
  \task $P(B\cond\co A)$
  \task $P(\co B\cond A)$
\end{tasks}\recorda{$P(A\cond B)=\dfrac{P(A\cap B)}{P(B)}$. El suceso que va detrás de la barra es la condición: va en el denominador.}

\ex{2} \pau{Madrid, EvAU julio 2019} Sean $A$ y $B$ dos sucesos con $P(A)=0{,}3$, $P(B\cond A)=0{,}4$ y $P(B\cond\co A)=0{,}6$. Calcula:\begin{tasks}(2)
  \task $P(A\cond B)$
  \task $P(\co A\cond\co B)$
\end{tasks}

\ex{1} \pau{Islas Baleares, PBAU septiembre 2017} Sean $A$ y $B$ dos sucesos con probabilidades $0{,}4$ y $0{,}6$, respectivamente. Se sabe que, dado $B$, la probabilidad de que ocurra $A$ es $0{,}3$.\begin{tasks}(1)

\seccio{6}{Independencia de sucesos}

\exf{1} Los sucesos $A$ y $B$ son independientes, con $P(A)=0{,}3$ y $P(B)=0{,}5$. Calcula $P(A\cap B)$, $P(A\cup B)$, $P(\co A\cap\co B)$, $P(A\cond B)$ y $P(A\cap\co B)$.\recorda{$A$ y $B$ son independientes si $P(A\cap B)=P(A)\cdot P(B)$. Entonces también lo son $A$ y $\co B$, $\co A$ y $B$, y $\co A$ y $\co B$.}

\ex{2} Se sabe que $P(A)=0{,}4$ y $P(A\cup B)=0{,}7$. Calcula $P(B)$ si:\begin{tasks}(2)
  \task $A$ y $B$ son incompatibles;
  \task $A$ y $B$ son independientes.
\end{tasks}

\ex{1} \pau{Madrid, EvAU extraordinaria 2021} Sean $A$ y $B$ dos sucesos con $P(A)=0{,}5$, $P(B)=0{,}8$ y $P(A\cup B)=0{,}9$.\begin{tasks}(1)

  \task Calcula $P(A\cond\co B)$.
\end{tasks}
