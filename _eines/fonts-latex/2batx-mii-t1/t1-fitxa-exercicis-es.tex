\seccio{1}{Matrices. Definiciones y tipos}

\ex{1} \label{ex:tipus}Escribe las matrices siguientes y di de qué tipo es cada una:\begin{tasks}(1)
  \task $A=(a_{ij})$, cuadrada de orden 3, con $a_{ij}=i^2-j^2$.
  \task $B=(b_{ij})$, cuadrada de orden 4, con $b_{ij}=1$ si $j=i+1$ \ y \ $b_{ij}=0$ en caso contrario.
  \task $C=(c_{ij})$, cuadrada de orden 3, con $c_{ij}=\min(i,j)$.
\end{tasks}

\ex{2} Sea $A$ una matriz cuadrada cualquiera.\begin{tasks}(1)
  \task Demuestra que $A+A^t$ es simétrica y que $A-A^t$ es antisimétrica.
  \task Deduce que toda matriz cuadrada es la suma de una matriz simétrica y de una antisimétrica, y escribe \ $M=\begin{pmatrix}1 & 4 & -2\\0 & 3 & 5\\6 & -1 & 2\end{pmatrix}$ \ de esta manera.
  \task Demuestra que la única matriz que es a la vez simétrica y antisimétrica es la matriz nula.
\end{tasks}\recorda{$(A+B)^t=A^t+B^t$, \ $(kA)^t=kA^t$ \ y \ $(A^t)^t=A$. \ $A$ es simétrica si $A^t=A$ y antisimétrica si $A^t=-A$.}

\ex{2} \label{ex:idem}Una matriz cuadrada $A$ es \emph{idempotente} si $A^2=A$, \emph{involutiva} si $A^2=I$ y \emph{nilpotente} si alguna potencia es la matriz nula (el menor exponente $k$ con $A^k=O$ se llama \emph{índice}). Clasifica las matrices:\begin{tasks}(2)
  \task $P=\begin{pmatrix}2 & -1\\2 & -1\end{pmatrix}$
  \task $Q=\begin{pmatrix}3 & -4\\2 & -3\end{pmatrix}$
  \task $R=\begin{pmatrix}2 & -4\\1 & -2\end{pmatrix}$
  \task $S=\begin{pmatrix}1 & 1 & 3\\5 & 2 & 6\\-2 & -1 & -3\end{pmatrix}$
\end{tasks}\par ¿Cuál es el índice de nilpotencia de la matriz $B$ del ejercicio \ref{ex:tipus}?

\ex{3} Sea $A$ una matriz idempotente.\begin{tasks}(1)
  \task Demuestra que $I-A$ también es idempotente y que $A(I-A)=O$.
  \task Demuestra que $B=2A-I$ es involutiva.
  \task Compruébalo con la matriz $P$ del ejercicio anterior.
\end{tasks}

\ex{2} \pau{Cataluña, ordinaria 2023} Considera las matrices \ $A=\begin{pmatrix}2 & -3 & -5\\-1 & 4 & 5\\1 & -3 & -4\end{pmatrix}$ \ y \ $B=\begin{pmatrix}2 & 2 & 0\\-1 & -1 & 0\\1 & 2 & 1\end{pmatrix}$.\begin{tasks}(1)
  \task Calcula $A\cdot B$ y $B\cdot A$.
  \task Sean $C$ y $D$ dos matrices cuadradas del mismo orden que cumplen $C\cdot D=C$ y $D\cdot C=D$. Demuestra que $C$ y $D$ son idempotentes.
\end{tasks}

\seccio{2}{Operaciones con matrices}

\ex{1} Dadas \ $A=\begin{pmatrix}1 & -2 & 0\\3 & 1 & -1\end{pmatrix}$, \ $B=\begin{pmatrix}2 & 1\\0 & -1\\4 & 3\end{pmatrix}$ \ y \ $C=\begin{pmatrix}1 & 2\\-1 & 0\end{pmatrix}$:\begin{tasks}(1)
  \task Calcula $(AB)^t-2C$ y $A\cdot B\cdot C$.
  \task Calcula $B\cdot A$. ¿Tiene sentido comparar $AB$ y $BA$?
  \task Comprueba que $(AB)^t=B^tA^t$ y explica por qué, en general, $A^tB^t$ no es igual a $(AB)^t$ (en este caso, ni siquiera tienen el mismo orden).
\end{tasks}

\ex{2} Sean $A$ y $B$ matrices cuadradas del mismo orden.\begin{tasks}(1)
  \task Demuestra que $(A+B)^2=A^2+2AB+B^2$ si y solo si $AB=BA$.
  \task Demuestra que $(A+B)(A-B)=A^2-B^2$ si y solo si $AB=BA$.
  \task Halla todas las matrices $X$ de orden 2 que conmutan con \ $A=\begin{pmatrix}1 & 2\\0 & 1\end{pmatrix}$, es decir, tales que $AX=XA$.
\end{tasks}

\ex{2} Halla todas las matrices de la forma \ $X=\begin{pmatrix}a & b\\ 0 & c\end{pmatrix}$ \ que cumplen $X^2=I$.

\ex{2} Halla los valores de $m$ para los que \ $A=\begin{pmatrix}1 & m\\ m & 1\end{pmatrix}$ \ cumple $A^2=2A$. Comprueba después que \ $B=\begin{pmatrix}1 & m\\ 0 & -1\end{pmatrix}$ \ cumple $B^2=I$ para cualquier $m$, y calcula $B^{2027}$.

\ex{2} Dada \ $A=\begin{pmatrix}1 & 1 & 1\\0 & 1 & 1\\0 & 0 & 1\end{pmatrix}$, calcula $A^2$ y $A^3$. Conjetura la expresión de $A^n$, comprueba que si es cierta para $n$ también lo es para $n+1$, y calcula $A^{10}$.\recorda{$1+2+\dots+n=\dfrac{n(n+1)}{2}$.}

\ex{2} Dada \ $A=\begin{pmatrix}a & 1\\ 0 & a\end{pmatrix}$, calcula $A^2$, $A^3$ y $A^4$, deduce la expresión de $A^n$ y calcula $A^{10}$ para $a=2$.

\ex{3} \label{ex:J}Sea \ $J=\begin{pmatrix}0 & -1\\1 & 0\end{pmatrix}$.\begin{tasks}(1)

  \task Calcula $J^{2026}$.
  \task Calcula la suma \ $J+J^2+J^3+\dots+J^{2026}$.

\ex{3} Sea \ $A=\begin{pmatrix}3 & -4\\1 & -1\end{pmatrix}$ \ y sea $N=A-I$.\begin{tasks}(1)
  \task Comprueba que $N^2=O$.
  \task Demuestra que $A^n=I+nN$ para todo $n\geq1$ (multiplica $I+nN$ por $A=I+N$).
  \task Calcula $A^{50}$.
\end{tasks}

\ex{3} \label{ex:pq}Sea \ $A=\begin{pmatrix}1 & 2\\3 & 4\end{pmatrix}$.\begin{tasks}(1)
  \task Halla los números $p$ y $q$ tales que $A^2=pA+qI$.
  \task Expresa $A^3$ y $A^4$ en la forma $\alpha A+\beta I$, sin calcularlas directamente.
\end{tasks}

\ex{3} La \emph{traza} de una matriz cuadrada, $\operatorname{tr}(A)$, es la suma de los elementos de la diagonal principal.\begin{tasks}(1)
  \task Comprueba que $\operatorname{tr}(AB)=\operatorname{tr}(BA)$ para dos matrices cualesquiera de orden 2.
  \task Deduce que no existe ningún par de matrices de orden 2 tales que $AB-BA=I$.
\end{tasks}

\ex{3} \pau{Madrid, extraordinaria 2025} Dada la matriz \ $A=\begin{pmatrix}1 & 5\\-1 & -3\end{pmatrix}$:\begin{tasks}(1)

\seccio{4}{Propiedades de los determinantes}

\ex{2} Sabiendo que \ $\begin{vmatrix}a & b & c\\ d & e & f\\ g & h & i\end{vmatrix}=3$, \ calcula, sin desarrollarlos:\begin{tasks}(2)
  \task $\begin{vmatrix}2a & 2b & 2c\\ d-3g & e-3h & f-3i\\ g & h & i\end{vmatrix}$
  \task $\begin{vmatrix}a+d & b+e & c+f\\ d+g & e+h & f+i\\ g+a & h+b & i+c\end{vmatrix}$
  \task $\begin{vmatrix}3c & a & b+2a\\ 3f & d & e+2d\\ 3i & g & h+2g\end{vmatrix}$
  \task $\begin{vmatrix}g & h & i\\ a & b & c\\ d & e & f\end{vmatrix}$
  \task $\begin{vmatrix}a-b & b-c & c-a\\ d-e & e-f & f-d\\ g-h & h-i & i-g\end{vmatrix}$
  \task $|-2A^t|$, donde $A$ es la matriz del determinante dado
\end{tasks}\recorda{Un determinante es lineal respecto a cada fila (o columna): si una fila es una suma, el determinante se descompone en una suma de dos determinantes.}

  \task $|3C|$
  \task $|C^tC^3|$
  \task $|(2C)^{-1}C^2|$
\end{tasks}

\seccio{6}{Matriz inversa}

\ex{2} Calcula la inversa de las matrices siguientes:\begin{tasks}(2)
  \task $A=\begin{pmatrix}2 & 1 & -1\\0 & 1 & 3\\1 & 0 & 2\end{pmatrix}$
  \task $B=\begin{pmatrix}1 & a & b\\ 0 & 1 & c\\ 0 & 0 & 1\end{pmatrix}$ \ ($a$, $b$, $c$ reales)
\end{tasks}\recorda{$A^{-1}=\dfrac{1}{|A|}\big(\mathrm{Adj}(A)\big)^t$, y solo existe si $|A|\neq0$.}

\ex{2} Dada \ $A=\begin{pmatrix}m & 1 & 0\\ 0 & m & 1\\ 0 & 0 & m\end{pmatrix}$, halla para qué valores de $m$ tiene inversa y calcúlala en función de $m$.
