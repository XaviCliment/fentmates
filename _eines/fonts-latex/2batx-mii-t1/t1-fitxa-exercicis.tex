\seccio{1}{Matrius. Definicions i tipus}

\ex{1} \label{ex:tipus}Escriu les matrius següents i digues de quin tipus és cadascuna:\begin{tasks}(1)
  \task $A=(a_{ij})$, quadrada d'ordre 3, amb $a_{ij}=i^2-j^2$.
  \task $B=(b_{ij})$, quadrada d'ordre 4, amb $b_{ij}=1$ si $j=i+1$ \ i \ $b_{ij}=0$ en cas contrari.
  \task $C=(c_{ij})$, quadrada d'ordre 3, amb $c_{ij}=\min(i,j)$.
\end{tasks}

\ex{2} Siga $A$ una matriu quadrada qualsevol.\begin{tasks}(1)
  \task Demostra que $A+A^t$ és simètrica i que $A-A^t$ és antisimètrica.
  \task Dedueix que tota matriu quadrada és la suma d'una matriu simètrica i d'una antisimètrica, i escriu \ $M=\begin{pmatrix}1 & 4 & -2\\0 & 3 & 5\\6 & -1 & 2\end{pmatrix}$ \ d'aquesta manera.
  \task Demostra que l'única matriu que és alhora simètrica i antisimètrica és la matriu nul·la.
\end{tasks}\recorda{$(A+B)^t=A^t+B^t$, \ $(kA)^t=kA^t$ \ i \ $(A^t)^t=A$. \ $A$ és simètrica si $A^t=A$ i antisimètrica si $A^t=-A$.}

\ex{2} \label{ex:idem}Una matriu quadrada $A$ és \emph{idempotent} si $A^2=A$, \emph{involutiva} si $A^2=I$ i \emph{nilpotent} si alguna potència és la matriu nul·la (el menor exponent $k$ amb $A^k=O$ s'anomena \emph{índex}). Classifica les matrius:\begin{tasks}(2)
  \task $P=\begin{pmatrix}2 & -1\\2 & -1\end{pmatrix}$
  \task $Q=\begin{pmatrix}3 & -4\\2 & -3\end{pmatrix}$
  \task $R=\begin{pmatrix}2 & -4\\1 & -2\end{pmatrix}$
  \task $S=\begin{pmatrix}1 & 1 & 3\\5 & 2 & 6\\-2 & -1 & -3\end{pmatrix}$
\end{tasks}\par Quin és l'índex de nilpotència de la matriu $B$ de l'exercici \ref{ex:tipus}?

\ex{3} Siga $A$ una matriu idempotent.\begin{tasks}(1)
  \task Demostra que $I-A$ també és idempotent i que $A(I-A)=O$.
  \task Demostra que $B=2A-I$ és involutiva.
  \task Comprova-ho amb la matriu $P$ de l'exercici anterior.
\end{tasks}

\ex{2} \pau{Catalunya, ordinària 2023} Considera les matrius \ $A=\begin{pmatrix}2 & -3 & -5\\-1 & 4 & 5\\1 & -3 & -4\end{pmatrix}$ \ i \ $B=\begin{pmatrix}2 & 2 & 0\\-1 & -1 & 0\\1 & 2 & 1\end{pmatrix}$.\begin{tasks}(1)
  \task Calcula $A\cdot B$ i $B\cdot A$.
  \task Siguen $C$ i $D$ dues matrius quadrades del mateix ordre que compleixen $C\cdot D=C$ i $D\cdot C=D$. Demostra que $C$ i $D$ són idempotents.
\end{tasks}

\seccio{2}{Operacions amb matrius}

\ex{1} Donades \ $A=\begin{pmatrix}1 & -2 & 0\\3 & 1 & -1\end{pmatrix}$, \ $B=\begin{pmatrix}2 & 1\\0 & -1\\4 & 3\end{pmatrix}$ \ i \ $C=\begin{pmatrix}1 & 2\\-1 & 0\end{pmatrix}$:\begin{tasks}(1)
  \task Calcula $(AB)^t-2C$ i $A\cdot B\cdot C$.
  \task Calcula $B\cdot A$. Té sentit comparar $AB$ i $BA$?
  \task Comprova que $(AB)^t=B^tA^t$ i explica per què, en general, $A^tB^t$ no és igual a $(AB)^t$ (en aquest cas, ni tan sols tenen el mateix ordre).
\end{tasks}

\ex{2} Siguen $A$ i $B$ matrius quadrades del mateix ordre.\begin{tasks}(1)
  \task Demostra que $(A+B)^2=A^2+2AB+B^2$ si i només si $AB=BA$.
  \task Demostra que $(A+B)(A-B)=A^2-B^2$ si i només si $AB=BA$.
  \task Troba totes les matrius $X$ d'ordre 2 que commuten amb \ $A=\begin{pmatrix}1 & 2\\0 & 1\end{pmatrix}$, és a dir, tals que $AX=XA$.
\end{tasks}

\ex{2} Troba totes les matrius de la forma \ $X=\begin{pmatrix}a & b\\ 0 & c\end{pmatrix}$ \ que compleixen $X^2=I$.

\ex{2} Troba els valors de $m$ per als quals \ $A=\begin{pmatrix}1 & m\\ m & 1\end{pmatrix}$ \ compleix $A^2=2A$. Comprova després que \ $B=\begin{pmatrix}1 & m\\ 0 & -1\end{pmatrix}$ \ compleix $B^2=I$ per a qualsevol $m$, i calcula $B^{2027}$.

\ex{2} Donada \ $A=\begin{pmatrix}1 & 1 & 1\\0 & 1 & 1\\0 & 0 & 1\end{pmatrix}$, calcula $A^2$ i $A^3$. Conjectura l'expressió de $A^n$, comprova que si és certa per a $n$ també ho és per a $n+1$, i calcula $A^{10}$.\recorda{$1+2+\dots+n=\dfrac{n(n+1)}{2}$.}

\ex{2} Donada \ $A=\begin{pmatrix}a & 1\\ 0 & a\end{pmatrix}$, calcula $A^2$, $A^3$ i $A^4$, dedueix l'expressió de $A^n$ i calcula $A^{10}$ per a $a=2$.

\ex{3} \label{ex:J}Siga \ $J=\begin{pmatrix}0 & -1\\1 & 0\end{pmatrix}$.\begin{tasks}(1)
  \task Calcula $J^2$, $J^3$ i $J^4$.
  \task Calcula $J^{2026}$.
  \task Calcula la suma \ $J+J^2+J^3+\dots+J^{2026}$.

\ex{3} Siga \ $A=\begin{pmatrix}3 & -4\\1 & -1\end{pmatrix}$ \ i siga $N=A-I$.\begin{tasks}(1)
  \task Comprova que $N^2=O$.
  \task Demostra que $A^n=I+nN$ per a tot $n\geq1$ (multiplica $I+nN$ per $A=I+N$).
  \task Calcula $A^{50}$.
\end{tasks}

\ex{3} \label{ex:pq}Siga \ $A=\begin{pmatrix}1 & 2\\3 & 4\end{pmatrix}$.\begin{tasks}(1)
  \task Troba els nombres $p$ i $q$ tals que $A^2=pA+qI$.
  \task Expressa $A^3$ i $A^4$ en la forma $\alpha A+\beta I$, sense calcular-les directament.

\ex{3} La \emph{traça} d'una matriu quadrada, $\operatorname{tr}(A)$, és la suma dels elements de la diagonal principal.\begin{tasks}(1)
  \task Comprova que $\operatorname{tr}(AB)=\operatorname{tr}(BA)$ per a dues matrius qualssevol d'ordre 2.
  \task Dedueix que no hi ha cap parella de matrius d'ordre 2 tals que $AB-BA=I$.
\end{tasks}

\ex{3} \pau{Madrid, extraordinària 2025} Donada la matriu \ $A=\begin{pmatrix}1 & 5\\-1 & -3\end{pmatrix}$:\begin{tasks}(1)

\seccio{4}{Propietats dels determinants}

\ex{2} Sabent que \ $\begin{vmatrix}a & b & c\\ d & e & f\\ g & h & i\end{vmatrix}=3$, \ calcula, sense desenvolupar-los:\begin{tasks}(2)
  \task $\begin{vmatrix}2a & 2b & 2c\\ d-3g & e-3h & f-3i\\ g & h & i\end{vmatrix}$
  \task $\begin{vmatrix}a+d & b+e & c+f\\ d+g & e+h & f+i\\ g+a & h+b & i+c\end{vmatrix}$
  \task $\begin{vmatrix}3c & a & b+2a\\ 3f & d & e+2d\\ 3i & g & h+2g\end{vmatrix}$
  \task $\begin{vmatrix}g & h & i\\ a & b & c\\ d & e & f\end{vmatrix}$
  \task $\begin{vmatrix}a-b & b-c & c-a\\ d-e & e-f & f-d\\ g-h & h-i & i-g\end{vmatrix}$
  \task $|-2A^t|$, on $A$ és la matriu del determinant donat
\end{tasks}\recorda{Un determinant és lineal respecte de cada fila (o columna): si una fila és una suma, el determinant es descompon en una suma de dos determinants.}

\ex{2} Siguen $A$ i $B$ matrius quadrades d'ordre 3 amb $|A|=2$ i $|B|=-\tfrac12$, i $C$ una matriu quadrada d'ordre 4 amb $|C|=-2$. Calcula:\begin{tasks}(4)
  \task $|2A^{-1}B^t|$
  \task $|(3A)^{-1}|$
  \task $|A^2B^{-3}|$
  \task $|A^t(2B)^{-1}A|$
  \task $|BAB^{-1}|$
  \task $|3C|$
  \task $|C^tC^3|$
  \task $|(2C)^{-1}C^2|$
\end{tasks}

\seccio{6}{Matriu inversa}

\ex{2} Calcula la inversa de les matrius següents:\begin{tasks}(2)
  \task $A=\begin{pmatrix}2 & 1 & -1\\0 & 1 & 3\\1 & 0 & 2\end{pmatrix}$
