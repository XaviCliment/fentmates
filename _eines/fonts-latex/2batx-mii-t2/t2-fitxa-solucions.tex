\ex{2} La 3a equació és la suma de les dues primeres, i $\begin{vmatrix}1 & 1\\ 1 & -1\end{vmatrix}=-2\neq0$: \ $\left\{\begin{aligned}x+y&=2-z\\ x-y&=1-2z\end{aligned}\right.$\[x=\frac{\begin{vmatrix}2-z & 1\\ 1-2z & -1\end{vmatrix}}{-2}=\frac{3-3z}{2},\qquad y=\frac{\begin{vmatrix}1 & 2-z\\ 1 & 1-2z\end{vmatrix}}{-2}=\frac{1+z}{2}\]\[\boxed{x=\tfrac{3-3\lambda}{2};\ \ y=\tfrac{1+\lambda}{2};\ \ z=\lambda,\quad\lambda\in\mathbb{R}}\]

\seccio{6}{Discussió de sistemes amb paràmetres}

\ex{2} $|A|=(a-1)^2(a+2)$.\begin{tasks}(1)

  \task Si $a=-2$: sumant les tres equacions, $0=1-2+4=3$: \ \rest{SI} (\,$\operatorname{rg}(A)=2$, $\operatorname{rg}(A^*)=3$).
\end{tasks}

  \task Si $a\neq3$: \ \rest{SCD}.
  \task Si $a=3$ i $b=3$: la 3a equació és la suma de les dues primeres: \rest{SCI}.
