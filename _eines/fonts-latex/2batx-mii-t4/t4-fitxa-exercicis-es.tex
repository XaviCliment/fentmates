\seccio{1}{Ángulos entre rectas y planos}

\ex{1} Calcula el ángulo que forman las rectas:\begin{tasks}(1)
  \task \mbox{$r:\dfrac{x-1}{2}=\dfrac{y+1}{1}=\dfrac{z}{-2}$} \ y \ $s:(x,\,y,\,z)=(0,\,1,\,3)+\lambda\,(4,\,0,\,3)$
  \task $r:\left\{\begin{aligned}x+y&=1\\y-z&=2\end{aligned}\right.$ \ y \ $s:x=y=z$
\end{tasks}\recorda{\mbox{$\cos\alpha=\dfrac{|\vec d_r\cdot\vec d_s|}{|\vec d_r|\,|\vec d_s|}$}, siempre entre $0^\circ$ y $90^\circ$.}

\ex{1} Calcula el ángulo que forma la recta \mbox{$r:\dfrac{x-2}{1}=\dfrac{y}{-1}=\dfrac{z+1}{2}$} con cada uno de los planos siguientes:\begin{tasks}(2)
  \task $\pi_1:x+y+z=4$
  \task $\pi_2:2x-2y+4z=1$
\end{tasks}

\ex{1} Calcula el ángulo que forman los planos:\begin{tasks}(2)
  \task $x+y=3$ \ y \ $x+z=1$
  \task $2x-y+2z=0$ \ y el plano $OXY$
\end{tasks}

\ex{2} Calcula los ángulos que forma la recta de vector director $(2,\,-1,\,2)$ con cada uno de los ejes de coordenadas.

\ex{2} Halla $m$ para que la recta $(x,\,y,\,z)=(1,\,0,\,0)+\lambda\,(1,\,m,\,1)$ forme un ángulo de $30^\circ$ con el plano $z=0$.

\ex{2} ¿Hay algún valor de $k$ para el que los planos $x+ky+z=0$ y $2x-y+kz=3$ sean perpendiculares? ¿Y paralelos?

\ex{2} \pau{Canarias, extraordinaria 2026} Se sabe que el plano $\pi:x+3y+2z=10$ y la recta \mbox{$r:\dfrac{x+21}{11}=3-y=\dfrac{5-z}{2}$} son secantes.\begin{tasks}(1)
  \task Halla el punto de corte del plano $\pi$ y la recta $r$.
  \task Calcula el ángulo que forman el plano $\pi$ y la recta $r$.
\end{tasks}

\ex{2} \pau{País Vasco, extraordinaria 2025} Sean $\pi$ el plano que pasa por los puntos $A(2,\,3,\,4)$, $B(3,\,-1,\,-2)$, $C(5,\,-1,\,2)$ y $r$ la recta que pasa por los puntos $D(6,\,-5,\,-4)$, $E(7,\,1,\,4)$.\begin{tasks}(1)
  \task Calcula el ángulo entre el plano $\pi$ y la recta $r$, expresando el resultado en grados, minutos y segundos.
  \task En caso de que $r$ y $\pi$ se corten, calcula el punto de intersección. En caso contrario, calcula la distancia entre la recta $r$ y el plano $\pi$.

\ex{2} \pau{Canarias, extraordinaria 2024}\begin{tasks}(1)

  \task Dados el plano $\pi:x+3y+2z+3=0$ y la recta $r:\left\{\begin{aligned}2x-3y-z&=4\\x+y+2z&=-3\end{aligned}\right.$, que son secantes, calcula el punto de corte de la recta y el plano y el ángulo que forman.
