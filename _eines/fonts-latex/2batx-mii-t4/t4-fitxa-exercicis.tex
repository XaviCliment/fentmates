\seccio{1}{Angles entre rectes i plans}

\ex{1} Calcula l'angle que formen les rectes:\begin{tasks}(1)
  \task \mbox{$r:\dfrac{x-1}{2}=\dfrac{y+1}{1}=\dfrac{z}{-2}$} \ i \ $s:(x,\,y,\,z)=(0,\,1,\,3)+\lambda\,(4,\,0,\,3)$
  \task $r:\left\{\begin{aligned}x+y&=1\\y-z&=2\end{aligned}\right.$ \ i \ $s:x=y=z$
\end{tasks}\recorda{\mbox{$\cos\alpha=\dfrac{|\vec d_r\cdot\vec d_s|}{|\vec d_r|\,|\vec d_s|}$}, sempre entre $0^\circ$ i $90^\circ$.}

\ex{1} Calcula l'angle que forma la recta \mbox{$r:\dfrac{x-2}{1}=\dfrac{y}{-1}=\dfrac{z+1}{2}$} amb cadascun dels plans següents:\begin{tasks}(2)
  \task $\pi_1:x+y+z=4$
  \task $\pi_2:2x-2y+4z=1$
\end{tasks}

\ex{1} Calcula l'angle que formen els plans:\begin{tasks}(2)
  \task $x+y=3$ \ i \ $x+z=1$
  \task $2x-y+2z=0$ \ i el pla $OXY$
\end{tasks}

\ex{2} Calcula els angles que forma la recta de vector director $(2,\,-1,\,2)$ amb cadascun dels eixos de coordenades.

\ex{2} Troba $m$ perquè la recta $(x,\,y,\,z)=(1,\,0,\,0)+\lambda\,(1,\,m,\,1)$ forme un angle de $30^\circ$ amb el pla $z=0$.

\ex{2} Hi ha algun valor de $k$ per al qual els plans $x+ky+z=0$ i $2x-y+kz=3$ siguen perpendiculars? I paral·lels?

\ex{2} \pau{Canàries, extraordinària 2026} Se sap que el pla $\pi:x+3y+2z=10$ i la recta \mbox{$r:\dfrac{x+21}{11}=3-y=\dfrac{5-z}{2}$} són secants.\begin{tasks}(1)
  \task Troba el punt de tall del pla $\pi$ i la recta $r$.
  \task Calcula l'angle que formen el pla $\pi$ i la recta $r$.
\end{tasks}

\ex{2} \pau{Castella-la Manxa, ordinària 2024} Siguen els vectors $\vec u=(1,\,a,\,a)$ i $\vec v=(-1,\,0,\,2)$, amb $a\in\mathbb R$. Determina el valor de $a$ perquè l'angle entre els vectors $\vec u$ i $\vec v$ siga de $60^\circ$.

\ex{2} \pau{País Basc, extraordinària 2025} Siguen $\pi$ el pla que passa pels punts $A(2,\,3,\,4)$, $B(3,\,-1,\,-2)$, $C(5,\,-1,\,2)$ i $r$ la recta que passa pels punts $D(6,\,-5,\,-4)$, $E(7,\,1,\,4)$.\begin{tasks}(1)
  \task Calcula l'angle entre el pla $\pi$ i la recta $r$, expressant el resultat en graus, minuts i segons.
  \task En cas que $r$ i $\pi$ es tallen, calcula el punt d'intersecció. En cas contrari, calcula la distància entre la recta $r$ i el pla $\pi$.

\ex{2} \pau{Canàries, extraordinària 2024}\begin{tasks}(1)

  \task Donats el pla $\pi:x+3y+2z+3=0$ i la recta $r:\left\{\begin{aligned}2x-3y-z&=4\\x+y+2z&=-3\end{aligned}\right.$, que són secants, calcula el punt de tall de la recta i el pla i l'angle que formen.

\ex{2} \pau{La Rioja, ordinària 2024} Troba l'equació contínua de la recta $s$ que està continguda en el pla $\pi:x+y-2z+1=0$ i talla perpendicularment la recta $r:\left\{\begin{aligned}x+y+z&=-1\\4x-y+z&=3\end{aligned}\right.$

\ex{2} \pau{Galícia, ordinària 2026} Considera els punts $P(1,\,0,\,1)$, $Q(0,\,1,\,0)$ i $R(1,\,1,\,1)$.\begin{tasks}(1)

\seccio{7}{Perpendicular comuna}

\ex{2} Troba els punts $X\in r$ i $Y\in s$ més pròxims de les rectes $r:\left\{\begin{aligned}x-y&=0\\z&=0\end{aligned}\right.$ \ i \ $s:\left\{\begin{aligned}x&=1\\y-z&=0\end{aligned}\right.$ \ de l'exercici \ref{ex:pc} i la seua perpendicular comuna.
