\seccio{1}{Límites en el infinito}

\ex{2} Calcula los límites siguientes:\begin{tasks}(3)
  \task $\displaystyle\lim_{x\to -\infty}(2x^5-x^2)$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{4x^3-x}{2x^3+7}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{1-3x^2}{x+2}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{x^2+1}{x^3-2}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{2^{x+1}+5}{2^{x}-3}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{3e^{2x}-e^{x}}{e^{2x}+4}$
\end{tasks}\recorda{Cuando $x\to\pm\infty$, en un cociente de polinomios mandan los términos de mayor grado. Con exponenciales, dividimos por la exponencial que crece más deprisa (y recuerda que $e^{x}\to0$ cuando $x\to-\infty$).}

\ex{2} Calcula los límites siguientes con raíces. Cuidado con el signo cuando $x\to-\infty$:\begin{tasks}(3)
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{\sqrt{9x^2+1}}{2x-5}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{\sqrt{9x^2+1}}{2x-5}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{3x+\sqrt{x^2+5}}{2x+1}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{\sqrt[3]{8x^3-1}}{x}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{\sqrt{x^3+x}}{x+4}$
  \task $\displaystyle\lim_{x\to -\infty}\dfrac{\sqrt{x^2-5x+3}}{3x-2}$
\end{tasks}\recorda{$\sqrt{x^2}=|x|$: cuando $x\to-\infty$, $\sqrt{a x^2+\dots}\approx-\sqrt a\,x$. Si dudas, haz el cambio $x=-t$ con $t\to+\infty$.}

\ex{2} Ordena de menor a mayor orden de infinito (cuando $x\to+\infty$) \ $x^3,\ 2^x,\ \ln x,\ \sqrt x,\ e^x,\ x^{10},\ (\ln x)^3$, \ y calcula:\begin{tasks}(3)
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{x^3}{e^x}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{\ln x}{x^2}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{2^x+x^4}{3\cdot2^x}$
  \task $\displaystyle\lim_{x\to +\infty}\left(x^{10}-1{,}5^x\right)$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{3^x+2^x}{3^x-x^2}$
  \task $\displaystyle\lim_{x\to +\infty}\dfrac{(\ln x)^3}{\sqrt x}$
\end{tasks}

\ex{2} Calcula, utilizando las propiedades de los logaritmos:\begin{tasks}(2)
  \task $\displaystyle\lim_{x\to +\infty}\big(\ln(2x+1)-\ln(x+3)\big)$
  \task $\displaystyle\lim_{x\to +\infty}\big(\ln(x^2+1)-\ln x\big)$
\end{tasks}

\seccio{2}{Indeterminaciones en el infinito}

\ex{2} Indeterminación $\infty-\infty$:\begin{tasks}(3)
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x^2}{x+1}-x\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x^3}{x^2+1}-\dfrac{x^2}{x+1}\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\sqrt{x^2+8x}-x\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\sqrt{x^2+3x}-\sqrt{x^2-x}\right)$
  \task $\displaystyle\lim_{x\to -\infty}\left(\sqrt{x^2+2x}+x\right)$
  \task $\displaystyle\lim_{x\to +\infty}\left(\sqrt{x^2+1}-3x\right)$
\end{tasks}\recorda{Con fracciones, hacemos la resta. Con raíces, multiplicamos y dividimos por el \textbf{conjugado}. Antes, comprueba si los términos principales se cancelan: si no lo hacen, no hay indeterminación real.}

\ex{3} Calcula $\displaystyle\lim_{x\to+\infty}\left(\sqrt[3]{x^3+3x^2}-x\right)$. \ \textit{Indicación}: $A^3-B^3=(A-B)(A^2+AB+B^2)$.

\ex{2} Indeterminación $0\cdot\infty$:\begin{tasks}(2)
  \task $\displaystyle\lim_{x\to +\infty}x\left(\sqrt{x^2+4}-x\right)$
  \task $\displaystyle\lim_{x\to +\infty}(2x+1)\left(\sqrt{x+1}-\sqrt x\right)$
  \task $\displaystyle\lim_{x\to +\infty}x\ln\left(1+\dfrac2x\right)$
  \task $\displaystyle\lim_{x\to +\infty}x^3e^{-x}$
\end{tasks}\recorda{Transformamos el producto en un cociente ($f\cdot g=\frac{f}{1/g}$) u operamos. Un exponente se puede poner dentro del logaritmo: $x\ln A=\ln A^{x}$.}

\ex{2} Indeterminación $1^{\infty}$:\begin{tasks}(3)
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x+2}{x-1}\right)^{2x}$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{2x-1}{2x+3}\right)^{x+1}$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x^2+3}{x^2-1}\right)^{x^2}$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{3x+1}{x+2}\right)^{x}$
  \task $\displaystyle\lim_{x\to -\infty}\left(\dfrac{x-4}{x}\right)^{3x}$
  \task $\displaystyle\lim_{x\to +\infty}\left(\dfrac{x^2-x+1}{x^2+2}\right)^{\frac{x^2+1}{x-3}}$
\end{tasks}\recorda{Si $f\to1$ y $g\to\pm\infty$: \ $\lim f^{\,g}=e^{\lim\,(f-1)\,g}$. Si la base no tiende a 1, no hay indeterminación.}

\ex{3} Indeterminaciones $\infty^0$. Calcula tomando logaritmos:\begin{tasks}(2)
  \task $\displaystyle\lim_{x\to +\infty}x^{1/x}$
  \task $\displaystyle\lim_{x\to +\infty}(3x)^{\frac{2}{\ln x}}$
\end{tasks}

\ex{2} \pau{Aragón, ordinaria 2022} Calcula $\displaystyle\lim_{x\to+\infty}\left[\sqrt{3x^2-2}-\left(\sqrt3\,x+5\right)\right]$.

\seccio{6}{Continuidad de funciones}

\ex{2} Estudia la continuidad en $x=0$ de las funciones siguientes y clasifica la discontinuidad, si la hay:\begin{tasks}(1)
  \task $f(x)=\begin{cases}\dfrac{\sin 3x}{x}&\text{si }x\neq0\\[3pt]3&\text{si }x=0\end{cases}$
  \task $g(x)=\dfrac{1}{1+e^{1/x}}$
  \task $h(x)=x\cos\dfrac1x$. ¿Qué valor habría que dar a $h(0)$ para que fuera continua?
\end{tasks}\recorda{$f$ es continua en $a$ si $\exists f(a)$, $\exists\lim_{x\to a}f(x)$ y son iguales. Discontinuidades: evitable (el límite existe), de salto finito (laterales finitos y distintos), de salto infinito (un lateral infinito) y asintótica (ambos infinitos).}

\ex{2} Estudia la continuidad de las funciones siguientes:\begin{tasks}(1)
  \task $f(x)=\begin{cases}\cos x&\text{si }x\leq0\\[3pt]\dfrac{e^x-1}{x}&\text{si }0<x<2\\[3pt]\log_2(x+2)&\text{si }x\geq2\end{cases}$
  \task $g(x)=\begin{cases}\dfrac{3}{x+2}&\text{si }x<-1\\[3pt]2-|x|&\text{si }-1\leq x\leq1\\[3pt]\sqrt{x+3}-1&\text{si }x>1\end{cases}$
  \task $h(x)=\begin{cases}\dfrac{\sin 2x}{x}&\text{si }x<0\\[3pt]x+2&\text{si }0\leq x<\pi\\[3pt]\cos x+\pi+3&\text{si }x\geq\pi\end{cases}$
\end{tasks}\recorda{Primero, cada trozo en su intervalo abierto (denominadores, logaritmos, raíces, $\tan$\ldots\ solo cuentan si el punto está dentro del trozo). Después, los puntos donde cambia la expresión.}

\ex{2} Estudia la continuidad de:\begin{tasks}(3)
  \task $f(x)=\dfrac{x}{1+e^{1/x}}$
  \task $g(x)=\dfrac{x^2-4}{|x-2|}$
  \task $h(x)=e^{\frac{1}{x-1}}$
\end{tasks}
