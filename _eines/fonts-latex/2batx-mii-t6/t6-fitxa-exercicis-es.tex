\seccio{1}{Tasa de variación y derivada por definición}

\ex{1} Calcula la tasa de variación media de las funciones siguientes en el intervalo indicado:\begin{tasks}(3)
  \task $f(x)=x^3-x$ en $[0,2]$
  \task $g(x)=\ln x$ en $[1,e]$
  \task $h(x)=\sin x$ en $\left[0,\frac\pi2\right]$
\end{tasks}\recorda{$\text{TVM}[a,b]=\dfrac{f(b)-f(a)}{b-a}$: es la pendiente de la recta secante. La derivada es el límite de la TVM cuando el intervalo se hace tan pequeño como queramos.}

\ex{2} Lanzamos una pelota hacia arriba y su altura, en metros, a los $t$ segundos es $s(t)=-5t^2+20t$. Calcula la velocidad media en los dos primeros segundos y, aplicando la definición de derivada, la velocidad instantánea en $t=1$.

\ex{2} Calcula, aplicando la definición de derivada:\begin{tasks}(2)
  \task $f'(2)$ si $f(x)=\dfrac{1}{x^2}$
  \task $f'(1)$ si $f(x)=\dfrac{x}{x+1}$
  \task $f'(0)$ si $f(x)=e^{3x}$
  \task $f'\!\left(\frac\pi2\right)$ si $f(x)=\sin x$
\end{tasks}\recorda{$f'(a)=\displaystyle\lim_{h\to0}\frac{f(a+h)-f(a)}{h}$. Con exponenciales y funciones trigonométricas, usa los infinitésimos equivalentes: $e^{u}-1\sim u$, \ $1-\cos u\sim\frac{u^2}{2}$, \ $\sin u\sim u$.}

\ex{3} Calcula la función derivada aplicando la definición:\begin{tasks}(2)
  \task $f(x)=x^3$
  \task $f(x)=\dfrac{1}{\sqrt x}$
  \task $f(x)=\cos x$
  \task $f(x)=\ln(2x)$
\end{tasks}

\seccio{2}{Rectas tangente y normal}

\ex{1} Escribe las ecuaciones de las rectas tangente y normal a la gráfica de cada función en el punto indicado:\begin{tasks}(2)

\seccio{7}{Continuidad y derivabilidad}

  \task $f(x)=|x^2-9|$ en $x=3$

  \task $f(x)=|\sin x|$ en $x=\pi$
  \task $f(x)=x^2|x|$ en $x=0$
\end{tasks}\recorda{$f'(a^\pm)=\displaystyle\lim_{h\to0^\pm}\frac{f(a+h)-f(a)}{h}$. Es derivable si ambas existen, son finitas y son iguales.}

\ex{1} Razona si son verdaderas o falsas. Si son falsas, pon un contraejemplo:\begin{tasks}(1)

  \task Si $f$ y $g$ no son derivables en $x=a$, $f+g$ tampoco lo es.
  \task Si $f$ es derivable en $x=a$, $|f|$ también lo es.
\end{tasks}

\ex{2} \pau{Galicia, ordinaria 2026} Dibuja, cuando sea posible, la gráfica de una función continua pero no derivable en $x=0$.
