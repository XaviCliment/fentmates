\seccio{1}{Rectas tangente y normal}

\ex{1} Calcula:\begin{tasks}(1)
  \task \pau{Castilla-La Mancha, junio 2017} La recta normal a la gráfica de $g(x)=x^2-2x-4$ en el punto de abscisa $x=-3$.
  \task \pau{Aragón, ordinaria 2020} La recta tangente a $f(x)=\ln(2x+1)$ en $x=\frac12$.
  \task \pau{Cataluña, extraordinaria 2024} La recta tangente a $y=e^{-2x}$ en $x=0$ y el punto donde corta el eje de abscisas.
  \task \pau{Aragón, extraordinaria 2021} La recta tangente a $f(x)=\dfrac{e^x}{x^3-x}$ en $x=2$.
\end{tasks}\recorda{Tangente: $y-f(a)=f'(a)(x-a)$. Normal: $y-f(a)=-\dfrac{1}{f'(a)}(x-a)$.}

\ex{2} \pau{Cataluña, septiembre 2019} Considera la función $f(x)=\dfrac{1}{1+x^2}$.\begin{tasks}(1)
  \task Calcula la ecuación de la recta tangente a la gráfica en los puntos en los que es horizontal.
  \task Calcula las coordenadas del punto de la gráfica en el que la pendiente de la recta tangente es máxima.
\end{tasks}\recorda{La pendiente de la tangente es $m(x)=f'(x)$: para hacerla máxima, derivamos $f'$ e igualamos a cero (es decir, $f''(x)=0$).}

\ex{2} \pau{Asturias, junio 2019} Dada la curva $y=\dfrac{1}{3+x^2}$, escribe la función $m(x)$ que da la pendiente de la recta tangente en cada punto y halla en qué $x$ la pendiente es máxima.

\ex{2} \pau{La Rioja, ordinaria 2022} Dada la curva $f(x)=\frac14x^2+4x+4$, halla los puntos en los que la recta tangente pasa por el origen y escribe sus ecuaciones.\recorda{Si el punto no es de la curva, escribe la tangente en un punto genérico $(a,f(a))$ e impón que pase por el punto dado.}

\ex{2} ¿Cuántas rectas tangentes a $f(x)=x^3-3x$ pasan por el punto $(0,2)$? Escríbelas.

\ex{2} Halla $a$ para que la recta $y=2x+a$ sea tangente a la parábola $f(x)=x^2-4x+10$. ¿En qué punto?\recorda{Una recta es tangente a una parábola si, al cortarlas, la ecuación de segundo grado tiene una raíz doble (discriminante cero).}

\ex{2} \pau{Galicia, extraordinaria 2022} Halla los vértices del triángulo rectángulo que tiene la hipotenusa sobre la recta tangente a $f(x)=x^2$ en $x=2$ y un cateto de longitud 2 sobre el eje $X$ (a la derecha del punto donde la tangente corta el eje).

\seccio{2}{Regla de L'Hôpital}

\ex{1} Calcula los límites siguientes:\begin{tasks}(1)
  \task \pau{Castilla-La Mancha, extraordinaria 2021} $\displaystyle\lim_{x\to1}\frac{x-1}{e^{x-1}-1}$
  \task \pau{Castilla-La Mancha, ordinaria 2021} $\displaystyle\lim_{x\to2}\frac{e^{x-2}-1}{2x-4}$
  \task \pau{Galicia, ordinaria 2022} $\displaystyle\lim_{x\to0}\frac{x\cos x}{\sin x}$
  \task \pau{Murcia, septiembre 2018} $\displaystyle\lim_{x\to0}\frac{\ln(\cos x+\sin x)}{x}$
  \task \pau{Cantabria, septiembre 2018} $\displaystyle\lim_{x\to0}\frac{\sin(2x^2)+x}{\ln(x+1)+x}$
  \task \pau{Castilla y León, ordinaria 2022} $\displaystyle\lim_{x\to0}\frac{x\sin x}{2-2\cos x}$
\end{tasks}\recorda{Si $\lim\frac fg$ es $\left[\frac00\right]$ o $\left[\frac\infty\infty\right]$, entonces $\lim\frac{f}{g}=\lim\frac{f'}{g'}$. Derivamos el numerador y el denominador \textbf{por separado}.}

\ex{2} Calcula estos límites del tipo $\left[\frac\infty\infty\right]$:\begin{tasks}(1)
  \task $\displaystyle\lim_{x\to+\infty}\frac{x^3}{e^{2x}}$
  \task $\displaystyle\lim_{x\to+\infty}\frac{(\ln x)^2}{x}$
  \task $\displaystyle\lim_{x\to0^+}\frac{\ln(\sin x)}{\ln x}$
  \task \pau{Castilla-La Mancha, ordinaria 2024} $\displaystyle\lim_{x\to+\infty}\frac{e^x-1}{x^2+3}$
\end{tasks}

\ex{2} \pau{Aragón, ordinaria 2025} Calcula $\displaystyle\lim_{x\to0}\frac{x^2-\sin(x^2)}{1-\cos x}$.

\ex{2} Límites con parámetros:\begin{tasks}(1)
  \task \pau{Aragón, extraordinaria 2023} Calcula $a$ para que $\displaystyle L=\lim_{x\to0}\frac{\ln(x+1)-a\sin x+3x\cos(2x)}{x^2}$ sea finito, y calcula $L$.
  \task \pau{Castilla y León, extraordinaria 2021} Calcula $m>0$ para que $\displaystyle\lim_{x\to0}\frac{1-\cos(mx)}{x^2}=2$.
  \task \pau{Galicia, ordinaria 2023} Si $f(x)=ae^x+b$, halla $a$ y $b$ para que $f(0)=0$ y $\displaystyle\lim_{x\to0}\frac{f(x)}{x}=3$.
\end{tasks}\recorda{Si el denominador tiende a 0 y el límite tiene que ser finito, el numerador también tiene que tender a 0: esto da una ecuación.}

\ex{3} \pau{Aragón, extraordinaria 2022} Calcula $a,b\in\mathbb R$ para que exista $\displaystyle\lim_{x\to1}\frac{2x^3+ax^2+bx+3}{-x^3+4x^2-5x+2}=L$ (finito) y calcula el límite.

\ex{2} ¿Dónde está el error? \ $\displaystyle\lim_{x\to1}\frac{x^2+1}{x+1}\overset{\text{L'H}}{=}\lim_{x\to1}\frac{2x}{1}=2$. \ ¿Y en este otro? \ $\displaystyle\lim_{x\to+\infty}\frac{x+\sin x}{x}\overset{\text{L'H}}{=}\lim_{x\to+\infty}(1+\cos x)$, que no existe, ``por tanto el límite no existe''.

  \task $f(x)=x^3-3x$ en $[-2,3]$
  \task $f(x)=x\,e^{-x}$ en $[0,3]$
  \task $f(x)=x-2\ln x$ en $[1,e]$
  \task $f(x)=\sin x+\cos x$ en $[0,2\pi]$
\end{tasks}\recorda{Candidatos: los extremos del intervalo, los puntos críticos interiores ($f'=0$) y los puntos no derivables. El valor más grande es el máximo absoluto y el más pequeño, el mínimo.}
