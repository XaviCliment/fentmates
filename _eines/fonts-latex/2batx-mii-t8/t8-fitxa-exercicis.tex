\seccio{1}{Domini d'una funció}

\ex{1} Calcula el domini de les funcions següents:\begin{tasks}(2)
  \task $f(x)=\dfrac{x+3}{x^2-5x+6}$
  \task $f(x)=\sqrt{9-x^2}$
  \task $f(x)=\sqrt{\dfrac{x-1}{x+2}}$
  \task $f(x)=\ln\left(4-x^2\right)$
  \task $f(x)=\dfrac{1}{\sqrt{x^2-2x-3}}$
  \task $f(x)=\sqrt[3]{\dfrac{x}{x^2-1}}$
  \task $f(x)=\dfrac{e^x}{e^{2x}-1}$
  \task $f(x)=\ln(\ln x)$
\end{tasks}\recorda{Denominadors $\neq0$; radicands d'índex parell $\geq0$ (i $>0$ si l'arrel és al denominador); arguments dels logaritmes $>0$. El domini és la intersecció de totes les condicions.}

\ex{2} Domini de:\begin{tasks}(2)
  \task $f(x)=\tan3x$
  \task $f(x)=\dfrac{1}{1-\cos x}$
  \task $f(x)=\arcsin\dfrac x2$
  \task $f(x)=\sqrt{\sin x}$, en $[0,2\pi]$
  \task $f(x)=\ln(\cos x)$, en $[0,2\pi]$
\end{tasks}

\ex{2} Domini de:\begin{tasks}(1)
  \task \pau{Aragó, ordinària 2023} $f(x)=\dfrac{2x-1}{\sqrt{x^2-x-2}}$
  \task \pau{Canàries, ordinària 2024} $f(x)=\dfrac{\ln(x+2)}{3x+4}$
\end{tasks}

\ex{3} Domini de:\begin{tasks}(1)
  \task $f(x)=\sqrt{\dfrac{x^2-4}{x^2-9}}$
  \task $f(x)=\log_2\left(x^2-x\right)-\sqrt{3-x}$
\end{tasks}

\seccio{2}{Talls amb els eixos i signe de la funció}

\ex{1} Troba els punts de tall amb els eixos i estudia el signe de:\begin{tasks}(2)
  \task $f(x)=x^3-x^2-6x$
  \task $f(x)=\dfrac{x^2-4x+3}{x+1}$
  \task $f(x)=\left(x^2-1\right)e^x$
  \task $f(x)=\ln\left(x^2-8\right)$
\end{tasks}\recorda{Els zeros de $f$ i els punts on no és contínua separen intervals de signe constant: n'hi ha prou amb un valor de prova en cada un.}

\ex{2} \pau{Extremadura, juny 2018} Demostra que $f(x)=x^3+x-3$ talla l'eix d'abscisses en un únic punt, que és positiu, i troba un interval de longitud $\frac12$ que el continga.

\ex{2} Estudia el signe de $f(x)=\dfrac{(x-2)^2(x+1)}{x^3}$. On talla o toca l'eix OX?

\ex{3} Quantes solucions té l'equació $x=2\sin x$? Localitza la positiva entre dos decimals consecutius.

\seccio{3}{Simetries i periodicitat}

\ex{1} Estudia la simetria de les funcions següents:\begin{tasks}(2)
  \task $f(x)=x^4-3x^2+1$
  \task $f(x)=\dfrac{x}{x^2-4}$
  \task $f(x)=e^{-x^2}\cos x$
  \task $f(x)=x^3+1$
  \task $f(x)=\ln\dfrac{1+x}{1-x}$
  \task $f(x)=\dfrac{\sin x\cos x}{x}$
\end{tasks}\recorda{Parella: $f(-x)=f(x)$ (simetria respecte de OY). Senar: $f(-x)=-f(x)$ (respecte de l'origen). El domini ha de ser simètric.}

\ex{1} Calcula el període de:\begin{tasks}(3)
  \task $\sin3x$
  \task $\cos\dfrac x2$
  \task $\sin x+\sin2x$
  \task $\sin^2(2x)$
  \task $\tan\dfrac{\pi x}{4}$
  \task $|\cos x|$
\end{tasks}

\ex{2} Comprova que la gràfica de $f(x)=\dfrac{x^2+1}{x-1}$ és simètrica respecte del punt en què es tallen les seues asímptotes, i que la de $g(x)=\dfrac{1}{x^2-6x+10}$ és simètrica respecte d'una recta vertical.

\ex{3} Demostra que $f(x)=\ln\left(x+\sqrt{x^2+1}\right)$ és senar.

\seccio{4}{Asímptotes verticals i horitzontals}

\ex{1} Calcula les asímptotes verticals i horitzontals (amb els límits laterals):\begin{tasks}(2)
  \task $f(x)=\dfrac{2x^2-3}{x^2-1}$
  \task $f(x)=\dfrac{x+2}{x^2-4}$
  \task $f(x)=\dfrac{3e^x}{e^x+1}$
  \task $f(x)=\dfrac{\sqrt{x^2+1}}{x}$
  \task $f(x)=\dfrac{\ln x}{x-1}$
\end{tasks}\recorda{AV $x=a$ si algun límit lateral en $a$ és infinit. AH $y=k$ si $\displaystyle\lim_{x\to\pm\infty}f(x)=k$: estudia cada costat per separat. Compte: $\sqrt{x^2}=|x|$.}

\ex{2} \pau{Andalusia, ordinària 2020} Estudia les asímptotes i la monotonia de $f(x)=\dfrac{x^2-2x-3}{x^2-1}$.

\ex{2} \pau{País Basc, juny 2018} Asímptotes, monotonia i extrems de $f(x)=\dfrac{x^2}{x^2-4}$.

\ex{2} \pau{Cantàbria, setembre 2019} Domini, asímptotes, monotonia i extrems de $f(x)=\dfrac{x+4}{x^2-7x-8}$.

\ex{2} \pau{País Basc, extraordinària 2025} Asímptotes i monotonia de $f(x)=\dfrac{x}{x^2-3x-4}$, i recta tangent en $x=0$.

\ex{3} \pau{La Rioja, juny 2018} Sigui $f(x)=x\,e^{-ax}$. Estudia'n les asímptotes segons els valors de $a$ i troba $a$ perquè tinga un extrem relatiu en $x=1$. És màxim o mínim?

\seccio{5}{Asímptotes obliqües i branques parabòliques}

\ex{1} Calcula totes les asímptotes de:\begin{tasks}(1)
  \task \pau{Andalusia, juny 2017} $f(x)=\dfrac{x^2}{x-1}$ (troba'n també els extrems).
  \task \pau{Astúries, extraordinària 2020} $f(x)=\dfrac{2x^3+1}{x^2}$ (i els extrems).
\end{tasks}\recorda{$m=\displaystyle\lim_{x\to\pm\infty}\frac{f(x)}{x}$ (finit i $\neq0$), $n=\displaystyle\lim_{x\to\pm\infty}\big(f(x)-mx\big)$. En les racionals, l'AO és el quocient de la divisió.}

\ex{2} \pau{Cantàbria, ordinària 2025} Sigui $f(x)=\dfrac{x^3-8}{x^2-4}$ si $x\neq2$ i $f(2)=e^k$.\begin{tasks}(1)
  \task Troba $k$ perquè sigui contínua en $x=2$.
  \task Calcula'n les asímptotes.
  \task Escriu la recta tangent en $x=1$.
\end{tasks}

\ex{2} \pau{Canàries, ordinària 2021} Sigui $f(x)=\dfrac{ax^2-2}{b-x}$.\begin{tasks}(1)
  \task Troba $a$ i $b$ perquè $f(-2)=2$ i $f$ siga contínua en $\mathbb R-\{5\}$.
  \task Per a $a=-1$, $b=-3$, calcula les asímptotes.
\end{tasks}

\ex{2} \pau{Castella i Lleó, juliol 2018} Troba $a$ i $b$ perquè $f(x)=\dfrac1x+ax+b$ tinga un mínim relatiu en $\left(\frac12,6\right)$ i, amb aquests valors, troba'n les asímptotes.

\ex{3} \pau{La Rioja, juliol 2018} Troba $a$ perquè $\displaystyle\lim_{x\to+\infty}\left(\sqrt{9x^2+ax+1}-(3x-1)\right)=2$. Quina és, aleshores, l'asímptota obliqua de $y=\sqrt{9x^2+ax+1}$ en $+\infty$?

\ex{2} Estudia les branques infinites (asímptotes o branques parabòliques) de:\begin{tasks}(2)
  \task $f(x)=\sqrt x$
  \task $f(x)=x+\sqrt x$
  \task $f(x)=x\ln x$
  \task $f(x)=\sqrt{x^2+x+1}-x$
\end{tasks}

\ex{2} Troba $b$ perquè $f(x)=\dfrac{2x^2+bx}{x+1}$ tinga l'asímptota obliqua $y=2x-1$.

\ex{2} Estudia la monotonia i la curvatura de $f(x)=x^2\ln x$ i de $g(x)=\dfrac{x-1}{x^2}$.

\ex{3} Troba $a$, $b$ i $c$ perquè $f(x)=x^3+ax^2+bx+c$ tinga un punt d'inflexió en $(1,2)$ amb tangent de pendent $-3$.

\seccio{11}{Funcions a trossos, amb valor absolut i transformacions}

\ex{1} Escriu a trossos i representa:\begin{tasks}(2)
  \task $f(x)=\left|x^2-2x-3\right|$
  \task $f(x)=x^2-4|x|+3$
  \task $f(x)=e^{-|x|}$
  \task $f(x)=|x|\,(x-2)$

\ex{2} \pau{Galícia, juny 2017} Calcula $a$ i $b$ perquè sigui derivable en $x=3$ la funció $f(x)=\begin{cases}ax^2+b&\text{si }x<3\\ \ln(x-2)&\text{si }x\geq3\end{cases}$ i representa-la.

\ex{3} \pau{Madrid, extraordinària 2022} Sigui $f(x)=\begin{cases}x&\text{si }x\leq0\\ x\ln x&\text{si }x>0\end{cases}$.\begin{tasks}(1)

\seccio{12}{Informació a partir de les gràfiques de f, f' i f''}

\ex{2} Sabem que $f'(x)=(x-1)^2(x+2)$. Sense calcular $f$, estudia'n la monotonia, els extrems i la curvatura, i dibuixa'n una gràfica possible.\recorda{$f'>0\Rightarrow f$ creix; canvi de signe de $f'$ $\Rightarrow$ extrem; $f'$ creix ($f''>0$) $\Rightarrow f$ convexa; extrem de $f'$ $\Rightarrow$ inflexió de $f$.}

\ex{2} \pau{Catalunya, extraordinària 2021} Calcula $a$ i $b$ perquè $g(x)=ax^3+bx^2+1$ tinga un punt d'inflexió en $x=\frac12$ i la seua derivada en aquest punt siga $-\frac32$.

\ex{2} La gràfica de la derivada d'un polinomi $p$ de grau 3 és una paràbola que talla l'eix OX en $x=-1$ i $x=2$. Si $p(0)=1$ i $p(2)=-19$, troba $p$ i digues on té els extrems i el punt d'inflexió.

\ex{2} \pau{Galícia, extraordinària 2026} Dibuixa la gràfica d'una funció $f:\mathbb R-\{2\}\to\mathbb R$ tal que $f$, $f'$ i $f''$ tinguen el mateix signe en $(-1,2)$ i $\displaystyle\lim_{x\to2^+}f(x)=-\infty$. Explica el dibuix.

\ex{3} \pau{Galícia, ordinària 2022} Dibuixa la gràfica d'una funció contínua i no negativa en $[0,3]$ tal que $f(0)=f(3)=0$, $f''>0$ en $(0,1)$, $f''<0$ en $(2,3)$ i constant en $(1,2)$. Pot ser derivable en $x=1$?
