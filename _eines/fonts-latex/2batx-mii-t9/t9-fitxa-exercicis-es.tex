\seccio{1}{Primitiva e integral indefinida}

\ex{1} Comprueba, derivando, que $F$ es una primitiva de $f$:\begin{tasks}(2)
  \task $f(x)=xe^x$, \ $F(x)=xe^x-e^x$
  \task $f(x)=\dfrac{1}{\sin x}$, \ $F(x)=\ln\left(\tan\dfrac x2\right)$
  \task $f(x)=\dfrac{2}{\left(1+x^2\right)^2}$, \ $F(x)=\arctan x+\dfrac{x}{1+x^2}$
  \task $f(x)=\dfrac{x}{\sqrt{x^2+1}}$, \ $F(x)=\sqrt{x^2+1}$
\end{tasks}\recorda{$\displaystyle\int f(x)\,dx=F(x)+C\iff F'(x)=f(x)$. Una primitiva siempre se puede comprobar derivándola.}

\ex{2} \pau{Asturias, extraordinaria 2025} $G(x)=\frac13x^3+ax^2+bx+5$ es una primitiva de una función $g$, donde $a$ y $b$ son constantes.\begin{tasks}(1)
  \task Determina $g(x)$ en función de $a$ y $b$.
  \task Escribe todas las primitivas de $g$.
\end{tasks}

\ex{2} Halla la función $f$ que cumple $f''(x)=12x^2-6$ y cuya recta tangente en $x=1$ es $y=2x-3$.

\ex{3} \pau{Aragón, ordinaria 2024} Calcula $a$, $b$ y $c$ para que $f(x)=ax+b\sin x\cos x+c$ sea una primitiva de $g(x)=\sin^2x$.

\seccio{2}{Integrales inmediatas}

\ex{1} Calcula:\begin{tasks}(2)
  \task $\displaystyle\int\left(4x^3-6x+2\right)dx$
  \task $\displaystyle\int\frac{x^2+1}{\sqrt x}\,dx$
  \task $\displaystyle\int\left(e^x-3^x\right)dx$
  \task $\displaystyle\int\left(\frac2x-\frac{5}{1+x^2}\right)dx$
  \task $\displaystyle\int(3\cos x-2\sin x)\,dx$
  \task $\displaystyle\int\left(1+\tan^2x\right)dx$
  \task $\displaystyle\int\frac{4}{\sqrt{1-x^2}}\,dx$
  \task $\displaystyle\int\frac{dx}{x^3}$
\end{tasks}\recorda{$\int x^n\,dx=\frac{x^{n+1}}{n+1}$ ($n\neq-1$), $\int\frac{dx}x=\ln|x|$, $\int a^x\,dx=\frac{a^x}{\ln a}$, $\int\frac{dx}{\cos^2x}=\tan x$, $\int\frac{dx}{1+x^2}=\arctan x$, $\int\frac{dx}{\sqrt{1-x^2}}=\arcsin x$.}

\ex{2} Transforma el integrando (operaciones o identidades trigonométricas) y calcula:\begin{tasks}(2)
  \task $\displaystyle\int\cos^2x\,dx$
  \task $\displaystyle\int\tan^2x\,dx$
  \task $\displaystyle\int(\sin x+\cos x)^2dx$
  \task $\displaystyle\int\frac{(x+1)^2}{x}\,dx$
  \task $\displaystyle\int\frac{2x-1}{\sqrt[3]{x}}\,dx$
\end{tasks}\recorda{$\cos^2x=\frac{1+\cos2x}{2}$, $\sin^2x=\frac{1-\cos2x}{2}$, $1+\tan^2x=\frac{1}{\cos^2x}$, $2\sin x\cos x=\sin2x$.}

\ex{1} Calcula:\begin{tasks}(2)

  \task $\displaystyle\int\frac{x^2}{x^3+1}\,dx$
  \task $\displaystyle\int e^{-x/2}\,dx$
  \task $\displaystyle\int\cos(3x+1)\,dx$
  \task $\displaystyle\int\frac{dx}{\cos^2(2x)}$
  \task $\displaystyle\int xe^{x^2}dx$
  \task $\displaystyle\int\frac{\sin x}{\cos^3x}\,dx$
  \task $\displaystyle\int\frac{dx}{9+x^2}$
  \task $\displaystyle\int\frac{x}{\sqrt{1-x^4}}\,dx$
  \task $\displaystyle\int\frac{\cos x}{\sin x}\,dx$
\end{tasks}\recorda{Si aparecen $u$ y $u'$ (o $u'$ multiplicada por una constante): $\int u'u^n=\frac{u^{n+1}}{n+1}$, $\int\frac{u'}u=\ln|u|$, $\int u'e^u=e^u$, $\int u'\cos u=\sin u$, $\int\frac{u'}{1+u^2}=\arctan u$…}

\ex{2} Calcula las integrales siguientes:\begin{tasks}(1)

\end{tasks}

\ex{2} \pau{Murcia, extraordinaria 2021}\begin{tasks}(1)

\ex{2} \pau{Murcia, extraordinaria 2023} Sea $f(x)=\dfrac{\sin x}{1+\cos^2x}$.\begin{tasks}(1)

\end{tasks}

\ex{3} Calcula:\begin{tasks}(1)

\seccio{4}{Integración por cambio de variable}

\ex{1} Calcula con el cambio indicado:\begin{tasks}(1)
  \task $\displaystyle\int\frac{x}{\sqrt{x+2}}\,dx$ \quad ($t=\sqrt{x+2}$)
  \task $\displaystyle\int\frac{dx}{x\ln x}$ \quad ($t=\ln x$)
  \task $\displaystyle\int\frac{dx}{\sqrt x+\sqrt[4]x}$ \quad ($x=t^4$)
\end{tasks}\recorda{Cambio $t=g(x)$, $dt=g'(x)\,dx$; todo tiene que quedar en $t$. Al final deshacemos el cambio (o, en una integral definida, cambiamos los límites).}

\ex{2} \pau{Asturias, ordinaria 2022} Sea $f(x)=x\sqrt{1-x^2}$.\begin{tasks}(1)
  \task Calcula una primitiva de $f$ que pase por el punto $(-1,0)$ (cambio $t=1-x^2$).
  \task Calcula $\displaystyle\int_0^1f(x)\,dx$.
\end{tasks}

\ex{2} \pau{Andalucía, extraordinaria 2022} Calcula $\displaystyle\int_3^8\frac{dx}{\sqrt{1+x}-1}$ (cambio $t=\sqrt{1+x}-1$).

\ex{2} Calcula las integrales siguientes con el cambio $t=e^x$:\begin{tasks}(1)
  \task \pau{Navarra, julio 2018} $\displaystyle\int\frac{e^x}{1+2e^x+e^{2x}}\,dx$
  \task \pau{Castilla-La Mancha, junio 2018} $\displaystyle\int\frac{e^x}{e^{2x}+e^x-2}\,dx$
\end{tasks}

\ex{2} \pau{Asturias, extraordinaria 2023} Calcula la primitiva de $f(x)=xe^{2x^2}$ que pasa por el punto $(0,-1)$ (puedes hacer el cambio $t=2x^2$).

\ex{3} Calcula $\displaystyle\int\sqrt{1-x^2}\,dx$ con el cambio $x=\sin t$ y deduce el área de un cuarto de círculo de radio 1.

\seccio{5}{Integración por partes}

\ex{1} Calcula por partes:\begin{tasks}(2)

\seccio{6}{Integración de funciones racionales}

\ex{1} Calcula (raíces reales simples):\begin{tasks}(1)
  \task $\displaystyle\int\frac{dx}{x^2-4}$
  \task $\displaystyle\int\frac{x+1}{x^2-3x+2}\,dx$
  \task \pau{Extremadura, ordinaria 2021} $\displaystyle\int\frac{3x}{x^2+x-2}\,dx$
  \task \pau{Navarra, julio 2017} $\displaystyle\int\frac{dx}{x^2-x-2}$
\end{tasks}\recorda{Si grado $P\geq$ grado $Q$, primero dividimos. Después factorizamos $Q$: $\frac{A}{x-a}$ para cada raíz simple, $\frac{A}{x-a}+\frac{B}{(x-a)^2}$ para una doble y $\frac{Mx+N}{x^2+px+q}$ para un factor sin raíces.}

\ex{2} \pau{Extremadura, extraordinaria 2024} Calcula $\displaystyle\int\frac{-x^2+7x+6}{x^3+x^2-2x}\,dx$.

\ex{2} Calcula (grado del numerador mayor o igual):\begin{tasks}(1)

  \task \pau{Islas Baleares, extraordinaria 2023} $\displaystyle\int\frac{x^4+2x-6}{x^2+x-2}\,dx$
\end{tasks}

\ex{2} \pau{País Vasco, junio 2018} Calcula $\displaystyle\int\frac{2x-1}{x(x+1)^2}\,dx$ (raíz doble).

\ex{3} Calcula $\displaystyle\int\frac{x+3}{x^3+4x}\,dx$ (factor de segundo grado sin raíces).

\ex{2} \pau{Comunidad Valenciana, extraordinaria 2023} Sea $f(x)=\dfrac{-2x^2+x+1}{2x^2+5x+2}$.\begin{tasks}(1)
  \task Comprueba que $x=-\frac12$ es una discontinuidad evitable.

\ex{2} \pau{Comunidad Valenciana, ordinaria 2020} Sea $f(x)=\dfrac{2x^2+1}{x(x-1)}$. Calcula $\displaystyle\int f(x)\,dx$ y la primitiva de $f$ cuya gráfica pasa por el punto $(2,0)$.

\ex{3} \pau{Comunidad Valenciana, julio 2019} Sea $h(x)=\dfrac{x^3+x^2+5x-3}{x^2+2x+5}$. Calcula la asíntota de la curva $y=h(x)$ y la primitiva $\int h(x)\,dx$.

\seccio{7}{La integral definida y la regla de Barrow}

\ex{1} Calcula:\begin{tasks}(2)
  \task $\displaystyle\int_0^1\left(x^3-2x\right)dx$
  \task $\displaystyle\int_1^e\left(\frac1x+x\right)dx$
  \task $\displaystyle\int_0^{\pi/2}\cos x\,dx$
  \task $\displaystyle\int_0^{\pi/4}\frac{dx}{\cos^2x}$
  \task $\displaystyle\int_0^1\frac{x}{x^2+1}\,dx$
  \task $\displaystyle\int_{-1}^1e^{2x}dx$
\end{tasks}\recorda{Regla de Barrow: $\int_a^bf(x)\,dx=F(b)-F(a)$, con $F$ una primitiva cualquiera de $f$ (continua en $[a,b]$).}

\ex{2} Calcula:\begin{tasks}(1)
  \task \pau{Galicia, ordinaria 2020} $\displaystyle\int_1^2x(\ln x-1)\,dx$
  \task \pau{Canarias, extraordinaria 2020} $\displaystyle\int_0^{\pi/2}x\cos x\,dx$

  \task \pau{Cantabria, junio 2019} $\displaystyle\int_0^5(x+10)e^{2x}dx$
\end{tasks}

\ex{3} \pau{Andalucía, septiembre 2018} Sea $f(x)=ax\ln x-bx$ ($x>0$). Determina $a$ y $b$ sabiendo que $f$ tiene un extremo relativo en $x=1$ y que $\displaystyle\int_1^2f(x)\,dx=8\ln2-9$.

\ex{2} \pau{Murcia, extraordinaria 2024} Sea $f(x)=\dfrac{\ln x}{\sqrt x}$. Calcula $\displaystyle\int f(x)\,dx$ y el valor de $a>0$ para el que $\displaystyle\int_1^af(x)\,dx=4$.

\ex{2} Halla el valor medio de $f(x)=x^2$ en $[0,3]$ y el punto $c$ que da el teorema del valor medio del cálculo integral.

\seccio{8}{Teorema fundamental del cálculo}

\ex{2} Calcula la derivada de:\begin{tasks}(2)
  \task $F(x)=\displaystyle\int_0^x\frac{dt}{1+t^4}$
  \task $G(x)=\displaystyle\int_1^{x^3}\ln t\,dt$ \ ($x>0$)
  \task $H(x)=\displaystyle\int_0^{x^2}e^{-t}\,dt$
  \task $K(x)=\displaystyle\int_{x}^{1}\cos\left(t^2\right)dt$
\end{tasks}\recorda{$\dfrac{d}{dx}\displaystyle\int_a^{x}f(t)\,dt=f(x)$ y, con la regla de la cadena, $\dfrac{d}{dx}\displaystyle\int_a^{g(x)}f(t)\,dt=f\big(g(x)\big)\,g'(x)$.}

\ex{2} Halla los extremos relativos de $F(x)=\displaystyle\int_0^x\left(t^2-4\right)e^{t}\,dt$.

\ex{2} \pau{Aragón, extraordinaria 2026} No toda función tiene una primitiva expresable con funciones elementales; por ejemplo $f(x)=\sin\left(x^2\right)$. Sea $F$ la primitiva de $f$ con $F(0)=0$.\begin{tasks}(1)
  \task ¿Cuál es la derivada de $F$?
  \task Calcula $\displaystyle\lim_{x\to0}\frac{F(x)}{x^3}$ con la regla de L'Hôpital.
\end{tasks}

\ex{3} Calcula $\displaystyle\lim_{x\to0}\frac{\int_0^xe^{t^2}dt-x}{x^3}$.

\ex{3} Halla una función continua $f$ tal que $\displaystyle\int_0^xf(t)\,dt=x^2e^x$ para todo $x$.
