\seccio{1}{Primitiva i integral indefinida}

\ex{1} Comprova, derivant, que $F$ és una primitiva de $f$:\begin{tasks}(2)
  \task $f(x)=xe^x$, \ $F(x)=xe^x-e^x$
  \task $f(x)=\dfrac{1}{\sin x}$, \ $F(x)=\ln\left(\tan\dfrac x2\right)$
  \task $f(x)=\dfrac{2}{\left(1+x^2\right)^2}$, \ $F(x)=\arctan x+\dfrac{x}{1+x^2}$
  \task $f(x)=\dfrac{x}{\sqrt{x^2+1}}$, \ $F(x)=\sqrt{x^2+1}$
\end{tasks}\recorda{$\displaystyle\int f(x)\,dx=F(x)+C\iff F'(x)=f(x)$. Una primitiva sempre es pot comprovar derivant-la.}

\ex{2} \pau{Astúries, extraordinària 2025} $G(x)=\frac13x^3+ax^2+bx+5$ és una primitiva d'una funció $g$, on $a$ i $b$ són constants.\begin{tasks}(1)
  \task Determina $g(x)$ en funció de $a$ i $b$.
  \task Escriu totes les primitives de $g$.
\end{tasks}

\ex{2} Troba la funció $f$ que compleix $f''(x)=12x^2-6$ i la recta tangent de la qual en $x=1$ és $y=2x-3$.

\ex{3} \pau{Aragó, ordinària 2024} Calcula $a$, $b$ i $c$ perquè $f(x)=ax+b\sin x\cos x+c$ siga una primitiva de $g(x)=\sin^2x$.

\seccio{2}{Integrals immediates}

\ex{1} Calcula:\begin{tasks}(2)
  \task $\displaystyle\int\left(4x^3-6x+2\right)dx$
  \task $\displaystyle\int\frac{x^2+1}{\sqrt x}\,dx$
  \task $\displaystyle\int\left(e^x-3^x\right)dx$
  \task $\displaystyle\int\left(\frac2x-\frac{5}{1+x^2}\right)dx$
  \task $\displaystyle\int(3\cos x-2\sin x)\,dx$
  \task $\displaystyle\int\left(1+\tan^2x\right)dx$
  \task $\displaystyle\int\frac{4}{\sqrt{1-x^2}}\,dx$
  \task $\displaystyle\int\frac{dx}{x^3}$
\end{tasks}\recorda{$\int x^n\,dx=\frac{x^{n+1}}{n+1}$ ($n\neq-1$), $\int\frac{dx}x=\ln|x|$, $\int a^x\,dx=\frac{a^x}{\ln a}$, $\int\frac{dx}{\cos^2x}=\tan x$, $\int\frac{dx}{1+x^2}=\arctan x$, $\int\frac{dx}{\sqrt{1-x^2}}=\arcsin x$.}

\ex{2} Transforma l'integrand (operacions o identitats trigonomètriques) i calcula:\begin{tasks}(2)
  \task $\displaystyle\int\cos^2x\,dx$
  \task $\displaystyle\int\tan^2x\,dx$
  \task $\displaystyle\int(\sin x+\cos x)^2dx$
  \task $\displaystyle\int\frac{(x+1)^2}{x}\,dx$
  \task $\displaystyle\int\frac{2x-1}{\sqrt[3]{x}}\,dx$
\end{tasks}\recorda{$\cos^2x=\frac{1+\cos2x}{2}$, $\sin^2x=\frac{1-\cos2x}{2}$, $1+\tan^2x=\frac{1}{\cos^2x}$, $2\sin x\cos x=\sin2x$.}

\ex{1} Calcula:\begin{tasks}(2)
  \task $\displaystyle\int(3x-2)^5dx$
  \task $\displaystyle\int\frac{x^2}{x^3+1}\,dx$
  \task $\displaystyle\int e^{-x/2}\,dx$
  \task $\displaystyle\int\cos(3x+1)\,dx$
  \task $\displaystyle\int\frac{dx}{\cos^2(2x)}$
  \task $\displaystyle\int xe^{x^2}dx$
  \task $\displaystyle\int\frac{\sin x}{\cos^3x}\,dx$
  \task $\displaystyle\int\frac{dx}{9+x^2}$
  \task $\displaystyle\int\frac{x}{\sqrt{1-x^4}}\,dx$
  \task $\displaystyle\int\frac{\cos x}{\sin x}\,dx$
\end{tasks}\recorda{Si apareixen $u$ i $u'$ (o $u'$ multiplicada per una constant): $\int u'u^n=\frac{u^{n+1}}{n+1}$, $\int\frac{u'}u=\ln|u|$, $\int u'e^u=e^u$, $\int u'\cos u=\sin u$, $\int\frac{u'}{1+u^2}=\arctan u$…}

\ex{2} Calcula les integrals següents:\begin{tasks}(1)

\end{tasks}

\ex{2} \pau{Múrcia, extraordinària 2021}\begin{tasks}(1)

\ex{2} \pau{Múrcia, extraordinària 2023} Sigui $f(x)=\dfrac{\sin x}{1+\cos^2x}$.\begin{tasks}(1)

\seccio{4}{Integració per canvi de variable}

\ex{1} Calcula amb el canvi indicat:\begin{tasks}(1)
  \task $\displaystyle\int\frac{x}{\sqrt{x+2}}\,dx$ \quad ($t=\sqrt{x+2}$)
  \task $\displaystyle\int\frac{dx}{x\ln x}$ \quad ($t=\ln x$)
  \task $\displaystyle\int\frac{dx}{\sqrt x+\sqrt[4]x}$ \quad ($x=t^4$)
\end{tasks}\recorda{Canvi $t=g(x)$, $dt=g'(x)\,dx$; tot ha de quedar en $t$. Al final desfem el canvi (o, en una integral definida, canviem els límits).}

\ex{2} \pau{Astúries, ordinària 2022} Sigui $f(x)=x\sqrt{1-x^2}$.\begin{tasks}(1)
  \task Calcula una primitiva de $f$ que passe pel punt $(-1,0)$ (canvi $t=1-x^2$).
  \task Calcula $\displaystyle\int_0^1f(x)\,dx$.
\end{tasks}

\ex{2} \pau{Andalusia, extraordinària 2022} Calcula $\displaystyle\int_3^8\frac{dx}{\sqrt{1+x}-1}$ (canvi $t=\sqrt{1+x}-1$).

\ex{2} Calcula les integrals següents amb el canvi $t=e^x$:\begin{tasks}(1)
  \task \pau{Navarra, juliol 2018} $\displaystyle\int\frac{e^x}{1+2e^x+e^{2x}}\,dx$
  \task \pau{Castella-la Manxa, juny 2018} $\displaystyle\int\frac{e^x}{e^{2x}+e^x-2}\,dx$
\end{tasks}

\ex{2} \pau{Astúries, extraordinària 2023} Calcula la primitiva de $f(x)=xe^{2x^2}$ que passa pel punt $(0,-1)$ (pots fer el canvi $t=2x^2$).

\ex{3} Calcula $\displaystyle\int\sqrt{1-x^2}\,dx$ amb el canvi $x=\sin t$ i dedueix l'àrea d'un quart de cercle de radi 1.

\seccio{5}{Integració per parts}

\ex{1} Calcula per parts:\begin{tasks}(2)
  \task $\displaystyle\int x\sin x\,dx$
  \task $\displaystyle\int xe^{-x}\,dx$
  \task $\displaystyle\int x^3\ln x\,dx$
  \task $\displaystyle\int\ln(x+1)\,dx$
\end{tasks}\recorda{$\int u\,dv=uv-\int v\,du$. Tria com a $u$ la funció que se simplifica en derivar-la (logaritmes, arctan, polinomis) i com a $dv$ la que sabem integrar (exponencials, sinus, cosinus).}

\ex{2} Calcula:\begin{tasks}(1)

\seccio{6}{Integració de funcions racionals}

\ex{1} Calcula (arrels reals simples):\begin{tasks}(1)
  \task $\displaystyle\int\frac{dx}{x^2-4}$
  \task $\displaystyle\int\frac{x+1}{x^2-3x+2}\,dx$
  \task \pau{Extremadura, ordinària 2021} $\displaystyle\int\frac{3x}{x^2+x-2}\,dx$
  \task \pau{Navarra, juliol 2017} $\displaystyle\int\frac{dx}{x^2-x-2}$
\end{tasks}\recorda{Si grau $P\geq$ grau $Q$, primer dividim. Després factoritzem $Q$: $\frac{A}{x-a}$ per a cada arrel simple, $\frac{A}{x-a}+\frac{B}{(x-a)^2}$ per a una de doble i $\frac{Mx+N}{x^2+px+q}$ per a un factor sense arrels.}

\ex{2} \pau{Extremadura, extraordinària 2024} Calcula $\displaystyle\int\frac{-x^2+7x+6}{x^3+x^2-2x}\,dx$.

\ex{2} Calcula (grau del numerador més gran o igual):\begin{tasks}(1)

\end{tasks}

\ex{2} \pau{País Basc, juny 2018} Calcula $\displaystyle\int\frac{2x-1}{x(x+1)^2}\,dx$ (arrel doble).

\ex{3} Calcula $\displaystyle\int\frac{x+3}{x^3+4x}\,dx$ (factor de segon grau sense arrels).

\ex{2} \pau{València, extraordinària 2023} Sigui $f(x)=\dfrac{-2x^2+x+1}{2x^2+5x+2}$.\begin{tasks}(1)

\ex{3} \pau{València, juliol 2019} Sigui $h(x)=\dfrac{x^3+x^2+5x-3}{x^2+2x+5}$. Calcula l'asímptota de la corba $y=h(x)$ i la primitiva $\int h(x)\,dx$.

\seccio{7}{La integral definida i la regla de Barrow}

\ex{1} Calcula:\begin{tasks}(2)
  \task $\displaystyle\int_0^1\left(x^3-2x\right)dx$
  \task $\displaystyle\int_1^e\left(\frac1x+x\right)dx$
  \task $\displaystyle\int_0^{\pi/2}\cos x\,dx$
  \task $\displaystyle\int_0^{\pi/4}\frac{dx}{\cos^2x}$
  \task $\displaystyle\int_0^1\frac{x}{x^2+1}\,dx$
  \task $\displaystyle\int_{-1}^1e^{2x}dx$
\end{tasks}\recorda{Regla de Barrow: $\int_a^bf(x)\,dx=F(b)-F(a)$, amb $F$ una primitiva qualsevol de $f$ (contínua en $[a,b]$).}

\ex{2} Calcula:\begin{tasks}(1)
  \task \pau{Galícia, ordinària 2020} $\displaystyle\int_1^2x(\ln x-1)\,dx$
  \task \pau{Canàries, extraordinària 2020} $\displaystyle\int_0^{\pi/2}x\cos x\,dx$
  \task \pau{Castella-la Manxa, juliol 2019} $\displaystyle\int_0^1(x+1)e^{-x}dx$
  \task \pau{Cantàbria, juny 2019} $\displaystyle\int_0^5(x+10)e^{2x}dx$
\end{tasks}

\ex{3} \pau{Andalusia, setembre 2018} Sigui $f(x)=ax\ln x-bx$ ($x>0$). Determina $a$ i $b$ sabent que $f$ té un extrem relatiu en $x=1$ i que $\displaystyle\int_1^2f(x)\,dx=8\ln2-9$.

\ex{2} \pau{Múrcia, extraordinària 2024} Sigui $f(x)=\dfrac{\ln x}{\sqrt x}$. Calcula $\displaystyle\int f(x)\,dx$ i el valor de $a>0$ per al qual $\displaystyle\int_1^af(x)\,dx=4$.

\seccio{8}{Teorema fonamental del càlcul}

\ex{2} Calcula la derivada de:\begin{tasks}(2)
  \task $F(x)=\displaystyle\int_0^x\frac{dt}{1+t^4}$
  \task $G(x)=\displaystyle\int_1^{x^3}\ln t\,dt$ \ ($x>0$)
  \task $H(x)=\displaystyle\int_0^{x^2}e^{-t}\,dt$
  \task $K(x)=\displaystyle\int_{x}^{1}\cos\left(t^2\right)dt$
\end{tasks}\recorda{$\dfrac{d}{dx}\displaystyle\int_a^{x}f(t)\,dt=f(x)$ i, amb la regla de la cadena, $\dfrac{d}{dx}\displaystyle\int_a^{g(x)}f(t)\,dt=f\big(g(x)\big)\,g'(x)$.}

\end{tasks}

\ex{3} Calcula $\displaystyle\lim_{x\to0}\frac{\int_0^xe^{t^2}dt-x}{x^3}$.

\ex{3} Troba una funció contínua $f$ tal que $\displaystyle\int_0^xf(t)\,dt=x^2e^x$ per a tot $x$.
