\seccio{1}{Concepte de funció}

\ex{1} En cada situació, digues quina és la variable independent ($x$) i quina la dependent ($y$):\begin{tasks}(2)
  \task El preu d'una pizza i el nombre d'ingredients.
  \task L'àrea d'un cercle i el seu radi.
  \task Els minuts parlats per telèfon i el preu.
  \task La temperatura d'un forn i el temps que fa que està encés.
\end{tasks}\recorda{La variable \textbf{independent} $x$ és la que triem o la que passa sola; la \textbf{dependent} $y$ és la que depén de l'altra. Una relació és una \textbf{funció} si a cada valor de $x$ li correspon \textbf{un únic} valor de $y$.}

  \task A cada quantitat de litres de gasolina, el preu que paguem.
  \task A cada edat, les persones de la classe que la tenen.
\end{tasks}
