\seccio{1}{Raons i proporcions}

\ex{1} $\frac{360}{24}=\res{15}$: hi ha 15 alumnes per cada professor.

\ex{1} \begin{tasks}(2)
  \task $3\cdot20=60=4\cdot15$: \rest{sí}
  \task $5\cdot15=75\neq6\cdot12=72$: \rest{no}
  \task $2{,}5\cdot8=20=2\cdot10$: \rest{sí}
  \task $7\cdot28=196\neq9\cdot21=189$: \rest{no}
\end{tasks}

\ex{2} \begin{tasks}(4)
  \task $x=\frac{12\cdot15}{20}=\res{9}$
  \task $x=\frac{8\cdot9}{6}=\res{12}$
  \task $x=\frac{4{,}5\cdot8}{6}=\res{6}$
  \task $x=\frac{35\cdot6}{14}=\res{15}$
\end{tasks}

\ex{2} Xiques: $84-35=49$. Xics/xiques: $\frac{35}{49}=\res{\frac{5}{7}}$; xiques/total: $\frac{49}{84}=\res{\frac{7}{12}}$.

\ex{3} $\frac{5}{2}=\frac{750}{x}\Rightarrow x=\frac{2\cdot750}{5}=\res{300\ \text{mL}}$ de maduixes; en total, $\res{1050\ \text{mL}}$ de batut.

\ex{3} Per cada 3 anys de la menuda, la gran en té 4: $3+4=7$ parts; cada part són $28:7=4$ anys. \rest{12 i 16 anys} (i $\frac{12}{16}=\frac{3}{4}$ \ding{51}).

  \task \rest{Sí} ($P=4\cdot c$).

\ex{1} \begin{tabular}{l|ccccc}Hores & 2 & 5 & 8 & \rest{12} & 20\\ \hline Km & 7 & \rest{17{,}5} & \rest{28} & 42 & \rest{70}\end{tabular}\quad Constant: $k=\frac{7}{2}=\res{3{,}5}$ km/h.

\ex{1} Una entrada: $30:4=7{,}50$ €. Set: $7{,}50\cdot7=\res{52{,}50\ \text{€}}$.

\ex{2} \begin{tasks}(2)
  \task $\begin{array}{rcl}100\ \text{km}&\longrightarrow&6{,}4\ \text{L}\\ 450\ \text{km}&\longrightarrow&x\end{array}$\quad $x=\frac{6{,}4\cdot450}{100}=\res{28{,}8\ \text{L}}$
  \task $\begin{array}{rcl}6{,}4\ \text{L}&\longrightarrow&100\ \text{km}\\ 40\ \text{L}&\longrightarrow&x\end{array}$\quad $x=\frac{100\cdot40}{6{,}4}=\res{625\ \text{km}}$
\end{tasks}

\ex{2} En 1 h: $1500:2{,}5=600$ botelles. En 8 h: $600\cdot8=\res{4800}$. Per a 2100: $2100:600=\res{3{,}5\ \text{h}}$ (3 h 30 min).

\ex{2} Per persona: 80 g d'arròs i 0,3 L de brou. Per a 10: \rest{800 g d'arròs i 3 L de brou}.

\ex{3} $1{,}6\cdot250=400$ cm $=4$ m; $\ 1{,}4\cdot250=350$ cm $=3{,}5$ m. Superfície: $4\cdot3{,}5=\res{14\ \text{m}^2}$.

\ex{3} a) 1a: 4, 8, 12, 24 €; \ 2a: 8, 11, 14, 23 €. b) \rest{Només en la primera} (quocient constant 4); en la segona $\frac{8}{1}\neq\frac{11}{2}$. c) Igualen a les 5 h (20 €); \rest{a partir de 6 hores} és més barata la segona.

\ex{1} \begin{tabular}{l|ccccc}Aixetes & 2 & 3 & 4 & \rest{6} & 12\\ \hline Hores & 18 & \rest{12} & \rest{9} & 6 & \rest{3}\end{tabular}\quad Constant: $k=2\cdot18=\res{36}$.

\ex{1} Una sola màquina: $12\cdot3=36$ h. Quatre: $36:4=\res{9\ \text{h}}$.

\ex{2} $\begin{array}{rcl}120\ \text{km/h}&\longrightarrow&3{,}5\ \text{h}\\ 140\ \text{km/h}&\longrightarrow&x\end{array}$\quad Inversa: $x=\frac{120\cdot3{,}5}{140}=\res{3\ \text{h}}$.

\ex{2} Constant: $40\cdot30=1200$. Amb 30 vaques: $1200:30=\res{40\ \text{dies}}$. Amb 48: $1200:48=\res{25\ \text{dies}}$.

\ex{2} El dipòsit té $45\cdot12=540$ L. $540:18=\res{30\ \text{cubs}}$; $\ 540:60=\res{9\ \text{L}}$.

\ex{3} Falten 12 dies de faena per a 10 obrers. Amb 8 obrers (inversa): $x=\frac{10\cdot12}{8}=15$ dies més. En total: $6+15=\res{21\ \text{dies}}$.

\ex{3} \begin{tasks}(1)
  \task Inversa: $\frac{30\cdot24}{20}=\res{36\ \text{files}}$.

  \task Inversa: $\frac{6\cdot85}{10}=\res{51\ \text{€}}$.
\end{tasks}

\seccio{4}{Proporcionalitat composta}

\ex{1} D i D: $x=24\cdot\frac{9}{6}\cdot\frac{15}{10}=\res{54\ \text{kg}}$. (Un gos menja $24:6:10=0{,}4$ kg al dia.)

\ex{1} Una aixeta en una hora: $6000:4:5=300$ L. Llavors: $300\cdot6\cdot3=\res{5400\ \text{L}}$.

\ex{2} $\begin{array}{ccc}\text{Obrers} & \text{Hores/dia} & \text{Dies}\\ \hline 12 & 8 & 20\\ 16 & 6 & x\end{array}$\\[3pt]Si només canvien els obrers: $20\cdot\frac{12}{15}=\res{16\ \text{dies}}$. Amb 16 obrers i 6 h (I i I): $20\cdot\frac{12}{16}\cdot\frac{8}{6}=\res{20\ \text{dies}}$.

\ex{2} D i D: $540\cdot\frac{6}{4}\cdot\frac{20}{30}=\res{540\ \text{€}}$. (Casualment, el mateix!)

\ex{2} Més tones → més viatges (D); més camions → menys viatges cadascun (I): $8\cdot\frac{540}{360}\cdot\frac{5}{6}=\res{10\ \text{viatges}}$.

\ex{3} $\begin{array}{cccc}\text{Màquines} & \text{Hores/dia} & \text{Peces} & \text{Dies}\\ \hline 8 & 9 & 3600 & 5\\ 6 & 10 & 4800 & x\end{array}$\\[3pt]I, I i D: $x=5\cdot\frac{8}{6}\cdot\frac{9}{10}\cdot\frac{4800}{3600}=\res{8\ \text{dies}}$. (Una màquina fa $3600:8:9:5=10$ peces per hora.)

\ex{3} Més hectàrees → més tractors (D); menys dies → més tractors (I): $3\cdot\frac{30}{18}\cdot\frac{6}{5}=\res{6\ \text{tractors}}$.
