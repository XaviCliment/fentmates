\seccio{1}{Figures semblants}

\ex{1} \begin{tasks}(2)
  \task \rest{Sí}: angles rectes i costats proporcionals.
  \task \rest{No}: poden ser més o menys allargats.
  \task \rest{Sí}: tots els angles fan 60°.
  \task \rest{No}: els angles són diferents.
\end{tasks}

\ex{1} \begin{tasks}(1)
  \task $k=\frac{10}{4}=\frac{15}{6}=\res{2{,}5}$: \rest{ampliació}
  \task $k=\frac{1{,}5}{3}=\frac24=\frac{2{,}5}{5}=\res{0{,}5}$: \rest{reducció}
\end{tasks}

\ex{1} a) $\res{15,\ 21\text{ i }27\text{ cm}}$. \quad b) $5\cdot0{,}4=2$, \ $7\cdot0{,}4=2{,}8$, \ $9\cdot0{,}4=3{,}6$: \ $\res{2,\ 2{,}8\text{ i }3{,}6\text{ cm}}$.

\ex{1} $k=\frac46=\res{\frac23}$; \quad $9\cdot\frac23=\res{6\text{ cm}}$.

\ex{2} $k=\frac64=\frac{4{,}5}{3}=\res{1{,}5}$; \quad $x=1\cdot1{,}5=\res{1{,}5}$; \quad $y=2\cdot1{,}5=\res{3}$.

\ex{2} \begin{tasks}(2)
  \task $\frac64=\frac{15}{10}=1{,}5$: \rest{sí}, $k=1{,}5$
  \task $\frac{10}{5}=2\neq\frac{18}{8}=2{,}25$: \rest{no}
\end{tasks}

\ex{2} $P'=24\cdot2{,}5=\res{60\text{ cm}}$; \quad $A'=32\cdot2{,}5^2=32\cdot6{,}25=\res{200\text{ cm}^2}$.

\ex{2} Costat $6\cdot\frac13=\res{2\text{ cm}}$; \ àrea $2^2=\res{4\text{ cm}^2}$ (abans, 36 cm²). L'àrea és $\res{9}$ vegades més menuda ($3^2=9$).

\ex{2} $P=6+8+10=24$; \ $k=\frac{72}{24}=3$: \ $\res{18,\ 24\text{ i }30\text{ cm}}$.

\ex{3} Costats $\sqrt9=\res{3\text{ cm}}$ i $\sqrt{81}=\res{9\text{ cm}}$; \ $k=\frac93=\res{3}$ (l'àrea es multiplica per $3^2=9$).

\ex{3} $A=150$ cm²; \ $k^2=\frac{600}{150}=4\Rightarrow k=\res{2}$. \ Foto de $\res{20\text{ cm}\times30\text{ cm}}$.

\ex{3} $k=\frac{15}{5}=3$: catets $\res{9\text{ cm}}$ i $\res{12\text{ cm}}$. \ Àrees: $\frac{3\cdot4}{2}=6$ cm² i $\frac{9\cdot12}{2}=54$ cm²: l'àrea és $\res{9}$ vegades més gran ($3^2$).

\seccio{2}{Plànols, mapes i maquetes}

\ex{1} \begin{tasks}(2)
  \task $350\,000\text{ cm}=\res{3{,}5\text{ km}}$
  \task $400\text{ cm}=\res{4\text{ m}}$
  \task $500\,000\text{ cm}=\res{5\text{ km}}$
  \task $3000\text{ cm}=\res{30\text{ m}}$
\end{tasks}

\ex{1} \begin{tasks}(1)
  \task $800:200=\res{4\text{ cm}}$
  \task $1\,500\,000:500\,000=\res{3\text{ cm}}$
  \task $12\,000:2000=\res{6\text{ cm}}$
\end{tasks}

\ex{1} \begin{tasks}(1)
  \task $150\,000:3=50\,000$: \ $\res{1:50\,000}$
  \task $1000:5=200$: \ $\res{1:200}$
  \task $4\,000\,000:2=2\,000\,000$: \ $\res{1:2\,000\,000}$
\end{tasks}

\ex{2} 1 cm són 500 m $=50\,000$ cm: \ $\res{1:50\,000}$. \ $6{,}5\cdot500=3250$ m $=\res{3{,}25\text{ km}}$.

\ex{2} $8\cdot50=400$ cm $=4$ m; \ $6{,}4\cdot50=320$ cm $=3{,}2$ m. \ $\res{4\text{ m}\times3{,}2\text{ m}}$, \ superfície $\res{12{,}8\text{ m}^2}$.

\ex{2} $18\cdot25\,000=450\,000$ cm $=\res{4{,}5\text{ km}}$. \ $4{,}5:5=0{,}9$ h $=\res{54\text{ minuts}}$.

\ex{2} $3600:72=\res{50\text{ cm}}$ de llarg; \ $2880:72=\res{40\text{ cm}}$ d'ala a ala.

\ex{2} $9\cdot500=4500$ cm $=\res{45\text{ m}}$.

\ex{3} $180\text{ km}=18\,000\,000$ cm; \ $18\,000\,000:6=3\,000\,000$: \ $\res{1:3\,000\,000}$. \ $2{,}5\cdot30=\res{75\text{ km}}$ (1 cm són 30 km).

\ex{3} 1 cm del plànol és 1 m real: $25\text{ m}\times16\text{ m}=\res{400\text{ m}^2}$. \ $400\cdot150=\res{60\,000\text{ €}}$.

\ex{3} $8:20=0{,}4$ cm $=\res{4\text{ mm}}$.

\seccio{3}{El teorema de Tales}

\ex{1} \begin{tasks}(4)
  \task $4x=36\Rightarrow\res{x=9}$
  \task $10x=70\Rightarrow\res{x=7}$
  \task $12x=48\Rightarrow\res{x=4}$
  \task $2{,}5x=15\Rightarrow\res{x=6}$
\end{tasks}

\ex{1} $\frac25=\frac3x\Rightarrow2x=15\Rightarrow\res{x=7{,}5\text{ cm}}$.

\ex{1} $\frac64=\frac{4{,}5}{x}\Rightarrow6x=18\Rightarrow\res{x=3}$.

\ex{2} $\frac x8=\frac36\Rightarrow6x=24\Rightarrow\res{x=4}$.

\ex{2} $\frac{3{,}6}{4{,}8}=\frac x6\Rightarrow4{,}8x=21{,}6\Rightarrow\res{\overline{A'B'}=4{,}5}$; \quad $\overline{A'C'}=4{,}5+6=\res{10{,}5}$.

\ex{2} \begin{tasks}(2)
  \task $3\cdot6=4{,}5\cdot4=18$: \rest{sí}
  \task $2\cdot7=14\neq5\cdot3=15$: \rest{no}
\end{tasks}

\ex{2} Cada segment de la segona recta és $\frac32=1{,}5$ vegades el de la primera: \ $\overline{B'C'}=3\cdot1{,}5=\res{4{,}5}$, \ $\overline{C'D'}=4\cdot1{,}5=\res{6}$.

\ex{2} a) $21:3=\res{7\text{ cm}}$ cada part. \quad b) $\frac{21}{1+2+4}=3$: \ $\res{3,\ 6\text{ i }12\text{ cm}}$.

\ex{3} $\frac{84}{18+24+30}=\frac{84}{72}=\frac76$: \ $18\cdot\frac76=\res{21\text{ m}}$, \ $24\cdot\frac76=\res{28\text{ m}}$, \ $30\cdot\frac76=\res{35\text{ m}}$.

\ex{3} $\frac{x}{x+2}=\frac69\Rightarrow9x=6x+12\Rightarrow3x=12\Rightarrow\res{x=4}$.

\ex{3} Des d'$A$ es traça una semirecta auxiliar i s'hi marquen 3 segments iguals. S'uneix l'última marca amb $B$ i, per les altres dues, es traça paral·leles a aquesta recta: tallen $AB$ en 3 parts iguals (teorema de Tales). Cada part fa $\frac{10}{3}\approx\res{3{,}33\text{ cm}}$.

\seccio{4}{Aplicació a la semblança de triangles}

\ex{1} \begin{tasks}(1)
  \task $\frac84=\frac{10}5=\frac{12}6=2$: \rest{sí}, $k=2$
  \task $\frac63=2$, \ $\frac94=2{,}25$: \rest{no}
  \task $\frac{7{,}5}{5}=\frac{18}{12}=\frac{19{,}5}{13}=1{,}5$: \rest{sí}, $k=1{,}5$
\end{tasks}

\ex{1} \begin{tasks}(1)
  \task El tercer angle del primer fa 75° i el del segon, 45°: tenen els mateixos angles: \rest{sí}.
  \task Angles 50°, 50°, 80° i 50°, 90°, 40°: \rest{no}.
\end{tasks}

\ex{1} $\overline{AB}=4+2=6$; \quad $\frac46=\frac x9\Rightarrow6x=36\Rightarrow\res{x=6}$.

\ex{2} Pel teorema de Tales: $\frac63=\frac8x\Rightarrow6x=24\Rightarrow\res{x=4}$.

\ex{2} $\overline{AB}=4+4=8$; \quad $\frac48=\frac5x\Rightarrow4x=40\Rightarrow\res{x=10}$.

\ex{2} \rest{Sí}: els angles del primer són 90°, 40° i 50°, i els del segon, 90°, 50° i 40°. Tenen els tres angles iguals.

\ex{2} $k=\frac96=1{,}5$: \ $8\cdot1{,}5=\res{12\text{ cm}}$ i $9\cdot1{,}5=\res{13{,}5\text{ cm}}$.

\ex{2} $\frac64=\frac96=1{,}5$ i l'angle comprés és igual: \rest{sí} (criteri 3r), $k=1{,}5$. \ $\overline{B'C'}=5{,}3\cdot1{,}5=\res{7{,}95}$.

\ex{3} $P=18$, \ $k=\frac{45}{18}=2{,}5$: \ $\res{10,\ 15\text{ i }20\text{ cm}}$.

\ex{3} $\overline{AB}=15$; \ $\frac5{15}=\frac{x}{12}\Rightarrow\res{\overline{B'C'}=4}$; \ $\frac5{15}=\frac6{\overline{AC}}\Rightarrow\overline{AC}=18$, \ $\res{\overline{C'C}=12}$. \ $P=10+12+12+4=\res{38}$.

\ex{3} $k=\frac{10}4=2{,}5$; \ altura $3\cdot2{,}5=\res{7{,}5\text{ cm}}$; \ àrea $\frac{10\cdot7{,}5}{2}=\res{37{,}5\text{ cm}^2}$ (abans, 6 cm²): s'ha multiplicat per $2{,}5^2=\res{6{,}25}$.

\seccio{5}{Problemes amb semblança de triangles}

\ex{1} $\frac{H}{9}=\frac{2}{1{,}5}\Rightarrow H=\frac{9\cdot2}{1{,}5}=\res{12\text{ m}}$.

\ex{1} $\frac{1{,}5}{s}=\frac{21}{14}\Rightarrow21s=21\Rightarrow\res{s=1\text{ m}}$.

\ex{1} $\frac{H}{15}=\frac{1{,}2}{0{,}9}\Rightarrow H=\frac{15\cdot1{,}2}{0{,}9}=\res{20\text{ m}}$.

\ex{2} $\frac{H}{15}=\frac{1{,}6}{2}\Rightarrow H=\frac{15\cdot1{,}6}{2}=\res{12\text{ m}}$.

\ex{2} Triangles en posició de Tales: $\frac h4=\frac{0{,}9}{6}\Rightarrow h=\frac{4\cdot0{,}9}{6}=\res{0{,}6\text{ m}}$.

\ex{2} $\frac{1{,}5}{s}=\frac{4{,}5}{s+2}\Rightarrow1{,}5s+3=4{,}5s\Rightarrow3s=3\Rightarrow\res{s=1\text{ m}}$.

\ex{2} Des del peu de l'escala, el punt està a $1{,}5-0{,}6=0{,}9$ m. \ $\frac h{0{,}9}=\frac4{1{,}5}\Rightarrow h=\frac{0{,}9\cdot4}{1{,}5}=\res{2{,}4\text{ m}}$.

\ex{2} $\frac d6=\frac{0{,}5}{0{,}1}\Rightarrow d=\frac{6\cdot0{,}5}{0{,}1}=\res{30\text{ m}}$.

\ex{3} Els triangles $ABC$ i $EDC$ són semblants (angle recte i angles en $C$ oposats pel vèrtex): $\frac x{2{,}5}=\frac93\Rightarrow\res{x=7{,}5\text{ m}}$.

\ex{3} $\frac{H}{219}=\frac{1{,}8}{2{,}7}\Rightarrow H=\frac{219\cdot1{,}8}{2{,}7}=\res{146\text{ m}}$.

\ex{3} L'ombra de l'arbre fa $1{,}6+6{,}4=8$ m (triangles en posició de Tales): $\frac H8=\frac{1{,}2}{1{,}6}\Rightarrow H=\frac{8\cdot1{,}2}{1{,}6}=\res{6\text{ m}}$.
