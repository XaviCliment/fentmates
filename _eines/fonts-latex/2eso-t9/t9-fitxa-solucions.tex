\seccio{1}{Poliedres. Fórmula d'Euler. Poliedres regulars}

\ex{1} \begin{center}\renewcommand{\arraystretch}{1.25}\begin{tabular}{|l|C{1.6cm}|C{1.6cm}|C{1.6cm}|l|}\hline Poliedre & $C$ & $V$ & $A$ & $C+V=A+2$ \\\hline prisma pentagonal & 7 & 10 & 15 & $17=17$ \\\hline piràmide hexagonal & 7 & 7 & 12 & $14=14$ \\\hline prisma triangular & 5 & 6 & 9 & $11=11$ \\\hline piràmide quadrangular & 5 & 5 & 8 & $10=10$ \\\hline ortoedre & 6 & 8 & 12 & $14=14$ \\\hline\end{tabular}\end{center}

\ex{1} \begin{tasks}(3)
  \task $6+8=A+2\Rightarrow\res{A=12}$
  \task $C+12=20\Rightarrow\res{C=8}$
  \task $7+V=14\Rightarrow\res{V=7}$
\end{tasks}

\ex{1} \begin{tasks}(3)
  \task \rest{sí}
  \task no (cares de dos tipus)
  \task \rest{sí}
  \task no (cares de dos tipus)
  \task \rest{sí}
  \task no (rectangles diferents)
\end{tasks}Un poliedre regular té totes les cares polígons regulars iguals i el mateix nombre de cares a cada vèrtex.

\ex{2} \begin{center}\renewcommand{\arraystretch}{1.2}\begin{tabular}{|l|c|c|c|}\hline & Cares & Forma & Cares per vèrtex \\\hline Tetraedre & 4 & triangles & 3 \\\hline Cub & 6 & quadrats & 3 \\\hline Octaedre & 8 & triangles & 4 \\\hline Dodecaedre & 12 & pentàgons & 3 \\\hline Icosaedre & 20 & triangles & 5 \\\hline\end{tabular}\end{center}

\ex{2} $3n=18\Rightarrow n=6$: base \rest{hexagonal}; \ $\res{C=8}$, \ $\res{V=12}$ \ ($8+12=18+2$).

\ex{2} $n+1=9\Rightarrow n=8$: base \rest{octogonal}; \ $\res{V=9}$, \ $\res{A=16}$ \ ($9+9=16+2$).

\ex{3} a) $8+12=20=18+2$: \rest{sí} (per exemple, un prisma hexagonal). \ b) $6+7=13\neq14$: \rest{no}.

\ex{3} Cada angle d'un hexàgon regular fa $120^\circ$. En un vèrtex s'hi han d'ajuntar com a mínim 3 cares, i $3\cdot120^\circ=360^\circ$: les cares quedarien \rest{planes} i no tancarien cap cos.

\seccio{2}{Prismes: elements i àrea}

\ex{1} \begin{tasks}(2)
  \task $6\cdot5^2=\res{150\text{ cm}^2}$
  \task $2(7\cdot4+7\cdot3+4\cdot3)=2\cdot61=\res{122\text{ cm}^2}$
\end{tasks}

\ex{1} $D=\sqrt{36+4+9}=\sqrt{49}=\res{7\text{ cm}}$.

\ex{1} $A_L=16\cdot10=\res{160\text{ cm}^2}$; \ $A_T=160+2\cdot16=\res{192\text{ cm}^2}$.

\ex{2} Hipotenusa: $\sqrt{25+144}=13$. \ $A_L=(5+12+13)\cdot10=300$; \ $A_{\text{base}}=\frac{5\cdot12}{2}=30$. \ $A_T=300+60=\res{360\text{ cm}^2}$.

\ex{2} $A_L=48\cdot5=240$; \ $a_b=\sqrt{64-16}=\sqrt{48}\approx6{,}93$; \ $A_{\text{base}}=\frac{48\cdot\sqrt{48}}{2}\approx166{,}28$. \ $A_T\approx240+332{,}55=\res{572{,}55\text{ cm}^2}$.

\ex{2} Base $30\cdot20=600$; \ cares laterals $2\cdot30\cdot15+2\cdot20\cdot15=1500$. \ Total: $\res{2100\text{ cm}^2}$ (només hi ha una base).

\ex{3} $6l^2=216\Rightarrow l^2=36\Rightarrow\res{l=6\text{ cm}}$; \ $D=\sqrt{36+36+36}=\sqrt{108}\approx\res{10{,}39\text{ cm}}$.

\ex{3} Parets: $2(4+3{,}5)\cdot2{,}5=37{,}5$ m²; \ llevem la porta i la finestra: $37{,}5-1{,}6-1{,}2=34{,}7$ m². \ Cost: $34{,}7\cdot4=\res{138{,}80\text{ €}}$.

\seccio{3}{Piràmides: elements i àrea}

\ex{1} $a_b=8$; \ $a_p=\sqrt{36+64}=\res{10\text{ cm}}$. \ $A_T=\frac{64\cdot10}{2}+16^2=320+256=\res{576\text{ cm}^2}$.

\ex{1} $A_L=\frac{36\cdot10}{2}=\res{180\text{ cm}^2}$.

\ex{1} $A_L=\frac{24\cdot12}{2}=\res{144\text{ cm}^2}$.

\ex{2} $a_b=5$; \ $h=\sqrt{13^2-5^2}=\sqrt{144}=\res{12\text{ cm}}$. \ $A_T=\frac{40\cdot13}{2}+100=\res{360\text{ cm}^2}$.

\ex{2} $a_p=\sqrt{16+9}=\res{5\text{ cm}}$. \ L'aresta lateral és la hipotenusa de l'apotema i mig costat: $\sqrt{25+9}=\sqrt{34}\approx\res{5{,}83\text{ cm}}$. \ $A_T=\frac{24\cdot5}{2}+36=\res{96\text{ cm}^2}$.

\ex{2} Són 4 triangles equilàters. Altura de cada un: $\sqrt{16-4}=\sqrt{12}\approx3{,}46$. \ $A=4\cdot\frac{4\cdot\sqrt{12}}{2}=8\sqrt{12}\approx\res{27{,}71\text{ cm}^2}$.

\ex{3} $a_b=1$; \ $a_p=\sqrt{1{,}5^2+1^2}=\sqrt{3{,}25}\approx1{,}80$ m. \ Només l'àrea lateral: $A_L=\frac{8\cdot\sqrt{3{,}25}}{2}\approx\res{7{,}21\text{ m}^2}$.

\ex{3} $a_b=\sqrt{16-4}=\sqrt{12}\approx3{,}46$; \ $a_p=\sqrt{36+12}=\sqrt{48}\approx6{,}93$. \ $A_L=\frac{24\cdot\sqrt{48}}{2}\approx83{,}14$; \ $A_{\text{base}}=\frac{24\cdot\sqrt{12}}{2}\approx41{,}57$. \ $A_T\approx\res{124{,}71\text{ cm}^2}$.

\ex{1} $A_L=2\pi\cdot3\cdot8=48\pi\approx\res{150{,}80\text{ cm}^2}$; \ $A_T=48\pi+18\pi=66\pi\approx\res{207{,}35\text{ cm}^2}$.

\ex{1} $A_L=\pi\cdot6\cdot10=60\pi\approx\res{188{,}50\text{ cm}^2}$; \ $A_T=60\pi+36\pi=96\pi\approx\res{301{,}59\text{ cm}^2}$.

\ex{2} $r=8$; \ $g=\sqrt{225+64}=\sqrt{289}=\res{17\text{ cm}}$. \ $A_T=136\pi+64\pi=200\pi\approx\res{628{,}32\text{ cm}^2}$.

\ex{2} $A_L=2\pi\cdot4\cdot11=88\pi\approx\res{276{,}46\text{ cm}^2}$.

\ex{2} $h=\sqrt{25^2-7^2}=\sqrt{576}=\res{24\text{ cm}}$.

\ex{3} $2\pi r\cdot10=314{,}16\Rightarrow r=\frac{314{,}16}{20\pi}\approx\res{5\text{ cm}}$. \ $A_T=100\pi+50\pi=150\pi\approx\res{471{,}24\text{ cm}^2}$.

\ex{3} Un got té la superfície lateral i un sol cercle: $54\pi+9\pi=63\pi\approx\res{197{,}92\text{ cm}^2}$. \ 1 m² = 10\,000 cm²: $\frac{10\,000}{197{,}92}\approx50{,}5\Rightarrow\res{50\text{ gots}}$.

\seccio{5}{L'esfera i la seua àrea}

\ex{1} \begin{tasks}(2)
  \task $4\pi\cdot49=196\pi\approx\res{615{,}75\text{ cm}^2}$
  \task $r=5$: \ $4\pi\cdot25=100\pi\approx\res{314{,}16\text{ cm}^2}$
\end{tasks}

\ex{1} $4\pi r^2=36\pi\Rightarrow r^2=9\Rightarrow\res{r=3\text{ cm}}$.

\ex{2} $r=3{,}25$: \ $A=4\pi\cdot3{,}25^2=42{,}25\pi\approx\res{132{,}73\text{ cm}^2}$.

\ex{2} La meitat de l'esfera: $\frac{4\pi\cdot36}{2}=72\pi\approx\res{226{,}19\text{ m}^2}$.

\ex{2} $2\pi r=50\Rightarrow r=\frac{50}{2\pi}\approx\res{7{,}96\text{ cm}}$; \ $A=4\pi r^2=\frac{2500}{\pi}\approx\res{795{,}77\text{ cm}^2}$.

\ex{3} $r=5$: \ $A=100\pi\approx\res{314{,}16\text{ cm}^2}$. \ $A_{\text{cub}}=600$: \ $\frac{314{,}16}{600}\cdot100\approx\res{52{,}36\,\%}$.

\ex{3} $A_{\text{esfera}}=4\pi\cdot16=64\pi$; \ $A_L=2\pi\cdot4\cdot8=64\pi$ \ \checkmark. \ $A_T=64\pi+32\pi=96\pi$: \ $\frac{64\pi}{96\pi}=\res{\tfrac23}$ (descobriment d'Arquimedes).

\ex{2} $A_{\text{base}}=\frac{3\cdot4}{2}=6$; \ $V=6\cdot20=\res{120\text{ cm}^3}$.

\ex{2} $60\cdot30\cdot40=72\,000\text{ cm}^3=\res{72\text{ L}}$; \ $60\cdot30\cdot35=63\,000\text{ cm}^3=\res{63\text{ L}}$.

\ex{3} 1 L = 1000 cm³: \ $6\cdot9{,}5\cdot h=1000\Rightarrow h=\frac{1000}{57}\approx\res{17{,}54\text{ cm}}$.

\ex{1} $V=\frac{\pi\cdot16\cdot9}{3}=48\pi\approx\res{150{,}80\text{ cm}^3}$.

\ex{1} $\frac43\pi\cdot27=36\pi\approx\res{113{,}10\text{ cm}^3}$; \ $\frac43\pi\cdot1000=\frac{4000\pi}{3}\approx\res{4188{,}79\text{ cm}^3}$.

\ex{2} $h=\sqrt{225-81}=\res{12\text{ cm}}$; \ $V=\frac{\pi\cdot81\cdot12}{3}=324\pi\approx\res{1017{,}88\text{ cm}^3}$.

\ex{2} $a_b=4$; \ $h=\sqrt{25-16}=\res{3\text{ cm}}$; \ $V=\frac{64\cdot3}{3}=\res{64\text{ cm}^3}$.

\ex{3} $V_{\text{cil}}=\pi\cdot9\cdot6=54\pi$; \ $V_{\text{esfera}}=\frac43\pi\cdot27=36\pi$. \ Vessa $\frac{36\pi}{54\pi}=\res{\tfrac23}$ de l'aigua.

\seccio{8}{Cossos compostos i problemes}

\ex{1} $V=5^3+\frac{25\cdot6}{3}=125+50=\res{175\text{ cm}^3}$.

\ex{2} $V=\frac{\pi\cdot9\cdot12}{3}+\frac12\cdot\frac43\pi\cdot27=36\pi+18\pi=54\pi\approx\res{169{,}65\text{ cm}^3}$.

\ex{2} $V=\pi\cdot9\cdot8+\frac{\pi\cdot9\cdot4}{3}=72\pi+12\pi=84\pi\approx\res{263{,}89\text{ m}^3}$.

\ex{2} $V=\pi(6^2-5^2)\cdot200=2200\pi\approx\res{6911{,}50\text{ cm}^3}$.

\ex{2} $V=12^3-\pi\cdot4\cdot12=1728-48\pi\approx\res{1577{,}20\text{ cm}^3}$.

\ex{3} a) $25\cdot10+2(25+10)\cdot2=250+140=\res{390\text{ m}^2}$. \ b) $25\cdot10\cdot2=500\text{ m}^3=\res{500\,000\text{ L}}$. \ c) $\frac{500\,000}{400}=1250$ min $\approx\res{20{,}83\text{ hores}}$ (20 h 50 min).

\ex{3} Amb $r=3{,}25$ i $h=6r$: \ $V_{\text{tub}}=\pi r^2\cdot6r=6\pi r^3$; \ $V_{\text{pilotes}}=3\cdot\frac43\pi r^3=4\pi r^3$. \ Ocupen $\res{\tfrac46=\tfrac23}$ del tub. \ Buit: $2\pi r^3=2\pi\cdot3{,}25^3\approx\res{215{,}69\text{ cm}^3}$.

\ex{3} Volum de les boles: $5\cdot\frac43\pi=\frac{20\pi}{3}$. \ L'aigua puja $h$: \ $16\pi h=\frac{20\pi}{3}\Rightarrow h=\frac{5}{12}\approx0{,}42$ cm. \ Arriba a $\res{10{,}42\text{ cm}}$.
