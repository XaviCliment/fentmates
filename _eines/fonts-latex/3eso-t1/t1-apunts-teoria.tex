\chip{fmK}{$\mathbb{N}$ \ naturals} \ $0,1,2,3\dots$ \ per a \textbf{comptar}.\\

\begin{idea}
{\centering\Large$\mathbb{N}\subset\mathbb{Z}\subset\mathbb{Q}$ \quad \normalsize tot natural és enter, i tot enter és racional: \ $5=\dfrac51=\dfrac{10}{2}$\par}
\smallskip Un racional en forma decimal és \textbf{exacte} o \textbf{periòdic} (1.11). Els que tenen infinites xifres \textbf{no periòdiques}, com $\pi$ o $\sqrt2$, són \textbf{irracionals} (unitat 2).
\end{idea}

\begin{minipage}[t]{0.54\linewidth}
\begin{compte}[Casos que enganyen]
$-\dfrac{12}{4}=-3$ \ és \textbf{enter}, encara que siga una fracció.\\[3pt]
$\dfrac{-18}{-6}=3$ \ és \textbf{natural}: menys entre menys és més.\\[3pt]
$0{,}75=\dfrac34$ \ i \ $0{,}\overline{6}=\dfrac23$ \ són \textbf{racionals}.
\end{compte}
\end{minipage}\hfill
\begin{minipage}[t]{0.43\linewidth}
\begin{truc}[I el 0?]
És \textbf{natural}, \textbf{enter} i \textbf{racional}.\\[3pt]
I no és ni positiu ni negatiu.
\end{truc}

\begin{minipage}[c]{0.46\linewidth}
\begin{idea}[Llevar parèntesis]
{\centering\renewcommand{\arraystretch}{1.4}\Large
\begin{tabular}{cccc}
$+(+a)$ & $+(-a)$ & $-(+a)$ & $-(-a)$\\
$\cK{+a}$ & $\cR{-a}$ & $\cR{-a}$ & $\cK{+a}$
\end{tabular}\par}\smallskip
Un $-$ davant del parèntesi canvia \textbf{tots} els signes: \ $-(3-5+2)=-3+5-2$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.51\linewidth}
\begin{exemple}[Exemple: \ $7-(-3)+(-5)-(+2)$]
$=7+3-5-2$ \ \nota{llevem parèntesis}\\[3pt]
$=10-7=\boxed{3}$ \ \nota{positius i negatius}
\end{exemple}
\begin{truc}[Agrupa]
Suma a part els positius i els negatius:\\[2pt]
$-4+9-12+3-7=(9+3)-(4+12+7)=12-23=-11$
\end{truc}
\end{minipage}

\begin{minipage}[t]{0.48\linewidth}
\begin{exemple}[Exemple: \ $-8+(4-9)-(-6+2)$]
$=-8+(-5)-(-4)$ \ \nota{dins dels parèntesis}\\[3pt]
$=-8-5+4$ \ \nota{llevem parèntesis}\\[3pt]
$=-13+4=\boxed{-9}$
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
\begin{exemple}[{Exemple: \ $15-\left[6-(2-9)\right]$}]
$=15-\left[6-(-7)\right]$ \ \nota{el de dins primer}\\[3pt]
$=15-\left[6+7\right]$\\[3pt]
$=15-13=\boxed{2}$
\end{exemple}
\end{minipage}

\be{5-(-3)=5+3=8}\qquad\quad \be{-2-6=-2+(-6)=-8}\qquad\quad \mal{-(4-9)=-4-9}
\end{compte}

\seccio{sec7}{Operacions amb fraccions}{Sumar demana el mateix denominador; multiplicar, no.}

\begin{tcbraster}[raster columns=2,raster equal height=rows,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faPlus\,\faMinus\ \ Suma i resta}]

{\centering$\dfrac56-\dfrac34=\dfrac{10}{12}-\dfrac{9}{12}=\dfrac{1}{12}$\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faTimes\ \ Multiplicació}]
\textbf{Dalt per dalt} i \textbf{baix per baix}.\par\smallskip
{\centering$\dfrac{a}{b}\cdot\dfrac{c}{d}=\dfrac{a\cdot c}{b\cdot d}\qquad \dfrac23\cdot\dfrac45=\dfrac{8}{15}$\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faDivide\ \ Divisió}]
Multipliquem \textbf{en creu} (o per la \textbf{inversa}).\par\smallskip
{\centering$\dfrac{a}{b}:\dfrac{c}{d}=\dfrac{a\cdot d}{b\cdot c}\qquad \dfrac23:\dfrac45=\dfrac{10}{12}=\dfrac56$\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faUndo\ \ Inversa}]
La inversa de $\dfrac{a}{b}$ és $\dfrac{b}{a}$: el seu producte és 1.\par\smallskip
{\centering La de $-2$ és $-\dfrac12$. \ El 0 no en té.\par}
\end{tcolorbox}
\end{tcbraster}

\begin{truc}[Un enter és una fracció]
Té denominador 1:\\[3pt]
$3-\dfrac75=\dfrac{15}{5}-\dfrac75=\dfrac85$
\end{truc}
\end{minipage}\hfill

\end{truc}

\seccio{sec8}{Potències de fraccions}{Elevem el numerador i el denominador.}

\begin{idea}
{\centering\Large$\left(\dfrac{a}{b}\right)^{n}=\dfrac{a^{n}}{b^{n}}$\qquad\normalsize
$\left(\dfrac23\right)^{3}=\dfrac{8}{27}\qquad \left(-\dfrac32\right)^{4}=\cK{+}\dfrac{81}{16}\qquad \left(-\dfrac12\right)^{5}=\cR{-}\dfrac{1}{32}$\par}
\smallskip Base negativa: exponent \textbf{parell} $\to$ positiu; \textbf{senar} $\to$ negatiu.
\end{idea}

\begin{minipage}[t]{0.42\linewidth}
\begin{truc}[Exponent 0 i negatiu]
$\left(\dfrac{a}{b}\right)^{0}=1$\\[4pt]
$\left(\dfrac{a}{b}\right)^{-n}=\left(\dfrac{\cB{b}}{\cB{a}}\right)^{n}$ \ \nota{la girem}\\[4pt]
$\left(\dfrac23\right)^{-2}=\left(\dfrac32\right)^{2}=\dfrac94$
\end{truc}
\end{minipage}\hfill
\begin{minipage}[t]{0.55\linewidth}\centering\renewcommand{\arraystretch}{1.9}
\begin{tcolorbox}[enhanced,hbox,arc=3mm,boxrule=1.2pt,colframe=fmblue,colback=white,coltitle=white,fonttitle=\heavy,colbacktitle=fmblue,
  title={Propietats (com amb els enters)},left=0pt,right=0pt,top=0pt,bottom=0pt,boxsep=0pt,toptitle=1mm,bottomtitle=1mm,before skip=0pt,after skip=0pt]
\rowcolors{1}{fmsurface}{white}
\begin{tabular}{@{\hspace{6pt}}l@{\quad}l@{\hspace{6pt}}}
$x^{m}\cdot x^{n}=x^{\cK{m+n}}$ & $\left(\frac13\right)^{2}\cdot\left(\frac13\right)^{3}=\left(\frac13\right)^{5}$\\
$x^{m}:x^{n}=x^{\cK{m-n}}$ & $\left(\frac34\right)^{5}:\left(\frac34\right)^{3}=\left(\frac34\right)^{2}$\\
$\left(x^{m}\right)^{n}=x^{\cK{m\cdot n}}$ & $\left(\left(-\frac12\right)^{2}\right)^{3}=\left(-\frac12\right)^{6}$\\
$x^{n}\cdot y^{n}=(x\cdot y)^{n}$ & $\left(\frac25\right)^{3}\cdot\left(\frac52\right)^{3}=1^{3}=1$
\end{tabular}
\end{tcolorbox}
\end{minipage}

\seccio{sec9}{Operacions combinades amb fraccions}{La mateixa jerarquia que amb els enters.}

\begin{minipage}[c]{0.4\linewidth}\raggedright
\pas{1} Parèntesis $\to$ \pas{2} potències $\to$ \pas{3} $\cdot$ i $:$ $\to$ \pas{4} $+$ i $-$.\\[8pt]
\faCheck\ \textbf{Simplifica} cada resultat parcial.\\[6pt]
\faCheck\ Comú denominador \textbf{només} per als termes que sumes en aquell pas.\\[6pt]
\faCheck\ Un $-$ davant d'un parèntesi afecta \textbf{tot} el resultat.
\end{minipage}\hfill
\begin{minipage}[c]{0.57\linewidth}
\begin{exemple}[Exemple: \ $\dfrac34-\dfrac12\cdot\left(\dfrac56-\dfrac13\right)$]
\vspace{-6pt}
\begin{alignat*}{2}
&=\dfrac34-\dfrac12\cdot\left(\dfrac56-\dfrac26\right)=\dfrac34-\dfrac12\cdot\dfrac12 &\quad& \text{\nota{\pas{1}}}\\
&=\dfrac34-\dfrac14 && \text{\nota{\pas{3}}}\\
&=\dfrac24=\boxed{\dfrac12} && \text{\nota{\pas{4}}}
\end{alignat*}
\end{exemple}
\end{minipage}

\begin{exemple}[Exemple: \ $\left(1-\dfrac25\right)^{2}:\dfrac{9}{10}+\dfrac13$]
$=\left(\dfrac35\right)^{2}:\dfrac{9}{10}+\dfrac13=\dfrac{9}{25}:\dfrac{9}{10}+\dfrac13=\dfrac{90}{225}+\dfrac13=\dfrac25+\dfrac13=\dfrac{6}{15}+\dfrac{5}{15}=\boxed{\dfrac{11}{15}}$
\end{exemple}

\seccio{sec10}{Problemes amb fraccions}{Un dibuix amb una barra ho aclareix tot.}

\begin{tcbraster}[raster columns=2,raster equal height=rows,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Fracció d'una quantitat}]
Dividim entre el denominador i multipliquem pel numerador.\par\smallskip
{\centering$\dfrac34$ de $200=\dfrac{3\cdot200}{4}=150$\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Fracció d'una fracció}]
Les \textbf{multipliquem}.\par\smallskip
{\centering$\dfrac23$ de $\dfrac35=\dfrac23\cdot\dfrac35=\dfrac25$\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={La part que queda}]
El total és $1=\dfrac{b}{b}$. Si gastem $\dfrac{a}{b}$, en queda $1-\dfrac{a}{b}$.\par\smallskip
{\centering Gastem $\dfrac25$: \ en queda $\dfrac55-\dfrac25=\dfrac35$\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Problema invers}]
Si $\dfrac{a}{b}$ del total és $P$, el total és $P:\dfrac{a}{b}$.\par\smallskip
{\centering$\dfrac23$ del total són 40: \ $40:\dfrac23=60$\par}
\end{tcolorbox}
\end{tcbraster}

\begin{exemple}[Exemple: \ l'Anna gasta $\frac25$ dels seus 60 € en roba i $\frac13$ del que li queda en un llibre. Quant li queda?]
\begin{minipage}[c]{0.45\linewidth}\centering
\begin{tikzpicture}[x=1.3cm]
  \foreach \i in {0,...,4} {\filldraw[fill=\ifnum\i<2 fmB!35\else white\fi,draw=fmB,line width=1pt] (\i,0) rectangle (\i+1,0.6);}
  \node[font=\small] at (1,0.3) {roba};
  \filldraw[fill=fmA!35,draw=fmA,line width=1pt] (2,0) rectangle (3,0.6); \node[font=\small] at (2.5,0.3) {llibre};
  \node[font=\small] at (4,0.3) {queda};
  \draw[decorate,decoration={brace,amplitude=5pt},line width=1pt] (0,0.7) -- (5,0.7) node[midway,above=6pt]{60 €};
  \foreach \i in {0,...,4} \node[font=\small,text=fmmuted] at (\i+0.5,-0.3) {12 €};
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.52\linewidth}
Roba: \ $\dfrac25$ de $60=24$ € \ $\to$ \ en queden 36 €\\[4pt]
Llibre: \ $\dfrac13$ de $36=12$ €\\[4pt]
Li queden \ $36-12=\boxed{24\text{ €}}$
\end{minipage}
\end{exemple}

\begin{exemple}[Exemple: \ un dipòsit està ple als $\frac23$ i té 40 litres. Quina capacitat té?]
\begin{minipage}[c]{0.4\linewidth}\centering
\begin{tikzpicture}[x=1.4cm]
  \foreach \i in {0,1,2} {\filldraw[fill=\ifnum\i<2 fmA!35\else white\fi,draw=fmA,line width=1pt] (\i,0) rectangle (\i+1,0.6);}
  \draw[decorate,decoration={brace,amplitude=5pt},line width=1pt] (0,0.7) -- (2,0.7) node[midway,above=6pt]{40 L};
  \foreach \i in {0,1,2} \node[font=\small,text=fmmuted] at (\i+0.5,-0.3) {20 L};
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.57\linewidth}
Problema \textbf{invers}: \ $40:\dfrac23=\dfrac{40\cdot3}{2}=\boxed{60\text{ L}}$\\[4pt]
\nota{Cada terç són 20 L; els tres terços, 60 L.}
\end{minipage}
\end{exemple}

\begin{minipage}[t]{0.5\linewidth}
\begin{compte}[De quina quantitat?]
El $\frac13$ és \textbf{del que queda} (36 €), no del total (60 €). Llig bé de què és cada fracció.
\end{compte}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{truc}[Comprova que té sentit]
Respon amb una \textbf{frase} i amb les \textbf{unitats}: «Li queden 24 €.»
\end{truc}
\end{minipage}

\subsection{La regla ràpida}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=3mm,bottom=3mm]
{\Large$N=\dfrac{\text{nombre sense coma}\ -\ \text{part que no es repeteix}}{\underbrace{\cA{9\dots9}}_{\text{\small tants 9 com xifres del període}}\qquad\underbrace{\cK{0\dots0}}_{\text{\small tants 0 com xifres de l'anteperíode}}}$\par}
\end{tcolorbox}

\begin{exemple}[Exemples amb la regla ràpida]
\begin{minipage}[t]{0.32\linewidth}\centering
$3{,}\overline{18}=\dfrac{318-3}{99}=\dfrac{315}{99}=\boxed{\dfrac{35}{11}}$
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}\centering
$0{,}1\overline{6}=\dfrac{16-1}{90}=\dfrac{15}{90}=\boxed{\dfrac16}$
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}\centering
$2{,}\overline{7}=\dfrac{27-2}{9}=\boxed{\dfrac{25}{9}}$
\end{minipage}
\end{exemple}

\begin{exemple}[Més exemples]
\begin{minipage}[t]{0.48\linewidth}\centering
$0{,}125=\dfrac{125}{1000}=\boxed{\dfrac18}$ \ \nota{exacte}
\end{minipage}\hfill
\begin{minipage}[t]{0.5\linewidth}\centering
$1{,}0\overline{45}=\dfrac{1045-10}{990}=\dfrac{1035}{990}=\boxed{\dfrac{23}{22}}$
\end{minipage}
\end{exemple}

\begin{truc}[Curiositat]
$0{,}\overline{9}=\dfrac99=1$. \ Sí: \ $0{,}999\dots$ \ i \ 1 \ són el \textbf{mateix} nombre!
\end{truc}

\seccio{sec13}{Operacions amb fraccions i decimals}{Passem els decimals a fracció i operem.}

\begin{minipage}[c]{0.45\linewidth}\raggedright
\pas{1} Passa tots els decimals a \textbf{fracció} (sobretot els periòdics).\\[6pt]
\pas{2} Opera amb la \textbf{jerarquia} de sempre.\\[6pt]
\pas{3} \textbf{Simplifica}. Si cal, torna-ho a decimal.
\end{minipage}\hfill
\begin{minipage}[c]{0.52\linewidth}
\begin{compte}[Per què no amb decimals?]
Amb periòdics no podem operar exactament:\\[3pt]
\be{0{,}\overline{3}\cdot3=\dfrac13\cdot3=1}\\[3pt]
\mal{0{,}333\cdot3=0{,}999}
\end{compte}
\end{minipage}

\begin{exemple}[Exemple: \ $0{,}\overline{3}+1{,}5\cdot\dfrac23$]
$0{,}\overline{3}=\dfrac39=\dfrac13$, \quad $1{,}5=\dfrac{15}{10}=\dfrac32$ \quad $\to$ \quad $\dfrac13+\dfrac32\cdot\dfrac23=\dfrac13+1=\boxed{\dfrac43}$
\end{exemple}

\begin{exemple}[Exemple: \ $\left(1{,}\overline{6}-0{,}5\right)^{2}:0{,}\overline{7}$]
$1{,}\overline{6}=\dfrac{16-1}{9}=\dfrac53$, \quad $0{,}5=\dfrac12$, \quad $0{,}\overline{7}=\dfrac79$\\[6pt]
$\left(\dfrac53-\dfrac12\right)^{2}:\dfrac79=\left(\dfrac76\right)^{2}:\dfrac79=\dfrac{49}{36}:\dfrac79=\dfrac{441}{252}=\boxed{\dfrac74}=1{,}75$
\end{exemple}

\begin{exemple}[Exemple: \ $2{,}25-0{,}1\overline{6}\cdot3+\dfrac14$]
$2{,}25=\dfrac94$, \quad $0{,}1\overline{6}=\dfrac16$ \quad $\to$ \quad $\dfrac94-\dfrac16\cdot3+\dfrac14=\dfrac94-\dfrac24+\dfrac14=\dfrac84=\boxed{2}$
\end{exemple}

\begin{exemple}[Més exemples]
\begin{minipage}[t]{0.42\linewidth}
$0{,}\overline{6}\cdot0{,}75=\dfrac23\cdot\dfrac34=\boxed{\dfrac12}$
\end{minipage}\hfill
\begin{minipage}[t]{0.55\linewidth}
$\left(0{,}5+0{,}\overline{3}\right):2{,}5=\left(\dfrac12+\dfrac13\right):\dfrac52=\dfrac56\cdot\dfrac25=\boxed{\dfrac13}$
\end{minipage}
\end{exemple}

\begin{truc}[Comprova amb la calculadora]
En acabar, fes la divisió del resultat: \ $\dfrac74=1{,}75$. Si coincideix amb el que dona la calculadora operant amb decimals (aproximadament), vas bé.
\end{truc}
