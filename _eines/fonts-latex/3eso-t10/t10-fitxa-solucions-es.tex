\ex{1} \begin{tasks}(3)
  \task $5x=60\Rightarrow\res{x=12}$
  \task $2x=56\Rightarrow\res{x=28}$
  \task $6x=60\Rightarrow\res{x=10}$
  \task $5x=20\Rightarrow\res{x=4}$
  \task $6x=90\Rightarrow\res{x=15}$

\ex{1} \begin{tasks}(2)

  \task $3\cdot9=27\neq5\cdot6=30$: \rest{no}

\ex{2} a) $\frac46=\frac{10}{x}\Rightarrow x=\frac{6\cdot10}{4}=\res{15\text{ cm}}$. \quad b) $\frac{2{,}5}{3}=\frac{7}{x}\Rightarrow x=\frac{3\cdot7}{2{,}5}=\res{8{,}4\text{ cm}}$.

\ex{3} $x^2=a\cdot b$: \ a) $x^2=100\Rightarrow\res{x=10}$; \ b) $x^2=16\Rightarrow\res{x=4}$; \ c) $x^2=18\Rightarrow x=\sqrt{18}\approx\res{4{,}24}$.

\seccio{2}{El teorema de Tales}

\ex{1} $\frac46=\frac6x\Rightarrow4x=36\Rightarrow\res{x=9}$.

\ex{1} \begin{tasks}(1)
  \task $\frac35=\frac{4{,}8}{x}\Rightarrow\res{\overline{B'C'}=8}$
  \task $\frac64=\frac{x}{5}\Rightarrow\res{\overline{BC}=7{,}5}$
  \task $\frac{4}{10}=\frac{x}{15}\Rightarrow\res{\overline{A'B'}=6}$
\end{tasks}

\ex{2} $\frac x6=\frac48\Rightarrow8x=24\Rightarrow\res{x=3}$.

\ex{2} \begin{tasks}(3)
  \task $\frac23=\frac46$: \rest{sí}
  \task $\frac34=0{,}75\neq\frac45=0{,}8$: \rest{no}
  \task $\frac{1{,}5}{2{,}5}=0{,}6=\frac{2{,}4}{4}$: \rest{sí}
\end{tasks}(Recíproco del teorema de Tales.)

\ex{2} $k=\frac{72}{15+20+25}=\frac{72}{60}=1{,}2$: \ $15\cdot1{,}2=\res{18\text{ m}}$, \ $20\cdot1{,}2=\res{24\text{ m}}$, \ $25\cdot1{,}2=\res{30\text{ m}}$.

\ex{3} $\frac{x}{x+4}=\frac35\Rightarrow5x=3x+12\Rightarrow\res{x=6}$ \ (y $\overline{BC}=10$).

\ex{3} $\frac{\overline{A'B'}}{5}=\frac{\overline{A'C'}}{13}=\frac{19{,}5}{13}=1{,}5$: \ $\res{\overline{A'B'}=7{,}5\text{ cm}}$, \ $\res{\overline{B'C'}=12\text{ cm}}$.

\ex{2} Semirrecta auxiliar, 5 marcas iguales, unimos la quinta con $B$ y trazamos la paralela por la marca 2. Partes: $\frac{10}{5}=2$, \ $\res{4\text{ cm y }6\text{ cm}}$.

\ex{2} $\frac{3{,}6}{8}=0{,}45$: \ $\res{0{,}45\text{ m}}$, \ $\res{0{,}9\text{ m}}$ y $\res{2{,}25\text{ m}}$.

\ex{3} Partes $3u$ y $7u$: $7u-3u=12\Rightarrow u=3$. \ $\res{9\text{ cm}}$ y $\res{21\text{ cm}}$; \ $\overline{AB}=\res{30\text{ cm}}$.

\ex{1} \begin{tasks}(1)

  \task $\frac58=\frac{7{,}5}{x}\Rightarrow\res{\overline{BC}=12}$

\ex{2} $\overline{AB}=10$: \ $\frac{6}{10}=\frac{7{,}2}{x}\Rightarrow\res{x=12}$. \ Tales: $\frac64=\frac9y\Rightarrow\res{y=6}$.

\ex{2} El triángulo pequeño y el grande están en posición de Tales y las alturas también son proporcionales: $\frac{4}{12}=\frac{x}{18}\Rightarrow\res{x=6\text{ cm}}$.

\ex{2} Los triángulos $ABC$ y $DEC$ tienen el ángulo en $C$ igual (opuestos por el vértice) y $AB\parallel DE$: \ $\frac{\overline{CE}}{\overline{CA}}=\frac96=1{,}5$. \ $x=8\cdot1{,}5=\res{12}$, \ $y=5\cdot1{,}5=\res{7{,}5}$.

\ex{3} Los triángulos $ABC$ y $EDC$ son semejantes (ángulo en $C$ opuesto por el vértice y ángulos rectos en $B$ y $D$): \ $\frac{\overline{AB}}{4}=\frac{15}{5}\Rightarrow\res{\overline{AB}=12\text{ m}}$.

\seccio{5}{Figuras semejantes. Razón de semejanza}

\ex{1} \begin{tasks}(2)
  \task $\frac64=\frac96=1{,}5$: \rest{sí}, $k=1{,}5$
  \task $\frac42=2\neq\frac85$: \rest{no}
  \task $\frac63=\frac84=2$: \rest{sí}, $k=2$
  \task \rest{sí} (todos los cuadrados lo son), $k=\frac73$
\end{tasks}

\ex{1} a) $\res{10,\ 15\text{ y }17{,}5\text{ cm}}$. \quad b) $\res{1{,}6,\ 2{,}4\text{ y }2{,}8\text{ cm}}$.

\ex{2} $k=\frac{27}{9}=3$; \ lado grande $13\cdot3=\res{39\text{ cm}}$; \ el área se multiplica por $k^2=\res{9}$.

\ex{2} $P'=20\cdot1{,}5=\res{30\text{ cm}}$; \ $A'=27{,}5\cdot1{,}5^2=27{,}5\cdot2{,}25\approx\res{61{,}88\text{ cm}^2}$.

\ex{2} $k^2=\frac{72}{18}=4\Rightarrow\res{k=2}$; \ $P'=16\cdot2=\res{32\text{ cm}}$.

\ex{3} $A=96$; \ $k^2=\frac{600}{96}=6{,}25\Rightarrow k=2{,}5$. \ Dimensions: $\res{20\text{ cm}\times30\text{ cm}}$.

\ex{3} Área del pequeño: $\frac{3\cdot4}{2}=6$; \ $k^2=\frac{54}{6}=9\Rightarrow k=3$. Hipotenusa del pequeño: 5. \ Lados: $\res{9,\ 12\text{ y }15\text{ cm}}$.

\seccio{6}{Criterios de semejanza de triángulos}

\ex{1} \begin{tasks}(2)
  \task \rest{sí}, $k=3$
  \task \rest{no}: $\frac63=\frac84=2\neq\frac{10}{6}$
  \task \rest{sí}, $k=1{,}5$
  \task \rest{no}: $\frac64=\frac96=1{,}5\neq\frac{12}{9}$
\end{tasks}

\ex{1} \begin{tasks}(1)
  \task El primero tiene 35°, 75°, 70°: \rest{sí} (criterio 1).
  \task El primero tiene 50°, 60°, 70° y el segundo 60°, 80°, 40°: \rest{no}.
  \task Los dos tienen 90°, 30° y 60°: \rest{sí}.
\end{tasks}

\ex{2} $\frac64=\frac96=1{,}5$ y el ángulo comprendido es igual: \rest{sí} (criterio 3), $k=1{,}5$. \ $\overline{B'C'}=5\cdot1{,}5=\res{7{,}5}$.

\ex{2} $k=\frac96=1{,}5$: \ lados $\res{12\text{ cm y }15\text{ cm}}$; \ razón de las áreas $k^2=\res{2{,}25}$.

\ex{2} Los ángulos iguales miden $\frac{180-40}{2}=70^\circ$: los dos tienen 40°, 70°, 70°, y son \rest{semejantes}. Dos isósceles cualesquiera, \rest{no} necesariamente.

\ex{2} $a=\sqrt{36+64}=10$. La altura divide el triángulo en dos triángulos semejantes al grande: $\frac h8=\frac6{10}\Rightarrow\res{h=4{,}8\text{ cm}}$.

\ex{3} $\overline{BC}=\sqrt{81+144}=15$. Los triángulos $HBA$ y $ABC$ son semejantes (rectángulos con el ángulo $\hat B$ común): $\frac{\overline{BH}}{9}=\frac{9}{15}\Rightarrow\res{\overline{BH}=5{,}4}$; \ $\res{\overline{HC}=9{,}6}$; \ $\frac{\overline{AH}}{12}=\frac{9}{15}\Rightarrow\res{\overline{AH}=7{,}2}$ cm.

\ex{3} Los triángulos $OAB$ y $OCD$ son semejantes (ángulos en $O$ opuestos por el vértice y $AB\parallel CD$): $\frac{\overline{OC}}{9}=\frac{8}{12}\Rightarrow\res{\overline{OC}=6\text{ cm}}$.

\ex{1} $\frac{s}{1{,}6}=\frac{18}{24}\Rightarrow\res{s=1{,}2\text{ m}}$.

\ex{3} $\frac{1{,}6}{s}=\frac{5}{s+3}\Rightarrow1{,}6s+4{,}8=5s\Rightarrow s=\frac{4{,}8}{3{,}4}\approx\res{1{,}41\text{ m}}$.

\ex{3} Los triángulos $ABC$ y $EDC$ son semejantes (ángulos rectos en $B$ y $D$ y ángulos en $C$ opuestos por el vértice): $\frac{\overline{AB}}{6}=\frac{20}{8}\Rightarrow\res{\overline{AB}=15\text{ m}}$.

\ex{3} Trazando la horizontal de los ojos quedan dos triángulos en posición de Tales: $\frac{H-1{,}5}{10}=\frac{3-1{,}5}{3}=0{,}5\Rightarrow H-1{,}5=5\Rightarrow\res{H=6{,}5\text{ m}}$.

\seccio{8}{Escalas. Áreas y volúmenes de figuras semejantes}

\ex{1} \begin{tasks}(3)
  \task $400\text{ cm}=\res{4\text{ m}}$
  \task $150\,000\text{ cm}=\res{1{,}5\text{ km}}$
  \task $700\,000\text{ cm}=\res{7\text{ km}}$
\end{tasks}

\ex{1} a) $1200:100=\res{12\text{ cm}}$. \quad b) $4\,500\,000:1\,500\,000=\res{3\text{ cm}}$.

\ex{2} a) $2\text{ km}=200\,000$ cm; \ $\frac{5}{200\,000}=\frac{1}{40\,000}$: \ $\res{1:40\,000}$. \quad b) $\frac{3}{15\,000}$: \ $\res{1:5000}$.

\ex{2} $3\cdot200=600$ cm $=6$ m; \ $2{,}5\cdot200=500$ cm $=5$ m. \ $\res{6\text{ m}\times5\text{ m}}$, \ $A=\res{30\text{ m}^2}$.

\ex{2} $500\text{ m}=50\,000$ cm; \ $\frac{2}{50\,000}=\frac{1}{25\,000}$: \ $\res{1:25\,000}$. \ $7\cdot25\,000=175\,000$ cm $=\res{1{,}75\text{ km}}$.

\ex{2} $450:18=\res{25\text{ cm}}$. \ $A=40\cdot18^2=40\cdot324=12\,960\text{ cm}^2\approx\res{1{,}30\text{ m}^2}$.

\ex{3} $k=2$: \ capacitat $0{,}25\cdot2^3=\res{2\text{ L}}$; \ etiqueta $60\cdot2^2=\res{240\text{ cm}^2}$.

\ex{3} 1 cm del mapa son $50\,000$ cm $=0{,}5$ km; \ 1 cm² son $0{,}5\cdot0{,}5=0{,}25$ km². \ $6\cdot0{,}25=\res{1{,}5\text{ km}^2}=\res{150\text{ ha}}$.
