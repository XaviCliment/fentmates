\begin{minipage}[c]{0.57\linewidth}
\begin{idea}[Poliedro]
Un \textbf{poliedro} es un cuerpo limitado por \textbf{polígonos}:\\[3pt]
\cA{\textbf{caras}}: los polígonos; \ \cR{\textbf{aristas}}: los lados de las caras; \ \cK{\textbf{vértices}}: donde se encuentran las aristas.\\[4pt]
Es \textbf{convexo} si todas las caras se pueden apoyar sobre una mesa; si no, \textbf{cóncavo}.
\end{idea}

\subsection{Los cinco poliedros regulares}
\begin{minipage}[c]{0.62\linewidth}\centering\small
\renewcommand{\arraystretch}{1.3}
\taularod{>{\heavy}l<{\hspace{4pt}}>{\hspace{4pt}}c<{\hspace{4pt}}>{\hspace{4pt}}c<{\hspace{4pt}}>{\hspace{4pt}}c<{\hspace{4pt}}>{\hspace{4pt}}l<{\hspace{4pt}}>{\hspace{4pt}}c<{\hspace{4pt}}}{
\rowcolor{fmblue}\color{white}Poliedro & \color{white}\heavy $C$ & \color{white}\heavy $V$ & \color{white}\heavy $A$ & \color{white}\heavy Caras & \color{white}\heavy Por vértice\\
Tetraedro & 4 & 4 & 6 & triángulos & 3\\
\rowcolor{fmsurface}Cubo & 6 & 8 & 12 & cuadrados & 3\\
Octaedro & 8 & 6 & 12 & triángulos & 4\\

}
\end{minipage}\hfill
\begin{minipage}[c]{0.35\linewidth}
\begin{truc}[¿Por qué solo cinco?]
En cada vértice, los ángulos de las caras tienen que sumar \textbf{menos de $360^\circ$}. Tres hexágonos ya suman $360^\circ$: queda plano.
\end{truc}
\end{minipage}

\vfill
\begin{exemple}[Ejemplos]
\begin{minipage}[t]{0.48\linewidth}
Pirámide hexagonal: \ $C=7$, \ $V=7$, \ $A=12$\\ $7+7=12+2$ \ \be{}\\[4pt]
9 caras y 16 aristas: \ $9+V=18\ \to\ \boxed{V=9}$
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
Prisma de 30 aristas: \ $3n=30\to n=10$\\
base \textbf{decágono}: \ $C=\boxed{12}$, \ $V=\boxed{20}$
\end{minipage}
\end{exemple}

\vfill

\begin{minipage}[c]{0.6\linewidth}
\begin{idea}[Prisma]
Dos \textbf{bases} iguales y paralelas, y \textbf{caras laterales} que son paralelogramos. La \textbf{altura} $h$ es la distancia entre las bases.\\[3pt]
\textbf{Recto}: caras laterales rectangulares. \ \textbf{Regular}: recto y con bases regulares.\\[3pt]
\textbf{Ortoedro}: una caja. \ \textbf{Cubo}: todas las caras cuadradas.\\[4pt]
Diagonal del ortoedro: \ $\boxed{D=\sqrt{a^2+b^2+c^2}}$
\end{idea}

\end{tikzpicture}\\
{\comic\small\color{fmmuted}desarrollo de un prisma hexagonal}

{\large$A_L=P_{\text{base}}\cdot h$}\par\medskip
{\large$A_T=A_L+2\,A_{\text{base}}$}\par\smallskip

\end{tcolorbox}

Apotema de la base: \ $a=\sqrt{4^2-2^2}=\sqrt{12}\approx3{,}46$ \qquad $A_L=24\cdot10=240$ cm$^2$\\[3pt]

\end{exemple}

\begin{minipage}[t]{0.48\linewidth}
Ortoedro $8\times5\times3$:\\ $2\,(40+24+15)=\boxed{158\text{ cm}^2}$\\[4pt]

\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
Pintar paredes y techo de una habitación $5\times4\times2{,}5$ m (puerta $0{,}8\times2$, ventana $1{,}5\times1{,}2$):\\
$45+20-1{,}6-1{,}8=61{,}6$ m$^2$; \ botes de 12 m$^2$: \boxed{6\text{ botes}}
\end{minipage}
\end{exemple}

\begin{minipage}[c]{0.57\linewidth}
\begin{idea}[Pirámide regular]
Base regular y caras laterales triángulos isósceles iguales.\\[3pt]
\cK{\textbf{Apotema de la pirámide} $a_p$}: la altura de cada cara lateral.\\
\cB{\textbf{Apotema de la base} $a_b$}. \ ¡No las confundas!\\[4pt]
Forman un triángulo rectángulo con la altura:\\[2pt]
{\centering\Large$a_p^2=h^2+a_b^2$\par}
\end{idea}

{\Large$A_L=\dfrac{P_{\text{base}}\cdot a_p}{2}\qquad\qquad A_T=A_L+A_{\text{base}}$}\par\smallskip
{\small el desarrollo es la base y $n$ triángulos de base $l$ y altura $a_p$}
\end{tcolorbox}

\vfill
\begin{exemple}[Ejemplo: \ pirámide cuadrangular regular de $l=6$ cm y $h=4$ cm]
$a_b=3$, \ $a_p=\sqrt{4^2+3^2}=5$ cm \qquad $A_L=\dfrac{24\cdot5}{2}=60$ cm$^2$ \qquad $A_T=60+36=\boxed{96\text{ cm}^2}$
\end{exemple}

\end{minipage}\hfill
\begin{minipage}[c]{0.63\linewidth}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Tronco de pirámide}]
Cortamos una pirámide con un plano paralelo a la base: dos bases semejantes y caras laterales \textbf{trapecios}.\\[3pt]
{\centering$A_L=\dfrac{(P+P')\cdot a_p}{2}\qquad A_T=A_L+A_B+A_{B'}$\par}
\end{tcolorbox}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ tronco cuadrangular regular, aristas de las bases 10 cm y 4 cm, altura 4 cm]
$a_p=\sqrt{4^2+3^2}=5$ \qquad $A_L=\dfrac{(40+16)\cdot5}{2}=140$ \qquad $A_T=140+100+16=\boxed{256\text{ cm}^2}$
\end{exemple}

Cilindro $r=3$, $h=10$: \ $90\pi\approx282{,}7$ cm$^3\approx0{,}28$ L
\end{minipage}
\end{exemple}

$V=25\cdot12\cdot1{,}8=540$ m$^3=\boxed{540\,000\text{ L}}$ \qquad $540\,000:500=1080$ min $=\boxed{18\text{ horas}}$
\end{exemple}

\seccio{sec7}{Volumen de pirámides, conos y esferas}{Una pirámide es la tercera parte del prisma.}

\vfill
\begin{minipage}[c]{0.42\linewidth}\centering
\begin{tikzpicture}[scale=0.62]
  \begin{scope}\cilindre{1}{2.6}{fmA}\con{1}{2.6}{fmB}\end{scope}
  \node[font=\Large] at (2.1,1.3) {$=$};
  \begin{scope}[xshift=3.4cm]\con{0.55}{2.6}{fmB}\end{scope}
  \begin{scope}[xshift=4.6cm]\con{0.55}{2.6}{fmB}\end{scope}
  \begin{scope}[xshift=5.8cm]\con{0.55}{2.6}{fmB}\end{scope}
  \node[below,font=\scriptsize] at (4.6,-0.4) {tres conos llenos};
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.55\linewidth}
\begin{idea}[Un tercio]
Con una pirámide llena de agua, hay que verter \textbf{tres} veces para llenar el prisma de la misma base y altura. Igual con el cono y el cilindro.
\end{idea}
\end{minipage}

\vfill
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=3mm,bottom=3mm]
{\Large$V_{\text{pirámide}}=\dfrac{A_{\text{base}}\cdot h}{3}\qquad V_{\text{cono}}=\dfrac{\pi r^2h}{3}\qquad V_{\text{esfera}}=\dfrac43\pi r^3$}
\end{tcolorbox}

\vfill
\begin{truc}[Arquímedes, otra vez]
La esfera es $\frac23$ del cilindro que la envuelve: \ $\frac23\cdot\pi r^2\cdot2r=\frac43\pi r^3$.
\end{truc}

\vfill
\begin{exemple}[Ejemplos]
Pirámide cuadrangular de $l=6$ cm y $h=4$ cm: \ $V=\dfrac{36\cdot4}{3}=\boxed{48\text{ cm}^3}$\\[5pt]
Cono de $r=5$ cm y $h=12$ cm: \ $V=\dfrac{\pi\cdot25\cdot12}{3}=100\pi\approx\boxed{314{,}16\text{ cm}^3}$\\[5pt]
Esfera de $r=3$ cm: \ $V=\frac43\pi\cdot27=36\pi\approx\boxed{113{,}10\text{ cm}^3}$ \qquad Pelota de $r=11$ cm: \ $\frac43\pi\cdot1331\approx\boxed{5{,}6\text{ L}}$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ cono de generatriz 13 cm y radio 5 cm. ¿Qué esfera tiene el mismo volumen?]
$h=\sqrt{13^2-5^2}=12$ \ $\to$ \ $V=\dfrac{\pi\cdot25\cdot12}{3}=100\pi$ \qquad $\dfrac43\pi r^3=100\pi\ \to\ r^3=75\ \to\ r\approx\boxed{4{,}22\text{ cm}}$
\end{exemple}

\vfill
\begin{compte}[No te olvides del 3]
Pirámide y cono: \textbf{entre 3}. Si no lo pones, calculas el volumen del prisma o del cilindro.
\end{compte}

\seccio{sec8}{Cuerpos compuestos y problemas}{Descomponemos en cuerpos conocidos y sumamos (o restamos).}

\vfill
\begin{minipage}[c]{0.3\linewidth}\centering
\begin{tikzpicture}[scale=0.62]
  \cilindre{1.2}{3.2}{fmA}
  \begin{scope}[yshift=3.2cm]\fill[fmB!25] (-1.2,0) -- (0,1.6) -- (1.2,0) -- cycle;\draw[fmB,line width=1.2pt] (-1.2,0) -- (0,1.6) -- (1.2,0);\end{scope}
  \node[right,font=\scriptsize] at (1.2,1.6) {6 m}; \node[right,font=\scriptsize] at (0.7,4) {3 m}; \node[below,font=\scriptsize] at (0.6,-0.4) {$r=2$};
\end{tikzpicture}\\
{\comic\small\color{fmmuted}silo}
\end{minipage}\hfill
\begin{minipage}[c]{0.67\linewidth}
\begin{idea}[Cuerpos compuestos]
\textbf{Volumen}: sumamos los volúmenes de los cuerpos simples (y \textbf{restamos} si hay agujeros).\\[3pt]
\textbf{Área}: sumamos solo las superficies que quedan \textbf{por fuera}.
\end{idea}
\begin{exemple}[Ejemplo: \ silo: cilindro + cono]
$V=\pi\cdot4\cdot6+\dfrac{\pi\cdot4\cdot3}{3}=24\pi+4\pi=28\pi\approx\boxed{87{,}96\text{ m}^3}$
\end{exemple}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ un helado: cono de $r=3$ cm y $h=10$ cm, lleno, con media esfera encima]
$V=\dfrac{\pi\cdot9\cdot10}{3}+\dfrac12\cdot\dfrac43\pi\cdot27=30\pi+18\pi=48\pi\approx\boxed{150{,}8\text{ cm}^3}$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ un cubo de madera de 10 cm con un agujero cilíndrico de 2 cm de radio que lo atraviesa]
$V=1000-\pi\cdot4\cdot10=1000-40\pi\approx\boxed{874{,}3\text{ cm}^3}$\\[3pt]
Área: \ $600-2\cdot4\pi+2\pi\cdot2\cdot10=600+32\pi\approx\boxed{700{,}5\text{ cm}^2}$ \ \nota{6 caras, menos los 2 círculos, más el interior}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ cubo en forma de tronco de cono: fondo $r=10$ cm, boca $r=15$ cm, altura 20 cm]
\begin{minipage}[c]{0.65\linewidth}
Es un cono grande menos un cono pequeño. Por Tales: \ $\dfrac{x}{10}=\dfrac{x+20}{15}\ \to\ x=40$\\[3pt]
$V=\dfrac{\pi\cdot225\cdot60}{3}-\dfrac{\pi\cdot100\cdot40}{3}=\dfrac{9500\pi}{3}\approx9948$ cm$^3\approx\boxed{9{,}95\text{ L}}$
\end{minipage}\hfill
