\seccio{1}{Poliedres. Fórmula d'Euler. Poliedres regulars}

\ex{1} \begin{center}\renewcommand{\arraystretch}{1.25}\begin{tabular}{|l|C{1.6cm}|C{1.6cm}|C{1.6cm}|l|}\hline Poliedre & $C$ & $V$ & $A$ & $C+V=A+2$ \\\hline prisma triangular & 5 & 6 & 9 & $11=11$ \\\hline piràmide pentagonal & 6 & 6 & 10 & $12=12$ \\\hline prisma hexagonal & 8 & 12 & 18 & $20=20$ \\\hline piràmide octogonal & 9 & 9 & 16 & $18=18$ \\\hline cub & 6 & 8 & 12 & $14=14$ \\\hline\end{tabular}\end{center}

\ex{1} \begin{tasks}(3)
  \task $8+12=A+2\Rightarrow\res{A=18}$
  \task $C+10=17\Rightarrow\res{C=7}$
  \task $12+V=32\Rightarrow\res{V=20}$
\end{tasks}

\ex{1} \begin{tasks}(3)
  \task \rest{sí}
  \task no (rectangles diferents)
  \task \rest{sí}
  \task no (cares de dos tipus)
  \task \rest{sí}
  \task \rest{sí}
\end{tasks}Un poliedre regular té totes les cares polígons regulars iguals i el mateix nombre de cares a cada vèrtex.

\ex{2} $3n=24\Rightarrow n=8$: base \rest{octogonal}; \ $\res{C=10}$, \ $\res{V=16}$ \ ($10+16=24+2$).

\ex{2} $n+1=11\Rightarrow n=10$: base \rest{decagonal}; \ $\res{C=11}$, \ $\res{A=20}$ \ ($11+11=20+2$).

\ex{2} a) $7+7=14=12+2$: \rest{sí} (per exemple, una piràmide hexagonal). \ b) $5+6=11\neq12$: \rest{no}.

\ex{3} Quadrangulars: $\res{C=8,\ V=6,\ A=12}$ (és un octaedre si les cares són equilàters). \ Pentagonals: $\res{C=10,\ V=7,\ A=15}$. \ En tots dos casos $C+V=A+2$.

\ex{3} Cada angle d'un hexàgon regular fa $120^\circ$. En un vèrtex s'hi han d'ajuntar com a mínim 3 cares, i $3\cdot120^\circ=360^\circ$: les cares quedarien \rest{planes} i no tancarien cap cos.

\seccio{2}{Prismes: elements i àrea}

\ex{1} \begin{tasks}(2)
  \task $6\cdot4^2=\res{96\text{ cm}^2}$
  \task $2(6\cdot3+6\cdot4+3\cdot4)=2\cdot54=\res{108\text{ cm}^2}$
\end{tasks}

\ex{1} $D=\sqrt{144+16+9}=\sqrt{169}=\res{13\text{ cm}}$.

\ex{1} $A_L=20\cdot12=\res{240\text{ cm}^2}$; \ $A_T=240+2\cdot25=\res{290\text{ cm}^2}$.

\ex{2} $A_L=12\cdot10=120$; altura de la base $\sqrt{16-4}=\sqrt{12}\approx3{,}46$; $A_{\text{base}}=\frac{4\cdot\sqrt{12}}{2}\approx6{,}93$. \ $A_T\approx120+2\cdot6{,}93=\res{133{,}86\text{ cm}^2}$.

\ex{2} $A_L=36\cdot8=288$; \ $a_b=\sqrt{36-9}=\sqrt{27}\approx5{,}20$; \ $A_{\text{base}}=\frac{36\cdot\sqrt{27}}{2}\approx93{,}53$. \ $A_T\approx288+187{,}06=\res{475{,}06\text{ cm}^2}$.

\ex{2} Base $40\cdot30=1200$; \ cares $2\cdot40\cdot20+2\cdot30\cdot20=2800$. \ Total: $\res{4000\text{ cm}^2}$.

\ex{3} $D^2=3l^2\Rightarrow3l^2=48\Rightarrow\res{l=4\text{ cm}}$; \ $A_T=6\cdot16=\res{96\text{ cm}^2}$.

\ex{3} Hipotenusa: $\sqrt{36+64}=10$. \ $A_L=(6+8+10)\cdot15=360$; \ $A_{\text{base}}=\frac{6\cdot8}{2}=24$. \ $A_T=360+48=\res{408\text{ cm}^2}$.

\seccio{3}{Piràmides i troncs de piràmide}

\ex{1} $a_b=5$; \ $a_p=\sqrt{144+25}=\res{13\text{ cm}}$. \ $A_T=\frac{40\cdot13}{2}+100=\res{360\text{ cm}^2}$.

\ex{1} $A_L=\frac{24\cdot9}{2}=\res{108\text{ cm}^2}$.

\ex{2} $h=\sqrt{5^2-3^2}=\res{4\text{ cm}}$; \ $A_T=\frac{24\cdot5}{2}+36=\res{96\text{ cm}^2}$.

\ex{2} $a_p=\sqrt{36+16}=\sqrt{52}\approx\res{7{,}21\text{ cm}}$; \ aresta: $\sqrt{a_p^2+4^2}=\sqrt{68}\approx\res{8{,}25\text{ cm}}$; \ $A_T=\frac{32\cdot\sqrt{52}}{2}+64\approx\res{179{,}38\text{ cm}^2}$.

\ex{2} Cada cara és un triangle equilàter d'altura $\sqrt{36-9}=\sqrt{27}\approx5{,}20$: \ $A=4\cdot\frac{6\cdot\sqrt{27}}{2}\approx\res{62{,}35\text{ cm}^2}$.

\ex{3} $a_p=\sqrt{4^2+3^2}=\res{5\text{ cm}}$ \ (catets: l'altura i $\frac{12-6}{2}=3$). \ $A_T=\frac{(48+24)\cdot5}{2}+144+36=\res{360\text{ cm}^2}$.

\ex{3} $a_p=\sqrt{146^2+115^2}=\sqrt{34\,541}\approx\res{185{,}85\text{ m}}$; \ $A_L=\frac{920\cdot185{,}85}{2}\approx\res{85\,492\text{ m}^2}$.

\ex{1} $g=\sqrt{144+25}=\res{13\text{ cm}}$; \ $A_L=\pi\cdot5\cdot13=65\pi\approx\res{204{,}20\text{ cm}^2}$; \ $A_T=65\pi+25\pi=90\pi\approx\res{282{,}74\text{ cm}^2}$.

\ex{1} $h=\sqrt{100-36}=\res{8\text{ cm}}$; \ $A_T=60\pi+36\pi=96\pi\approx\res{301{,}59\text{ cm}^2}$.

\ex{2} $r=4$: \ $A_T=2\pi\cdot4\cdot12+2\pi\cdot16=96\pi+32\pi=128\pi\approx\res{402{,}12\text{ cm}^2}$.

\ex{2} $g=\sqrt{81+144}=15$; \ només la superfície lateral: $A_L=\pi\cdot9\cdot15=135\pi\approx\res{424{,}12\text{ cm}^2}$.

\ex{2} $2\pi r\cdot10=120\pi\Rightarrow\res{r=6\text{ cm}}$; \ $A_T=120\pi+72\pi=192\pi\approx\res{603{,}19\text{ cm}^2}$.

\ex{3} L'arc del sector fa $2\pi\cdot4=8\pi$ i la circumferència sencera de radi 12 faria $24\pi$: és $\frac{8\pi}{24\pi}=\frac13$ del cercle. \ Angle: $\frac{360^\circ}{3}=\res{120^\circ}$.

\ex{3} $A=12\cdot2\pi\cdot0{,}4\cdot5=48\pi\approx150{,}80\text{ m}^2$; \ $\frac{150{,}80}{4}\approx\res{37{,}70\text{ L}}$.

\ex{2} $V=\frac{\pi\cdot2{,}5^2\cdot12}{3}+\frac12\cdot\frac43\pi\cdot2{,}5^3=25\pi+\frac{125\pi}{12}=\frac{425\pi}{12}\approx\res{111{,}26\text{ cm}^3}$.

\ex{2} $V=\pi(6^2-5^2)\cdot50=550\pi\approx\res{1727{,}88\text{ cm}^3}$.

\ex{2} $V=1000-\pi\cdot9\cdot10=1000-90\pi\approx\res{717{,}26\text{ cm}^3}$.

\ex{3} $V=\pi\cdot4\cdot5+\frac12\cdot\frac43\pi\cdot8=20\pi+\frac{16\pi}{3}=\frac{76\pi}{3}\approx\res{79{,}59\text{ m}^3}$. \ $\frac{79\,587}{50}\approx1591{,}7$ min $\approx\res{26{,}5\text{ hores}}$.
