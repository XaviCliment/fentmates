\begin{idea}[Espacio muestral]
Cada resultado posible es un \textbf{resultado elemental}. Todos juntos forman el \textbf{espacio muestral} $E$:\\[4pt]
\begin{minipage}[t]{0.5\linewidth}
Moneda: \ $E=\{C,\,X\}$\\[2pt]
Dado: \ $E=\{1,2,3,4,5,6\}$
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}
Bola roja, azul o verde: \ $E=\{R,\,A,\,V\}$\\[2pt]
Dos monedas: \ $E=\{CC,\,CX,\,XC,\,XX\}$
\end{minipage}
\end{idea}

\vfill
\begin{center}
\taularod{>{\heavy}l<{\hspace{5pt}}>{\hspace{5pt}}l<{\hspace{5pt}}>{\hspace{5pt}}l<{\hspace{4pt}}}{
\rowcolor{fmblue}\color{white}Suceso & \color{white}\heavy Qué es & \color{white}\heavy Ejemplo con un dado\\
Elemental & un solo resultado & «sacar un 5» $=\{5\}$\\
\rowcolor{fmsurface}Compuesto & dos o más resultados & «sacar par» $=\{2,4,6\}$\\
Seguro & ocurre siempre: es $E$ & «sacar menos de 7» $=E$\\
\rowcolor{fmsurface}Imposible & no ocurre nunca: $\varnothing$ & «sacar un 8» $=\varnothing$
}
\end{center}

Cada camino es un resultado: $E$ tiene $2\cdot2\cdot2=8$.\\[4pt]
«Al menos dos» quiere decir dos o tres:\\[3pt]
$A=\{CCC,\,CCX,\,CXC,\,XCC\}$
\end{minipage}
\end{exemple}

\clearpage
\seccio{sec2}{Probabilidad. Propiedades}{Un número entre 0 y 1 que dice cómo de esperable es.}

\vfill
\begin{idea}[Frecuencias]
Repetimos un experimento $n$ veces y $A$ ocurre $f$ veces: $f$ es la \textbf{frecuencia absoluta} y \ $h(A)=\dfrac{f}{n}$ \ la \textbf{frecuencia relativa}.
\end{idea}

\vfill
\begin{center}\small
\taularod{>{\heavy}l<{\hspace{3pt}}*{6}{>{\hspace{3pt}}c<{\hspace{3pt}}}>{\hspace{3pt}}c<{\hspace{3pt}}}{
\rowcolor{fmblue}\color{white}Dado, 60 lanzamientos & \color{white}1 & \color{white}2 & \color{white}3 & \color{white}4 & \color{white}5 & \color{white}6 & \color{white}Total\\
Frec. absoluta $f$ & 8 & 11 & 9 & 12 & 10 & 10 & 60\\
\rowcolor{fmsurface}Frec. relativa $h$ & $0{,}13$ & $0{,}18$ & $0{,}15$ & $0{,}20$ & $0{,}17$ & $0{,}17$ & 1
}
\end{center}
{\centering\comic\color{fmmuted}frecuencia relativa de «par»: \ $\frac{11+12+10}{60}=0{,}55$\par}

\vfill
\begin{center}
\begin{tikzpicture}[x=1.45cm]
  \shade[left color=fmR!45,right color=fmK!55] (0,0) rectangle (10,0.45);
  \draw[fmnavy,line width=1pt] (0,0) rectangle (10,0.45);
  \foreach \x/\v/\t in {0/0/imposible,2.5/{0,25}/poco probable,5/{0,5}/tan probable como no,7.5/{0,75}/muy probable,10/1/seguro} {
    \draw[fmnavy,line width=1pt] (\x,0) -- (\x,-0.15); \node[below,font=\small\heavy] at (\x,-0.15) {$\v$}; \node[below=14pt,font=\scriptsize,align=center,text width=2cm] at (\x,-0.15) {\t};}
\end{tikzpicture}
\end{center}

\vfill
\begin{minipage}[t]{0.52\linewidth}
\begin{idea}[Propiedades]
\faCheck\ $0\le P(A)\le1$\\[2pt]
\faCheck\ $P(E)=1$ \ y \ $P(\varnothing)=0$\\[2pt]
\faCheck\ Las probabilidades de los resultados elementales \textbf{suman 1}.\\[2pt]
\faCheck\ $P(\text{compuesto})$ = suma de las de sus resultados:\\ \hspace*{1em}$P(\text{par})=P(2)+P(4)+P(6)$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[t]{0.45\linewidth}
\begin{truc}[¿Cómo la calculamos?]
Si los resultados son \textbf{equiprobables}: ley de \textbf{Laplace} (12.3).\\[3pt]
Si no: con la \textbf{frecuencia relativa} de muchas repeticiones (12.4).
\end{truc}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ dado trucado con $P(1)=0{,}1$, $P(2)=0{,}2$, $P(3)=0{,}1$, $P(4)=0{,}2$, $P(5)=0{,}1$]
$P(6)=1-(0{,}1+0{,}2+0{,}1+0{,}2+0{,}1)=\boxed{0{,}3}$ \qquad $P(\text{par})=0{,}2+0{,}2+0{,}3=\boxed{0{,}7}$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ ruleta con $P(\text{rojo})=0{,}35$, $P(\text{azul})=0{,}25$, $P(\text{verde})=0{,}1$ y amarillo]
$P(\text{amarillo})=1-0{,}7=\boxed{0{,}3}$ \qquad $P(\text{azul o amarillo})=0{,}25+0{,}3=\boxed{0{,}55}$
\end{exemple}

\seccio{sec3}{Experimentos regulares. Ley de Laplace}{Si todos los resultados son igual de probables, contamos.}

\vfill
\begin{minipage}[c]{0.6\linewidth}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=3mm,bottom=3mm]
{\Large$P(A)=\dfrac{\text{casos favorables}}{\text{casos posibles}}$}\par\smallskip
{\small solo en experimentos \textbf{regulares}: resultados equiprobables}
\end{tcolorbox}
\end{minipage}\hfill
\begin{minipage}[c]{0.37\linewidth}\centering
\foreach \i in {1,...,5}{\bola{fmR}\,}\\[2pt]
\foreach \i in {1,...,3}{\bola{fmA}\,}\ \bola{fmK}\,\bola{fmK}\\[3pt]
{\comic\small\color{fmmuted}5 rojas, 3 azules, 2 verdes}
\end{minipage}

\vfill
\begin{tcbraster}[raster columns=3,raster equal height,raster column skip=8pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faDiceSix\ \ Dado},before upper=\centering]
$P(\text{par})=\dfrac36=\dfrac12$\par\smallskip
$P(\text{múltiple de 3})=\dfrac26=\dfrac13$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faShoppingBag\ \ Bolsa},before upper=\centering]
$P(R)=\dfrac{5}{10}=\dfrac12$\par\smallskip
$P(\text{no verde})=\dfrac{8}{10}=\dfrac45$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faIcon{clone}\ \ Baraja (40)},before upper=\centering]
$P(\text{oros})=\dfrac{10}{40}=\dfrac14$\par\smallskip
$P(\text{figura})=\dfrac{12}{40}=\dfrac{3}{10}$
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{compte}[Solo si son equiprobables]
Una ruleta con sectores distintos, un dado trucado o una chincheta \textbf{no} son regulares. ¡«Aprobar o suspender» tiene dos resultados, pero no tiene por qué ser $\frac12$ cada uno!
\end{compte}

\vfill
\begin{exemple}[Ejemplos]
\begin{minipage}[t]{0.48\linewidth}
Urna con 4 rojas, 6 azules y 5 verdes (15):\\
$P(R)=\frac{4}{15}$, \ $P(\text{azul})=\frac{6}{15}=\frac25$, \ $P(\text{no azul})=\frac{9}{15}=\frac35$\\[5pt]
Dado: \ $P(\text{primo})=\frac36=\frac12$ \ \nota{2, 3, 5}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
Letras de PROBABILIDAD (12 tarjetas):\\
$P(A)=\frac{2}{12}=\frac16$, \ $P(\text{vocal})=\frac{5}{12}$\\[5pt]
Copas \textbf{o} figura: \ $10+12-3=19$ \ \nota{sin repetir}\\
$P=\boxed{\tfrac{19}{40}}$
\end{minipage}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ 3 bolas rojas y unas cuantas azules, con $P(R)=\frac15$. ¿Cuántas azules hay?]
$\dfrac3n=\dfrac15\ \to\ n=15$ bolas \ $\to$ \ $15-3=\boxed{12\text{ azules}}$\\[5pt]
¿Cuántas rojas hay que añadir para que $P(R)=\frac12$? \ $\dfrac{3+x}{15+x}=\dfrac12\ \to\ 6+2x=15+x\ \to\ \boxed{x=9}$
\end{exemple}

\seccio{sec4}{Ley de los grandes números}{Si no podemos contar, experimentamos muchas veces.}

\vfill
\begin{idea}[Ley de los grandes números]
Si repetimos un experimento \textbf{muchas veces}, la frecuencia relativa se estabiliza alrededor de un valor: la \textbf{probabilidad}.\\[3pt]
{\centering\Large$P(A)\approx h(A)$ \ \normalsize si $n$ és gran\par}
\end{idea}

\vfill
\begin{minipage}[c]{0.5\linewidth}\centering\small
\taularod{>{\heavy}l<{\hspace{3pt}}*{6}{>{\hspace{3pt}}c<{\hspace{3pt}}}}{
\rowcolor{fmblue}\color{white}Chincheta: $n$ & \color{white}10 & \color{white}50 & \color{white}100 & \color{white}200 & \color{white}500 & \color{white}1000\\
Punta arriba & 7 & 32 & 61 & 126 & 309 & 621\\
\rowcolor{fmsurface}$h$ & $0{,}7$ & $0{,}64$ & $0{,}61$ & $0{,}63$ & $0{,}618$ & $0{,}621$
}
\end{minipage}\hfill
\begin{minipage}[c]{0.46\linewidth}\centering
\begin{tikzpicture}[x=0.95cm,y=9cm]

{\large Al principio la $h$ cambia mucho; después se acerca a $0{,}62$: \ $P(\text{punta arriba})\approx0{,}62$ \ y \ $P(\text{de lado})\approx0{,}38$.}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={\faBullseye\ \ Hacer previsiones}]
Si repetimos $n$ veces, esperamos que $A$ ocurra unas \ $\boxed{n\cdot P(A)}$ \ veces.\par\smallskip
600 dados: \ $600\cdot\frac16=100$ seises \textbf{aproximadamente}.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmR!6,colframe=fmR,coltitle=white,title={\faBrain\ \ El azar no tiene memoria}]
Después de cinco cruces seguidas, la probabilidad de cara sigue siendo $\frac12$. La ley habla de muchísimas repeticiones, no de «compensar».
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplos]
\begin{minipage}[t]{0.48\linewidth}
Acierta 168 de 240 tiros libres:\\
$P\approx\frac{168}{240}=0{,}7$ \ $\to$ \ en 50 tiros, unos $\boxed{35}$\\[5pt]
Chincheta 500 veces: de lado unas $500\cdot0{,}38=\boxed{190}$
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
20 bolas; en 500 extracciones (devolviéndola), 198 rojas:\\
$\frac{x}{20}\approx\frac{198}{500}\approx0{,}4$ \ $\to$ \ unas $\boxed{8\text{ rojas}}$\\[5pt]
Dado con 502 seises de 1000: \ esperaríamos unos 167 \ $\to$ \textbf{trucado}
\end{minipage}
\end{exemple}

\vfill
\begin{truc}[Buffon y Pearson]
Buffon lanzó una moneda 4040 veces: 2048 caras ($h\approx0{,}507$). Pearson, 24\,000 veces: 12\,012 caras ($h=0{,}5005$). Las dos se acercan a $\frac12$.
\end{truc}

\seccio{sec5}{Experimentos compuestos}{Varios experimentos seguidos: árbol o tabla.}

\vfill
\begin{idea}[Experimentos compuestos]
Están formados por varios experimentos simples: dos monedas, una moneda y un dado, dos dados\dots\ \ El número de resultados es el \textbf{producto}: \ moneda y dado, $2\cdot6=12$; \ dos dados, $6\cdot6=36$.
\end{idea}

\vfill
\begin{minipage}[c]{0.45\linewidth}\centering\small
\newcommand{\sv}{\cellcolor{fmyellow!70}\heavy 7}
\renewcommand{\arraystretch}{1.25}
\taularod{>{\columncolor{fmblue}\color{white}\heavy}c*{6}{w{c}{0.55cm}}}{

2 & 3 & 4 & 5 & 6 & \sv & 8\\
3 & 4 & 5 & 6 & \sv & 8 & 9\\
4 & 5 & 6 & \sv & 8 & 9 & 10\\
5 & 6 & \sv & 8 & 9 & 10 & 11\\
6 & \sv & 8 & 9 & 10 & 11 & 12
}
\end{minipage}\hfill
\begin{minipage}[c]{0.52\linewidth}
\textbf{Dos dados}: 36 resultados equiprobables.\\[4pt]
$P(\text{suma }7)=\dfrac{6}{36}=\dfrac16$\qquad $P(\text{suma }2)=\dfrac{1}{36}$\\[6pt]
\begin{compte}[Las sumas no son equiprobables]
El 7 es la más probable. Con $E=\{2,\dots,12\}$ y Laplace saldría $\frac1{11}$: \mal{\text{incorrecto}}
\end{compte}
\end{minipage}

\vfill
\begin{minipage}[c]{0.5\linewidth}
\begin{idea}[Regla del producto]
Si los experimentos son \textbf{independientes}:\\[3pt]
\pas{1} En cada rama, su probabilidad.\\[2pt]
\pas{2} Un camino: \textbf{multiplicamos} las ramas.\\[2pt]
\pas{3} Varios caminos: \textbf{sumamos}.
\end{idea}
\end{minipage}\hfill

{\comic\small\color{fmmuted}dos tiros libres al 70\,\%}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ dos tiros libres, acierta el 70\,\% ($A$ acierto, $F$ fallo)]
Los dos: \ $0{,}7\cdot0{,}7=\boxed{0{,}49}$ \qquad Ninguno: \ $0{,}3\cdot0{,}3=\boxed{0{,}09}$ \qquad Exactamente uno: \ $0{,}21+0{,}21=\boxed{0{,}42}$
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
\begin{minipage}[t]{0.48\linewidth}
Moneda y dado: \ $P(C\text{ y }6)=\frac12\cdot\frac16=\boxed{\tfrac{1}{12}}$\\[4pt]
Tres monedas, exactamente dos caras:\\ 3 caminos de $\frac18$: \ $\boxed{\tfrac38}$; \ al menos dos: $\boxed{\tfrac48=\tfrac12}$

Suma 10 o más: \ $\frac{3+2+1}{36}=\boxed{\tfrac16}$\\[3pt]
Doble: \ $\frac{6}{36}=\boxed{\tfrac16}$ \quad Al menos un 6: \ $\boxed{\tfrac{11}{36}}$
\end{minipage}
\end{exemple}

\begin{minipage}[c]{0.42\linewidth}\raggedright

\pas{1} ¿Cuál es el experimento? ¿Simple o compuesto? ¿Equiprobable?\\[3pt]

\end{exemple}
