\seccio{2}{Probabilitat d'un esdeveniment. Propietats}

\ex{1} $h$: \ $\res{0{,}14}$, \ $\res{0{,}18}$, \ $\res{0{,}16}$, \ $\res{0{,}2}$, \ $\res{0{,}12}$, \ $\res{0{,}2}$ \ (sumen 1). \quad $h(\text{parell})=\frac{9+10+10}{50}=\res{0{,}58}$.

\ex{1} Una probabilitat sempre està entre 0 i 1. Poden ser-ho: \rest{$0{,}7$, \ $0$, \ $100\,\%$ i $\frac{99}{100}$}. No poden ser-ho: $1{,}2$, $-0{,}1$, $\frac32$ i $\frac54$.

\ex{1} $P(\text{verd})=1-(0{,}5+0{,}3)=\res{0{,}2}$. \ $P(\text{roig o verd})=0{,}5+0{,}2=\res{0{,}7}$.

\ex{2} $P(\text{groc})=1-(0{,}35+0{,}25+0{,}1)=\res{0{,}3}$. \ $P(\text{no roig})=P(\text{blau})+P(\text{verd})+P(\text{groc})=0{,}25+0{,}1+0{,}3=\res{0{,}65}$. \ $P(\text{blau o groc})=0{,}25+0{,}3=\res{0{,}55}$.

\ex{2} Si $P(1)=\dots=P(5)=x$: \ $5x+0{,}5=1\Rightarrow x=\res{0{,}1}$. \ $P(\text{parell})=0{,}1+0{,}1+0{,}5=\res{0{,}7}$. \ $P(\text{menys de 3})=0{,}1+0{,}1=\res{0{,}2}$.

\ex{3} Les sis probabilitats sumen 1: $p+2p+\dots+6p=21p=1\Rightarrow p=\res{\frac1{21}}$. \ $P(6)=\frac6{21}=\res{\frac27}$. \ $P(\text{parell})=\frac{2+4+6}{21}=\frac{12}{21}=\res{\frac47}$.

\ex{3} Si $P(\text{blava})=x$: \ $2x+x+0{,}4=1\Rightarrow x=0{,}2$. \ $P(\text{roja})=\res{0{,}4}$, \ $P(\text{blava})=\res{0{,}2}$. \ Amb 20 boles: \res{8 roges, 4 blaves i 8 verdes}.
