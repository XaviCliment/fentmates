\seccio{2}{Operacions amb esdeveniments}

\ex{1} $A=\{3,\,6\}$, $B=\{1,\,2\}$: \ $A\cup B=\res{\{1,\,2,\,3,\,6\}}$, \ $A\cap B=\res{\varnothing}$, \ $\overline{A}=\res{\{1,\,2,\,4,\,5\}}$, \ $\overline{B}=\res{\{3,\,4,\,5,\,6\}}$. \ Són \rest{incompatibles}, perquè no tenen cap element comú.

\ex{1} $A=\{2,4,6,8,10\}$, $B=\{2,3,5,7\}$: \ $A\cup B=\res{\{2,3,4,5,6,7,8,10\}}$, \ $A\cap B=\res{\{2\}}$, \ $\overline{A}=\res{\{1,3,5,7,9\}}$, \ $\overline{B}=\res{\{1,4,6,8,9,10\}}$.

\ex{1} $A\cup B$: «parell o major que 4» $=\res{\{2,4,5,6\}}$. \ $A\cap B$: «parell i major que 4» $=\res{\{6\}}$. \ $\overline{A}$: «no parell», és a dir, senar $=\res{\{1,3,5\}}$.

\ex{2} $O\cap A$ = l'as d'oros: \res{1 carta}. \ $O\cup A$ = els 10 oros i els altres 3 asos: $10+4-1=\res{13\text{ cartes}}$. \ $\overline{O}$ = les que no són d'oros: \res{30 cartes}.

\ex{2} $A=\{CCC,\,CCX,\,CXC,\,XCC\}$, \ $B=\{XCC,\,XCX,\,XXC,\,XXX\}$. \ $A\cap B=\res{\{XCC\}}$; \ $A\cup B=\res{E-\{CXX\}}$ (7 resultats); \ $\overline{A}=\res{\{CXX,\,XCX,\,XXC,\,XXX\}}$.

\seccio{3}{Probabilitat d'un esdeveniment. Propietats}

\ex{1} $h$: \ $\res{0{,}14}$, \ $\res{0{,}18}$, \ $\res{0{,}16}$, \ $\res{0{,}2}$, \ $\res{0{,}12}$, \ $\res{0{,}2}$ \ (sumen 1). \quad $h(\text{parell})=\frac{9+10+10}{50}=\res{0{,}58}$.

\ex{1} \begin{tasks}(3)
  \task $P(\overline{A})=\res{0{,}38}$
  \task $P(\overline{B})=\res{\frac47}$
  \task $P(\overline{C})=\res{85\,\%}$
\end{tasks}

\ex{1} Una probabilitat sempre està entre 0 i 1. Poden ser-ho: \rest{$0{,}7$, \ $0$, \ $100\,\%$ i $\frac{99}{100}$}. No poden ser-ho: $1{,}2$, $-0{,}1$, $\frac32$ i $\frac54$.

\ex{2} $P(\text{groc})=1-(0{,}35+0{,}25+0{,}1)=\res{0{,}3}$. \ $P(\text{no roig})=1-0{,}35=\res{0{,}65}$. \ Són incompatibles: $P(\text{blau o groc})=0{,}25+0{,}3=\res{0{,}55}$.

\ex{2} $P(A\cup B)=0{,}4+0{,}5-0{,}15=\res{0{,}75}$. \ $P(\overline{A\cup B})=1-0{,}75=\res{0{,}25}$. \ $P(A\cap\overline{B})=0{,}4-0{,}15=\res{0{,}25}$.

\ex{2} a) $P(A\cup B)=0{,}3+0{,}45=\res{0{,}75}$. \ b) \rest{No}: si foren incompatibles, $P(A\cup B)=0{,}6+0{,}5=1{,}1>1$, i això és impossible.

\ex{2} a) $0{,}45+0{,}60-0{,}25=\res{0{,}8}$. \ b) $1-0{,}8=\res{0{,}2}$. \ c) $0{,}60-0{,}25=\res{0{,}35}$.

\ex{3} $P(A)=1-0{,}3=\res{0{,}7}$, \ $P(B)=1-0{,}4=\res{0{,}6}$. \ $0{,}9=0{,}7+0{,}6-P(A\cap B)\Rightarrow P(A\cap B)=\res{0{,}4}$. \ \rest{No} són incompatibles ($P(A\cap B)\neq0$).

\ex{3} Les sis probabilitats sumen 1: $p+2p+\dots+6p=21p=1\Rightarrow p=\res{\frac1{21}}$. \ $P(6)=\frac6{21}=\res{\frac27}$. \ $P(\text{parell})=\frac{2+4+6}{21}=\frac{12}{21}=\res{\frac47}$.
