\begin{minipage}[t]{0.5\linewidth}
\begin{idea}[Regla dels signes]
Base \textbf{positiva} $\to$ sempre positiu.\\[2pt]
Base \textbf{negativa}: exponent \textbf{parell} $\to$ \cK{positiu}; \textbf{senar} $\to$ \cR{negatiu}.\\[4pt]
{\centering$(-2)^4=\cK{16}\qquad (-2)^3=\cR{-8}$\par}
\end{idea}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{compte}[Compte amb els parèntesis!]
\be{(-2)^4=16}\qquad \be{-2^4=-16}\\[3pt]
Sense parèntesi, l'exponent afecta \textbf{només el nombre}, no el signe.
\end{compte}
\end{minipage}

$10^n$: un 1 i $n$ zeros\par\smallskip
$10^6=1\,000\,000$
\end{tcolorbox}
\end{tcbraster}

\begin{exemple}[Exemples]
\begin{minipage}[t]{0.32\linewidth}
$(-5)^2=25$\\[3pt]
$-(-3)^3=-(-27)=27$
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}
$(-1)^{7}=-1$ \ \nota{senar}\\[3pt]
$(-1)^{20}=1$ \ \nota{parell}
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}
$\left(-\dfrac25\right)^{3}=-\dfrac{8}{125}$\\[3pt]
$0^5=0\qquad 1^{37}=1$
\end{minipage}
\end{exemple}

\begin{exemple}[Exemple: \ calcula]
\begin{minipage}[t]{0.19\linewidth}\centering $2^5=32$\end{minipage}\hfill
\begin{minipage}[t]{0.19\linewidth}\centering $(-3)^4=81$\end{minipage}\hfill
\begin{minipage}[t]{0.19\linewidth}\centering $-3^4=\cR{-81}$\end{minipage}\hfill
\begin{minipage}[t]{0.19\linewidth}\centering $\left(\dfrac12\right)^{4}=\dfrac{1}{16}$\end{minipage}\hfill
\begin{minipage}[t]{0.19\linewidth}\centering $10^3=1000$\end{minipage}
\end{exemple}

\begin{truc}[Com es llig?]
$a^n$ es llig «$a$ elevat a $n$». \ $5^2$: «cinc al quadrat». \ $2^3$: «dos al cub». \ $7^4$: «set elevat a quatre».
\end{truc}

\clearpage
\seccio{sec2}{Propietats de les potències}{Per a multiplicar i dividir, juguem amb els exponents.}

\newcommand{\prop}[3]{\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={#1},before upper=\centering]
{\Large$#2$}\par\smallskip{\small$#3$}
\end{tcolorbox}}
\begin{tcbraster}[raster columns=3,raster equal height=rows,raster column skip=8pt]
\prop{\pas{1}\ Producte}{a^{m}\cdot a^{n}=a^{\cK{m+n}}}{2^{5}\cdot2^{3}=2^{8}=256}
\prop{\pas{2}\ Quocient}{a^{m}:a^{n}=a^{\cK{m-n}}}{7^{9}:7^{7}=7^{2}=49}
\prop{\pas{3}\ Potència de potència}{\left(a^{m}\right)^{n}=a^{\cK{m\cdot n}}}{\left(5^{2}\right)^{3}=5^{6}=15\,625}
\prop{\pas{4}\ Mateix exponent}{a^{n}\cdot b^{n}=(\cA{a\cdot b})^{n}}{2^{4}\cdot5^{4}=10^{4}=10\,000}
\prop{\pas{5}\ Mateix exponent}{a^{n}:b^{n}=(\cA{a:b})^{n}}{15^{3}:5^{3}=3^{3}=27}
\prop{Exponent zero}{a^{0}=1\ \ (a\neq0)}{a^{n}:a^{n}=a^{0}=1}
\end{tcbraster}

\begin{compte}[Només per a productes i quocients, no per a sumes!]
\mal{2^{2}+2^{3}=2^{5}}\ \ {\small$4+8=12$, \ però $2^5=32$}\qquad\quad
\mal{(2+3)^{2}=2^{2}+3^{2}}\ \ {\small$25$, \ però $4+9=13$}
\end{compte}

\subsection{Bases diferents: descompon en factors primers}
\begin{minipage}[c]{0.42\linewidth}\raggedright
\pas{1} \textbf{Descompon} cada base: \ $12=2^2\cdot3$, \ $8=2^3$\dots\\[6pt]
\pas{2} \textbf{Reparteix} l'exponent a cada factor: \ $12^{3}=2^{6}\cdot3^{3}$.\\[6pt]
\pas{3} \textbf{Agrupa} les potències de la mateixa base i calcula.
\end{minipage}\hfill
\begin{minipage}[c]{0.55\linewidth}
\begin{exemple}[Exemple: \ $\dfrac{8^{2}\cdot4^{3}}{16^{2}}$]
$=\dfrac{\left(2^{3}\right)^{2}\cdot\left(2^{2}\right)^{3}}{\left(2^{4}\right)^{2}}=\dfrac{2^{6}\cdot2^{6}}{2^{8}}=\dfrac{2^{12}}{2^{8}}=2^{4}=\boxed{16}$
\end{exemple}
\end{minipage}

\begin{exemple}[Exemple: \ $\dfrac{6^{3}\cdot9^{2}}{4\cdot27^{2}}$]
$=\dfrac{(\cA{2}\cdot\cB{3})^{3}\cdot\left(\cB{3}^{2}\right)^{2}}{\cA{2}^{2}\cdot\left(\cB{3}^{3}\right)^{2}}=\dfrac{\cA{2}^{3}\cdot\cB{3}^{3}\cdot\cB{3}^{4}}{\cA{2}^{2}\cdot\cB{3}^{6}}=\dfrac{\cA{2}^{3}\cdot\cB{3}^{7}}{\cA{2}^{2}\cdot\cB{3}^{6}}=\cA{2}^{3-2}\cdot\cB{3}^{7-6}=2\cdot3=\boxed{6}$
\end{exemple}

\begin{exemple}[Exemple: \ $\dfrac{12^{2}\cdot18^{3}}{2^{5}\cdot27^{2}}$]
$=\dfrac{\left(2^{2}\cdot3\right)^{2}\cdot\left(2\cdot3^{2}\right)^{3}}{2^{5}\cdot\left(3^{3}\right)^{2}}=\dfrac{2^{4}\cdot3^{2}\cdot2^{3}\cdot3^{6}}{2^{5}\cdot3^{6}}=\dfrac{2^{7}\cdot3^{8}}{2^{5}\cdot3^{6}}=2^{2}\cdot3^{2}=\boxed{36}$
\end{exemple}

% ============================================================ 2.3
\clearpage
\seccio{sec3}{Potències d'exponent negatiu}{Un exponent negatiu «gira» la base.}

\begin{minipage}[c]{0.5\linewidth}
D'on ix? \ Per la propietat del quocient, \ $2^{3}:2^{5}=2^{-2}$.\\ I si simplifiquem:\par\smallskip
{\centering$\dfrac{2^{3}}{2^{5}}=\dfrac{\cancel{2}\cdot\cancel{2}\cdot\cancel{2}}{\cancel{2}\cdot\cancel{2}\cdot\cancel{2}\cdot2\cdot2}=\dfrac{1}{2^{2}}$\par}\smallskip
Per tant, \ $2^{-2}=\dfrac{1}{2^{2}}$.
\end{minipage}\hfill
\begin{minipage}[c]{0.46\linewidth}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=3mm,bottom=3mm]
{\Large$a^{-n}=\dfrac{1}{a^{n}}$}\par\medskip
{\Large$\left(\dfrac{a}{b}\right)^{-n}=\left(\dfrac{b}{a}\right)^{n}$}
\end{tcolorbox}
\end{minipage}

\begin{compte}[El resultat NO es fa negatiu]
L'exponent negatiu \textbf{gira} la base, però no canvia el signe: \qquad \be{2^{-3}=\dfrac18}\qquad \mal{2^{-3}=-8}
\end{compte}

\begin{exemple}[Exemples]
\begin{minipage}[t]{0.32\linewidth}
$2^{-3}=\dfrac{1}{2^{3}}=\dfrac18$\\[5pt]
$(-3)^{-2}=\dfrac{1}{(-3)^{2}}=\dfrac19$
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}
$\left(\dfrac25\right)^{-2}=\left(\dfrac52\right)^{2}=\dfrac{25}{4}$\\[5pt]
$\left(-\dfrac12\right)^{-3}=(-2)^{3}=-8$
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}
$10^{-4}=\dfrac{1}{10\,000}=0{,}0001$\\[5pt]
I al revés: \ $\dfrac18=2^{-3}$
\end{minipage}
\end{exemple}

\begin{idea}[Les propietats també valen amb exponents negatius]
Compte amb els signes en sumar i restar exponents:\\[4pt]
{\centering$3^{-2}\cdot3^{5}=3^{-2+5}=3^{3}\qquad 5^{-3}:5^{-7}=5^{-3-(-7)}=5^{4}\qquad \left(2^{-3}\right)^{-2}=2^{6}\qquad 6^{-2}:3^{-2}=2^{-2}=\dfrac14$\par}
\end{idea}

\begin{minipage}[t]{0.49\linewidth}
\begin{exemple}[Exemple: \ $\dfrac{4^{-2}\cdot8^{3}}{16^{-1}\cdot2^{5}}$]
$=\dfrac{2^{-4}\cdot2^{9}}{2^{-4}\cdot2^{5}}=\dfrac{2^{5}}{2^{1}}=2^{4}=\boxed{16}$
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}
\begin{exemple}[Exemple: \ $\left(\dfrac23\right)^{-2}\cdot\left(\dfrac49\right)^{3}$]
$=\dfrac{3^{2}}{2^{2}}\cdot\dfrac{2^{6}}{3^{6}}=\dfrac{2^{4}}{3^{4}}=\boxed{\dfrac{16}{81}}$
\end{exemple}
\end{minipage}

\begin{exemple}[Exemple: \ $\dfrac{6^{-2}\cdot9^{3}}{4^{-1}\cdot3^{5}}$]
$=\dfrac{(2\cdot3)^{-2}\cdot\left(3^{2}\right)^{3}}{\left(2^{2}\right)^{-1}\cdot3^{5}}=\dfrac{2^{-2}\cdot3^{-2}\cdot3^{6}}{2^{-2}\cdot3^{5}}=\dfrac{2^{-2}\cdot3^{4}}{2^{-2}\cdot3^{5}}=2^{0}\cdot3^{-1}=\boxed{\dfrac13}$
\end{exemple}

\end{minipage}

\seccio{sec5}{Radicals: extracció i introducció de factors}{L'arrel desfà la potència.}

\begin{minipage}[c]{0.5\linewidth}\centering
\begin{tikzpicture}
  \node[font=\fontsize{40}{40}\selectfont,inner sep=3pt] (r) {$\sqrt[\cB{n}]{\cA{a}}=\cK{b}$};
  \node[font=\Large,anchor=north] at ([yshift=-0.2cm]r.south) {perquè \ $\cK{b}^{\cB{n}}=\cA{a}$};
  \node[text=fmB,font=\large\heavy,anchor=east] (i) at ([xshift=-0.5cm,yshift=0.6cm]r.west) {índex};
  \node[text=fmA,font=\large\heavy,anchor=south] (ra) at ([xshift=-0.4cm,yshift=0.4cm]r.north) {radicand};
  \draw[-{Latex[length=2.5mm]},fmB,line width=1.2pt] (i.east) -- ([xshift=14pt,yshift=8pt]r.west);
  \draw[-{Latex[length=2.5mm]},fmA,line width=1.2pt] (ra.south) -- ([xshift=-0.15cm,yshift=-4pt]r.north);
\end{tikzpicture}\\[4pt]
$\sqrt{49}=7\qquad \sqrt[3]{-8}=-2\qquad \sqrt[4]{81}=3$
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
\begin{idea}[Quantes arrels hi ha?]
\textbf{Índex parell}, radicand positiu: dues, oposades ($\pm7$); el símbol $\sqrt{\ }$ indica la positiva.\\[3pt]
\textbf{Índex parell}, radicand negatiu: \textbf{cap} real: \ $\sqrt{-4}\notin\mathbb{R}$.\\[3pt]
\textbf{Índex senar}: una, amb el signe del radicand.
\end{idea}
\end{minipage}

\subsection{Extreure factors}
\begin{minipage}[c]{0.45\linewidth}\raggedright
\pas{1} \textbf{Descompon} el radicand en factors primers.\\[6pt]
\pas{2} \textbf{Divideix} cada exponent entre l'índex.\\[6pt]
\pas{3} El \textbf{quocient} ix fora; el \textbf{residu} es queda dins.
\end{minipage}\hfill
\begin{minipage}[c]{0.52\linewidth}\centering
\begin{tikzpicture}
  \node[font=\fontsize{18}{18}\selectfont] (a) {$\sqrt{2^{5}}=\sqrt{\cA{2^{2}}\cdot\cA{2^{2}}\cdot2}=\cA{2\cdot2}\,\sqrt{2}=\boxed{4\sqrt2}$};
  \node[font=\comic,text=fmmuted,anchor=north] at ([yshift=-2pt]a.south) {$5:2=$ \textbf{2} (fora), residu \textbf{1} (dins)};
\end{tikzpicture}\\[2pt]
{\comic\color{fmblue}cada grup de $n$ factors iguals ix fora com un de sol}
\end{minipage}

\begin{exemple}[Exemples d'extracció]
\begin{minipage}[t]{0.32\linewidth}
$\sqrt{72}=\sqrt{2^{3}\cdot3^{2}}=\boxed{6\sqrt2}$\\[5pt]
$\sqrt[3]{54}=\sqrt[3]{2\cdot3^{3}}=\boxed{3\sqrt[3]{2}}$
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}
$\sqrt[4]{2^{9}\cdot3^{5}}=2^{2}\cdot3\sqrt[4]{2\cdot3}=\boxed{12\sqrt[4]{6}}$\\[5pt]
$\sqrt{a^{5}\,b^{2}}=\boxed{a^{2}\,b\sqrt{a}}$
\end{minipage}\hfill
\begin{minipage}[t]{0.32\linewidth}
$\sqrt{\dfrac{50}{9}}=\dfrac{\sqrt{2\cdot5^{2}}}{\sqrt{3^{2}}}=\boxed{\dfrac{5\sqrt2}{3}}$
\end{minipage}
\end{exemple}

\subsection{Introduir factors}
\begin{minipage}[t]{0.45\linewidth}
\begin{idea}
{\centering\Large$a\sqrt[n]{b}=\sqrt[n]{a^{\cB{n}}\cdot b}$\par}\smallskip
El factor de fora entra \textbf{elevat a l'índex}:\\[3pt]
$3\sqrt2=\sqrt{3^2\cdot2}=\sqrt{18}$\qquad $2\sqrt[3]{5}=\sqrt[3]{40}$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[t]{0.52\linewidth}
\begin{truc}[Per a comparar radicals]
Quin és major, \ $3\sqrt2$ \ o \ $2\sqrt5$?\\[3pt]
$3\sqrt2=\sqrt{18}$ \ i \ $2\sqrt5=\sqrt{20}$ \ $\to$ \ és major \ $2\sqrt5$.\\[3pt]
\notab{Amb índex parell, el signe $-$ es queda fora: $-3\sqrt2=-\sqrt{18}$.}
\end{truc}
\end{minipage}

% ============================================================ 2.6
\clearpage
\seccio{sec6}{Operacions amb radicals}{Sumar, només els semblants; multiplicar, si tenen el mateix índex.}

\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faPlus\,\faMinus\ \ Suma i resta: radicals semblants}]
Mateix \textbf{índex} i mateix \textbf{radicand}. Sumem els coeficients, com \ $3x+5x=8x$:\par\smallskip
{\centering$3\sqrt2+5\sqrt2-\sqrt2=\boxed{7\sqrt2}$\par}\smallskip
Si no són semblants, primer \textbf{extraiem factors}.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faTimes\,\faDivide\ \ Producte i quocient: mateix índex}]
Multipliquem (o dividim) els \textbf{radicands}:\par\smallskip
{\centering$\sqrt[n]{a}\cdot\sqrt[n]{b}=\sqrt[n]{a\cdot b}\qquad \dfrac{\sqrt[n]{a}}{\sqrt[n]{b}}=\sqrt[n]{\dfrac{a}{b}}$\par}\smallskip
Al final, extraiem factors si es pot.
\end{tcolorbox}
\end{tcbraster}

\begin{compte}[Error molt freqüent]
\mal{\sqrt{a}+\sqrt{b}=\sqrt{a+b}}\qquad\quad $\sqrt9+\sqrt{16}=3+4=\cK{7}$, \ però \ $\sqrt{9+16}=\sqrt{25}=\cR{5}$
\end{compte}

\begin{exemple}[Exemple: \ $\sqrt{50}-\sqrt{18}+\sqrt{8}$]
$=\sqrt{2\cdot5^{2}}-\sqrt{2\cdot3^{2}}+\sqrt{2^{3}}=5\sqrt2-3\sqrt2+2\sqrt2=(5-3+2)\sqrt2=\boxed{4\sqrt2}$
\end{exemple}

\begin{exemple}[Exemple: \ $\sqrt{12}+\sqrt{20}-\sqrt{27}+\sqrt{45}$ \ \ (dos tipus de radicals)]
$=\cA{2\sqrt3}+\cB{2\sqrt5}-\cA{3\sqrt3}+\cB{3\sqrt5}=(2-3)\cA{\sqrt3}+(2+3)\cB{\sqrt5}=\boxed{-\sqrt3+5\sqrt5}$
\end{exemple}

\begin{exemple}[Exemples de productes i quocients]
\begin{minipage}[t]{0.48\linewidth}
$\sqrt2\cdot\sqrt8=\sqrt{16}=4$\\[3pt]
$\sqrt6\cdot\sqrt{15}=\sqrt{90}=3\sqrt{10}$\\[3pt]
$\left(3\sqrt{10}\right)\cdot\left(2\sqrt5\right)=6\sqrt{50}=30\sqrt2$
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
$\sqrt{72}:\sqrt2=\sqrt{36}=6$\\[3pt]
$\dfrac{6\sqrt{15}}{2\sqrt3}=\dfrac62\sqrt{\dfrac{15}{3}}=3\sqrt5$\\[3pt]
$\left(\sqrt[3]{2}\right)^{5}=\sqrt[3]{2^{5}}=2\sqrt[3]{4}$
\end{minipage}
\end{exemple}

\begin{exemple}[{Exemple: \ $2\sqrt[3]{16}+\sqrt[3]{54}$}]
$=2\sqrt[3]{2^{4}}+\sqrt[3]{2\cdot3^{3}}=2\cdot2\sqrt[3]{2}+3\sqrt[3]{2}=4\sqrt[3]{2}+3\sqrt[3]{2}=\boxed{7\sqrt[3]{2}}$
\end{exemple}

\begin{truc}[Per a saber-ne més: índexs diferents]
Primer els passem a \textbf{índex comú} (el m.c.m.): \quad $\sqrt2\cdot\sqrt[3]{3}=\sqrt[6]{2^{3}}\cdot\sqrt[6]{3^{2}}=\sqrt[6]{72}$
\end{truc}

\seccio{sec7}{Intervals}{Un «tros» de la recta real, amb infinits nombres.}

\begin{minipage}[c]{0.5\linewidth}
\begin{idea}[Claudàtors]
\chip{fmK}{$[\ \ ]$} \ \textbf{tancat}: l'extrem \textbf{sí} hi és \ ($\le$, $\ge$) \ \ {\color{fmK}\faCircle}\\[6pt]
\chip{fmR}{$]\ \ [$} \ \textbf{obert}: l'extrem \textbf{no} hi és \ ($<$, $>$) \ \ {\color{fmR}\faCircle[regular]}\\[6pt]
$+\infty$ i $-\infty$ no són nombres: sempre amb claudàtor \textbf{obert}.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.46\linewidth}
\begin{exemple}[Exemples]
$x\ge-3\ \to\ [-3,+\infty\ro$\\[3pt]
$-1<x\le4\ \to\ \lo-1,4]$\\[3pt]
«Majors que 2 i menors que 7» $\to\ \lo2,7\ro$\\[3pt]
El 4 és de $\lo-1,4]$? \ \textbf{Sí}. \ I de $[-2,4\ro$? \ \textbf{No}.
\end{exemple}
\end{minipage}

\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{truc}[Una altra manera]
Alguns llibres, en lloc de punts, dibuixen directament el \textbf{claudàtor} sobre la recta. Les dues maneres són correctes.
\end{truc}
\end{minipage}

% ============================================================ 2.9
\clearpage
\seccio{sec9}{Unió i intersecció d'intervals}{Tot el que està pintat, o només la part comuna.}

\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmA!8,colframe=fmA,coltitle=white,title={Unió \ $A\cup B$: «o»}]
Els nombres que són d'$A$ \textbf{o} de $B$ (almenys d'un). És \textbf{tot el que està pintat}.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={Intersecció \ $A\cap B$: «i»}]
Els nombres que són d'$A$ \textbf{i} de $B$ alhora. És \textbf{la part comuna}. Si no n'hi ha: $\varnothing$.
\end{tcolorbox}
\end{tcbraster}

{\large Dibuixem els dos intervals \textbf{un davall de l'altre}, amb la mateixa escala, i mirem.} Fixa't bé en els extrems!

\newcommand{\ui}[7]{\begin{exemple}[{Exemple: \ #1}]
\begin{minipage}[c]{0.52\linewidth}
\begin{tabular}{@{}r@{\ \ }l@{}}
$A$ & #2\\[2pt] $B$ & #3\\[2pt] \cA{$A\cup B$} & #4\\[2pt] \cK{$A\cap B$} & #5
\end{tabular}
\end{minipage}\hfill
\begin{minipage}[c]{0.45\linewidth}
{\large$A\cup B=#6$}\\[6pt]
{\large$A\cap B=#7$}
\end{minipage}
\end{exemple}}

\begin{compte}[Els extrems]
En l'últim exemple, el 2 \textbf{no} és d'$A$ però \textbf{sí} de $B$: està en la unió, però no en la intersecció.
\end{compte}

\begin{idea}
{\centering\Large$\mathbb{N}\subset\mathbb{Z}\subset\mathbb{Q}\subset\mathbb{R}\qquad \mathbb{I}\subset\mathbb{R}\qquad \mathbb{Q}\cap\mathbb{I}=\varnothing$\par}
\smallskip A cada nombre real li correspon un \textbf{punt} de la recta, i a cada punt, un nombre real.
\end{idea}

\begin{compte}[Casos que enganyen]
\begin{tabular}{@{}p{0.47\linewidth}p{0.47\linewidth}@{}}
$\sqrt9=3$ és \textbf{natural}, encara que tinga arrel. \ $-\sqrt{16}=-4$ és \textbf{enter}. &
$\dfrac{22}{7}$ és \textbf{racional}, encara que s'assemble a $\pi$.\\[8pt]
$0{,}121\,212\dots=0{,}\overline{12}$ \ té període $\to$ \textbf{racional}. &
$0{,}121\,121\,112\dots$ \ (un 1 més cada vegada) $\to$ \textbf{irracional}.\\[4pt]
\multicolumn{2}{@{}l}{$\sqrt{-4}$ \ \textbf{no} és un nombre real.}
\end{tabular}
\end{compte}

\chip{fmR}{$\sqrt7\in\mathbb{I}$}\ \ \chip{fmV}{$0{,}1\overline{6}\in\mathbb{Q}$}\ \ \chip{fmR}{$\tfrac{\pi}{2}\in\mathbb{I}$}
\end{exemple}

\end{truc}

\rtile{2.1}{Potències}{$a^n$: $n$ factors iguals. Base negativa: parell $+$, senar $-$. \ $-2^4\neq(-2)^4$.}
\rtile{2.2}{Propietats}{$a^m\cdot a^n=a^{m+n}$, \ $a^m:a^n=a^{m-n}$, \ $(a^m)^n=a^{m\cdot n}$, \ $a^0=1$.}
\rtile{2.3}{Exponent negatiu}{$a^{-n}=\dfrac{1}{a^n}$, \ $\left(\frac{a}{b}\right)^{-n}=\left(\frac{b}{a}\right)^{n}$.\\[1ex] No fa negatiu el resultat.}
\rtile{2.4}{Notació científica}{$a\cdot10^n$, amb $1\le|a|<10$. Gran: $n>0$. Xicotet: $n<0$.}
\rtile{2.5}{Radicals}{Extraure: exponent entre índex. Introduir: elevat a l'índex.}
\rtile{2.6}{Operacions}{Sumar: només semblants. Multiplicar: mateix índex. \ $\sqrt a+\sqrt b\neq\sqrt{a+b}$.}
\rtile{2.7}{Intervals}{$[\ ]$: l'extrem hi és. \ $]\ [$: no hi és. \ $\pm\infty$: sempre obert.}
\rtile{2.8}{Gràfica}{Punt ple: hi és. Punt buit: no hi és. Fletxa: fins a l'infinit.}
\rtile{2.9}{Unió i intersecció}{$A\cup B$: tot el pintat. \ $A\cap B$: la part comuna.}
\rtile{2.10}{Reals}{$\mathbb{R}=\mathbb{Q}\cup\mathbb{I}$. Irracionals: decimals infinits no periòdics.}
\end{tcbraster}
