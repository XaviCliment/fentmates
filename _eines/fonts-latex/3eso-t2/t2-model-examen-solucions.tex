\begin{tcolorbox}[colback=soft,colframe=soft,arc=2mm,left=3mm,right=3mm,top=1.5mm,bottom=1.5mm]
\small\textbf{Criteris generals}
\begin{itemize}[leftmargin=1.3em,itemsep=1pt,topsep=2pt]
  \item Un resultat sense cap procediment o justificació no puntua.
  \item Si el procediment és correcte però hi ha un error de càlcul aïllat, l'apartat val la meitat. Si l'error canvia el tipus de problema (per exemple, sumar exponents en una suma de potències), l'apartat val 0.
  \item Un error arrossegat d'un apartat anterior només es penalitza una vegada.
  \item Els resultats s'han de donar simplificats: fraccions irreductibles i radicals amb tots els factors extrets.
\end{itemize}
\end{tcolorbox}

\pregunta{1 punt}{Calcula les potències.}
\begin{tasks}(4)
  \task $(-2)^{5}=\res{-32}$
  \task $-2^{4}=\res{-16}$
  \task $\left(-\dfrac13\right)^{3}=\res{-\dfrac{1}{27}}$
  \task $(-1)^{48}=\res{1}$
\end{tasks}
\criteri{0,25 per apartat. En b), si s'eleva també el signe ($-2^{4}=16$), 0 punts. En c), si el signe és incorrecte, 0 punts.}

\pregunta{1,5 punts}{Aplica les propietats de les potències i calcula.}
\begin{tasks}(2)
  \task $5^{7-3-2}=5^{2}=\res{25}$
  \task $2^{-6}\cdot2^{8}=2^{2}=\res{4}$
\end{tasks}
\[\text{c)}\quad \frac{12^{3}\cdot9^{-1}}{8\cdot6^{2}}=\frac{\left(2^{2}\cdot3\right)^{3}\cdot\left(3^{2}\right)^{-1}}{2^{3}\cdot(2\cdot3)^{2}}=\frac{2^{6}\cdot3^{3}\cdot3^{-2}}{2^{3}\cdot2^{2}\cdot3^{2}}=\frac{2^{6}\cdot3}{2^{5}\cdot3^{2}}=2\cdot3^{-1}=\res{\frac{2}{3}}\]
\criteri{0,5 per apartat: 0,25 per aplicar correctament les propietats (en c), per descompondre bé les bases) i 0,25 pel resultat final.}

\pregunta{1 punt}{Calcula i dona el resultat en forma de fracció.}
\begin{tasks}(2)
  \task $\dfrac{1}{2^{2}}+3=\dfrac14+\dfrac{12}{4}=\res{\dfrac{13}{4}}$
  \task $\left(\dfrac52\right)^{2}-1=\dfrac{25}{4}-\dfrac44=\res{\dfrac{21}{4}}$
\end{tasks}
\criteri{0,5 per apartat: 0,25 per interpretar bé l'exponent negatiu (o l'exponent 0) i 0,25 per l'operació amb fraccions. Si s'interpreta l'exponent negatiu com un resultat negatiu ($2^{-2}=-4$), 0 punts.}

\pregunta{1 punt}{Notació científica.}
\begin{tasks}(1)
  \task $0,000\,38=\res{3,8\cdot10^{-4}}$\qquad $52\,300\,000=\res{5,23\cdot10^{7}}$
  \task $6,07\cdot10^{5}=\res{607\,000}$\qquad $1,2\cdot10^{-3}=\res{0,0012}$
\end{tasks}
\criteri{0,25 per cada nombre. Si el valor és correcte però $a$ no compleix $1\le|a|<10$ (per exemple, $38\cdot10^{-5}$), 0 punts.}

\pregunta{1 punt}{Radicals.}
\begin{tasks}(1)
  \task $\sqrt{300}=\sqrt{2^{2}\cdot3\cdot5^{2}}=2\cdot5\sqrt{3}=\res{10\sqrt{3}}$\qquad $\sqrt[3]{2^{5}\cdot3^{4}}=2\cdot3\sqrt[3]{2^{2}\cdot3}=\res{6\sqrt[3]{12}}$
  \task $4\sqrt{3}=\sqrt{4^{2}\cdot3}=\res{\sqrt{48}}$\qquad $2\sqrt[3]{5}=\sqrt[3]{2^{3}\cdot5}=\res{\sqrt[3]{40}}$
\end{tasks}
\criteri{0,25 per cada radical. Si només s'extrau una part dels factors (per exemple, $\sqrt{300}=5\sqrt{12}$), 0,125.}

\pregunta{1,5 punts}{Opera i simplifica.}
\begin{tasks}(1)
  \task $\sqrt{75}-2\sqrt{12}+\sqrt{27}=5\sqrt{3}-4\sqrt{3}+3\sqrt{3}=\res{4\sqrt{3}}$
  \task $\sqrt[3]{54}+\sqrt[3]{16}-\sqrt[3]{250}=3\sqrt[3]{2}+2\sqrt[3]{2}-5\sqrt[3]{2}=\res{0}$
  \task $3\sqrt{6}\cdot2\sqrt{15}=6\sqrt{90}=6\cdot3\sqrt{10}=\res{18\sqrt{10}}$
  \task $\sqrt{200}:\sqrt{8}=\sqrt{\dfrac{200}{8}}=\sqrt{25}=\res{5}$
\end{tasks}
\criteri{a) i b): 0,25 per extraure bé els factors i 0,25 per agrupar els radicals semblants. Si se sumen els radicands ($\sqrt{a}+\sqrt{b}=\sqrt{a+b}$), 0 punts. c) i d): 0,25 cadascun. En c), si no s'extrau el factor de $\sqrt{90}$, 0,125.}

\pregunta{1 punt}{Siguen $A=[-4,\ 1]$ i $B=(-1,\ 3)$. Representa'ls i calcula $A\cup B$ i $A\cap B$.}
\smallskip
{\centering
\begin{tabular}{@{}l@{\ \ }l@{\qquad}l@{\ \ }l@{}}
$A$ & \rectaint{-4}{1}{p}{p} & $A\cup B$ & \rectaint{-4}{3}{p}{b} \\[14pt]
$B$ & \rectaint{-1}{3}{b}{b} & $A\cap B$ & \rectaint{-1}{1}{b}{p}
\end{tabular}\par}
\medskip
$A\cup B=\res{[-4,\ 3)}\qquad\qquad A\cap B=\res{(-1,\ 1]}$
\criteri{0,5 per la unió i 0,5 per la intersecció. Si els extrems són correctes però un d'ells té el parèntesi o el claudàtor equivocat, 0,25. La representació gràfica és necessària per a obtindre la puntuació màxima.}
