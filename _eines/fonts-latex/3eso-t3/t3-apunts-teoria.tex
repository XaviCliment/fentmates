\begin{tcbraster}[raster columns=4,raster equal height,raster column skip=8pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,before upper=\centering,top=2mm,bottom=2mm]
{\Large\heavy\color{fmblue}$50\pc$}\par la \textbf{meitat}\par{\small$50\pc$ de $84=42$}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,before upper=\centering,top=2mm,bottom=2mm]
{\Large\heavy\color{fmblue}$25\pc$}\par la meitat de la meitat\par{\small$25\pc$ de $84=21$}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,before upper=\centering,top=2mm,bottom=2mm]
{\Large\heavy\color{fmblue}$10\pc$}\par dividim entre 10\par{\small$10\pc$ de $84=8{,}4$}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,before upper=\centering,top=2mm,bottom=2mm]
{\Large\heavy\color{fmblue}$1\pc$}\par dividim entre 100\par{\small$1\pc$ de $84=0{,}84$}
\end{tcolorbox}
\end{tcbraster}

\begin{minipage}[t]{0.5\linewidth}
\begin{truc}[Combina'ls]
$5\pc$: la meitat del $10\pc$. \ $20\pc$: el doble del $10\pc$.\\[4pt]
$15\pc$ de $60=\underbrace{6}_{10\pc}+\underbrace{3}_{5\pc}=9$
\end{truc}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{truc}[Es poden intercanviar]
El $p\pc$ de $C$ és igual que el $C\pc$ de $p$:\\[4pt]
el $8\pc$ de $50$ = el $50\pc$ de $8=\boxed{4}$
\end{truc}
\end{minipage}

\begin{exemple}[Exemple: \ en un institut de 640 alumnes, el $45\pc$ són xiques. Quantes n'hi ha?]
$45\pc$ de $640=640\cdot0{,}45=\boxed{288}$ xiques \qquad \nota{o bé: $\ 10\pc=64,\ \ 40\pc=256,\ \ 5\pc=32\ \to\ 256+32=288$}
\end{exemple}

\seccio{sec4}{Augments i disminucions percentuals}{Multipliquem directament per l'índex de variació.}

\begin{center}
\begin{tikzpicture}[x=0.075cm]
  \filldraw[fill=fmA!25,draw=fmA,line width=1.2pt] (0,0) rectangle (100,0.8); \node at (50,0.4) {\large$100\pc$ \ (inicial)};
  \filldraw[fill=fmK!30,draw=fmK,line width=1.2pt] (100,0) rectangle (115,0.8); \node at (107.5,0.4) {$+15$};
  \draw[decorate,decoration={brace,amplitude=5pt},fmK,line width=1.1pt] (0,0.95) -- (115,0.95) node[midway,above=6pt,font=\large,text=fmK!80!black]{augment del $15\pc$: \ $115\pc\ \to\ I_V=1{,}15$};
  \begin{scope}[yshift=-1.9cm]
  \filldraw[fill=fmA!25,draw=fmA,line width=1.2pt] (0,0) rectangle (80,0.8); \node at (40,0.4) {\large$80\pc$ \ (queda)};
  \filldraw[fill=fmR!8,draw=fmR,line width=1.2pt,dashed] (80,0) rectangle (100,0.8); \node[text=fmR] at (90,0.4) {$-20$};

  \end{scope}
\end{tikzpicture}
\end{center}

\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=3mm,bottom=3mm]
{\Large$C_f=C_i\cdot I_V$}\qquad\qquad augment: \ $I_V=1+\dfrac{p}{100}$\qquad\quad disminució: \ $I_V=1-\dfrac{p}{100}$
\end{tcolorbox}

\begin{minipage}[t]{0.5\linewidth}
\begin{exemple}[Exemples]
Un preu de 80 € puja un $15\pc$:\\[2pt]
\hspace*{1em}$80\cdot1{,}15=\boxed{92\text{ €}}$\\[4pt]
Una jaqueta de 50 € té un $20\pc$ de descompte:\\[2pt]
\hspace*{1em}$50\cdot0{,}8=\boxed{40\text{ €}}$
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{idea}[Llegir un índex de variació]
$I_V>1$: \textbf{augmenta}. \ $I_V<1$: \textbf{disminueix}.\\[4pt]
$1{,}15\to\cK{+15\pc}\qquad 0{,}92\to\cR{-8\pc}$\\[3pt]
$1{,}05\to\cK{+5\pc}\qquad\ \ 0{,}7\to\cR{-30\pc}$\\[3pt]
$2\to\cK{+100\pc}$
\end{idea}
\end{minipage}

\begin{minipage}[t]{0.5\linewidth}
\begin{truc}[I si coneixem l'inici i el final?]
$I_V=\dfrac{C_f}{C_i}$: \ de 40 € a 46 €, \ $\dfrac{46}{40}=1{,}15\to+15\pc$
\end{truc}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{compte}[Un sol pas]
No cal calcular l'augment i sumar-lo: multiplica per $I_V$ i ja ho tens.
\end{compte}
\end{minipage}

\begin{exemple}[Més exemples]
\begin{minipage}[t]{0.48\linewidth}
\textbf{IVA del $21\pc$}: un article val 50 € sense IVA.\\[3pt]
$50\cdot1{,}21=\boxed{60{,}50\text{ €}}$
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
Un poble de 25\,000 habitants perd el $4\pc$.\\[3pt]
$25\,000\cdot0{,}96=\boxed{24\,000\text{ habitants}}$
\end{minipage}
\end{exemple}

\seccio{sec7}{Interés compost}{Els interessos també produeixen interessos.}

\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Interés simple}]
Els interessos es calculen sempre sobre el capital \textbf{inicial} i es retiren.\par\smallskip
{\centering$I=\dfrac{C_0\cdot r\cdot t}{100}$\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={Interés compost}]
Cada any els interessos s'\textbf{afigen} al capital: és una variació encadenada.\par\smallskip
{\centering\Large$C_f=C_0\cdot\left(1+\dfrac{r}{100}\right)^{t}$\par}
\end{tcolorbox}
\end{tcbraster}
{\comic\color{fmmuted}\centering $C_0$: capital inicial \quad $r$: rèdit (\% anual) \quad $t$: anys \quad $I=C_f-C_0$: interessos\par}

\begin{exemple}[Exemple: \ dipositem 1000 € al $5\pc$ anual durant 3 anys]
\begin{minipage}[c]{0.48\linewidth}\centering
\begin{tikzpicture}[x=1.15cm,y=0.0042cm]

  \foreach \t/\v/\l in {0/1000/1000,1/1050/1050,2/1102.5/{1102{,}50},3/1157.625/{1157{,}63}} {
    \filldraw[fill=fmK!35,draw=fmK,line width=1pt] (\t-0.32,0) rectangle (\t+0.32,\v-800);
    \node[above,font=\small] at (\t,\v-800) {\l};
    \node[below,font=\small] at (\t,0) {any \t};}
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.49\linewidth}
\textbf{Compost}: \ $1000\cdot1{,}05^{3}=\boxed{1157{,}63\text{ €}}$\\[2pt]
\hspace*{1em}interessos: $157{,}63$ €\\[6pt]
\textbf{Simple}: \ $I=\dfrac{1000\cdot5\cdot3}{100}=150$ €\\[4pt]
\nota{Amb el compost guanyem més: els interessos creixen cada any.}
\end{minipage}
\end{exemple}

\begin{minipage}[t]{0.55\linewidth}
\begin{idea}[Si s'abona $n$ vegades l'any]
Mensual $n=12$, trimestral $n=4$, semestral $n=2$:\\[4pt]
{\centering\large$C_f=C_0\cdot\left(1+\dfrac{r}{100\cdot n}\right)^{n\cdot t}$\par}
\end{idea}
\end{minipage}\hfill
\begin{minipage}[t]{0.42\linewidth}
\begin{exemple}[Exemple]
1000 € al $6\pc$, trimestral, 2 anys:\\[3pt]
$1000\cdot\left(1+\dfrac{6}{400}\right)^{8}=$\\[2pt]
$=1000\cdot1{,}015^{8}\approx\boxed{1126{,}49\text{ €}}$
\end{exemple}
\end{minipage}

\begin{exemple}[Exemple: \ 2000 € al $3\pc$ anual durant 4 anys. Quants interessos guanyem?]
\begin{minipage}[t]{0.5\linewidth}
\textbf{Compost}: \ $2000\cdot1{,}03^{4}\approx2251{,}02$ €\\[2pt]
\hspace*{1em}$I=2251{,}02-2000=\boxed{251{,}02\text{ €}}$
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\textbf{Simple}: \ $I=\dfrac{2000\cdot3\cdot4}{100}=240$ €\\[2pt]
\nota{la diferència creix com més anys passen}
\end{minipage}
\end{exemple}

\seccio{sec8}{Repartiments proporcionals}{Repartir de manera justa: segons el que aporta cadascú.}

\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={\faArrowUp\ \ Directament proporcional}]
\textbf{Més} nombre, \textbf{més} part (el doble, el doble).\par\smallskip
\pas{1} Una part: \ $k=\dfrac{N}{a+b+c}$\\[3pt]
\pas{2} Cadascú: \ $k\cdot a$, \ $k\cdot b$, \ $k\cdot c$\\[3pt]
\pas{3} Comprovem que sumen $N$.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmB!8,colframe=fmB,coltitle=white,title={\faArrowDown\ \ Inversament proporcional}]
\textbf{Més} nombre, \textbf{menys} part (el doble, la meitat).\par\smallskip
És repartir directament als \textbf{inversos}: $\dfrac1a,\ \dfrac1b,\ \dfrac1c$\\[3pt]
{\centering$k=\dfrac{N}{\frac1a+\frac1b+\frac1c}\qquad x=\dfrac{k}{a}\dots$\par}
\end{tcolorbox}
\end{tcbraster}

\begin{exemple}[Exemple: \ repartim 900 € directament proporcional a 2, 3 i 4 (hores treballades)]

    \node[font=\scriptsize] at (\i+0.5,0.32) {100};}
  \node[below,font=\small,text=fmA] at (1,0) {2 parts};
  \node[below,font=\small,text=fmB] at (3.5,0) {3 parts};

\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
$k=\dfrac{900}{2+3+4}=100$ € cada part\\[4pt]
\cA{$200$ €}, \ \cB{$300$ €}, \ \cK{$400$ €}\\[2pt]
\nota{$200+300+400=900$ \be{}}
\end{minipage}
\end{exemple}

\begin{minipage}[t]{0.52\linewidth}
\begin{exemple}[Exemple: \ 600 € inversament a 2, 3 i 6]
Inversos: \ $\dfrac12,\ \dfrac13,\ \dfrac16=\dfrac36,\ \dfrac26,\ \dfrac16$\\[4pt]
Directament a \textbf{3, 2 i 1}: \ $k=\dfrac{600}{6}=100$\\[4pt]
$\boxed{300\text{ €}},\ \boxed{200\text{ €}},\ \boxed{100\text{ €}}$\\[2pt]
\nota{qui té el 2 (el menor) rep més}
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.45\linewidth}

Reduïm els inversos a \textbf{comú denominador} (m.c.m.) i repartim \textbf{directament} als numeradors.
\end{truc}
\end{minipage}

\begin{exemple}[Exemple: \ tres amigues aporten 10 €, 15 € i 25 € a una butlleta i guanyen 150 €]
\begin{minipage}[c]{0.5\linewidth}

\end{minipage}\hfill

$10\cdot3=\boxed{30\text{ €}}$, \ $15\cdot3=\boxed{45\text{ €}}$, \ $25\cdot3=\boxed{75\text{ €}}$\\[2pt]
\nota{$30+45+75=150$ \be{}}
\end{minipage}
\end{exemple}

\begin{minipage}[t]{0.5\linewidth}
\begin{idea}[Aliatges: la llei]
{\centering$\text{llei}=\dfrac{\text{massa de metall preciós}}{\text{massa total}}$\par}\smallskip
Un nombre entre 0 i 1. També serveix per a concentracions (\% de suc, de sal\dots).
\end{idea}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{truc}[Els quirats]
L'or pur té \textbf{24 quirats}. Un or de 18 k té llei \ $\dfrac{18}{24}=0{,}750$.\\[3pt]
\notab{Una peça de 40 g d'or de 18 k té $40\cdot0{,}75=30$ g d'or pur.}
\end{truc}
\end{minipage}

\begin{exemple}[Exemple: \ mesclem 2 L de suc al $40\pc$ amb 3 L al $20\pc$. Quina concentració té la mescla?]
$\dfrac{2\cdot0{,}40+3\cdot0{,}20}{2+3}=\dfrac{0{,}8+0{,}6}{5}=0{,}28\ \to\ \boxed{28\pc}$ \qquad \nota{entre el 20 i el 40, més a prop del 20}
\end{exemple}

\seccio{sec10}{Problemes de moviments}{Distància = velocitat $\times$ temps.}

\begin{minipage}[c]{0.45\linewidth}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=3mm,bottom=3mm]
{\fontsize{24}{28}\selectfont$d=v\cdot t$}\par\medskip
$v=\dfrac{d}{t}\qquad t=\dfrac{d}{v}$
\end{tcolorbox}
\end{minipage}\hfill
\begin{minipage}[c]{0.52\linewidth}
\begin{compte}[Compte amb les unitats!]
Si $v$ està en km/h, $t$ en \textbf{hores}: \ 2 h 15 min $=2{,}25$ h.\\[4pt]
De km/h a m/s, \textbf{dividim entre 3,6}:\\[2pt]
$72$ km/h $=\dfrac{72\,000\text{ m}}{3\,600\text{ s}}=20$ m/s
\end{compte}
\end{minipage}

\subsection{Dos mòbils}
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faArrowRight\,\faArrowLeft\ \ Es troben (sentits contraris)}]
{\centering\begin{tikzpicture}[x=0.6cm]
  \draw[fmmuted,line width=1pt] (0,0) -- (10,0);
  \node[text=fmA,font=\Large] at (0.4,0.4) {\faCarSide}; \node[text=fmB,font=\Large,xscale=-1] at (9.6,0.4) {\faCarSide};
  \draw[-{Latex},fmA,line width=1.2pt] (1.3,0.4) -- (3,0.4); \draw[-{Latex},fmB,line width=1.2pt] (8.7,0.4) -- (7,0.4);
  \draw[decorate,decoration={brace,amplitude=4pt,mirror},line width=0.8pt] (0,-0.15) -- (10,-0.15) node[midway,below=4pt,font=\small]{$d$};
\end{tikzpicture}\par}\smallskip
La distància es redueix a $v_1+v_2$:\par\smallskip
{\centering\large$t=\dfrac{d}{v_1+v_2}$\par}
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faArrowRight\,\faArrowRight\ \ Un abasta l'altre (mateix sentit)}]
{\centering\begin{tikzpicture}[x=0.6cm]
  \draw[fmmuted,line width=1pt] (0,0) -- (10,0);
  \node[text=fmA,font=\Large] at (0.4,0.4) {\faCarSide}; \node[text=fmB,font=\Large] at (5.4,0.4) {\faCarSide};
  \draw[-{Latex},fmA,line width=1.2pt] (1.3,0.4) -- (3.4,0.4); \draw[-{Latex},fmB,line width=1.2pt] (6.3,0.4) -- (7.5,0.4);
  \draw[decorate,decoration={brace,amplitude=4pt,mirror},line width=0.8pt] (0,-0.15) -- (5,-0.15) node[midway,below=4pt,font=\small]{$d$};
\end{tikzpicture}\par}\smallskip
La distància es redueix a $v_1-v_2$:\par\smallskip
{\centering\large$t=\dfrac{d}{v_1-v_2}$\par}
\end{tcolorbox}
\end{tcbraster}

\begin{minipage}[t]{0.49\linewidth}
\begin{exemple}[Exemple: \ es troben]
Dues ciutats estan a 300 km. Ixen alhora l'un cap a l'altre a 80 km/h i a 70 km/h.\\[3pt]
$t=\dfrac{300}{80+70}=\dfrac{300}{150}=\boxed{2\text{ h}}$\\[2pt]
\nota{el primer ha fet $80\cdot2=160$ km}
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}
\begin{exemple}[Exemple: \ l'abasta]
Un camió va a 60 km/h i porta 30 km d'avantatge a un cotxe que va a 90 km/h.\\[3pt]
$t=\dfrac{30}{90-60}=\dfrac{30}{30}=\boxed{1\text{ h}}$
\end{exemple}
\end{minipage}

\begin{exemple}[Exemple: \ si no ixen alhora]
\begin{minipage}[c]{0.55\linewidth}
Un cotxe ix a les 9 h a 60 km/h. Un altre ix a les 10 h del mateix lloc a 90 km/h. Quan l'abasta?
\end{minipage}\hfill
\begin{minipage}[c]{0.42\linewidth}
En 1 h el primer fa \textbf{60 km}: és l'avantatge.\\[3pt]
$t=\dfrac{60}{90-60}=2$ h \ $\to$ \ \boxed{\text{a les 12 h}}
\end{minipage}
\end{exemple}

\begin{exemple}[Exemple: \ un ciclista va a 18 km/h. Quant tarda a fer 45 km?]
$t=\dfrac{d}{v}=\dfrac{45}{18}=2{,}5$ h $=\boxed{2\text{ h }30\text{ min}}$ \qquad \nota{0,5 h són 30 minuts, no 50!}
\end{exemple}

\seccio{sec11}{Repàs: proporcionalitat simple}{El doble, el doble? O el doble, la meitat?}

\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={\faArrowUp\ \ Directament proporcionals},before upper=\centering]
El doble $\to$ el doble. \ Es conserva el \textbf{quocient}.\par\smallskip
\taularod[fmK]{>{\columncolor{fmK}\color{white}\heavy}l*{4}{w{c}{0.9cm}}}{
kg & 1 & 2 & 3 & 5\\
\rowcolor{fmK!8}\cellcolor{fmK}€ & 2,40 & 4,80 & 7,20 & 12
}\par\smallskip
$\dfrac{2{,}40}{1}=\dfrac{4{,}80}{2}=\dots=\cK{2{,}40}=k$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmB!8,colframe=fmB,coltitle=white,title={\faArrowDown\ \ Inversament proporcionals},before upper=\centering]
El doble $\to$ la meitat. \ Es conserva el \textbf{producte}.\par\smallskip
\taularod[fmB]{>{\columncolor{fmB}\color{white}\heavy}l*{4}{w{c}{0.9cm}}}{
pintors & 1 & 2 & 3 & 4\\
\rowcolor{fmB!8}\cellcolor{fmB}dies & 60 & 30 & 20 & 15
}\par\smallskip
$1\cdot60=2\cdot30=\dots=\cB{60}$
\end{tcolorbox}
\end{tcbraster}

\begin{center}
\taularod{>{\heavy}l<{\hspace{10pt}}>{\hspace{10pt}}c<{\hspace{10pt}}>{\hspace{10pt}}c<{\hspace{6pt}}}{
\rowcolor{fmblue} & \color{white}\heavy Directa (D) & \color{white}\heavy Inversa (I)\\
Regla de tres & $\left.\begin{array}{l}a\to b\\ c\to x\end{array}\right\}\ x=\dfrac{b\cdot c}{a}$ & $\left.\begin{array}{l}a\to b\\ c\to x\end{array}\right\}\ x=\dfrac{a\cdot b}{c}$\\[10pt]
\rowcolor{fmsurface}\rule[-15pt]{0pt}{36pt}Reducció a la unitat & $1\to\dfrac{b}{a}$, \ i multipliquem per $c$ & $1\to a\cdot b$, \ i dividim entre $c$
}
\end{center}

\begin{minipage}[t]{0.49\linewidth}
\begin{exemple}[Exemples]
3 kg costen 7,20 €. I 5 kg? \ \chip{fmK}{D}\\[2pt]
\hspace*{1em}$x=\dfrac{7{,}20\cdot5}{3}=\boxed{12\text{ €}}$\\[4pt]
4 pintors tarden 15 dies. I 6 pintors? \ \chip{fmB}{I}\\[2pt]
\hspace*{1em}$x=\dfrac{4\cdot15}{6}=\boxed{10\text{ dies}}$
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}
\begin{compte}[No totes són proporcionals!]
L'\textbf{edat} i l'\textbf{altura} creixen alhora, però amb el doble d'edat no tenim el doble d'altura.\\[3pt]
El costat d'un quadrat i la seua àrea tampoc: el doble de costat, l'àrea \textbf{quàdruple}.
\end{compte}
\end{minipage}

\begin{exemple}[Més exemples]
\begin{minipage}[t]{0.48\linewidth}
A 90 km/h tardem 4 h. I a 120 km/h? \ \chip{fmB}{I}\\[3pt]
\hspace*{1em}$x=\dfrac{90\cdot4}{120}=\boxed{3\text{ h}}$
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
5 bolígrafs costen 3,50 €. I 8? \ \chip{fmK}{D} \ \nota{a la unitat}\\[3pt]
\hspace*{1em}1 boli: $3{,}50:5=0{,}70$ € \ $\to$ \ $8\cdot0{,}70=\boxed{5{,}60\text{ €}}$
\end{minipage}
\end{exemple}

\seccio{sec12}{Proporcionalitat composta}{Tres o més magnituds: les comparem d'una en una.}

\begin{minipage}[c]{0.42\linewidth}\raggedright
\pas{1} Fem una \textbf{taula} amb les dades; la $x$ al final.\\[6pt]
\pas{2} Comparem \textbf{cada} magnitud amb la de la $x$, per separat: \chip{fmK}{D} o \chip{fmB}{I}?\\[6pt]
\pas{3} Multipliquem el valor conegut per cada raó:\\[2pt]
\hspace*{1.6em}\chip{fmK}{D}: \ $\dfrac{\text{nou}}{\text{antic}}$\qquad \chip{fmB}{I}: \ $\dfrac{\text{antic}}{\text{nou}}$
\end{minipage}\hfill
\begin{minipage}[c]{0.55\linewidth}
\begin{exemple}[Exemple: \ 5 màquines, 8 h al dia, fan 400 peces. I 3 màquines, 10 h al dia?]
\centering
\taularod[fmex]{lccc}{
\rowcolor{fmex}\color{white}\heavy & \color{white}\heavy màquines & \color{white}\heavy hores/dia & \color{white}\heavy peces\\
abans & 5 & 8 & 400\\
\rowcolor{fmexbg}ara & 3 & 10 & $x$\\
 & \chip{fmK}{D} & \chip{fmK}{D} &
}\par\smallskip
$x=400\cdot\dfrac{3}{5}\cdot\dfrac{10}{8}=\boxed{300\text{ peces}}$
\end{exemple}
\end{minipage}

\begin{exemple}[Exemple: \ 6 obrers, treballant 8 h al dia, tarden 10 dies. Quants dies tardaran 4 obrers treballant 6 h al dia?]
\begin{minipage}[c]{0.5\linewidth}\centering
\taularod[fmex]{lccc}{
\rowcolor{fmex}\color{white}\heavy & \color{white}\heavy obrers & \color{white}\heavy hores/dia & \color{white}\heavy dies\\
abans & 6 & 8 & 10\\
\rowcolor{fmexbg}ara & 4 & 6 & $x$\\
 & \chip{fmB}{I} & \chip{fmB}{I} &
}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
\nota{menys obrers o menys hores $\to$ \textbf{més} dies}\\[4pt]
$x=10\cdot\dfrac{6}{4}\cdot\dfrac{8}{6}=\boxed{20\text{ dies}}$\\[2pt]
\nota{les raons inverses es giren: antic entre nou}
\end{minipage}
\end{exemple}

\begin{exemple}[Exemple: \ 12 vaques mengen 900 kg de pinso en 15 dies. Quant en mengen 20 vaques en 30 dies?]
\begin{minipage}[c]{0.5\linewidth}\centering
\taularod[fmex]{lccc}{
\rowcolor{fmex}\color{white}\heavy & \color{white}\heavy vaques & \color{white}\heavy dies & \color{white}\heavy kg\\
abans & 12 & 15 & 900\\
\rowcolor{fmexbg}ara & 20 & 30 & $x$\\
 & \chip{fmK}{D} & \chip{fmK}{D} &
}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
\nota{més vaques o més dies $\to$ \textbf{més} pinso}\\[4pt]
$x=900\cdot\dfrac{20}{12}\cdot\dfrac{30}{15}=\boxed{3000\text{ kg}}$
\end{minipage}
\end{exemple}

\end{truc}
\end{minipage}\hfill
