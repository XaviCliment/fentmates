\ex{Porcentajes, fracciones y decimales}{Completa la tabla y ordena.}
\recorda{$p\,\%=\dfrac{p}{100}$. \textbf{\% $\to$ decimal:} dividimos entre 100 ($35\,\%=0,35$). \textbf{Decimal $\to$ \%:} multiplicamos por 100. \textbf{\% $\to$ fracción:} escribimos $\frac{p}{100}$ y simplificamos. \textbf{Fracción $\to$ decimal:} dividimos el numerador entre el denominador. Para \textbf{ordenar}, lo pasamos todo a la misma forma (por ejemplo, a porcentaje).}

\smallskip
{\renewcommand{\arraystretch}{2.8}\centering
\begin{tabular}{|C{0.6cm}|C{3.6cm}|C{3.6cm}|C{3.6cm}|}
\hline
\rowcolor{soft} & \textbf{Porcentaje} & \textbf{Fracción} & \textbf{Decimal} \\ \hline
a) & $45\,\%$ & $\dfrac{45}{100}=\dfrac{9}{20}$ & $0,45$ \\ \hline
b) & $37,5\,\%$ & $\dfrac38$ & $3:8=0,375$ \\ \hline
c) & $6\,\%$ & $\dfrac{6}{100}=\dfrac{3}{50}$ & $0,06$ \\ \hline
d) & $120\,\%$ & $\dfrac{120}{100}=\dfrac65$ & $1,2$ \\ \hline
\end{tabular}\par}

\smallskip
\pas{a) Simplificamos $\frac{45}{100}$ dividiendo entre 5. \ b) $3:8=0,375$ y, multiplicando por 100, $37,5\,\%$. \ c) $0,06\cdot100=6\,\%$; $\frac{6}{100}$ se simplifica entre 2. \ d) Más del 100\,\%: el decimal es mayor que 1.}

\ap{e} \pas{Lo pasamos todo a porcentaje: $\frac{8}{25}=\frac{32}{100}=32\,\%$; \ $0,\overline{3}=33,33\ldots\,\%$; \ $0,335=33,5\,\%$.}
\[32\,\%<33\,\%<33,\overline{3}\,\%<33,5\,\%\quad\Longrightarrow\quad\res{\frac{8}{25}<33\,\%<0,\overline{3}<0,335}\]

\ex{Cálculo de porcentajes}{Calcula.}
\recorda{El $p\,\%$ de $C$ es $C\cdot\dfrac{p}{100}$: multiplicamos por la expresión decimal del porcentaje ($35\,\%\to0,35$; \ $4\,\%\to0,04$; \ $0,5\,\%\to0,005$). Cuando un porcentaje es «\emph{de lo que queda}», primero calculamos lo que queda y después le aplicamos el porcentaje.}

\begin{tasks}(2)
  \task $260\cdot0,35=\res{91}$
  \task $1\,250\cdot0,04=\res{50}$
  \task $48\cdot1,25=\res{60}$ \ \pas{(más del total)}
  \task $3\,000\cdot0,005=\res{15}$
\end{tasks}
\ap{e} \pas{Van a dibujo: $32\cdot0,25=8$ alumnos. No van: $32-8=24$. El $75\,\%$ \emph{de esos 24} hace música:}
\[24\cdot0,75=\res{18\ \text{alumnos}}\]

% -----------------------------------------------------
\ex{Tanto por ciento que representa un número}{Resuelve.}
\recorda{\textbf{Porcentaje:} $p=\dfrac{\text{parte}}{\text{total}}\cdot100$ (el total es la cantidad de referencia, la que hace de $100\,\%$). \textbf{Total:} si conocemos la parte y el porcentaje, $\text{total}=\text{parte}:\dfrac{p}{100}$.}

\ap{a} \[\frac{18}{120}=0,15\ \longrightarrow\ 0,15\cdot100=\res{15\,\%}\]
\ap{b} \[\frac{63}{84}=0,75\ \longrightarrow\ \res{75\,\%}\]
\ap{c} \pas{Conocemos la parte (84) y el porcentaje (35\,\%); buscamos el total.}
\[84:0,35=\res{240}\qquad(\text{comprobación: }240\cdot0,35=84)\]
\ap{d} \pas{El total son los 640 alumnos.}
\[\frac{208}{640}=0,325\ \longrightarrow\ \res{32,5\,\%}\]
\ap{e} \pas{Los 54 € son la parte; el precio del viaje es el total.}
\[54:0,18=\res{300\text{ €}}\]

% -----------------------------------------------------
\ex{Aumentos y disminuciones porcentuales}{Aumentos y disminuciones.}
\recorda{$C_f=C_i\cdot I_V$, con $I_V=1+\dfrac{p}{100}$ si aumenta e $I_V=1-\dfrac{p}{100}$ si disminuye. Si conocemos las dos cantidades, $I_V=\dfrac{C_f}{C_i}$: si $I_V>1$ ha aumentado (por ejemplo, $1,15\to+15\,\%$) y si $I_V<1$ ha disminuido ($0,92\to-8\,\%$).}

\ap{a} \[I_V=1+0,12=1,12\qquad 250\cdot1,12=\res{280}\]
\ap{b} \[I_V=1-0,35=0,65\qquad 1\,800\cdot0,65=\res{1\,170}\]
\ap{c} \[84\cdot1,21=\res{101,64\text{ €}}\]
\ap{d} \[I_V=\frac{414}{360}=1,15\ \longrightarrow\ \text{ha subido un }\res{15\,\%}\]
\ap{e} \pas{El índice es menor que 1: $1-0,78125=0,21875$.}
\[I_V=\frac{1\,250}{1\,600}=0,78125\ \longrightarrow\ \text{ha bajado un }\res{21,875\,\%}\approx21,9\,\%\]

% -----------------------------------------------------
\ex{Cálculo de la cantidad inicial}{Cantidad inicial.}
\recorda{Deshacemos la variación \textbf{dividiendo} entre el índice de variación: $C_i=\dfrac{C_f}{I_V}$. \textbf{Error típico:} aplicar el porcentaje a la cantidad final; el porcentaje siempre se calcula sobre la cantidad \emph{inicial}.}

\ap{a} \pas{Un descuento del 30\,\% significa que pagamos el 70\,\%: $I_V=0,7$.}
\[C_i=\frac{87,50}{0,7}=\res{125\text{ €}}\]
\ap{b} \[C_i=\frac{72,60}{1,21}=\res{60\text{ €}}\]
\ap{c} \[C_i=\frac{1\,620}{1,08}=\res{1\,500\text{ €}}\]
\ap{d} \pas{Dividimos entre el índice de variación $1,25$:}
\[C_i=\frac{100}{1,25}=80\ne75\ \Longrightarrow\ \res{\text{Falso}}\]
\pas{El 75 se obtiene si (mal) restamos el $25\,\%$ de 100. Comprobación: $80\cdot1,25=100$.}

% -----------------------------------------------------
\ex{Variaciones porcentuales encadenadas}{Variaciones encadenadas.}
\recorda{Cada variación se aplica sobre el resultado de la anterior, por eso el índice de variación total es el \textbf{producto} de los índices: $I_{V\,\text{total}}=I_{V_1}\cdot I_{V_2}\cdots$ \ Los porcentajes encadenados \textbf{no se suman}.}

\ap{a} \[I_V=1,2\cdot0,9=1,08\ \longrightarrow\ \res{+8\,\%}\quad\text{(no un }10\,\%\text{)}\]
\ap{b} \[I_V=0,75\cdot0,9=0,675\qquad 300\cdot0,675=\res{202,50\text{ €}}\]
\pas{Descuento total: $1-0,675=0,325$, es decir, un $\res{32,5\,\%}$ (no un 35\,\%).}
\ap{c} \pas{Buscamos el índice que compensa $1,25$: el producto tiene que ser 1.}
\[I_V=\frac{1}{1,25}=0,8\ \longrightarrow\ \text{tiene que bajar un }\res{20\,\%}\]
\ap{d} \[I_V=1,05\cdot1,05\cdot0,9=0,99225<1\]
\pas{Ha \textbf{bajado}: $1-0,99225=0,00775$, es decir, un $\res{0,775\,\%}$ (aunque $5+5-10=0$).}

% -----------------------------------------------------
\ex{Problemas de porcentajes}{Problemas de porcentajes.}
\recorda{Lee bien el orden de las operaciones: primero calculamos la cantidad sobre la que se aplican los porcentajes y después multiplicamos por los índices de variación, uno detrás de otro.}

\ap{a} \pas{Precio sin IVA ni descuento: $3\cdot15+40=85$~€. Aplicamos el descuento ($I_V=0,8$) y el IVA ($I_V=1,21$):}
\[85\cdot0,8\cdot1,21=68\cdot1,21=\res{82,28\text{ €}}\]
\ap{b} \[150\,000\cdot1,06\cdot0,94=150\,000\cdot0,9964=\res{149\,460\text{ €}}\]
\pas{Ha \textbf{perdido} $150\,000-149\,460=540$~€, es decir, $1-0,9964=0,0036\to\res{0,36\,\%}$.}

% -----------------------------------------------------
\ex{Repartos directamente proporcionales}{Repartos directos.}
\recorda{Repartir $N$ directamente a $a,\ b,\ c$: calculamos el valor de una parte, $k=\dfrac{N}{a+b+c}$, y cada uno recibe $k\cdot a$, $k\cdot b$, $k\cdot c$. Si el reparto depende de dos magnitudes (dinero y tiempo), repartimos directamente al \textbf{producto}. Comprueba siempre que la suma de las partes es $N$.}

\ap{a} \[k=\frac{120}{6+9+15}=\frac{120}{30}=4\qquad 6\cdot4=\res{24\text{ €}}\quad 9\cdot4=\res{36\text{ €}}\quad 15\cdot4=\res{60\text{ €}}\]
\pas{Comprobación: $24+36+60=120$.}
\ap{b} \pas{Repartimos directamente a capital $\times$ tiempo: Ana $4\,000\cdot9=36\,000$; Berta $6\,000\cdot5=30\,000$.}
\[k=\frac{2\,200}{36\,000+30\,000}=\frac{2\,200}{66\,000}=\frac{1}{30}\qquad\text{Ana: }\frac{36\,000}{30}=\res{1\,200\text{ €}}\quad\text{Berta: }\frac{30\,000}{30}=\res{1\,000\text{ €}}\]

% -----------------------------------------------------
\ex{Repartos inversamente proporcionales}{Repartos inversos.}
\recorda{Quien tiene el número mayor recibe \textbf{menos}. Repartir inversamente a $a,\ b,\ c$ es repartir directamente a $\frac1a,\ \frac1b,\ \frac1c$. \textbf{Truco:} reducimos los inversos a común denominador (el m.c.m.) y repartimos directamente a los numeradores.}

\ap{a} \pas{m.c.m.$(2,3,4)=12$:}
\[\frac12,\ \frac13,\ \frac14=\frac{6}{12},\ \frac{4}{12},\ \frac{3}{12}\ \longrightarrow\ \text{directamente a }6,\ 4,\ 3\]
\[k=\frac{1\,300}{6+4+3}=100\qquad\res{600},\quad\res{400},\quad\res{300}\]
\ap{b} \pas{Menos tiempo $\to$ más premio. m.c.m.$(20,30,60)=60$:}
\[\frac{1}{20},\ \frac{1}{30},\ \frac{1}{60}=\frac{3}{60},\ \frac{2}{60},\ \frac{1}{60}\ \longrightarrow\ \text{directamente a }3,\ 2,\ 1\]
\[k=\frac{2\,700}{3+2+1}=450\qquad\res{1\,350\text{ €}},\quad\res{900\text{ €}},\quad\res{450\text{ €}}\]

% -----------------------------------------------------
\ex{Proporcionalidad simple directa e inversa}{Proporcionalidad simple.}
\recorda{\textbf{Directa:} el doble $\to$ el doble; se conserva el cociente: $x=\dfrac{b\cdot c}{a}$. \textbf{Inversa:} el doble $\to$ la mitad; se conserva el producto: $x=\dfrac{a\cdot b}{c}$. Para decidirlo, pregúntate: \emph{si una aumenta, ¿la otra aumenta o disminuye?}}

\ap{a} \pas{(1) Más pintores $\to$ menos tiempo: $\res{\text{inversa}}$. \ (2) El doble de litros cuesta el doble: $\res{\text{directa}}$. \ (3) Crecen los dos, pero un bebé del doble de edad no pesa el doble: $\res{\text{no proporcionales}}$.}
\ap{b} \pas{Más kilómetros $\to$ más gasolina (directa).}
\[x=\frac{7,2\cdot250}{100}=\res{18\ \text{L}}\]
\ap{c} \pas{Más obreros $\to$ menos días (inversa). Un obrero solo tardaría $9\cdot20=180$ días.}
\[x=\frac{9\cdot20}{12}=\frac{180}{12}=\res{15\ \text{días}}\]
\ap{d} \pas{Después de 3 días, a 40 personas les queda comida para $15-3=12$ días. Ahora son 48 (inversa):}
\[x=\frac{40\cdot12}{48}=\res{10\ \text{días}}\]

% -----------------------------------------------------
\ex{Proporcionalidad compuesta}{Proporcionalidad compuesta.}
\recorda{Hacemos una tabla con las magnitudes (la incógnita al final). Comparamos \textbf{cada magnitud con la de la incógnita por separado} y decidimos si es directa (D) o inversa (I). Multiplicamos el valor conocido por cada razón: si es D, por $\frac{\text{nuevo}}{\text{antiguo}}$; si es I, por $\frac{\text{antiguo}}{\text{nuevo}}$.}

\ap{a} \pas{Más impresoras $\to$ más hojas (D). Más horas $\to$ más hojas (D).}
\[x=6\,000\cdot\frac64\cdot\frac85=6\,000\cdot1,5\cdot1,6=\res{14\,400\ \text{hojas}}\]
\ap{b} \pas{Más obreros $\to$ menos días (I). Más horas al día $\to$ menos días (I). Invertimos las dos razones.}
\[x=15\cdot\frac{8}{10}\cdot\frac{6}{8}=15\cdot0,8\cdot0,75=\res{9\ \text{días}}\]
\ap{c} \pas{Menos operarios $\to$ más días (I). Más horas al día $\to$ menos días (I). Más piezas $\to$ más días (D).}
\[x=10\cdot\frac{15}{12}\cdot\frac{8}{10}\cdot\frac{900}{600}=10\cdot1,25\cdot0,8\cdot1,5=\res{15\ \text{días}}\]
