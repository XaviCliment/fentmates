\ex{Percentatges, fraccions i decimals}{Completa la taula i ordena.}
\recorda{$p\,\%=\dfrac{p}{100}$. \textbf{\% $\to$ decimal:} dividim entre 100 ($35\,\%=0,35$). \textbf{Decimal $\to$ \%:} multipliquem per 100. \textbf{\% $\to$ fracció:} escrivim $\frac{p}{100}$ i simplifiquem. \textbf{Fracció $\to$ decimal:} dividim el numerador entre el denominador. Per a \textbf{ordenar}, ho passem tot a la mateixa forma (per exemple, a percentatge).}

\smallskip
{\renewcommand{\arraystretch}{2.8}\centering
\begin{tabular}{|C{0.6cm}|C{3.6cm}|C{3.6cm}|C{3.6cm}|}
\hline
\rowcolor{soft} & \textbf{Percentatge} & \textbf{Fracció} & \textbf{Decimal} \\ \hline
a) & $45\,\%$ & $\dfrac{45}{100}=\dfrac{9}{20}$ & $0,45$ \\ \hline
b) & $37,5\,\%$ & $\dfrac38$ & $3:8=0,375$ \\ \hline
c) & $6\,\%$ & $\dfrac{6}{100}=\dfrac{3}{50}$ & $0,06$ \\ \hline
d) & $120\,\%$ & $\dfrac{120}{100}=\dfrac65$ & $1,2$ \\ \hline
\end{tabular}\par}

\smallskip
\pas{a) Simplifiquem $\frac{45}{100}$ dividint entre 5. \ b) $3:8=0,375$ i, multiplicant per 100, $37,5\,\%$. \ c) $0,06\cdot100=6\,\%$; $\frac{6}{100}$ es simplifica entre 2. \ d) Més del 100\,\%: el decimal és més gran que 1.}

\ap{e} \pas{Ho passem tot a percentatge: $\frac{8}{25}=\frac{32}{100}=32\,\%$; \ $0,\overline{3}=33,33\ldots\,\%$; \ $0,335=33,5\,\%$.}
\[32\,\%<33\,\%<33,\overline{3}\,\%<33,5\,\%\quad\Longrightarrow\quad\res{\frac{8}{25}<33\,\%<0,\overline{3}<0,335}\]

% -----------------------------------------------------
\ex{Càlcul de percentatges}{Calcula.}
\recorda{El $p\,\%$ de $C$ és $C\cdot\dfrac{p}{100}$: multipliquem per l'expressió decimal del percentatge ($35\,\%\to0,35$; \ $4\,\%\to0,04$; \ $0,5\,\%\to0,005$). Quan un percentatge és «\emph{del que queda}», primer calculem el que queda i després hi apliquem el percentatge.}

\begin{tasks}(2)
  \task $260\cdot0,35=\res{91}$
  \task $1\,250\cdot0,04=\res{50}$
  \task $48\cdot1,25=\res{60}$ \ \pas{(més del total)}
  \task $3\,000\cdot0,005=\res{15}$
\end{tasks}
\ap{e} \pas{Fan dibuix: $32\cdot0,25=8$ alumnes. No en fan: $32-8=24$. El $75\,\%$ \emph{d'aquests 24} fa música:}
\[24\cdot0,75=\res{18\ \text{alumnes}}\]

% -----------------------------------------------------
\ex{Tant per cent que representa un nombre}{Resol.}
\recorda{\textbf{Percentatge:} $p=\dfrac{\text{part}}{\text{total}}\cdot100$ (el total és la quantitat de referència, la que fa de $100\,\%$). \textbf{Total:} si coneixem la part i el percentatge, $\text{total}=\text{part}:\dfrac{p}{100}$.}

\ap{a} \[\frac{18}{120}=0,15\ \longrightarrow\ 0,15\cdot100=\res{15\,\%}\]
\ap{b} \[\frac{63}{84}=0,75\ \longrightarrow\ \res{75\,\%}\]
\ap{c} \pas{Coneixem la part (84) i el percentatge (35\,\%); busquem el total.}
\[84:0,35=\res{240}\qquad(\text{comprovació: }240\cdot0,35=84)\]
\ap{d} \pas{El total són els 640 alumnes.}
\[\frac{208}{640}=0,325\ \longrightarrow\ \res{32,5\,\%}\]
\ap{e} \pas{Els 54 € són la part; el preu del viatge és el total.}
\[54:0,18=\res{300\text{ €}}\]

% -----------------------------------------------------
\ex{Augments i disminucions percentuals}{Augments i disminucions.}
\recorda{$C_f=C_i\cdot I_V$, amb $I_V=1+\dfrac{p}{100}$ si augmenta i $I_V=1-\dfrac{p}{100}$ si disminueix. Si coneixem les dues quantitats, $I_V=\dfrac{C_f}{C_i}$: si $I_V>1$ ha augmentat (per exemple, $1,15\to+15\,\%$) i si $I_V<1$ ha disminuït ($0,92\to-8\,\%$).}

\ap{a} \[I_V=1+0,12=1,12\qquad 250\cdot1,12=\res{280}\]
\ap{b} \[I_V=1-0,35=0,65\qquad 1\,800\cdot0,65=\res{1\,170}\]
\ap{c} \[84\cdot1,21=\res{101,64\text{ €}}\]
\ap{d} \[I_V=\frac{414}{360}=1,15\ \longrightarrow\ \text{ha pujat un }\res{15\,\%}\]
\ap{e} \pas{L'índex és menor que 1: $1-0,78125=0,21875$.}
\[I_V=\frac{1\,250}{1\,600}=0,78125\ \longrightarrow\ \text{ha baixat un }\res{21,875\,\%}\approx21,9\,\%\]

% -----------------------------------------------------
\ex{Càlcul de la quantitat inicial}{Quantitat inicial.}
\recorda{Desfem la variació \textbf{dividint} entre l'índex de variació: $C_i=\dfrac{C_f}{I_V}$. \textbf{Error típic:} aplicar el percentatge a la quantitat final; el percentatge sempre es calcula sobre la quantitat \emph{inicial}.}

\ap{a} \pas{Un descompte del 30\,\% vol dir que paguem el 70\,\%: $I_V=0,7$.}
\[C_i=\frac{87,50}{0,7}=\res{125\text{ €}}\]
\ap{b} \[C_i=\frac{72,60}{1,21}=\res{60\text{ €}}\]
\ap{c} \[C_i=\frac{1\,620}{1,08}=\res{1\,500\text{ €}}\]
\ap{d} \pas{Dividim entre l'índex de variació $1,25$:}
\[C_i=\frac{100}{1,25}=80\ne75\ \Longrightarrow\ \res{\text{Fals}}\]
\pas{El 75 s'obté si (malament) restem el $25\,\%$ de 100. Comprovació: $80\cdot1,25=100$.}

% -----------------------------------------------------
\ex{Variacions percentuals encadenades}{Variacions encadenades.}
\recorda{Cada variació s'aplica sobre el resultat de l'anterior, per això l'índex de variació total és el \textbf{producte} dels índexs: $I_{V\,\text{total}}=I_{V_1}\cdot I_{V_2}\cdots$ \ Els percentatges encadenats \textbf{no se sumen}.}

\ap{a} \[I_V=1,2\cdot0,9=1,08\ \longrightarrow\ \res{+8\,\%}\quad\text{(no un }10\,\%\text{)}\]
\ap{b} \[I_V=0,75\cdot0,9=0,675\qquad 300\cdot0,675=\res{202,50\text{ €}}\]
\pas{Descompte total: $1-0,675=0,325$, és a dir, un $\res{32,5\,\%}$ (no un 35\,\%).}
\ap{c} \pas{Busquem l'índex que compensa $1,25$: el producte ha de ser 1.}
\[I_V=\frac{1}{1,25}=0,8\ \longrightarrow\ \text{ha de baixar un }\res{20\,\%}\]
\ap{d} \[I_V=1,05\cdot1,05\cdot0,9=0,99225<1\]
\pas{Ha \textbf{baixat}: $1-0,99225=0,00775$, és a dir, un $\res{0,775\,\%}$ (encara que $5+5-10=0$).}

% -----------------------------------------------------
\ex{Problemes de percentatges}{Problemes de percentatges.}
\recorda{Llig bé l'ordre de les operacions: primer calculem la quantitat sobre la qual s'apliquen els percentatges i després multipliquem pels índexs de variació, un darrere de l'altre.}

\ap{a} \pas{Preu sense IVA ni descompte: $3\cdot15+40=85$~€. Apliquem el descompte ($I_V=0,8$) i l'IVA ($I_V=1,21$):}
\[85\cdot0,8\cdot1,21=68\cdot1,21=\res{82,28\text{ €}}\]
\ap{b} \[150\,000\cdot1,06\cdot0,94=150\,000\cdot0,9964=\res{149\,460\text{ €}}\]
\pas{Ha \textbf{perdut} $150\,000-149\,460=540$~€, és a dir, $1-0,9964=0,0036\to\res{0,36\,\%}$.}

% -----------------------------------------------------
\ex{Repartiments directament proporcionals}{Repartiments directes.}
\recorda{Repartir $N$ directament a $a,\ b,\ c$: calculem el valor d'una part, $k=\dfrac{N}{a+b+c}$, i cadascú rep $k\cdot a$, $k\cdot b$, $k\cdot c$. Si el repartiment depén de dues magnituds (diners i temps), repartim directament al \textbf{producte}. Comprova sempre que la suma de les parts és $N$.}

\ap{a} \[k=\frac{120}{6+9+15}=\frac{120}{30}=4\qquad 6\cdot4=\res{24\text{ €}}\quad 9\cdot4=\res{36\text{ €}}\quad 15\cdot4=\res{60\text{ €}}\]
\pas{Comprovació: $24+36+60=120$.}
\ap{b} \pas{Repartim directament a capital $\times$ temps: Anna $4\,000\cdot9=36\,000$; Berta $6\,000\cdot5=30\,000$.}
\[k=\frac{2\,200}{36\,000+30\,000}=\frac{2\,200}{66\,000}=\frac{1}{30}\qquad\text{Anna: }\frac{36\,000}{30}=\res{1\,200\text{ €}}\quad\text{Berta: }\frac{30\,000}{30}=\res{1\,000\text{ €}}\]

% -----------------------------------------------------
\ex{Repartiments inversament proporcionals}{Repartiments inversos.}
\recorda{Qui té el nombre més gran rep \textbf{menys}. Repartir inversament a $a,\ b,\ c$ és repartir directament a $\frac1a,\ \frac1b,\ \frac1c$. \textbf{Truc:} reduïm els inversos a comú denominador (el m.c.m.) i repartim directament als numeradors.}

\ap{a} \pas{m.c.m.$(2,3,4)=12$:}
\[\frac12,\ \frac13,\ \frac14=\frac{6}{12},\ \frac{4}{12},\ \frac{3}{12}\ \longrightarrow\ \text{directament a }6,\ 4,\ 3\]
\[k=\frac{1\,300}{6+4+3}=100\qquad\res{600},\quad\res{400},\quad\res{300}\]
\ap{b} \pas{Menys temps $\to$ més premi. m.c.m.$(20,30,60)=60$:}
\[\frac{1}{20},\ \frac{1}{30},\ \frac{1}{60}=\frac{3}{60},\ \frac{2}{60},\ \frac{1}{60}\ \longrightarrow\ \text{directament a }3,\ 2,\ 1\]
\[k=\frac{2\,700}{3+2+1}=450\qquad\res{1\,350\text{ €}},\quad\res{900\text{ €}},\quad\res{450\text{ €}}\]

\ex{Proporcionalitat simple directa i inversa}{Proporcionalitat simple.}
\recorda{\textbf{Directa:} el doble $\to$ el doble; es conserva el quocient: $x=\dfrac{b\cdot c}{a}$. \textbf{Inversa:} el doble $\to$ la meitat; es conserva el producte: $x=\dfrac{a\cdot b}{c}$. Per a decidir-ho, pregunta't: \emph{si una augmenta, l'altra augmenta o disminueix?}}

\ap{a} \pas{(1) Més pintors $\to$ menys temps: $\res{\text{inversa}}$. \ (2) El doble de litres costa el doble: $\res{\text{directa}}$. \ (3) Creixen tots dos, però un bebé del doble d'edat no pesa el doble: $\res{\text{no proporcionals}}$.}
\ap{b} \pas{Més quilòmetres $\to$ més gasolina (directa).}
\[x=\frac{7,2\cdot250}{100}=\res{18\ \text{L}}\]
\ap{c} \pas{Més obrers $\to$ menys dies (inversa). Un obrer sol tardaria $9\cdot20=180$ dies.}
\[x=\frac{9\cdot20}{12}=\frac{180}{12}=\res{15\ \text{dies}}\]
\ap{d} \pas{Després de 3 dies, a 40 persones els queda menjar per a $15-3=12$ dies. Ara són 48 (inversa):}
\[x=\frac{40\cdot12}{48}=\res{10\ \text{dies}}\]

% -----------------------------------------------------
\ex{Proporcionalitat composta}{Proporcionalitat composta.}
\recorda{Fem una taula amb les magnituds (la incògnita al final). Comparem \textbf{cada magnitud amb la de la incògnita per separat} i decidim si és directa (D) o inversa (I). Multipliquem el valor conegut per cada raó: si és D, per $\frac{\text{nou}}{\text{antic}}$; si és I, per $\frac{\text{antic}}{\text{nou}}$.}

\ap{a} \pas{Més impressores $\to$ més fulls (D). Més hores $\to$ més fulls (D).}
\[x=6\,000\cdot\frac64\cdot\frac85=6\,000\cdot1,5\cdot1,6=\res{14\,400\ \text{fulls}}\]
\ap{b} \pas{Més obrers $\to$ menys dies (I). Més hores al dia $\to$ menys dies (I). Girem les dues raons.}
\[x=15\cdot\frac{8}{10}\cdot\frac{6}{8}=15\cdot0,8\cdot0,75=\res{9\ \text{dies}}\]
\ap{c} \pas{Menys operaris $\to$ més dies (I). Més hores al dia $\to$ menys dies (I). Més peces $\to$ més dies (D).}
\[x=10\cdot\frac{15}{12}\cdot\frac{8}{10}\cdot\frac{900}{600}=10\cdot1,25\cdot0,8\cdot1,5=\res{15\ \text{dies}}\]
