\seccio{1}{Ecuaciones lineales con dos incógnitas}

\ex{1} ¿Cuáles de estas ecuaciones son lineales con dos incógnitas?\begin{tasks}(3)
  \task $3x-y=2$
  \task $x^2+y=1$
  \task $xy=4$
  \task $\dfrac{x}{2}+y=3$
  \task $2x+5=3x$
  \task $y=4x-1$
\end{tasks}

\ex{1} Comprueba cuáles de estos pares son solución de $2x+3y=12$:\begin{tasks}(4)
  \task $(3,\,2)$
  \task $(0,\,4)$
  \task $(6,\,1)$
  \task $(-3,\,6)$
\end{tasks}

\ex{1} Despeja la $y$ y completa una tabla de valores con $x=-2,\,-1,\,0,\,1,\,2$:\begin{tasks}(2)
  \task $x+y=5$
  \task $2x-y=1$
\end{tasks}\recorda{Cada par $(x,\,y)$ de la tabla es una solución de la ecuación. Una ecuación con dos incógnitas tiene infinitas soluciones.}

\ex{2} Escribe tres soluciones de cada ecuación:\begin{tasks}(2)
  \task $x-3y=6$
  \task $4x+2y=10$
\end{tasks}

\ex{2} Responde:\begin{tasks}(1)
  \task Halla $k$ para que $(-1,\,3)$ sea solución de $4x+ky=8$.
  \task Halla $m$ para que $(2,\,m)$ sea solución de $5x-2y=4$.
\end{tasks}

\ex{2} Haz una tabla de valores y representa cada recta en unos ejes:\begin{tasks}(2)
  \task $x+2y=4$
  \task $3x-y=3$
\end{tasks}

\ex{3} Responde:\begin{tasks}(1)
  \task Halla la solución de $3x+2y=21$ en la que la $y$ vale 3 unidades más que la $x$.
  \task ¿En qué puntos corta la recta $x-2y=6$ a los ejes de coordenadas?
\end{tasks}

\seccio{2}{Sistemas de ecuaciones y soluciones}

\ex{1} Comprueba si el par $(2,\,-1)$ es solución de cada sistema:\begin{tasks}(2)
  \task $\begin{cases}x+y=1\\ 2x-y=5\end{cases}$
  \task $\begin{cases}3x+y=5\\ x-2y=3\end{cases}$
\end{tasks}\recorda{Un par es solución del sistema si cumple \textbf{las dos} ecuaciones a la vez.}

\ex{1} ¿Cuál de estos pares es la solución del sistema $\begin{cases}x+y=3\\ 2x-y=3\end{cases}$? \quad $(1,\,2)$, \ $(2,\,1)$, \ $(3,\,0)$

\ex{2} Escribe un sistema de dos ecuaciones que tenga por solución $x=-2,\ y=5$.

\ex{3} Halla $a$ y $b$ para que $(3,\,-2)$ sea la solución del sistema $\begin{cases}ax+2y=5\\ 4x+by=6\end{cases}$.

\seccio{3}{Método de sustitución}

\ex{1} Resuelve por sustitución (una incógnita ya está despejada):\begin{tasks}(2)
  \task $\begin{cases}x=3y\\ x+y=8\end{cases}$
  \task $\begin{cases}y=2x+1\\ 3x+y=11\end{cases}$
\end{tasks}

\ex{1} Resuelve por sustitución:\begin{tasks}(2)
  \task $\begin{cases}x+y=7\\ 2x-y=2\end{cases}$
  \task $\begin{cases}x-y=1\\ 3x+2y=13\end{cases}$
\end{tasks}\recorda{Despeja la incógnita que tiene coeficiente 1 o $-1$ y sustitúyela, \textbf{entre paréntesis}, en la otra ecuación.}

\ex{2} Resuelve por sustitución:\begin{tasks}(2)
  \task $\begin{cases}3x+y=1\\ 5x-2y=20\end{cases}$
  \task $\begin{cases}2x-y=-4\\ 4x+3y=2\end{cases}$
\end{tasks}

\ex{2} Resuelve por sustitución y comprueba la solución:\begin{tasks}(2)
  \task $\begin{cases}x+4y=-5\\ 3x-2y=13\end{cases}$
  \task $\begin{cases}5x-y=13\\ 2x+3y=12\end{cases}$
\end{tasks}

\ex{3} Resuelve por sustitución (ningún coeficiente es 1: aparecerán fracciones):\begin{tasks}(2)
  \task $\begin{cases}2x+3y=12\\ 5x-4y=7\end{cases}$
  \task $\begin{cases}3x-2y=-1\\ 4x+5y=14\end{cases}$
\end{tasks}

\ex{3} Al resolver $\begin{cases}y=3-x\\ 2x-3y=1\end{cases}$ un alumno escribe $2x-9-3x=1$. ¿Qué error ha cometido? Resuelve el sistema correctamente.

\seccio{4}{Método de igualación}

\ex{1} Resuelve por igualación:\begin{tasks}(2)
  \task $\begin{cases}y=x+5\\ y=3x-1\end{cases}$
  \task $\begin{cases}x=2y-3\\ x=y+1\end{cases}$
\end{tasks}\recorda{Despeja \textbf{la misma} incógnita en las dos ecuaciones e iguala las expresiones.}

\ex{1} Resuelve por igualación:\begin{tasks}(2)
  \task $\begin{cases}x+y=6\\ x-y=2\end{cases}$
  \task $\begin{cases}y-2x=1\\ y+x=7\end{cases}$
\end{tasks}

\ex{2} Resuelve por igualación:\begin{tasks}(2)
  \task $\begin{cases}2x+y=8\\ 3x+y=11\end{cases}$
  \task $\begin{cases}x+3y=10\\ x-2y=0\end{cases}$
\end{tasks}

\ex{2} Resuelve por igualación y comprueba la solución:\begin{tasks}(2)
  \task $\begin{cases}3x-y=7\\ x+2y=0\end{cases}$
  \task $\begin{cases}4x+y=-2\\ 2x-y=-4\end{cases}$
\end{tasks}

\ex{3} Resuelve por igualación (aparecerán denominadores):\begin{tasks}(2)
  \task $\begin{cases}2x+3y=7\\ 3x-2y=4\end{cases}$
  \task $\begin{cases}5x-2y=1\\ 3x+4y=11\end{cases}$
\end{tasks}

\ex{3} Resuelve el sistema $\begin{cases}y=\dfrac{x+1}{2}\\[6pt] y=\dfrac{2x-1}{3}\end{cases}$ por el método más adecuado.

\seccio{5}{Método de reducción}

\ex{1} Resuelve por reducción (los coeficientes de una incógnita ya son opuestos):\begin{tasks}(2)
  \task $\begin{cases}x+y=9\\ x-y=3\end{cases}$
  \task $\begin{cases}3x+2y=13\\ x-2y=-1\end{cases}$
\end{tasks}\recorda{Si los coeficientes de una incógnita son opuestos, al sumar las ecuaciones esa incógnita desaparece.}

\ex{1} Resuelve por reducción:\begin{tasks}(2)
  \task $\begin{cases}2x+y=7\\ 2x-3y=-5\end{cases}$
  \task $\begin{cases}5x+3y=11\\ 2x-3y=-4\end{cases}$
\end{tasks}

\ex{2} Resuelve por reducción (hay que multiplicar una de las ecuaciones):\begin{tasks}(2)
  \task $\begin{cases}x+2y=4\\ 3x-y=5\end{cases}$
  \task $\begin{cases}4x+3y=10\\ 2x+5y=12\end{cases}$
\end{tasks}

\ex{2} Resuelve por reducción (hay que multiplicar las dos ecuaciones):\begin{tasks}(2)
  \task $\begin{cases}3x+4y=18\\ 5x-2y=4\end{cases}$
  \task $\begin{cases}2x-5y=-1\\ 3x+2y=8\end{cases}$
\end{tasks}

\ex{3} Resuelve por reducción:\begin{tasks}(2)
  \task $\begin{cases}4x+3y=-1\\ 5x+2y=4\end{cases}$
  \task $\begin{cases}6x-5y=7\\ 4x-3y=5\end{cases}$
\end{tasks}

\ex{3} Resuelve el sistema $\begin{cases}2x+3y=4\\ 3x-2y=1\end{cases}$ por \textbf{doble reducción}: elimina primero la $y$ para hallar la $x$ y después elimina la $x$ para hallar la $y$. ¿Por qué es útil aquí?

\seccio{7}{Clasificación de los sistemas}

\ex{1} Clasifica estos sistemas sin resolverlos (SCD, SCI o SI):\begin{tasks}(2)
  \task $\begin{cases}x+3y=2\\ 2x+6y=4\end{cases}$
  \task $\begin{cases}2x-y=1\\ 4x-2y=3\end{cases}$
  \task $\begin{cases}x+y=5\\ 2x-y=1\end{cases}$
  \task $\begin{cases}3x-6y=9\\ -x+2y=-3\end{cases}$
\end{tasks}\recorda{$\dfrac{a}{a'}\neq\dfrac{b}{b'}$: compatible determinado (secantes). \ $\dfrac{a}{a'}=\dfrac{b}{b'}=\dfrac{c}{c'}$: compatible indeterminado (coincidentes). \ $\dfrac{a}{a'}=\dfrac{b}{b'}\neq\dfrac{c}{c'}$: incompatible (paralelas).}

\ex{1} Completa: si las rectas de un sistema\ldots\begin{tasks}(1)
  \task se cortan en un punto, el sistema tiene \rule{2.5cm}{0.4pt} y es \rule{3.5cm}{0.4pt}.
  \task son paralelas, el sistema tiene \rule{2.5cm}{0.4pt} y es \rule{3.5cm}{0.4pt}.
  \task coinciden, el sistema tiene \rule{2.5cm}{0.4pt} y es \rule{3.5cm}{0.4pt}.
\end{tasks}

\ex{2} Resuelve por reducción y clasifica:\begin{tasks}(2)
  \task $\begin{cases}x+y=3\\ 2x+2y=5\end{cases}$
  \task $\begin{cases}x-2y=1\\ 3x-6y=3\end{cases}$
\end{tasks}

\ex{2} Resuelve y clasifica:\begin{tasks}(3)
  \task $\begin{cases}2x+3y=5\\ 4x+6y=10\end{cases}$
  \task $\begin{cases}x-y=2\\ -2x+2y=4\end{cases}$
  \task $\begin{cases}3x+y=2\\ x-y=2\end{cases}$
\end{tasks}

\ex{2} Escribe una ecuación que forme con $2x-y=3$ un sistema:\begin{tasks}(3)
  \task compatible determinado
  \task compatible indeterminado
  \task incompatible
\end{tasks}

\ex{3} ¿Para qué valor de $m$ el sistema $\begin{cases}mx+2y=3\\ 3x+y=1\end{cases}$ es incompatible? ¿Cómo es para los demás valores de $m$?

\ex{3} Halla $a$ y $b$ para que el sistema $\begin{cases}2x+ay=b\\ 4x-6y=10\end{cases}$ tenga infinitas soluciones.

\seccio{8}{Sistemas con paréntesis y denominadores}

\ex{1} Escribe cada ecuación en la forma $ax+by=c$ y resuelve:\begin{tasks}(2)
  \task $\begin{cases}2(x-1)+y=7\\ x-(y-3)=3\end{cases}$
  \task $\begin{cases}3(x+y)-2x=10\\ 2(x-y)+y=-1\end{cases}$
\end{tasks}\recorda{Primero quitamos paréntesis y denominadores, transponemos y reducimos. Después elegimos el método más adecuado.}

\ex{1} Quita denominadores y resuelve:\begin{tasks}(2)
  \task $\begin{cases}\dfrac{x}{2}+y=4\\ x-\dfrac{y}{3}=1\end{cases}$
  \task $\begin{cases}\dfrac{x}{3}+\dfrac{y}{2}=2\\ \dfrac{x}{2}-\dfrac{y}{4}=1\end{cases}$
\end{tasks}

\ex{2} Resuelve:\begin{tasks}(2)
  \task $\begin{cases}\dfrac{x+1}{2}+\dfrac{y-1}{3}=3\\ x-y=-1\end{cases}$
  \task $\begin{cases}\dfrac{x-2}{3}=\dfrac{y+1}{4}\\ 2x-y=7\end{cases}$
\end{tasks}

\ex{2} Resuelve (multiplica por 10 para eliminar los decimales):\begin{tasks}(1)
  \task $\begin{cases}0{,}5x+0{,}2y=2\\ x-y=-3\end{cases}$
\end{tasks}

\ex{3} Resuelve:\begin{tasks}(1)
  \task $\begin{cases}\dfrac{2x-1}{3}-\dfrac{y+2}{2}=-1\\ 3(x-y)-2(x+1)=-6\end{cases}$
\end{tasks}

\ex{3} ¿Qué método es el más adecuado para cada sistema? Justifícalo y resuélvelos.\begin{tasks}(3)
  \task $\begin{cases}y=3x-2\\ y=x+4\end{cases}$
  \task $\begin{cases}x+5y=9\\ 2x-3y=5\end{cases}$
  \task $\begin{cases}4x+3y=6\\ 6x-3y=24\end{cases}$
\end{tasks}

\seccio{9}{Problemas con sistemas de ecuaciones}

\ex{1} La suma de dos números es 56 y su diferencia es 12. ¿Cuáles son?\recorda{Elige las dos incógnitas y escribe qué representan, busca dos relaciones para plantear el sistema, resuélvelo y comprueba que la solución tiene sentido en el problema.}

\ex{1} Un número es el triple de otro y entre los dos suman 48. ¿Cuáles son?

\ex{1} En un corral hay gallinas y conejos. En total hay 40 cabezas y 112 patas. ¿Cuántos animales hay de cada tipo?

\ex{1} 3 kg de manzanas y 2 kg de peras cuestan 8 €, y 1 kg de manzanas y 3 kg de peras cuestan 5 €. ¿Cuánto cuesta el kilo de cada fruta?

\ex{2} En una hucha hay 22 monedas, unas de 2 € y otras de 50 céntimos, que suman 26 €. ¿Cuántas hay de cada tipo?

\ex{2} La entrada de un cine cuesta 7 € para los adultos y 4 € para los niños. Un día se vendieron 85 entradas y se recaudaron 445 €. ¿Cuántas entradas de cada tipo se vendieron?

\ex{2} La edad de una madre es el cuádruple de la de su hijo. Dentro de 5 años solo será el triple. ¿Cuántos años tienen ahora?

\ex{2} El perímetro de un rectángulo mide 46 m y la base mide 7 m más que la altura. ¿Qué dimensiones tiene?

\ex{2} Un test tiene 30 preguntas y las has contestado todas. Cada acierto vale 2 puntos y cada fallo resta 1. Si has sacado 39 puntos, ¿cuántas has acertado?

\ex{2} En una clase hay 28 alumnos. Si llegaran 2 chicas más, habría tantas chicas como chicos. ¿Cuántos chicos y cuántas chicas hay?

\ex{3} Queremos 60 kg de café de 7 €/kg mezclando café de 9 €/kg con café de 6 €/kg. ¿Cuántos kilos de cada tipo tenemos que poner?

\ex{3} Las cifras de un número de dos cifras suman 11. Si las intercambiamos, el número que resulta es 27 unidades menor. ¿Cuál es el número?

\ex{3} Dos ciudades están a 75 km. Dos ciclistas salen a la vez, cada uno de una ciudad, y van el uno hacia el otro. Uno va 5 km/h más rápido que el otro y se encuentran al cabo de 3 horas. ¿A qué velocidad va cada uno?

\ex{3} Si sumamos 1 al numerador y al denominador de una fracción, obtenemos $\tfrac23$; si restamos 1 a los dos, obtenemos $\tfrac12$. ¿Cuál es la fracción?

\ex{3} Un hotel tiene 50 habitaciones, unas dobles y otras individuales, y en total 87 camas. ¿Cuántas habitaciones hay de cada tipo?

\ex{3} El perímetro de un triángulo isósceles mide 40 cm y cada uno de los lados iguales mide 5 cm más que la base. ¿Cuánto miden los lados?
