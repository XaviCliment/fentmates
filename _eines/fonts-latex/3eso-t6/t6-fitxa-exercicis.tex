\seccio{1}{Equacions lineals amb dues incògnites}

\ex{1} Quines d'aquestes equacions són lineals amb dues incògnites?\begin{tasks}(3)
  \task $3x-y=2$
  \task $x^2+y=1$
  \task $xy=4$
  \task $\dfrac{x}{2}+y=3$
  \task $2x+5=3x$
  \task $y=4x-1$
\end{tasks}

\ex{1} Comprova quins d'aquests parells són solució de $2x+3y=12$:\begin{tasks}(4)
  \task $(3,\,2)$
  \task $(0,\,4)$
  \task $(6,\,1)$
  \task $(-3,\,6)$
\end{tasks}

\ex{1} Aïlla la $y$ i completa una taula de valors amb $x=-2,\,-1,\,0,\,1,\,2$:\begin{tasks}(2)
  \task $x+y=5$
  \task $2x-y=1$
\end{tasks}\recorda{Cada parell $(x,\,y)$ de la taula és una solució de l'equació. Una equació amb dues incògnites té infinites solucions.}

\ex{2} Escriu tres solucions de cada equació:\begin{tasks}(2)
  \task $x-3y=6$
  \task $4x+2y=10$
\end{tasks}

\ex{2} Respon:\begin{tasks}(1)
  \task Troba $k$ perquè $(-1,\,3)$ siga solució de $4x+ky=8$.
  \task Troba $m$ perquè $(2,\,m)$ siga solució de $5x-2y=4$.
\end{tasks}

\ex{2} Fes una taula de valors i representa cada recta en uns eixos:\begin{tasks}(2)
  \task $x+2y=4$
  \task $3x-y=3$
\end{tasks}

\ex{3} Respon:\begin{tasks}(1)
  \task Troba la solució de $3x+2y=21$ en què la $y$ val 3 unitats més que la $x$.
  \task En quins punts talla la recta $x-2y=6$ els eixos de coordenades?
\end{tasks}

\seccio{2}{Sistemes d'equacions i solucions}

\ex{1} Comprova si el parell $(2,\,-1)$ és solució de cada sistema:\begin{tasks}(2)
  \task $\begin{cases}x+y=1\\ 2x-y=5\end{cases}$
  \task $\begin{cases}3x+y=5\\ x-2y=3\end{cases}$
\end{tasks}\recorda{Un parell és solució del sistema si compleix \textbf{les dues} equacions alhora.}

\ex{1} Quin d'aquests parells és la solució del sistema $\begin{cases}x+y=3\\ 2x-y=3\end{cases}$? \quad $(1,\,2)$, \ $(2,\,1)$, \ $(3,\,0)$

\ex{2} Escriu un sistema de dues equacions que tinga per solució $x=-2,\ y=5$.

\ex{3} Troba $a$ i $b$ perquè $(3,\,-2)$ siga la solució del sistema $\begin{cases}ax+2y=5\\ 4x+by=6\end{cases}$.

\seccio{3}{Mètode de substitució}

\ex{1} Resol per substitució (una incògnita ja està aïllada):\begin{tasks}(2)
  \task $\begin{cases}x=3y\\ x+y=8\end{cases}$
  \task $\begin{cases}y=2x+1\\ 3x+y=11\end{cases}$
\end{tasks}

\ex{1} Resol per substitució:\begin{tasks}(2)
  \task $\begin{cases}x+y=7\\ 2x-y=2\end{cases}$
  \task $\begin{cases}x-y=1\\ 3x+2y=13\end{cases}$
\end{tasks}\recorda{Aïlla la incògnita que té coeficient 1 o $-1$ i substitueix-la, \textbf{entre parèntesis}, en l'altra equació.}

\ex{2} Resol per substitució:\begin{tasks}(2)
  \task $\begin{cases}3x+y=1\\ 5x-2y=20\end{cases}$
  \task $\begin{cases}2x-y=-4\\ 4x+3y=2\end{cases}$
\end{tasks}

\ex{2} Resol per substitució i comprova la solució:\begin{tasks}(2)
  \task $\begin{cases}x+4y=-5\\ 3x-2y=13\end{cases}$
  \task $\begin{cases}5x-y=13\\ 2x+3y=12\end{cases}$
\end{tasks}

\ex{3} Resol per substitució (cap coeficient és 1: apareixeran fraccions):\begin{tasks}(2)
  \task $\begin{cases}2x+3y=12\\ 5x-4y=7\end{cases}$
  \task $\begin{cases}3x-2y=-1\\ 4x+5y=14\end{cases}$
\end{tasks}

\ex{3} En resoldre $\begin{cases}y=3-x\\ 2x-3y=1\end{cases}$ un alumne escriu $2x-9-3x=1$. Quin error ha comés? Resol el sistema correctament.

\seccio{4}{Mètode d'igualació}

\ex{1} Resol per igualació:\begin{tasks}(2)
  \task $\begin{cases}y=x+5\\ y=3x-1\end{cases}$
  \task $\begin{cases}x=2y-3\\ x=y+1\end{cases}$
\end{tasks}\recorda{Aïlla \textbf{la mateixa} incògnita en les dues equacions i iguala les expressions.}

\ex{1} Resol per igualació:\begin{tasks}(2)
  \task $\begin{cases}x+y=6\\ x-y=2\end{cases}$
  \task $\begin{cases}y-2x=1\\ y+x=7\end{cases}$
\end{tasks}

\ex{2} Resol per igualació:\begin{tasks}(2)
  \task $\begin{cases}2x+y=8\\ 3x+y=11\end{cases}$
  \task $\begin{cases}x+3y=10\\ x-2y=0\end{cases}$
\end{tasks}

\ex{2} Resol per igualació i comprova la solució:\begin{tasks}(2)
  \task $\begin{cases}3x-y=7\\ x+2y=0\end{cases}$
  \task $\begin{cases}4x+y=-2\\ 2x-y=-4\end{cases}$
\end{tasks}

\ex{3} Resol per igualació (apareixeran denominadors):\begin{tasks}(2)
  \task $\begin{cases}2x+3y=7\\ 3x-2y=4\end{cases}$
  \task $\begin{cases}5x-2y=1\\ 3x+4y=11\end{cases}$
\end{tasks}

\ex{3} Resol el sistema $\begin{cases}y=\dfrac{x+1}{2}\\[6pt] y=\dfrac{2x-1}{3}\end{cases}$ pel mètode més adequat.

\seccio{5}{Mètode de reducció}

\ex{1} Resol per reducció (els coeficients d'una incògnita ja són oposats):\begin{tasks}(2)
  \task $\begin{cases}x+y=9\\ x-y=3\end{cases}$
  \task $\begin{cases}3x+2y=13\\ x-2y=-1\end{cases}$
\end{tasks}\recorda{Si els coeficients d'una incògnita són oposats, en sumar les equacions aquesta incògnita desapareix.}

\ex{1} Resol per reducció:\begin{tasks}(2)
  \task $\begin{cases}2x+y=7\\ 2x-3y=-5\end{cases}$
  \task $\begin{cases}5x+3y=11\\ 2x-3y=-4\end{cases}$
\end{tasks}

\ex{2} Resol per reducció (cal multiplicar una de les equacions):\begin{tasks}(2)
  \task $\begin{cases}x+2y=4\\ 3x-y=5\end{cases}$
  \task $\begin{cases}4x+3y=10\\ 2x+5y=12\end{cases}$
\end{tasks}

\ex{2} Resol per reducció (cal multiplicar les dues equacions):\begin{tasks}(2)
  \task $\begin{cases}3x+4y=18\\ 5x-2y=4\end{cases}$
  \task $\begin{cases}2x-5y=-1\\ 3x+2y=8\end{cases}$
\end{tasks}

\ex{3} Resol per reducció:\begin{tasks}(2)
  \task $\begin{cases}4x+3y=-1\\ 5x+2y=4\end{cases}$
  \task $\begin{cases}6x-5y=7\\ 4x-3y=5\end{cases}$
\end{tasks}

\ex{3} Resol el sistema $\begin{cases}2x+3y=4\\ 3x-2y=1\end{cases}$ per \textbf{reducció doble}: elimina primer la $y$ per a trobar la $x$ i després elimina la $x$ per a trobar la $y$. Per què és útil ací?

\seccio{7}{Classificació dels sistemes}

\ex{1} Classifica aquests sistemes sense resoldre'ls (SCD, SCI o SI):\begin{tasks}(2)
  \task $\begin{cases}x+3y=2\\ 2x+6y=4\end{cases}$
  \task $\begin{cases}2x-y=1\\ 4x-2y=3\end{cases}$
  \task $\begin{cases}x+y=5\\ 2x-y=1\end{cases}$
  \task $\begin{cases}3x-6y=9\\ -x+2y=-3\end{cases}$
\end{tasks}\recorda{$\dfrac{a}{a'}\neq\dfrac{b}{b'}$: compatible determinat (secants). \ $\dfrac{a}{a'}=\dfrac{b}{b'}=\dfrac{c}{c'}$: compatible indeterminat (coincidents). \ $\dfrac{a}{a'}=\dfrac{b}{b'}\neq\dfrac{c}{c'}$: incompatible (paral·leles).}

\ex{1} Completa: si les rectes d'un sistema\ldots\begin{tasks}(1)
  \task es tallen en un punt, el sistema té \rule{2.5cm}{0.4pt} i és \rule{3.5cm}{0.4pt}.
  \task són paral·leles, el sistema té \rule{2.5cm}{0.4pt} i és \rule{3.5cm}{0.4pt}.
  \task coincideixen, el sistema té \rule{2.5cm}{0.4pt} i és \rule{3.5cm}{0.4pt}.
\end{tasks}

\ex{2} Resol per reducció i classifica:\begin{tasks}(2)
  \task $\begin{cases}x+y=3\\ 2x+2y=5\end{cases}$
  \task $\begin{cases}x-2y=1\\ 3x-6y=3\end{cases}$
\end{tasks}

\ex{2} Resol i classifica:\begin{tasks}(3)
  \task $\begin{cases}2x+3y=5\\ 4x+6y=10\end{cases}$
  \task $\begin{cases}x-y=2\\ -2x+2y=4\end{cases}$
  \task $\begin{cases}3x+y=2\\ x-y=2\end{cases}$
\end{tasks}

\ex{2} Escriu una equació que forme amb $2x-y=3$ un sistema:\begin{tasks}(3)
  \task compatible determinat
  \task compatible indeterminat
  \task incompatible
\end{tasks}

\ex{3} Per a quin valor de $m$ el sistema $\begin{cases}mx+2y=3\\ 3x+y=1\end{cases}$ és incompatible? Com és per als altres valors de $m$?

\ex{3} Troba $a$ i $b$ perquè el sistema $\begin{cases}2x+ay=b\\ 4x-6y=10\end{cases}$ tinga infinites solucions.

\seccio{8}{Sistemes amb parèntesis i denominadors}

\ex{1} Escriu cada equació en la forma $ax+by=c$ i resol:\begin{tasks}(2)
  \task $\begin{cases}2(x-1)+y=7\\ x-(y-3)=3\end{cases}$
  \task $\begin{cases}3(x+y)-2x=10\\ 2(x-y)+y=-1\end{cases}$
\end{tasks}\recorda{Primer traiem parèntesis i denominadors, transposem i reduïm. Després triem el mètode més adequat.}

\ex{1} Traiem denominadors i resol:\begin{tasks}(2)
  \task $\begin{cases}\dfrac{x}{2}+y=4\\ x-\dfrac{y}{3}=1\end{cases}$
  \task $\begin{cases}\dfrac{x}{3}+\dfrac{y}{2}=2\\ \dfrac{x}{2}-\dfrac{y}{4}=1\end{cases}$
\end{tasks}

\ex{2} Resol:\begin{tasks}(2)
  \task $\begin{cases}\dfrac{x+1}{2}+\dfrac{y-1}{3}=3\\ x-y=-1\end{cases}$
  \task $\begin{cases}\dfrac{x-2}{3}=\dfrac{y+1}{4}\\ 2x-y=7\end{cases}$
\end{tasks}

\ex{2} Resol (multiplica per 10 per a eliminar els decimals):\begin{tasks}(1)
  \task $\begin{cases}0{,}5x+0{,}2y=2\\ x-y=-3\end{cases}$
\end{tasks}

\ex{3} Resol:\begin{tasks}(1)
  \task $\begin{cases}\dfrac{2x-1}{3}-\dfrac{y+2}{2}=-1\\ 3(x-y)-2(x+1)=-6\end{cases}$
\end{tasks}

\ex{3} Quin mètode és el més adequat per a cada sistema? Justifica-ho i resol-los.\begin{tasks}(3)
  \task $\begin{cases}y=3x-2\\ y=x+4\end{cases}$
  \task $\begin{cases}x+5y=9\\ 2x-3y=5\end{cases}$
  \task $\begin{cases}4x+3y=6\\ 6x-3y=24\end{cases}$
\end{tasks}

\seccio{9}{Problemes amb sistemes d'equacions}

\ex{1} La suma de dos nombres és 56 i la seua diferència és 12. Quins són?\recorda{Tria les dues incògnites i escriu què representen, busca dues relacions per a plantejar el sistema, resol-lo i comprova que la solució té sentit en el problema.}

\ex{1} Un nombre és el triple d'un altre i entre els dos sumen 48. Quins són?

\ex{1} En un corral hi ha gallines i conills. En total hi ha 40 caps i 112 potes. Quants animals hi ha de cada tipus?

\ex{1} 3 kg de pomes i 2 kg de peres costen 8 €, i 1 kg de pomes i 3 kg de peres costen 5 €. Quant costa el quilo de cada fruita?

\ex{2} En una guardiola hi ha 22 monedes, unes de 2 € i altres de 50 cèntims, que sumen 26 €. Quantes n'hi ha de cada tipus?

\ex{2} L'entrada d'un cine costa 7 € per als adults i 4 € per als xiquets. Un dia es van vendre 85 entrades i es van recaptar 445 €. Quantes entrades de cada tipus es van vendre?

\ex{2} L'edat d'una mare és el quàdruple de la del seu fill. D'ací a 5 anys només serà el triple. Quants anys tenen ara?

\ex{2} El perímetre d'un rectangle fa 46 m i la base fa 7 m més que l'altura. Quines dimensions té?

\ex{2} Un test té 30 preguntes i les has contestades totes. Cada encert val 2 punts i cada errada en resta 1. Si has tret 39 punts, quantes n'has encertat?

\ex{2} En una classe hi ha 28 alumnes. Si arribaren 2 xiques més, hi hauria tantes xiques com xics. Quants xics i quantes xiques hi ha?

\ex{3} Volem 60 kg de cafè de 7 €/kg barrejant cafè de 9 €/kg amb cafè de 6 €/kg. Quants quilos de cada tipus hi hem de posar?

\ex{3} Les xifres d'un nombre de dues xifres sumen 11. Si les intercanviem, el nombre que resulta és 27 unitats menor. Quin és el nombre?

\ex{3} Dues ciutats estan a 75 km. Dos ciclistes ixen alhora, cadascun d'una ciutat, i van l'un cap a l'altre. Un va 5 km/h més ràpid que l'altre i es troben al cap de 3 hores. A quina velocitat va cadascun?

\ex{3} Si sumem 1 al numerador i al denominador d'una fracció, obtenim $\tfrac23$; si restem 1 a tots dos, obtenim $\tfrac12$. Quina és la fracció?

\ex{3} Un hotel té 50 habitacions, unes dobles i altres individuals, i en total 87 llits. Quantes habitacions hi ha de cada tipus?

\ex{3} El perímetre d'un triangle isòsceles fa 40 cm i cadascun dels costats iguals fa 5 cm més que la base. Quant fan els costats?
