\seccio{3}{Mètode de substitució}

\ex{1} \begin{tasks}(1)
  \task $x=3y$\newline $(3y)+y=8\Rightarrow 3y+y=8\Rightarrow 4y=8\Rightarrow y=\tfrac{8}{4}=2$\newline $x=3\cdot 2=6$\newline $\res{x=6,\ \ y=2}$
  \task $y=2x+1$\newline $3x+(2x+1)=11\Rightarrow 3x+2x+1=11\Rightarrow 5x=10\Rightarrow x=\tfrac{10}{5}=2$\newline $y=2\cdot 2+1=5$\newline $\res{x=2,\ \ y=5}$
\end{tasks}

\ex{1} \begin{tasks}(1)
  \task $x=7-y$\newline $2(7-y)-y=2\Rightarrow 14-2y-y=2\Rightarrow -3y=-12\Rightarrow y=\tfrac{-12}{-3}=4$\newline $x=7-4=3$\newline $\res{x=3,\ \ y=4}$
  \task $x=y+1$\newline $3(y+1)+2y=13\Rightarrow 3y+3+2y=13\Rightarrow 5y=10\Rightarrow y=\tfrac{10}{5}=2$\newline $x=2+1=3$\newline $\res{x=3,\ \ y=2}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $y=1-3x$\newline $5x-2(1-3x)=20\Rightarrow 5x-2+6x=20\Rightarrow 11x=22\Rightarrow x=\tfrac{22}{11}=2$\newline $y=1-3\cdot 2=-5$\newline $\res{x=2,\ \ y=-5}$
  \task $-y=-4-2x\Rightarrow y=2x+4$\newline $4x+3(2x+4)=2\Rightarrow 4x+6x+12=2\Rightarrow 10x=-10\Rightarrow x=\tfrac{-10}{10}=-1$\newline $y=2\cdot (-1)+4=2$\newline $\res{x=-1,\ \ y=2}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $x=-5-4y$\newline $3(-5-4y)-2y=13\Rightarrow -15-12y-2y=13\Rightarrow -14y=28\Rightarrow y=\tfrac{28}{-14}=-2$\newline $x=-5-4\cdot (-2)=3$\newline $\res{x=3,\ \ y=-2}$
  \task $-y=13-5x\Rightarrow y=5x-13$\newline $2x+3(5x-13)=12\Rightarrow 2x+15x-39=12\Rightarrow 17x=51\Rightarrow x=\tfrac{51}{17}=3$\newline $y=5\cdot 3-13=2$\newline $\res{x=3,\ \ y=2}$
\end{tasks}

\ex{3} \begin{tasks}(1)
  \task $2x=12-3y\Rightarrow x=\tfrac{12-3y}{2}$\newline $5\cdot \tfrac{12-3y}{2}-4y=7\Rightarrow 5(12-3y)-8y=14\Rightarrow 60-15y-8y=14\Rightarrow -23y=-46\Rightarrow y=\tfrac{-46}{-23}=2$\newline $x=\tfrac{12-3\cdot 2}{2}=3$\newline $\res{x=3,\ \ y=2}$
  \task $-2y=-1-3x\Rightarrow y=\tfrac{3x+1}{2}$\newline $4x+5\cdot \tfrac{3x+1}{2}=14\Rightarrow 8x+5(3x+1)=28\Rightarrow 8x+15x+5=28\Rightarrow 23x=23\Rightarrow x=\tfrac{23}{23}=1$\newline $y=\tfrac{3\cdot 1+1}{2}=2$\newline $\res{x=1,\ \ y=2}$
\end{tasks}

\ex{3} $-3(3-x)=-9+3x$: el $-3$ multiplica els dos termes. \ $y=3-x$\newline $2x-3(3-x)=1\Rightarrow 2x-9+3x=1\Rightarrow 5x=10\Rightarrow x=\tfrac{10}{5}=2$\newline $y=3-2=1$\newline $\res{x=2,\ \ y=1}$
