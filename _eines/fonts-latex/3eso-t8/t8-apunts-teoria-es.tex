\seccio{sec1}{Función de proporcionalidad directa}{El doble de $x$, el doble de $y$: una recta por el origen.}

\begin{minipage}[c]{0.5\linewidth}
\begin{idea}[Función de proporcionalidad directa]
{\centering\fontsize{22}{26}\selectfont$y=\cA{m}\,x$\par}\smallskip
\chip{fmA}{$m$} es la \textbf{constante de proporcionalidad}: \ $m=\dfrac{y}{x}$, igual para todos los pares.\\[3pt]
La gráfica es una \textbf{recta que pasa por el origen} $(0,\,0)$.\\[3pt]
$m$ también es la \textbf{pendiente}: lo que aumenta la $y$ cuando la $x$ aumenta 1.
\end{idea}
\end{minipage}\hfill

  \draw[fmR,line width=1.6pt] (-1,3) -- (1,-3); \draw[fmR!60,line width=1.6pt] (-3,1) -- (3,-1);
  \node[above,font=\small,text=fmK!80!black] at (1.5,3) {$y=2x$}; \node[right,font=\small,text=fmK!60!black] at (3,1.5) {$y=\frac12x$};
  \node[above,font=\small,text=fmR] at (-1,3) {$y=-3x$}; \node[left,font=\small,text=fmR!70!black] at (-3,1) {$y=-\frac13x$};

\pas{1} Calculamos la \textbf{pendiente}.\\[4pt]
\pas{2} Punto-pendiente con \textbf{uno} de los puntos.\\[4pt]
\pas{3} \textbf{Despejamos} la $y$.\\[4pt]

\end{tcolorbox}

{\centering$y-4=-2\,(x-3)\ \to\ \boxed{y=-2x+10}$\par}
\end{tcolorbox}

\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Desplazamiento: \ $y=ax^2+c$}]
Es la parábola $y=ax^2$ desplazada \textbf{$c$ unidades} hacia arriba (o abajo si $c<0$). El vértice es $(0,\,c)$.\par\smallskip
{\centering$y=x^2-4$: \ vértice $(0,\,-4)$, cortes $x^2=4\to x=\pm2$\par}
\end{tcolorbox}

\begin{exemple}[Ejemplo: \ $a$ para que $y=ax^2$ pase por $(2,\,-6)$]
$-6=a\cdot2^2\ \to\ \boxed{a=-\tfrac32}$ \qquad Pasa también por $(-2,\,-6)$, por \textbf{simetría}: $x=-2$ tiene la misma imagen que $x=2$.
\end{exemple}

\seccio{sec7}{Representación de $y=ax^2+bx+c$}{Vértice, eje y cortes: con eso ya la tenemos.}

\begin{idea}[Elementos de la parábola]
\begin{minipage}[t]{0.5\linewidth}
\textbf{Forma}: el signo de $a$ ($\cup$ o $\cap$) y $|a|$.\\[3pt]
\textbf{Vértice}: \ $x_v=\dfrac{-b}{2a}$, \ $y_v=f(x_v)$.\\[3pt]
\textbf{Eje de simetría}: la recta $x=x_v$.
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}
\textbf{Corte con el eje $Y$}: \ $(0,\,c)$.\\[3pt]
\textbf{Cortes con el eje $X$}: resolvemos $ax^2+bx+c=0$.\\[3pt]
¿Cuántos? Lo dice $\Delta=b^2-4ac$.
\end{minipage}
\end{idea}

\vfill

\end{tcolorbox}

\begin{tikzpicture}[scale=0.42]\eixos{0}{4}{-1}{3}\draw[fmblue,line width=1.5pt,domain=0.3:3.7,samples=40] plot (\x,{(\x-2)*(\x-2)});\fill[fmgold] (2,0) circle (5pt);\end{tikzpicture}
\end{tcolorbox}

\pas{4} Eje $X$: \ $x=\dfrac{2\pm\sqrt{4+12}}{2}=\dfrac{2\pm4}{2}$ \ $\to$ \ $(-1,\,0)$, \ $(3,\,0)$\\[3pt]
\pas{5} Tabla alrededor del vértice, \textbf{simétrica}:\\[3pt]
\hspace*{1.5em}\tv[fmex]{$x$ & $-1$ & 0 & 1 & 2 & 3}{$y$ & 0 & $-3$ & $-4$ & $-3$ & 0}

\seccio{sec8}{Problemas con parábolas}{¿Cuál es el valor máximo (o mínimo)? Mira el vértice.}

\begin{tcbraster}[raster columns=3,raster equal height,raster column skip=8pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faBasketballBall\ \ Lanzamientos}]

\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\faCoins\ \ Ingresos}]

{\large El \textbf{vértice} responde: \ \cA{$x_v$} dice \textbf{cuándo} (o con qué valor) \ y \ \cB{$y_v$}, \textbf{cuánto} vale.}
\end{tcolorbox}

\rtile{8.1}{Proporcionalidad}{$y=mx$: recta por el origen.\\[1ex] $m=\frac{y}{x}$. \ $m>0$ crece, $m<0$ decrece.}
\rtile{8.2}{Función lineal}{$y=mx+n$. \ $m$: pendiente; \ $n$: corte con el eje $Y$.\\[1ex] $m=0$: horizontal. \ $x=k$: no es función.}
\rtile{8.3}{Ecuación de una recta}{$m=\frac{y_2-y_1}{x_2-x_1}$.\\[1ex] $y-y_0=m(x-x_0)$. Paralelas: misma $m$.}
\rtile{8.4}{Aplicaciones}{$n$: valor inicial. \ $m$: ritmo de cambio.\\[1ex] Dominio con sentido.}
\rtile{8.5}{Dos rectas}{$m$ distinta: secantes. Misma $m$: paralelas o coincidentes.\\[1ex] Punto de corte: igualamos.}
\rtile{8.6}{$y=ax^2$}{$a>0$: $\cup$, mínimo. \ $a<0$: $\cap$, máximo.\\[1ex] $|a|$ grande: más cerrada.}
\rtile{8.7}{$y=ax^2+bx+c$}{$x_v=\frac{-b}{2a}$. Eje $Y$: $(0,\,c)$.\\[1ex] Eje $X$: $ax^2+bx+c=0$ ($\Delta$).}
\rtile{8.8}{Problemas}{El máximo o el mínimo es el \textbf{vértice}: $x_v$ cuándo, $y_v$ cuánto.}
