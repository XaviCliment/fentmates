\seccio{1}{El teorema de Pitàgores}

\ex{1} \begin{tasks}(3)
  \task $\sqrt{100}=\res{10\text{ cm}}$
  \task $\sqrt{169}=\res{13\text{ cm}}$
  \task $\sqrt{1681}=\res{41\text{ m}}$
  \task $\sqrt{58}\approx\res{7{,}62\text{ cm}}$
  \task $\sqrt{6{,}25}=\res{2{,}5\text{ m}}$
  \task $\sqrt{200}\approx\res{14{,}14\text{ cm}}$
\end{tasks}

\ex{1} \begin{tasks}(3)
  \task $\sqrt{576}=\res{24}$
  \task $\sqrt{225}=\res{15}$
  \task $\sqrt{36}=\res{6}$
  \task $\sqrt{119}\approx\res{10{,}91}$
  \task $\sqrt{36}=\res{6}$
  \task $\sqrt{65}\approx\res{8{,}06}$
\end{tasks}

\ex{1} a) $x=\sqrt{144+256}=\res{20}$ \qquad b) $x=\sqrt{169-25}=\res{12}$ \qquad c) $x=\sqrt{289-225}=\res{8}$

\ex{2} L'escala és la hipotenusa: $h=\sqrt{6{,}5^2-2{,}5^2}=\sqrt{42{,}25-6{,}25}=\sqrt{36}=\res{6\text{ m}}$.

\ex{2} $h=\sqrt{50^2-30^2}=\sqrt{1600}=\res{40\text{ m}}$.

\ex{2} Els dos trams són perpendiculars: $d=\sqrt{9^2+12^2}=\sqrt{225}=\res{15\text{ km}}$.

\ex{2} $\sqrt{6^2+0{,}8^2}=\sqrt{36{,}64}\approx\res{6{,}05\text{ m}}$.

\ex{3} $x^2+(x+2)^2=100\Rightarrow 2x^2+4x-96=0\Rightarrow x^2+2x-48=0\Rightarrow x=\frac{-2\pm14}{2}$: $x=6$ ($x=-8$ no té sentit). Catets: $\res{6\text{ cm i }8\text{ cm}}$.

\ex{3} Si es trenca a altura $x$, el tros caigut mesura $9-x$ i és la hipotenusa: $x^2+3^2=(9-x)^2\Rightarrow x^2+9=81-18x+x^2\Rightarrow18x=72\Rightarrow\res{x=4\text{ m}}$.

\ex{1} \begin{tasks}(3)

\end{tasks}

\ex{1} Isòsceles: $h=\sqrt{17^2-8^2}=\sqrt{225}=\res{15\text{ cm}}$. \ Equilàter: $h=\sqrt{12^2-6^2}=\sqrt{108}\approx\res{10{,}39\text{ cm}}$.

\ex{1} $l=\sqrt{9^2+12^2}=\sqrt{225}=\res{15\text{ cm}}$; \ $P=4\cdot15=\res{60\text{ cm}}$.

\ex{2} $l^2+l^2=144\Rightarrow l=\sqrt{72}\approx\res{8{,}49\text{ cm}}$; \ $P=4\sqrt{72}\approx\res{33{,}94\text{ cm}}$.

\ex{2} $c=\sqrt{(15-9)^2+8^2}=\sqrt{100}=\res{10\text{ cm}}$; \ $P=15+9+8+10=\res{42\text{ cm}}$.

\ex{2} $\frac{22-12}{2}=5$; \ $h=\sqrt{13^2-5^2}=\sqrt{144}=\res{12\text{ cm}}$.

\ex{2} \begin{tasks}(3)
  \task $\sqrt{9+16}=\res{5}$
  \task $\sqrt{36+64}=\res{10}$
  \task $\sqrt{16+25}=\sqrt{41}\approx\res{6{,}40}$
\end{tasks}

\ex{3} $AB=6$, \ $AC=\sqrt{9+16}=5$, \ $BC=\sqrt{9+16}=5$: \rest{isòsceles}, $P=\res{16}$. \ $36<25+25$: \rest{acutangle}.

\ex{3} Meitat de l'altra diagonal: $\sqrt{10^2-6^2}=8$; \ diagonal: $\res{16\text{ cm}}$.

\ex{3} $d=32\cdot2{,}54=81{,}28$ cm. \ $(16k)^2+(9k)^2=81{,}28^2\Rightarrow337k^2=6606{,}44\Rightarrow k\approx4{,}43$. \ Amplària $\approx\res{70{,}8\text{ cm}}$, altura $\approx\res{39{,}8\text{ cm}}$.

\ex{3} $A=\frac{20\cdot15}{2}=\res{150\text{ cm}^2}$; \ $c=\sqrt{625}=\res{25\text{ cm}}$. Prenent la hipotenusa com a base: $\frac{25\cdot h}{2}=150\Rightarrow\res{h=12\text{ cm}}$.

\ex{3} $b+h=17$ i $b^2+h^2=169$: \ $b^2+(17-b)^2=169\Rightarrow b^2-17b+60=0\Rightarrow b=12$ o $b=5$. Costats $\res{12\text{ cm i }5\text{ cm}}$; \ $A=\res{60\text{ cm}^2}$.

\ex{2} $a=\sqrt{10{,}2^2-6^2}=\sqrt{68{,}04}\approx\res{8{,}25\text{ cm}}$; \ $A\approx\frac{60\cdot8{,}25}{2}\approx\res{247{,}5\text{ cm}^2}$.

\ex{2} La diagonal és un diàmetre: $l^2+l^2=12^2\Rightarrow l=\sqrt{72}\approx\res{8{,}49\text{ cm}}$; \ $a=\frac l2\approx\res{4{,}24\text{ cm}}$; \ $A=l^2=\res{72\text{ cm}^2}$.

\ex{2} $a=\sqrt{100-25}=\sqrt{75}$; \ $A=\frac{60\cdot\sqrt{75}}{2}\approx259{,}81\text{ cm}^2$. \ $\frac{30\,000}{259{,}81}\approx115{,}5$: \ $\res{116\text{ rajoles}}$.
