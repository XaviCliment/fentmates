\seccio{4}{Operaciones con intervalos}

\exf{1} Dados $A=[-2,\, 3]$ y $B=(1,\, 5)$, represéntalos en la misma recta y calcula $A\cup B$ y $A\cap B$.\recorda{La unión $A\cup B$ son los números que están en $A$ \textbf{o} en $B$; la intersección $A\cap B$, los que están en $A$ \textbf{y} en $B$ a la vez.}

\ex{2} Sean $A=(-\infty,\, 1]$, $B=[-3,\, 4)$ y $C=(2,\, +\infty)$. Calcula:\begin{tasks}(3)
  \task $A\cup B$
  \task $A\cap B$
  \task $B\cap C$
  \task $A\cap C$
  \task $B\cup C$
  \task $A\cup C$
\end{tasks}

\ex{2} Calcula y representa:\begin{tasks}(3)
  \task $[0,\, 4)\cup[4,\, 7]$
  \task $[0,\, 4)\cap[4,\, 7]$
  \task $(-5,\, -1]\cap[-1,\, 2)$
  \task $(-\infty,\, 3)\cap(-\infty,\, 0]$
  \task $(-2,\, 6)\cap[1,\, 3]$
\end{tasks}

\ex{2} Escribe con un intervalo o una semirrecta los números que cumplen a la vez las dos condiciones:\begin{tasks}(2)

\ex{2} \begin{tasks}(1)

\end{tasks}

\seccio{6}{Radicales y potencias de exponente fraccionario}

\exf{1} Escribe en forma de potencia:\begin{tasks}(6)

\ex{1} Escribe en forma de radical:\begin{tasks}(6)

\end{tasks}

\ex{2} Calcula sin calculadora:\begin{tasks}(4)
  \task $9^{\frac12}$

\seccio{10}{Racionalización}

\exf{1} Racionaliza y simplifica:\begin{tasks}(6)
  \task $\dfrac{1}{\sqrt2}$

  \task $\dfrac{6}{\sqrt2}$
  \task $\dfrac{5}{\sqrt{10}}$
  \task $\dfrac{2}{3\sqrt5}$
  \task $\dfrac{\sqrt2}{\sqrt6}$
\end{tasks}\recorda{Si en el denominador hay $\sqrt b$, multiplica el numerador y el denominador por $\sqrt b$: $\ \dfrac{6}{\sqrt3}=\dfrac{6\sqrt3}{3}=2\sqrt3$.}

\ex{2} Racionaliza y simplifica:\begin{tasks}(6)
  \task $\dfrac{1}{\sqrt[3]2}$

  \task $\dfrac{4}{\sqrt[3]4}$
  \task $\dfrac{10}{\sqrt[3]{25}}$
  \task $\dfrac{1}{\sqrt[4]2}$
  \task $\dfrac{3}{\sqrt[4]{27}}$
\end{tasks}\recorda{Con $\sqrt[n]{b^m}$ en el denominador, multiplica por $\sqrt[n]{b^{n-m}}$, porque $\sqrt[n]{b^m}\cdot\sqrt[n]{b^{n-m}}=\sqrt[n]{b^n}=b$.}

\ex{2} Racionaliza multiplicando por el conjugado:\begin{tasks}(4)

\end{tasks}\recorda{El conjugado de $a+\sqrt b$ es $a-\sqrt b$, y $\left(a+\sqrt b\right)\left(a-\sqrt b\right)=a^2-b$.}
