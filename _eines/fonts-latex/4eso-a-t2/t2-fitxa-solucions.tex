\ex{1} \begin{tasks}(3)
  \task $\frac{75}{100}=\res{\frac34}$
  \task $\frac{14-1}{9}=\res{\frac{13}{9}}$
  \task $\frac{218-2}{99}=\frac{216}{99}=\res{\frac{24}{11}}$
  \task $\frac{16-1}{90}=\frac{15}{90}=\res{\frac16}$
  \task $\frac{325-32}{90}=\res{\frac{293}{90}}$
  \task $\frac{3}{90}=\res{\frac1{30}}$
\end{tasks}

\ex{2} \begin{tasks}(2)
  \task $\frac39+\frac69=\res{1}$
  \task $\frac{23}{9}-\frac{11}{9}=\frac{12}{9}=\res{\frac43}$
  \task $\frac{11}{9}\cdot\frac69=\res{\frac{22}{27}}$
  \task $\frac16\cdot\frac65=\res{\frac15}$
\end{tasks}

\ex{3} $0,\widehat9=\frac99=\res{1}$; \ $2,4\widehat9=\frac{249-24}{90}=\frac{225}{90}=\res{\frac52=2,5}$. Un decimal amb període 9 és igual a un decimal exacte: no hi ha cap nombre entre $0,\widehat9$ i $1$.

\seccio{11}{Problemes}

\exf{2} Costat: $\sqrt{50}=\res{5\sqrt2\ \text{cm}}$; \ perímetre: $4\cdot5\sqrt2=\res{20\sqrt2\approx28,28\ \text{cm}}$; \ diagonal (Pitàgores): $\sqrt{50+50}=\sqrt{100}=\res{10\ \text{cm}}$.

\ex{2} Perímetre: $2\left(2\sqrt3+3\sqrt3\right)=\res{10\sqrt3\ \text{cm}}$; \ àrea: $\sqrt{12\cdot27}=\sqrt{324}=\res{18\ \text{cm}^2}$; \ diagonal: $\sqrt{12+27}=\res{\sqrt{39}\ \text{cm}}$.

\ex{2} Hipotenusa: $\sqrt{8+12}=\sqrt{20}=\res{2\sqrt5\ \text{cm}}$; \ àrea: $\dfrac{\sqrt8\cdot\sqrt{12}}{2}=\dfrac{\sqrt{96}}{2}=\dfrac{4\sqrt6}{2}=\res{2\sqrt6\ \text{cm}^2}$.

\ex{2} Aresta: $\sqrt[3]{54}=\res{3\sqrt[3]2\approx3,78\ \text{cm}}$; \ superfície: $6\left(3\sqrt[3]2\right)^2=6\cdot9\sqrt[3]4=\res{54\sqrt[3]4\approx85,72\ \text{cm}^2}$.

\ex{3} Altura (Pitàgores amb la meitat del costat): $h=\sqrt{\left(2\sqrt3\right)^2-\left(\sqrt3\right)^2}=\sqrt{12-3}=\res{3\ \text{cm}}$; \ àrea: $\dfrac{2\sqrt3\cdot3}{2}=\res{3\sqrt3\ \text{cm}^2}$.

\ex{3} Llarg: $\res{21\sqrt2\approx29,70\ \text{cm}}$. \ Plegat: llarg 21 cm i curt $\frac{21\sqrt2}{2}$ cm; \ $\dfrac{21}{\frac{21\sqrt2}{2}}=\dfrac{2}{\sqrt2}=\dfrac{2\sqrt2}{2}=\res{\sqrt2}$.
