\seccio{4}{Producte de polinomis}

\exf{1} Multiplica:\begin{tasks}(2)
  \task $3x\left(x^2-2x+5\right)$
  \task $-2x^2\left(x^3-4x+1\right)$
  \task $\frac12x\left(4x^2-6x+2\right)$
  \task $5x^3\left(-x^2+x-3\right)$
  \task $\left(x^2-3x+1\right)\cdot(-4x)$
  \task $-x\left(2x^4-x^2+7\right)$
\end{tasks}\recorda{Es multiplica el monomi per cada terme del polinomi. Compte amb els signes: $(-)\cdot(-)=+$.}

\ex{1} Multiplica i redueix:\begin{tasks}(2)
  \task $(x+2)(x-5)$
  \task $(2x-1)(3x+4)$
  \task $\left(x^2+1\right)(x-3)$
  \task $\left(x^2-2x+3\right)(x+1)$
  \task $\left(2x^2-x+4\right)(x-2)$
  \task $\left(x^2+x-1\right)\left(x^2-x+2\right)$
\end{tasks}\recorda{Cada terme del primer polinomi es multiplica per cada terme del segon; després es redueixen els termes semblants.}

\ex{2} Donats $P(x)=x^2-3x+2$, $Q(x)=2x+1$ i $R(x)=x-4$, calcula:\begin{tasks}(3)
  \task $P(x)\cdot Q(x)$
  \task $P(x)\cdot R(x)-Q(x)$
  \task $[Q(x)+R(x)]\cdot P(x)$
\end{tasks}

\ex{2} Opera i redueix:\begin{tasks}(1)
  \task $2x(x-3)-x(x+1)$
  \task $(x+2)(x-1)-(x-3)(x+4)$
  \task $3\left(x^2-1\right)-(x+1)(2x-3)$
\end{tasks}

\ex{3} Un rectangle fa $2x+3$ cm de base i $x-1$ cm d'altura. Escriu el polinomi que dona l'àrea i calcula-la per a $x=5$. I el volum d'una caixa d'arestes $x$, $x+2$ i $x+5$? Calcula'l per a $x=3$.

\seccio{5}{Identitats notables}

\exf{1} Desenvolupa aplicant les identitats notables:\begin{tasks}(3)
  \task $(x+3)^2$
  \task $(x-4)^2$
  \task $(x+5)(x-5)$
  \task $(2x+1)^2$
  \task $(3x-2)^2$
  \task $(2x+7)(2x-7)$
  \task $\left(x^2+3\right)^2$
  \task $(5-x)^2$
  \task $\left(\frac12x+1\right)^2$
\end{tasks}\recorda{$(a+b)^2=a^2+2ab+b^2$, \quad $(a-b)^2=a^2-2ab+b^2$, \quad $(a+b)(a-b)=a^2-b^2$.}

\ex{2} Opera i simplifica:\begin{tasks}(2)
  \task $(x+1)^2-(x-1)^2$

  \task $2(x-3)^2-(x+1)^2$
\end{tasks}

\ex{2} Completa els buits:\begin{tasks}(1)
  \task $(x+\square)^2=x^2+\square+36$
  \task $(\square-5)^2=4x^2-\square+25$
  \task $(3x+\square)(3x-\square)=\square-16$
\end{tasks}

\ex{3} Calcula sense calculadora, fent servir les identitats notables: \ $101^2$, \ $99^2$ \ i \ $51\cdot49$.
