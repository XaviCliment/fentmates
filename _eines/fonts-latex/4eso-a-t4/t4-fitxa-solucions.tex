\seccio{3}{Problemes de primer grau}

\exf{1} $x+(x+1)+(x+2)=147\Rightarrow3x=144\Rightarrow x=48$: $\res{48,\ 49,\ 50}$

\ex{1} $0,8x=68\Rightarrow x=\res{85\ \text{€}}$.

\ex{2} $41-x=4(14-x)\Rightarrow3x=15\Rightarrow x=5$: $\res{\text{fa 5 anys}}$ (36 i 9 anys).

\ex{2} $2x+(x-4)=50\Rightarrow x=18$: $\res{18,\ 18\ \text{i}\ 14\ \text{cm}}$

\ex{2} Després de dilluns queden $\frac23x$; dimarts se'n gasten $\frac25\cdot\frac23x=\frac4{15}x$: $x-\dfrac x3-\dfrac{4x}{15}=240\Rightarrow\dfrac{6x}{15}=240\Rightarrow x=\res{600\ \text{L}}$.

\ex{2} $8\cdot30+12x=9,5(30+x)\Rightarrow240+12x=285+9,5x\Rightarrow2,5x=45\Rightarrow x=\res{18\ \text{kg}}$.

\ex{2} $x+2x+(2x+200)=1400\Rightarrow5x=1200\Rightarrow x=240$: $\res{240\ \text{€},\ 480\ \text{€}\ \text{i}\ 680\ \text{€}}$

\ex{3} Si el segon tarda $x$ hores, $120x=90(x+1,5)\Rightarrow30x=135\Rightarrow x=4,5$: $\res{4\ \text{h}\ 30\ \text{min}}$, a $\res{540\ \text{km}}$.

\ex{3} En una hora omplin $\frac16$ i $\frac13$ del dipòsit. En $x$ hores: $\dfrac x6+\dfrac x3=1\Rightarrow x+2x=6\Rightarrow x=\res{2\ \text{h}}$.

\seccio{10}{Equacions amb radicals}

\exf{1} \begin{tasks}(3)
  \task $x+3=16$: $\res{x=13}$
  \task $2x-1=9$: $\res{x=5}$
  \task $x^2=16$: $\res{x=\pm4}$
  \task Una arrel no pot ser negativa: $\res{\text{no té solució}}$
  \task $\sqrt x=3$: $\res{x=9}$
  \task $5x-1=4$: $\res{x=1}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $x+7=x^2+2x+1\Rightarrow x^2+x-6=0\Rightarrow x=2,\ x=-3$; $x=-3$ no val ($\sqrt4\neq-2$): $\res{x=2}$
  \task $x-4=\sqrt{x+2}\Rightarrow x^2-9x+14=0\Rightarrow x=7,\ x=2$; $x=2$ no val ($2-2\neq4$): $\res{x=7}$
  \task $\sqrt{2x+1}=x-1\Rightarrow x^2-4x=0\Rightarrow x=0,\ x=4$; $x=0$ no val ($1+1\neq0$): $\res{x=4}$
  \task $3x-2=x^2\Rightarrow x^2-3x+2=0$; totes dues valen: $\res{x=1,\ \ x=2}$
  \task $\sqrt{x-1}=3-x\Rightarrow x^2-7x+10=0\Rightarrow x=2,\ x=5$; $x=5$ no val ($5+2\neq3$): $\res{x=2}$
  \task $4(x+1)=(x+1)^2\Rightarrow x^2-2x-3=0$; totes dues valen: $\res{x=-1,\ \ x=3}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $2x+3=x+5$: $\res{x=2}$
  \task $5x-4=x+8$: $\res{x=3}$
  \task $9(x-1)=x+7$: $\res{x=2}$
  \task $x^2-x-12=0$; totes dues valen ($\sqrt9=\sqrt9$, $\sqrt2=\sqrt2$): $\res{x=4,\ \ x=-3}$
  \task $x^2+3x=4x\Rightarrow x^2-x=0$; totes dues valen: $\res{x=0,\ \ x=1}$
  \task $x^2-2x-3=0\Rightarrow x=3,\ x=-1$; $x=-1$ no val ($\sqrt{-2}$ no existeix): $\res{x=3}$
\end{tasks}

\ex{3} \begin{tasks}(1)
  \task $\sqrt{25-x^2}=x-1\Rightarrow 25-x^2=x^2-2x+1\Rightarrow x^2-x-12=0\Rightarrow x=4,\ x=-3$; $x=-3$ no val ($-3-4\neq1$): $\res{x=4}$
  \task $\sqrt{x^2+5}=2x-1\Rightarrow x^2+5=4x^2-4x+1\Rightarrow3x^2-4x-4=0\Rightarrow x=2,\ x=-\frac23$; $x=-\frac23$ no val ($\frac73+1\neq-\frac43$): $\res{x=2}$
\end{tasks}

\ex{3} $\sqrt{x+2}=x-4\Rightarrow x^2-9x+14=0\Rightarrow x=7,\ x=2$; $x=2$ no val: $\res{x=7}$
