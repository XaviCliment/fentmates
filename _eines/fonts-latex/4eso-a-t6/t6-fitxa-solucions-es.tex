\seccio{4}{Continuidad y tipos de discontinuidades}

\exf{1} \begin{tasks}(1)
  \task Discontinuidad $\res{\text{evitable}}$: los dos lados llegan a 3, pero $f(2)$ no existe. Sería continua si $\res{f(2)=3}$.
  \task Discontinuidad de $\res{\text{salto finito}}$: por la izquierda llega a 1 (y $f(2)=1$) y por la derecha a 3.
  \task $\res{\text{Continua}}$: los dos lados llegan a 1 y $f(2)=1$. Tiene un pico, pero se dibuja sin levantar el lápiz.
  \task Discontinuidad $\res{\text{asintótica}}$: por la izquierda se va a $-\infty$ y por la derecha a $+\infty$ (asíntota vertical $x=2$).
\end{tasks}

\ex{2} $x=-3$: $\res{\text{evitable}}$ (el punto está fuera de lugar). \ $x=0$: $\res{\text{salto finito}}$ (de 3 a 1). \ $x=2$: $\res{\text{salto infinito}}$ (por la izquierda llega a 0 y por la derecha se va a $+\infty$).
