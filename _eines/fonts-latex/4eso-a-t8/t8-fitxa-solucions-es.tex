\seccio{1}{Figuras semejantes}

  \task \rest{No}: pueden ser más o menos alargados.
  \task \rest{Sí}: todos los ángulos miden 60°.
  \task \rest{No}: los ángulos son distintos.
\end{tasks}

\ex{1} \begin{tasks}(1)
  \task $k=\frac{10}{4}=\frac{15}{6}=\res{2{,}5}$: \rest{ampliación}
  \task $k=\frac{1{,}5}{3}=\frac24=\frac{2{,}5}{5}=\res{0{,}5}$: \rest{reducción}
\end{tasks}

\ex{1} a) $\res{15,\ 21\text{ y }27\text{ cm}}$. \quad b) $5\cdot0{,}4=2$, \ $7\cdot0{,}4=2{,}8$, \ $9\cdot0{,}4=3{,}6$: \ $\res{2,\ 2{,}8\text{ y }3{,}6\text{ cm}}$.

\ex{1} $k=\frac46=\res{\frac23}$; \quad $9\cdot\frac23=\res{6\text{ cm}}$.

\ex{2} $k=\frac64=\frac{4{,}5}{3}=\res{1{,}5}$; \quad $x=1\cdot1{,}5=\res{1{,}5}$; \quad $y=2\cdot1{,}5=\res{3}$.

\ex{2} \begin{tasks}(2)
  \task $\frac64=\frac{15}{10}=1{,}5$: \rest{sí}, $k=1{,}5$
  \task $\frac{10}{5}=2\neq\frac{18}{8}=2{,}25$: \rest{no}
\end{tasks}

\ex{2} $P'=24\cdot2{,}5=\res{60\text{ cm}}$; \quad $A'=32\cdot2{,}5^2=32\cdot6{,}25=\res{200\text{ cm}^2}$.

\ex{2} Lado $6\cdot\frac13=\res{2\text{ cm}}$; \ área $2^2=\res{4\text{ cm}^2}$ (antes, 36 cm²). El área es $\res{9}$ veces más pequeña ($3^2=9$).

\ex{2} $P=6+8+10=24$; \ $k=\frac{72}{24}=3$: \ $\res{18,\ 24\text{ y }30\text{ cm}}$.

\ex{2} $k=\frac{24}{12}=2$, \ $k^3=8$: \ $0{,}25\cdot8=\res{2\text{ L}}$.

\ex{2} Arista $\res{6\text{ cm}}$ ($\times3$); \ área $6\cdot6^2=\res{216\text{ cm}^2}$ (antes 24 cm²: $\times9$); \ volumen $6^3=\res{216\text{ cm}^3}$ (antes 8 cm³: $\times27$).

\ex{3} Lados $\sqrt9=\res{3\text{ cm}}$ y $\sqrt{81}=\res{9\text{ cm}}$; \ $k=\frac93=\res{3}$ (el área se multiplica por $3^2=9$).

\ex{3} $A=150$ cm²; \ $k^2=\frac{600}{150}=4\Rightarrow k=\res{2}$. \ Foto de $\res{20\text{ cm}\times30\text{ cm}}$.

\ex{3} $k=\frac{15}{5}=3$: catetos $\res{9\text{ cm}}$ y $\res{12\text{ cm}}$. \ Áreas: $\frac{3\cdot4}{2}=6$ cm² y $\frac{9\cdot12}{2}=54$ cm²: el área es $\res{9}$ veces mayor ($3^2$).

\ex{3} $k^3=\frac{54}{2}=27\Rightarrow k=\sqrt[3]{27}=\res{3}$; \ superficie $6\cdot3^2=\res{54\text{ cm}^2}$.

  \task $1\,500\,000:500\,000=\res{3\text{ cm}}$

\ex{1} \begin{tasks}(1)
  \task $150\,000:3=50\,000$: \ $\res{1:50\,000}$
  \task $1000:5=200$: \ $\res{1:200}$
  \task $4\,000\,000:2=2\,000\,000$: \ $\res{1:2\,000\,000}$
\end{tasks}

\ex{2} 1 cm son 500 m $=50\,000$ cm: \ $\res{1:50\,000}$. \ $6{,}5\cdot500=3250$ m $=\res{3{,}25\text{ km}}$.

\ex{2} $8\cdot50=400$ cm $=4$ m; \ $6{,}4\cdot50=320$ cm $=3{,}2$ m. \ $\res{4\text{ m}\times3{,}2\text{ m}}$, \ superficie $\res{12{,}8\text{ m}^2}$.

\ex{2} $18\cdot25\,000=450\,000$ cm $=\res{4{,}5\text{ km}}$. \ $4{,}5:5=0{,}9$ h $=\res{54\text{ minutos}}$.

\ex{2} $3600:72=\res{50\text{ cm}}$ de largo; \ $2880:72=\res{40\text{ cm}}$ de ala a ala.

\ex{2} $9\cdot500=4500$ cm $=\res{45\text{ m}}$.

\ex{3} $180\text{ km}=18\,000\,000$ cm; \ $18\,000\,000:6=3\,000\,000$: \ $\res{1:3\,000\,000}$. \ $2{,}5\cdot30=\res{75\text{ km}}$ (1 cm son 30 km).

\ex{3} 1 cm del plano es 1 m real: $25\text{ m}\times16\text{ m}=\res{400\text{ m}^2}$. \ $400\cdot150=\res{60\,000\text{ €}}$.

\ex{3} $8:20=0{,}4$ cm $=\res{4\text{ mm}}$.

\ex{3} $V=1{,}5\cdot200^3=12\,000\,000\text{ dm}^3=\res{12\,000\text{ m}^3}$; \ $A=4\cdot200^2=160\,000\text{ dm}^2=\res{1600\text{ m}^2}$.

\seccio{3}{El teorema de Tales}

\ex{1} \begin{tasks}(4)
  \task $4x=36\Rightarrow\res{x=9}$
  \task $10x=70\Rightarrow\res{x=7}$
  \task $12x=48\Rightarrow\res{x=4}$
  \task $2{,}5x=15\Rightarrow\res{x=6}$
\end{tasks}

\ex{1} $\frac25=\frac3x\Rightarrow2x=15\Rightarrow\res{x=7{,}5\text{ cm}}$.

\ex{1} $\frac64=\frac{4{,}5}{x}\Rightarrow6x=18\Rightarrow\res{x=3}$.

\ex{2} $\frac x8=\frac36\Rightarrow6x=24\Rightarrow\res{x=4}$.

\ex{2} $\frac{3{,}6}{4{,}8}=\frac x6\Rightarrow4{,}8x=21{,}6\Rightarrow\res{\overline{A'B'}=4{,}5}$; \quad $\overline{A'C'}=4{,}5+6=\res{10{,}5}$.

\ex{2} \begin{tasks}(2)
  \task $3\cdot6=4{,}5\cdot4=18$: \rest{sí}
  \task $2\cdot7=14\neq5\cdot3=15$: \rest{no}
\end{tasks}

\ex{2} Cada segmento de la segunda recta es $\frac32=1{,}5$ veces el de la primera: \ $\overline{B'C'}=3\cdot1{,}5=\res{4{,}5}$, \ $\overline{C'D'}=4\cdot1{,}5=\res{6}$.

\ex{2} a) $21:3=\res{7\text{ cm}}$ cada parte. \quad b) $\frac{21}{1+2+4}=3$: \ $\res{3,\ 6\text{ y }12\text{ cm}}$.

\ex{3} $\frac{84}{18+24+30}=\frac{84}{72}=\frac76$: \ $18\cdot\frac76=\res{21\text{ m}}$, \ $24\cdot\frac76=\res{28\text{ m}}$, \ $30\cdot\frac76=\res{35\text{ m}}$.

\ex{3} $\frac{x}{x+2}=\frac69\Rightarrow9x=6x+12\Rightarrow3x=12\Rightarrow\res{x=4}$.

\ex{3} Desde $A$ se traza una semirrecta auxiliar y se marcan en ella 3 segmentos iguales. Se une la última marca con $B$ y, por las otras dos, se trazan paralelas a esa recta: cortan $AB$ en 3 partes iguales (teorema de Tales). Cada parte mide $\frac{10}{3}\approx\res{3{,}33\text{ cm}}$.

\seccio{4}{Semejanza de triángulos. Criterios}

\ex{1} \begin{tasks}(1)
  \task $\frac84=\frac{10}5=\frac{12}6=2$: \rest{sí}, $k=2$
  \task $\frac63=2$, \ $\frac94=2{,}25$: \rest{no}
  \task $\frac{7{,}5}{5}=\frac{18}{12}=\frac{19{,}5}{13}=1{,}5$: \rest{sí}, $k=1{,}5$
\end{tasks}

\ex{1} \begin{tasks}(1)
  \task El tercer ángulo del primero mide 75° y el del segundo, 45°: tienen los mismos ángulos: \rest{sí}.
  \task Ángulos 50°, 50°, 80° y 50°, 90°, 40°: \rest{no}.
\end{tasks}

\ex{1} $\overline{AB}=4+2=6$; \quad $\frac46=\frac x9\Rightarrow6x=36\Rightarrow\res{x=6}$.

\ex{2} Por el teorema de Tales: $\frac63=\frac8x\Rightarrow6x=24\Rightarrow\res{x=4}$.

\ex{2} $\overline{AB}=4+4=8$; \quad $\frac48=\frac5x\Rightarrow4x=40\Rightarrow\res{x=10}$.

\ex{2} \rest{Sí}: los ángulos del primero son 90°, 40° y 50°, y los del segundo, 90°, 50° y 40°. Tienen los tres ángulos iguales.

\ex{2} $k=\frac96=1{,}5$: \ $8\cdot1{,}5=\res{12\text{ cm}}$ y $9\cdot1{,}5=\res{13{,}5\text{ cm}}$.

\ex{2} $\frac64=\frac96=1{,}5$ y el ángulo comprendido es igual: \rest{sí} (criterio 3.º), $k=1{,}5$. \ $\overline{B'C'}=5{,}3\cdot1{,}5=\res{7{,}95}$.

\ex{3} $P=18$, \ $k=\frac{45}{18}=2{,}5$: \ $\res{10,\ 15\text{ y }20\text{ cm}}$.

\ex{3} $\overline{AB}=15$; \ $\frac5{15}=\frac{x}{12}\Rightarrow\res{\overline{B'C'}=4}$; \ $\frac5{15}=\frac6{\overline{AC}}\Rightarrow\overline{AC}=18$, \ $\res{\overline{C'C}=12}$. \ $P=10+12+12+4=\res{38}$.

\ex{3} $k=\frac{10}4=2{,}5$; \ altura $3\cdot2{,}5=\res{7{,}5\text{ cm}}$; \ área $\frac{10\cdot7{,}5}{2}=\res{37{,}5\text{ cm}^2}$ (antes, 6 cm²): se ha multiplicado por $2{,}5^2=\res{6{,}25}$.

\seccio{5}{Aplicación: distancia a un punto inaccesible}

\ex{1} $\frac x3=\frac{x+12}{4}\Rightarrow4x=3x+36\Rightarrow\res{x=36\text{ m}}$.

\ex{1} Los triángulos $ABC$ y $EDC$ son semejantes (ángulo recto y ángulos en $C$ opuestos por el vértice): $\frac x{2{,}5}=\frac93\Rightarrow\res{x=7{,}5\text{ m}}$.

\ex{2} Triángulos $ABC$ y $EDC$ semejantes: $\frac x{3{,}5}=\frac{24}4\Rightarrow x=\frac{3{,}5\cdot24}{4}=\res{21\text{ m}}$.

\ex{2} $\frac x6=\frac{x+15}{6{,}5}\Rightarrow6{,}5x=6x+90\Rightarrow0{,}5x=90\Rightarrow\res{x=180\text{ m}}$.

\ex{2} $\frac{40}{\overline{DE}}=\frac{16}{2}\Rightarrow\overline{DE}=\frac{40\cdot2}{16}=\res{5\text{ m}}$.

\ex{3} $\frac{60}{4}=\frac{75}{b}\Rightarrow b=\frac{4\cdot75}{60}=\res{5\text{ m}}$.

\ex{3} $\frac x9=\frac{x+6}{10}\Rightarrow10x=9x+54\Rightarrow\res{x=54\text{ m}}$; \ Pau: $\overline{AC}=\sqrt{54^2+9^2}=\sqrt{2997}\approx\res{54{,}74\text{ m}}$.

\seccio{6}{Semejanza de triángulos rectángulos}

\ex{1} \begin{tasks}(2)
  \task $90^\circ-40^\circ=50^\circ$: \rest{sí}
  \task $\frac63=\frac{10}5=2$: \rest{sí}, $k=2$
  \task $\frac42=2\neq\frac53$: \rest{no}
  \task $\frac{26}{13}=\frac{10}5=2$: \rest{sí}, $k=2$
\end{tasks}

\ex{1} $k=\frac{20}8=\res{2{,}5}$; \ cateto $6\cdot2{,}5=\res{15\text{ cm}}$; \ hipotenusa $\sqrt{6^2+8^2}\cdot2{,}5=10\cdot2{,}5=\res{25\text{ cm}}$.

\ex{2} $h=\sqrt{4\cdot9}=\res{6\text{ cm}}$; \ $A=\frac{13\cdot6}{2}=\res{39\text{ cm}^2}$.

\ex{2} $h=\sqrt{2\cdot8}=\res{4\text{ cm}}$; \ $c=\sqrt{10\cdot2}=\sqrt{20}\approx\res{4{,}47\text{ cm}}$; \ $b=\sqrt{10\cdot8}=\sqrt{80}\approx\res{8{,}94\text{ cm}}$.

\ex{2} $c=\sqrt{20\cdot7{,}2}=\sqrt{144}=\res{12\text{ cm}}$; \ $n=20-7{,}2=12{,}8$, \ $b=\sqrt{20\cdot12{,}8}=\sqrt{256}=\res{16\text{ cm}}$; \ $h=\sqrt{7{,}2\cdot12{,}8}=\sqrt{92{,}16}=\res{9{,}6\text{ cm}}$.

\ex{2} $\sqrt{6{,}5^2-2{,}5^2}=\sqrt{36}=\res{6\text{ m}}$. \ El peldaño forma un triángulo rectángulo semejante: $\frac h{2{,}6}=\frac6{6{,}5}\Rightarrow\res{h=2{,}4\text{ m}}$.

\ex{3} $a=\sqrt{81+144}=\res{15\text{ cm}}$; \ $81=15m\Rightarrow\res{m=5{,}4\text{ cm}}$; \ $144=15n\Rightarrow\res{n=9{,}6\text{ cm}}$; \ $h=\sqrt{5{,}4\cdot9{,}6}=\sqrt{51{,}84}=\res{7{,}2\text{ cm}}$ \ (también $h=\frac{9\cdot12}{15}$).

\ex{3} $144=9n\Rightarrow\res{n=16\text{ cm}}$; \ $\res{a=25\text{ cm}}$; \ $c=\sqrt{25\cdot9}=\res{15\text{ cm}}$, \ $b=\sqrt{25\cdot16}=\res{20\text{ cm}}$; \ $P=\res{60\text{ cm}}$; \ $A=\frac{25\cdot12}2=\res{150\text{ cm}^2}$.

\seccio{7}{Problemas de semejanza}

\ex{1} $\frac{H}{9}=\frac{2}{1{,}5}\Rightarrow H=\frac{9\cdot2}{1{,}5}=\res{12\text{ m}}$.

\ex{1} $\frac{1{,}5}{s}=\frac{21}{14}\Rightarrow21s=21\Rightarrow\res{s=1\text{ m}}$.

\ex{1} $\frac{H}{15}=\frac{1{,}2}{0{,}9}\Rightarrow H=\frac{15\cdot1{,}2}{0{,}9}=\res{20\text{ m}}$.

\ex{2} $\frac{H}{15}=\frac{1{,}6}{2}\Rightarrow H=\frac{15\cdot1{,}6}{2}=\res{12\text{ m}}$.

\ex{2} Triángulos en posición de Tales: $\frac h4=\frac{0{,}9}{6}\Rightarrow h=\frac{4\cdot0{,}9}{6}=\res{0{,}6\text{ m}}$.

\ex{2} $\frac{1{,}5}{s}=\frac{4{,}5}{s+2}\Rightarrow1{,}5s+3=4{,}5s\Rightarrow3s=3\Rightarrow\res{s=1\text{ m}}$.

\ex{2} Desde el pie de la escalera, el punto está a $1{,}5-0{,}6=0{,}9$ m. \ $\frac h{0{,}9}=\frac4{1{,}5}\Rightarrow h=\frac{0{,}9\cdot4}{1{,}5}=\res{2{,}4\text{ m}}$.

\ex{2} $\frac d6=\frac{0{,}5}{0{,}1}\Rightarrow d=\frac{6\cdot0{,}5}{0{,}1}=\res{30\text{ m}}$.

\ex{2} $k=\frac{180}{4{,}5}=40$; \ $12\cdot40=480\text{ cm}=\res{4{,}8\text{ m}}$.

\ex{3} $\frac{H}{219}=\frac{1{,}8}{2{,}7}\Rightarrow H=\frac{219\cdot1{,}8}{2{,}7}=\res{146\text{ m}}$.

\ex{3} La sombra del árbol mide $1{,}6+6{,}4=8$ m (triángulos en posición de Tales): $\frac H8=\frac{1{,}2}{1{,}6}\Rightarrow H=\frac{8\cdot1{,}2}{1{,}6}=\res{6\text{ m}}$.

\ex{3} Desde el anclaje del cable (triángulos en posición de Tales): $\frac t{10}=\frac58\Rightarrow t=6{,}25$ m; \ a $10-6{,}25=\res{3{,}75\text{ m}}$ del pie.
