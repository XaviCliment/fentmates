\seccio{sec1}{Població, mostra i variables estadístiques}{Abans de comptar, cal saber a qui i què preguntem.}

\vfill
\begin{minipage}[c]{0.55\linewidth}
\begin{idea}[Els conceptes bàsics]
L'\textbf{estadística} arreplega, ordena, representa i interpreta dades per a traure'n conclusions.\\[3pt]
\textbf{Població}: tots els elements que volem estudiar. \ \textbf{Individu}: cadascun.\\[2pt]
\textbf{Mostra}: la part de la població que estudiem quan no podem preguntar a tothom. El nombre d'individus és la seua \textbf{grandària}.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.42\linewidth}
\begin{exemple}[Exemple: \ quantes hores dormen?]
Preguntem a 60 dels 600 alumnes d'un institut, triats a l'atzar.\\[3pt]

\seccio{sec2}{Taules de freqüències amb dades aïllades}{Comptem quantes vegades apareix cada valor.}

\vfill
\begin{idea}[Freqüències]
\textbf{Absoluta} $f_i$: quantes vegades apareix el valor. La suma de totes és $N$, el nombre de dades.\\[2pt]
\textbf{Relativa} $h_i=\dfrac{f_i}{N}$: la part del total (fracció, decimal o \%). Totes sumen 1 (el 100\,\%).\\[2pt]
\textbf{Acumulades}: $F_i$, les dades menors o iguals que $x_i$ (sumem les $f_i$ des de dalt); \ $H_i=\dfrac{F_i}{N}$. L'última $F_i$ és $N$.
\end{idea}

\vfill
\begin{exemple}[Exemple: \ nombre de germans de 20 alumnes]
\begin{minipage}[c]{0.4\linewidth}
\small 1 \ 2 \ 0 \ 1 \ 1 \ 3 \ 2 \ 1 \ 0 \ 2\\ 1 \ 1 \ 4 \ 0 \ 2 \ 1 \ 3 \ 1 \ 2 \ 1\\[6pt]
\normalsize Recompte amb palets: una ratlla per dada; la cinquena travessa les quatre anteriors.\\[4pt]
9 dels 20 tenen un germà: \ $h=\frac9{20}=0{,}45$, el \boxed{45\,\%}
\end{minipage}\hfill

\rowcolor{fmex}\color{white}$x_i$ & \color{white}\heavy Recompte & \color{white}$f_i$ & \color{white}$h_i$ & \color{white}\%\\
0 & $|||$ & 3 & $0{,}15$ & 15\\
\rowcolor{fmexbg}1 & $\cancel{||||}\ ||||$ & 9 & $0{,}45$ & 45\\
2 & $\cancel{||||}$ & 5 & $0{,}25$ & 25\\
\rowcolor{fmexbg}3 & $||$ & 2 & $0{,}1$ & 10\\
4 & $|$ & 1 & $0{,}05$ & 5\\
\rowcolor{fmexbg}Total & & 20 & 1 & 100
}

\seccio{sec3}{Taules amb dades agrupades en intervals}{Si hi ha molts valors diferents, els agrupem.}

\vfill
\begin{minipage}[c]{0.52\linewidth}
\begin{idea}[Intervals i marca de classe]
Per a variables contínues o amb molts valors fem \textbf{intervals} (classes) de la mateixa amplitud.\\[3pt]
$[150,\ 155\ro$: hi entra el 150, però el 155 va a l'interval següent.\\[3pt]
\textbf{Marca de classe}: el punt mitjà, \ $\dfrac{150+155}{2}=152{,}5$. Representa totes les dades de l'interval.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.45\linewidth}
\passosT{Com es fan els intervals}{\pas{1} \textbf{Rang}: $x_{\max}-x_{\min}$.\\[3pt]
\pas{2} Nombre d'intervals: entre 5 i 10, a prop de $\sqrt N$.\\[3pt]
\pas{3} \textbf{Amplitud}: rang entre el nombre d'intervals, arrodonit cap amunt a un nombre còmode.\\[3pt]
\pas{4} Comencem en un valor rodó $\le x_{\min}$.}
\end{minipage}

\pas{3} $22:5=4{,}4\ \to$ amplitud \textbf{5}\\[3pt]
\pas{4} Comencem en 140
\end{minipage}\hfill

\seccio{sec5}{Paràmetres de centralització}{Un sol valor «del mig» que resumeix totes les dades.}

\vfill
\begin{tcbraster}[raster columns=3,raster equal height,raster column skip=8pt]
\begin{tcolorbox}[fmbase,colback=fmA!8,colframe=fmA,coltitle=white,title={Moda $Mo$}]
El valor que \textbf{més es repeteix}. L'únic per a variables qualitatives.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmB!8,colframe=fmB,coltitle=white,title={Mitjana $\bar x$}]
La suma de les dades entre $N$: el \textbf{repartiment a parts iguals}.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={Mediana $Me$}]
El valor del \textbf{centre} amb les dades ordenades.
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{exemple}[Exemple: \ minuts que tarden 7 alumnes a arribar a l'institut: \ 20, 40, 15, 46, 20, 30, 25]
\begin{minipage}[c]{0.6\linewidth}
$Mo=\boxed{20}$ min \ \nota{apareix dues vegades}\\[4pt]
$\bar x=\dfrac{20+40+15+46+20+30+25}{7}=\dfrac{196}{7}=\boxed{28}$ min\\[4pt]
Ordenades: \ $15,\ 20,\ 20,\ \boxed{25},\ 30,\ 40,\ 46$: \ $Me=\boxed{25}$ min
\end{minipage}\hfill
\begin{minipage}[c]{0.36\linewidth}
Amb $N$ parell: \ $3,\ 5,\ 6,\ \boxed{7,\ 8},\ 8,\ 9,\ 10$\\[3pt]
$Me=\dfrac{7+8}{2}=7{,}5$
\end{minipage}
\end{exemple}

\vfill
\begin{compte}[La mitjana i les dades extremes]
Les dades extremes mouen molt la mitjana, però no la mediana: si n'hi ha, la \textbf{mediana} representa millor el conjunt. I abans de buscar la mediana, \textbf{ordena}!
\end{compte}

\vfill

\rowcolor{fmexbg}1 & 9 & 12 & 9\\
2 & 5 & 17 & 10\\
\rowcolor{fmexbg}3 & 2 & 19 & 6\\
4 & 1 & 20 & 4\\
\rowcolor{fmexbg}Total & 20 & & 29
}
\end{minipage}\hfill

$Me$: les centrals són la 10a i l'11a; la primera $F_i\ge10$ és 12, de l'1: \ $Me=\boxed{1}$\\[4pt]
$Mo=\boxed{1}$ \ \nota{$f=9$, la més gran}\\[3pt]

\end{minipage}

\seccio{sec6}{Paràmetres de dispersió}{Com d'escampades estan les dades al voltant de la mitjana.}

\vfill
\begin{idea}[Mateixa mitjana, dades molt diferents]
Punts en cinc partits: \ \textbf{Anna} $8,\,9,\,10,\,11,\,12$ \ i \ \textbf{Bea} $2,\,6,\,10,\,14,\,18$. Totes dues fan una mitjana de 10, però Anna és molt més regular.
\end{idea}

\vfill
\begin{minipage}[c]{0.54\linewidth}\small
\textbf{Rang}: \ $x_{\max}-x_{\min}$. Fàcil, però només depén de dues dades.\\[4pt]
\textbf{Desviació mitjana}: \ $DM=\dfrac{\sum|x_i-\bar x|\cdot f_i}{N}$\\[4pt]
\textbf{Variància}: \ $\sigma^2=\dfrac{\sum(x_i-\bar x)^2\cdot f_i}{N}$\\[4pt]
\textbf{Desviació típica}: \ $\sigma=\sqrt{\sigma^2}$, amb les mateixes unitats que les dades: la més usada.\\[4pt]
Com més grans, més escampades i \textbf{menys representativa} és la mitjana.
\end{minipage}\hfill

rang 16; \ $DM=4{,}8$; \ $\sigma^2=32$; \ $\sigma=\sqrt{32}\approx\boxed{5{,}66}$\\[5pt]
Tots diuen el mateix: les de Bea estan molt més escampades.
\end{minipage}
\end{exemple}

\vfill
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=2mm,bottom=2mm]
{\Large$\sigma^2=\dfrac{\sum x_i^2\cdot f_i}{N}-\bar x^{\,2}$}\par\smallskip
{\small la mitjana dels quadrats menys el quadrat de la mitjana: més còmoda si $\bar x$ no és entera}
\end{tcolorbox}

\vfill
\begin{exemple}[Exemple: \ germans de 20 alumnes]
\begin{minipage}[c]{0.4\linewidth}\centering\small
\taularod[fmex]{>{\heavy}c*{3}{>{\hspace{5pt}}c<{\hspace{5pt}}}}{
\rowcolor{fmex}\color{white}$x_i$ & \color{white}$f_i$ & \color{white}$x_if_i$ & \color{white}$x_i^2f_i$\\
0 & 3 & 0 & 0\\
\rowcolor{fmexbg}1 & 9 & 9 & 9\\
2 & 5 & 10 & 20\\
\rowcolor{fmexbg}3 & 2 & 6 & 18\\
4 & 1 & 4 & 16\\
\rowcolor{fmexbg}$\sum$ & 20 & 29 & 63
}
\end{minipage}\hfill
\begin{minipage}[c]{0.57\linewidth}
$\bar x=\dfrac{29}{20}=1{,}45$\\[4pt]
$\sigma^2=\dfrac{63}{20}-1{,}45^2=3{,}15-2{,}1025=1{,}0475$\\[4pt]
$\sigma=\sqrt{1{,}0475}\approx\boxed{1{,}02}$ germans\\[3pt]
\nota{una variància negativa vol dir que t'has equivocat}
\end{minipage}
\end{exemple}

\vfill
\begin{exemple}[Exemple: \ dades agrupades (diners de 40 alumnes, 9.3)]
Amb les marques de classe: \ $\sum x_i^2f_i=37{,}5+562{,}5+1875+2450+2025=6950$; \ $\bar x=11{,}75$\\[4pt]
$\sigma^2=\dfrac{6950}{40}-11{,}75^2=173{,}75-138{,}06=35{,}69$ \qquad $\sigma\approx\boxed{5{,}97\text{ €}}$ \qquad $CV=\dfrac{5{,}97}{11{,}75}\approx\boxed{51\,\%}$ \ \nota{molt escampades}
\end{exemple}

\vfill
\begin{minipage}[c]{0.5\linewidth}
\begin{idea}[Coeficient de variació]
{\centering\Large$CV=\dfrac{\sigma}{\bar x}$\par}\smallskip
Sense unitats (sovint en \%): compara la dispersió de grups amb mitjanes o unitats molt diferents.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
\begin{exemple}[Exemple: \ nadons i adults]
Nadons: $\bar x=3{,}2$, $\sigma=0{,}4$ kg: \ $CV=\boxed{12{,}5\,\%}$\\[3pt]
Adults: $\bar x=70$, $\sigma=8$ kg: \ $CV\approx\boxed{11{,}4\,\%}$\\[3pt]
En proporció, els nadons tenen \textbf{més} dispersió.
\end{exemple}
\end{minipage}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{exemple}[Exemple: \ dues marques de bombetes]
A: $\bar x=1000$ h, $\sigma=50$ h \quad B: $\bar x=1000$ h, $\sigma=200$ h\\[3pt]
Duren el mateix de mitjana, però les de \textbf{A} són molt més regulars: és més fàcil saber quant duraran.
\end{exemple}
\begin{truc}[Amb la calculadora]
En mode estadístic, la nostra $\sigma$ és la que es diu $\sigma_x$ (o $\sigma_n$). La $s_x$ divideix entre $N-1$: es fa servir en cursos posteriors.
\end{truc}
\end{tcbraster}

\seccio{sec7}{Paràmetres de posició: quartils i percentils}{Quin lloc ocupa una dada dins de les dades ordenades.}

\vfill
\begin{minipage}[c]{0.55\linewidth}
\begin{idea}[Quartils i percentils]
\textbf{Quartils}: divideixen les dades ordenades en quatre parts iguals. $Q_1$ deixa per davall el 25\,\%; \ $Q_2=Me$, el 50\,\%; \ $Q_3$, el 75\,\%.\\[3pt]
\textbf{Percentil} $P_k$: deixa per davall el $k$\,\%. \ $Q_1=P_{25}$, $Me=P_{50}$, $Q_3=P_{75}$.\\[3pt]
\textbf{Rang interquartílic}: $RI=Q_3-Q_1$, on hi ha el 50\,\% central. No el mouen les dades extremes.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.42\linewidth}
\passosT{Com es calculen}{\pas{1} \textbf{Ordenem} les $N$ dades.\\[3pt]
\pas{2} Posició: \ $\dfrac{k\cdot N}{100}$ \ ($Q_1$: $\frac N4$; \ $Q_3$: $\frac{3N}4$).\\[3pt]
\pas{3} \textbf{No entera}: la dada de la posició següent. \textbf{Entera}: la mitjana d'eixa dada i la següent.}
\end{minipage}

\vfill
\begin{exemple}[Exemple: \ notes de 12 alumnes, ja ordenades]
\centering\small
\taularod[fmex]{>{\heavy}l*{12}{>{\hspace{3pt}}c<{\hspace{3pt}}}}{
\rowcolor{fmex}\color{white}Posició & \color{white}1 & \color{white}2 & \color{white}3 & \color{white}4 & \color{white}5 & \color{white}6 & \color{white}7 & \color{white}8 & \color{white}9 & \color{white}10 & \color{white}11 & \color{white}12\\
Dada & 3 & 4 & \cellcolor{fmA!20}5 & \cellcolor{fmA!20}5 & 6 & \cellcolor{fmK!20}6 & \cellcolor{fmK!20}7 & 7 & \cellcolor{fmB!20}8 & \cellcolor{fmB!20}8 & 9 & 10
}\par\medskip\raggedright\normalsize
$Q_1$: $\frac{12}4=3$, entera: \ $\frac{5+5}2=\boxed{5}$ \qquad $Me$: $\frac{12}2=6$: \ $\frac{6+7}2=\boxed{6{,}5}$ \qquad $Q_3$: $\frac{36}4=9$: \ $\frac{8+8}2=\boxed{8}$ \qquad $RI=8-5=3$\\[4pt]
$P_{90}$: \ $\frac{90\cdot12}{100}=10{,}8$, no entera: la dada 11a, \ $P_{90}=\boxed{9}$
\end{exemple}

\vfill
\begin{exemple}[Exemple: \ amb una taula (notes de 25 alumnes, 9.2): \ la primera $F_i$ que arriba a la posició]
\begin{minipage}[c]{0.4\linewidth}\centering\small
\taularod[fmex]{>{\heavy}c*{7}{>{\hspace{2pt}}c<{\hspace{2pt}}}}{
\rowcolor{fmex}\color{white}$x_i$ & \color{white}4 & \color{white}5 & \color{white}6 & \color{white}7 & \color{white}8 & \color{white}9 & \color{white}10\\
$F_i$ & 2 & 7 & 13 & 18 & 22 & 24 & 25
}
\end{minipage}\hfill
\begin{minipage}[c]{0.57\linewidth}
$Q_1$: $\frac{25}4=6{,}25\to F=7$: \ $Q_1=\boxed{5}$\\[3pt] $Me$: $12{,}5\to F=13$: \ $Me=\boxed{6}$\\[3pt]
$Q_3$: $18{,}75\to F=22$: \ $Q_3=\boxed{8}$\\[3pt] $P_{90}$: $22{,}5\to F=24$: \ $P_{90}=\boxed{9}$\\[3pt]
\nota{si una $F_i$ coincideix exactament amb la posició, mitjana d'eixe valor i el següent}
\end{minipage}
\end{exemple}

\vfill
\begin{truc}[Percentils en la vida real]
Un nadó en el \textbf{percentil 80} de pes pesa més que el 80\,\% dels nadons de la seua edat (i menys que el 20\,\%). Amb dades agrupades, el quartil o percentil és dins del primer interval amb $F_i\ge\frac{kN}{100}$.
\end{truc}

\seccio{sec8}{Diagrama de caixes}{Cinc nombres que resumeixen tota la distribució.}

\vfill
\begin{minipage}[c]{0.5\linewidth}
\begin{idea}[El diagrama de caixes (caixa i bigotis)]
\pas{1} Una escala horitzontal amb totes les dades.\\[3pt]
\pas{2} Una \textbf{caixa} de $Q_1$ a $Q_3$, amb la \textbf{mediana} marcada.\\[3pt]
\pas{3} Els \textbf{bigotis}: de la caixa fins al mínim i fins al màxim.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}\centering
\begin{tikzpicture}[x=0.75cm]

  \foreach \x/\t in {4/mín,5/$Q_1$,6/$Me$,8/$Q_3$,10/màx} \node[above,font=\scriptsize] at (\x,1.5) {\t};
  \foreach \a/\b in {4/5,5/6,6/8,8/10} \node[font=\scriptsize,text=fmR] at ({(\a+\b)/2},2.05) {25\,\%};
\end{tikzpicture}\\
{\comic\small\color{fmmuted}notes de 25 alumnes (9.7)}
\end{minipage}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Com s'interpreta}]
Cada part (bigoti, mitja caixa, mitja caixa, bigoti) té el \textbf{25\,\%} de les dades. Una part \textbf{curta}: dades concentrades; \textbf{llarga}: escampades.\par\smallskip
La caixa fa el $RI$; el total, el rang.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Simetria}]
Mediana al centre de la caixa i bigotis iguals: \textbf{simètrica}. Si no, \textbf{asimètrica}, cap al costat de la part més llarga.\par\smallskip
En l'exemple, de 5 a 6 molt juntes; de 6 a 10, escampades.
\end{tcolorbox}
\end{tcbraster}
