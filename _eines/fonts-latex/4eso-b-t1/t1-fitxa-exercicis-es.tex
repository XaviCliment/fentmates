\seccio{2}{Intervalos y semirrectas}

\exf{1} Representa estos intervalos en la recta real y exprésalos como inecuaciones:\begin{tasks}(4)
  \task $A=(-\infty,\, 2]$
  \task $B=[1,\, 5]$
  \task $C=(2,\, 4]$
  \task $D=(-7,\, +\infty)$
\end{tasks}

\ex{2} Calcula las uniones e intersecciones siguientes, donde $A$, $B$, $C$ y $D$ son los intervalos del ejercicio anterior:\begin{tasks}(4)
  \task $A\cup B$
  \task $B\cup C$
  \task $A\cup D$
  \task $A\cup B\cup C$
  \task $A\cap B$
  \task $B\cap D$
  \task $A\cap C$
  \task $A\cap B\cap C$
\end{tasks}\recorda{La unión son los números que están en alguno de los intervalos; la intersección, los que están en todos a la vez.}


% =====================================================
\seccio{3}{Valor absoluto}

\exf{2} Averigua para qué valores de $x$ se cumplen las desigualdades siguientes. Expresa la solución gráficamente y con intervalos:\begin{tasks}(4)
  \task $|x|<5$
  \task $|x|\ge4$
  \task $|x+2|\le9$
  \task $|x+1|\ge6$
\end{tasks}\recorda{Si $r>0$: \ $|A|<r\iff -r<A<r$ \quad y \quad $|A|>r\iff A<-r$ o $A>r$.}

\seccio{4}{Potencias}

\exf{1} Calcula las potencias siguientes:\begin{tasks}(5)
  \task $(-2)^5$
  \task $-2^5$
  \task $53^0$
  \task $(-1)^{35}$
  \task $(-3)^2$
  \task $\left(\frac12\right)^{-3}$
  \task $\left(\frac32\right)^2$
  \task $\left(\frac{-1}{2}\right)^4$
  \task $\left(\frac23\right)^{-2}$
  \task $\left(\frac15\right)^{-1}$
\end{tasks}

\ex{2} Simplifica las expresiones siguientes con potencias:\begin{tasks}(2)
  \task $\left[\left(\dfrac{-1}{3}\right)^3\cdot\left(\dfrac32\right)^{-2}\right]^2$
  \task $\dfrac{(-2)^5\cdot(2^3)^4}{-2^4\cdot(-2)^{-6}}$
  \task $\dfrac{2^4\cdot3^{-2}\cdot5^{-1}}{2^{-3}\cdot3^2\cdot5^{-2}}$
  \task $\dfrac{3^{-4}\cdot16\cdot9^{-1}}{8^2\cdot3^{-5}\cdot2^{-3}}$
\end{tasks}\recorda{Descompón las bases en factores primos y aplica $a^m\cdot a^n=a^{m+n}$, $a^m:a^n=a^{m-n}$, $(a^m)^n=a^{m\cdot n}$ y $a^{-n}=\frac1{a^n}$.}

\ex{2} Extrae todos los factores que se pueda de las raíces siguientes:\begin{tasks}(4)
  \task $\sqrt{1280}$

\ex{2} Introduce los factores en las raíces y simplifica:\begin{tasks}(4)
  \task $\frac35\sqrt2$

\ex{2} Expresa con un solo radical y simplifica si es posible:\begin{tasks}(4)

\ex{2} Calcula:\begin{tasks}(2)

\seccio{7}{Logaritmos}

\exf{1} Calcula, utilizando la definición, los logaritmos siguientes:\begin{tasks}(4)
  \task $\log_2 8$
  \task $\log_3 81$
  \task $\log 1000$
  \task $\ln e^{-4}$
  \task $\log 0,0001$
  \task $\log_4 0,25$
  \task $\log_2\sqrt[6]4$
  \task $\ln e^3$
\end{tasks}\recorda{$\log_a b=c\iff a^c=b$. Además, $\log_a a=1$ y $\log_a 1=0$.}

  \task $\log 10\,000=x$
  \task $\log_a a=x$
  \task $\log_a 1=x$
\end{tasks}

\ex{2} Sabiendo que $\log_5 a=0,31$ y que $\log_5 b=1,4$, calcula:\begin{tasks}(2)

\seccio{10}{Ampliación: logaritmos y problemas}

  \task $\log_{\frac12}x=-3$
  \task $\log_x\frac18=-3$

\ex{2} Sabiendo que $\log2=0,301$ y $\log3=0,477$, calcula sin calculadora:\begin{tasks}(5)
  \task $\log12$
  \task $\log5$
  \task $\log1,5$
  \task $\log\sqrt{18}$
  \task $\log0,06$
\end{tasks}

\ex{2} Desarrolla con las propiedades de los logaritmos y escribe $E$ sin logaritmos:\begin{tasks}(2)
  \task $\log\dfrac{100x^3}{y}$
  \task $\log\sqrt{\dfrac{x}{y}}$
  \task $\log E=\log3+2\log x-\log y$
  \task $\log E=2-\log x$
\end{tasks}

\ex{1} Con la calculadora (cambio de base), calcula con tres decimales:\begin{tasks}(4)
  \task $\log_3 20$
  \task $\log_{0,5}60$
  \task $\log_2 100$
  \task $\ln7$
\end{tasks}\recorda{$\log_a x=\dfrac{\log x}{\log a}=\dfrac{\ln x}{\ln a}$.}

\ex{2} Una colonia de bacterias se duplica cada hora. ¿Cuántas horas tarda en pasar de 100 bacterias a un millón?

\ex{2} Ingresamos 1500 € en una cuenta al 4\,\% de interés compuesto anual. ¿Cuántos años deben pasar, como mínimo, para que haya 2000 €?

\ex{3} El pH de una disolución es $\text{pH}=-\log[\text{H}^+]$. Calcula el pH si $[\text{H}^+]=10^{-3}$ y si $[\text{H}^+]=2,5\cdot10^{-5}$. ¿Cuál es más ácida (pH menor)?

\ex{3} Sin calculadora, ordena de menor a mayor $\log_2 10$, \ $3$ \ y \ $\log_3 10$. Justifícalo.
