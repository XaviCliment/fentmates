\vfill
\begin{center}
\begin{tikzpicture}[t/.style={font=\fontsize{30}{30}\selectfont,inner sep=3pt,text height=24pt,text depth=6pt}]

  \node[fill=fmA!14,draw=fmA,rounded corners=2mm,font=\large,anchor=east] (lc) at ([xshift=-1.2cm,yshift=0.6cm]c.west) {{\heavy\color{fmA}coeficiente}};
  \node[fill=fmB!14,draw=fmB,rounded corners=2mm,font=\large,anchor=west,align=center] (ll) at ([xshift=1.2cm,yshift=0.6cm]l.east) {{\heavy\color{fmB}parte literal}\\[-2pt]\small grado: 3};
  \draw[-{Latex[length=3mm]},fmA,line width=1.3pt] (lc.east) to[out=0,in=150] ([xshift=-2pt]c.north west);

\end{tikzpicture}
\end{center}

\vfill
\begin{minipage}[t]{0.5\linewidth}
\begin{idea}[Monomios]
Coeficiente por letras con exponentes naturales. \textbf{Grado}: la suma de los exponentes: \ $4x^2y$ tiene grado 3.\\[3pt]
\textbf{Semejantes}: misma parte literal. Solo los semejantes se suman: \ $3x^2-x^2=2x^2$.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{idea}[Polinomios]
{\centering$P(x)=a_nx^n+\cdots+a_1x+a_0$\par}\smallskip
\textbf{Grado} $n$; \ \textbf{coeficiente principal} $a_n$; \ \textbf{término independiente} $a_0$.\\[2pt]
\textbf{Ordenado} y \textbf{completo} (todos los grados).
\end{idea}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ $P(x)=3x^2-5+2x^4-x$]
Ordenado: \ $P(x)=2x^4+3x^2-x-5$ \qquad grado 4, \ coeficiente principal 2, \ término independiente $-5$\\[3pt]
\textbf{No} es completo: falta el grado 3. \ Completo: \ $2x^4+\cR{0x^3}+3x^2-x-5$ \ \nota{importante para Ruffini}
\end{exemple}

\vfill
\begin{center}
\taularod{>{\heavy}c<{\hspace{8pt}}>{\hspace{8pt}}c<{\hspace{8pt}}>{\hspace{8pt}}c<{\hspace{8pt}}>{\hspace{8pt}}c<{\hspace{6pt}}}{
\rowcolor{fmblue}\color{white}Monomio & \color{white}\heavy Coeficiente & \color{white}\heavy Parte literal & \color{white}\heavy Grado\\
$-5x^3$ & $-5$ & $x^3$ & 3\\
\rowcolor{fmsurface}$\frac23\,x$ & $\frac23$ & $x$ & 1\\
$4x^2y$ & 4 & $x^2y$ & $2+1=3$\\
\rowcolor{fmsurface}$7$ & 7 & ninguna & 0
}
\end{center}

\vfill
\begin{exemple}[Ejemplo: \ reduce $5x^2-3x+2x^2+8x-1$]
Agrupamos los semejantes: \ $(5+2)x^2+(-3+8)x-1=\boxed{7x^2+5x-1}$
\end{exemple}

\vfill
\begin{truc}[Binomios y trinomios]
2 términos, \textbf{binomio} ($x^2-9$); 3 términos, \textbf{trinomio} ($x^2+2x+1$).
\end{truc}

\seccio{sec2}{Operaciones con polinomios}{Suma, producto, binomio de Newton y división.}

\vfill
\begin{minipage}[t]{0.5\linewidth}
\begin{idea}[Suma, resta y producto]
\textbf{Restar}: cambiamos el signo de \textbf{todos} los términos del segundo.\\[3pt]
\textbf{Multiplicar}: cada término por cada término; el grado es la \textbf{suma} de los grados.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,top=2mm,bottom=2mm]
$(a+b)^2=a^2+2ab+b^2$\\[2pt]
$(a-b)^2=a^2-2ab+b^2$\\[2pt]
$(a+b)(a-b)=a^2-b^2$
\end{tcolorbox}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ $P=2x^3-x+4$, \ $Q=x^2-3x$]
$P-Q=2x^3-x^2+2x+4$ \qquad $P\cdot Q=2x^5-6x^4-x^3+7x^2-12x$ \ \nota{grado $3+2=5$}\\[3pt]
$3(x-1)^2-(2x+3)(2x-3)=3x^2-6x+3-4x^2+9=\boxed{-x^2-6x+12}$
\end{exemple}

\vfill
\begin{minipage}[c]{0.3\linewidth}\centering
{\renewcommand{\arraystretch}{1.25}$\begin{array}{c}1\\ 1\quad1\\ 1\quad2\quad1\\ 1\quad3\quad3\quad1\\ 1\quad4\quad6\quad4\quad1\\ 1\quad5\quad10\quad10\quad5\quad1\end{array}$}\\[2pt]
{\comic\small\color{fmmuted}triángulo de Tartaglia}
\end{minipage}\hfill
\begin{minipage}[c]{0.67\linewidth}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Binomio de Newton}]
{\centering$(a+b)^n=\binom n0a^n+\binom n1a^{n-1}b+\cdots+\binom nnb^n$\par}\smallskip
Coeficientes: el triángulo. Los exponentes de $a$ bajan y los de $b$ suben. Con $(a-b)^n$, signos \textbf{alternos}.\\[3pt]
$(2x-3)^3=8x^3-36x^2+54x-27$
\end{tcolorbox}
\end{minipage}

\subsection{División y regla de Ruffini}
\begin{minipage}[c]{0.45\linewidth}
\begin{idea}[División entera]
{\centering$D(x)=d(x)\cdot c(x)+r(x)$\par}\smallskip
con $\text{gr}\,r<\text{gr}\,d$.\\[3pt]
Si el divisor es $x-a$: \textbf{Ruffini}. Cuidado: $x+2=x-(-2)$, \ $a=-2$.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.52\linewidth}\centering
{\small$(x^4-5x^2+3x-2):(x+2)$}\\[2pt]
\ruffini{-2}{1,0,-5,3,-2}{-2,4,2,-10}{1,-2,-1,5,-12}{5}\\[2pt]
$c(x)=x^3-2x^2-x+5$, \quad $r=-12$
\end{minipage}

\vfill
\begin{compte}[Los ceros de los grados que faltan]
Escribe \textbf{todos} los coeficientes, con un 0 para cada grado que falta: \ $x^4-5x^2+3x-2\ \to\ 1,\ \cR{0},\ -5,\ 3,\ -2$.
\end{compte}

\clearpage
\seccio{sec3}{Valor numérico y teorema del resto}{El resto de dividir entre $x-a$ es $P(a)$.}

\vfill
\begin{minipage}[c]{0.5\linewidth}
\begin{idea}[Valor numérico]
$P(a)$: sustituimos $x$ por $a$.\\[3pt]
$P(x)=x^3-2x+5$: \ $P(-2)=-8+4+5=\boxed{1}$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=3mm,bottom=3mm]
{\Large Teorema del resto}\par\smallskip
el resto de $P(x):(x-a)$ es\par\smallskip
{\fontsize{22}{26}\selectfont$r=P(a)$}
\end{tcolorbox}
\end{minipage}

\vfill
\begin{truc}[¿Por qué?]
$P(x)=(x-a)\cdot c(x)+r$. \ Si ponemos $x=a$, el primer sumando vale 0: \ $P(a)=r$.\\[2pt]
Dos maneras de calcular $P(a)$: sustituyendo, o haciendo Ruffini y mirando el resto (más rápido con grados altos).
\end{truc}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{exemple}[Ejemplo: \ resto de $(x^5-3x^2+7):(x+1)$]
Sin dividir: \ $r=P(-1)=-1-3+7=\boxed{3}$
\end{exemple}
\begin{exemple}[Ejemplo: \ $k$ para que $(x^3+kx^2-5x+6):(x-3)$ sea exacta]
$P(3)=0$: \ $27+9k-15+6=0\ \to\ \boxed{k=-2}$
\end{exemple}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplo: \ $P(3)$ con Ruffini, para $P(x)=x^4-3x^3+2x-5$]
\begin{minipage}[c]{0.5\linewidth}\centering
\scalebox{0.78}{\ruffini{3}{1,-3,0,2,-5}{3,0,0,6}{1,0,0,2,1}{5}}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
No hay término en $x^2$: ponemos un \textbf{0}.\\[4pt]
El resto es el valor numérico: \ $P(3)=\boxed{1}$\\[4pt]
Comprobamos: \ $81-81+6-5=1$ \ \be{}
\end{minipage}
\end{exemple}

\vfill
\begin{compte}[El signo de $a$]
Dividir entre $x+2$ es dividir entre $x-(-2)$: el resto es \be{P(-2)}, no \mal{P(2)}.
\end{compte}

\vfill
\begin{exemple}[Ejemplo: \ $a$ y $b$ para que $x^3+ax^2+bx-6$ sea divisible por $x-1$ y, al dividirlo por $x+1$, el resto sea $-4$]
$P(1)=0$: \ $1+a+b-6=0\ \to\ a+b=5$ \qquad $P(-1)=-4$: \ $-1+a-b-6=-4\ \to\ a-b=3$\\[4pt]
Resolvemos el sistema: \ $2a=8$ \ $\to$ \ \boxed{a=4,\ b=1}
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$P(x)=2x^3-x^2+3$: \ $P(0)=3$, \ $P(1)=4$, \ $P(-1)=0$, \ $P(2)=15$\\[4pt]
Resto de $(x^3-mx+4):(x+2)$ igual a 10: \ $P(-2)=-8+2m+4=10\ \to\ \boxed{m=7}$\\[4pt]
¿Es exacta $(x^4-16):(x+2)$? \ $P(-2)=16-16=0$ \ \be{} sí: \ $x+2$ es un factor de $x^4-16$
\end{exemple}

\clearpage
\seccio{sec4}{Raíces de un polinomio}{Los valores que anulan el polinomio.}

\vfill
\begin{minipage}[c]{0.52\linewidth}
\begin{idea}[Raíz y teorema del factor]
$a$ es una \textbf{raíz} de $P(x)$ si $P(a)=0$.\\[4pt]
{\centering$P(a)=0\iff(x-a)$ es un \textbf{factor} de $P(x)$\par}\smallskip
Grado $n$: $n$ raíces como mucho. \ $x^2+1$: ninguna.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.45\linewidth}
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={\faSearch\ \ Raíces enteras}]
\pas{1} Candidatos: divisores del \textbf{término indep.}\\[2pt]
\pas{2} Probamos: si $P(a)=0$, es raíz.\\[2pt]
\pas{3} Sin término indep.: factor común $x$.
\end{tcolorbox}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ raíces enteras de $P(x)=x^3-2x^2-5x+6$]
Candidatos: divisores de 6: \ $\pm1,\ \pm2,\ \pm3,\ \pm6$\\[4pt]
$P(1)=0$ \be{} \qquad $P(-1)=8$ \qquad $P(2)=-4$ \qquad $P(-2)=0$ \be{} \qquad $P(3)=0$ \be{}\\[4pt]
Grado 3 y ya tenemos 3: \ \boxed{x=1,\ x=-2,\ x=3}
\end{exemple}

\vfill
\begin{minipage}[c]{0.44\linewidth}\centering
\begin{tikzpicture}[x=0.7cm,y=0.7cm]
  \draw[fmmuted!30,line width=0.4pt] (-3,-1.5) grid (4,2.5);
  \draw[-{Latex[length=2mm]},fmmuted,line width=0.8pt] (-3,0) -- (4.2,0) node[right,font=\scriptsize] {$x$};
  \draw[-{Latex[length=2mm]},fmmuted,line width=0.8pt] (0,-1.5) -- (0,2.7) node[above,font=\scriptsize] {$y$};
  \draw[fmA,line width=1.6pt,domain=-2.3:3.3,samples=80] plot (\x,{0.25*(\x*\x*\x-2*\x*\x-5*\x+6)});
  \foreach \r in {-2,1,3} {\fill[fmR] (\r,0) circle (3pt); \node[below left=-1pt,font=\scriptsize\heavy,text=fmR] at (\r,0) {$\r$};}
\end{tikzpicture}\\
{\comic\small\color{fmmuted}$y=x^3-2x^2-5x+6$}
\end{minipage}\hfill
\begin{minipage}[c]{0.53\linewidth}
\begin{truc}[Raíces y gráfica]
Las raíces son los puntos donde la gráfica \textbf{corta el eje $X$}: allí $y=P(x)=0$.\\[3pt]
Una parábola, $x^2+1$, que no corta el eje no tiene raíces.
\end{truc}
\end{minipage}

\vfill
\begin{minipage}[t]{0.5\linewidth}
\begin{exemple}[Ejemplo: \ $x^3+x^2-4x-4$]
Candidatos $\pm1,\pm2,\pm4$: \ $\boxed{x=-1,\ 2,\ -2}$
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{compte}[¿Puede ser raíz el 4?]
De $x^3+2x^2-x+6$: \textbf{no}, porque 4 no es divisor de 6.
\end{compte}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ raíces de $x^3-3x+2$]
\begin{minipage}[c]{0.55\linewidth}
Candidatos $\pm1,\pm2$: \ $P(1)=0$ \be{}, \ $P(-1)=4$, \ $P(2)=4$, \ $P(-2)=0$ \be{}\\[4pt]
¿Solo dos raíces y el grado es 3? \ Una se \textbf{repite}: \\[2pt]
$x^3-3x+2=(x-1)^2(x+2)$ \ \nota{$x=1$ es raíz doble}
\end{minipage}\hfill
\begin{minipage}[c]{0.42\linewidth}\centering
\scalebox{0.7}{\ruffini{1}{1,0,-3,2}{1,1,-2}{1,1,-2,0}{4}}
\end{minipage}
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
\begin{minipage}[t]{0.48\linewidth}
$x^4-x^3-2x^2=x^2(x^2-x-2)$:\\ raíces \boxed{x=0\text{ (doble)},\ 2,\ -1} \ \nota{factor común}\\[4pt]
$2x^2-x-1$: \ $x=1$ y $x=-\frac12$ \ \nota{no todas son enteras}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
Grado 3, raíces $1,\,-2,\,3$ y $P(0)=12$:\\
$P(x)=a(x-1)(x+2)(x-3)$, \ $P(0)=6a=12$\\
$\boxed{P(x)=2(x-1)(x+2)(x-3)}$
\end{minipage}
\end{exemple}

\seccio{sec5}{Factorización de polinomios}{Producto de factores del menor grado posible.}

\vfill
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=2mm,bottom=2mm]
{\Large$P(x)=a_n\,(x-x_1)(x-x_2)\cdots(x-x_n)$}\par\smallskip
{\small $x_1,\dots,x_n$ son las raíces; si un factor se repite, la raíz es \textbf{doble}, \textbf{triple}\dots}
\end{tcolorbox}

\vfill
\begin{tcbraster}[raster columns=4,raster equal height,raster column skip=6pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\pas{1}\ Factor común},left=2mm,right=2mm]
$2x^4-6x^3$\\ $=2x^3(x-3)$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\pas{2}\ Identidades},left=2mm,right=2mm]
$x^2-9=(x+3)(x-3)$\\ $x^2+6x+9=(x+3)^2$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\pas{3}\ Grado 2},left=2mm,right=2mm]
Ecuación:\\ $a(x-x_1)(x-x_2)$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={\pas{4}\ Grado mayor},left=2mm,right=2mm]
Ruffini hasta llegar a grado 2.
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplo: \ $P(x)=x^4-x^3-7x^2+x+6$]
\begin{minipage}[c]{0.45\linewidth}\centering
\ruffini{1}{1,-1,-7,1,6}{1,0,-7,-6}{1,0,-7,-6,0}{5}
\end{minipage}\hfill
\begin{minipage}[c]{0.52\linewidth}
$x=1$: resto 0. Con el cociente y $x=-1$: \ $x^3-7x-6=(x+1)(x^2-x-6)$\\[3pt]
$x^2-x-6=0\ \to\ x=3,\ x=-2$\\[3pt]
$\boxed{P(x)=(x-1)(x+1)(x-3)(x+2)}$
\end{minipage}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $P(x)=2x^3-3x^2-3x+2$ \ (coeficiente principal 2)]
\begin{minipage}[c]{0.4\linewidth}\centering
\scalebox{0.72}{\ruffini{-1}{2,-3,-3,2}{-2,5,-2}{2,-5,2,0}{4}}
\end{minipage}\hfill
\begin{minipage}[c]{0.57\linewidth}
$P(-1)=0$: \ $P(x)=(x+1)(2x^2-5x+2)$\\[3pt]
$2x^2-5x+2=0\ \to\ x=\dfrac{5\pm3}{4}=2,\ \dfrac12$\\[3pt]
$P(x)=2(x+1)(x-2)\left(x-\tfrac12\right)=\boxed{(x+1)(x-2)(2x-1)}$\\[2pt]
\notab{el 2 entra en el último paréntesis: \ $2\left(x-\frac12\right)=2x-1$}
\end{minipage}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $P(x)=x^3-x^2+x-6$ \ (queda un factor de grado 2 sin raíces)]
\begin{minipage}[c]{0.4\linewidth}\centering
\scalebox{0.72}{\ruffini{2}{1,-1,1,-6}{2,2,6}{1,1,3,0}{4}}
\end{minipage}\hfill
\begin{minipage}[c]{0.57\linewidth}
$P(2)=0$: \ $P(x)=(x-2)(x^2+x+3)$\\[3pt]
$x^2+x+3=0$: \ $\Delta=1-12<0$, \ \textbf{no tiene solución}\\[3pt]
\raggedright Se queda tal cual en la factorización:\\[3pt] $\boxed{P(x)=(x-2)(x^2+x+3)}$
\end{minipage}
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$x^3-6x^2+9x=x(x^2-6x+9)=\boxed{x(x-3)^2}$ \ \nota{$x=3$ doble}\\[4pt]
$x^2-16=(x+4)(x-4)$ \qquad $3x^2-12x+12=3(x-2)^2$ \qquad $x^3-x=x(x+1)(x-1)$\\[4pt]
$x^4-5x^2+4=(x-1)(x+1)(x-2)(x+2)$ \qquad $x^3+x^2-5x+3=(x-1)^2(x+3)$
\end{exemple}

\clearpage
\seccio{sec6}{Divisibilidad. M.c.d. y m.c.m.}{Como con los números, pero con los polinomios factorizados.}

\vfill
\begin{idea}[Divisores e irreducibles]
$Q(x)$ es \textbf{divisor} de $P(x)$ si la división es exacta: $P=Q\cdot C$. \ $x^2-1=(x+1)(x-1)$.\\[3pt]
Un polinomio sin divisores de grado menor, como $x+3$ o $x^2+1$, es \textbf{irreducible} (como un primo).
\end{idea}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmA!8,colframe=fmA,coltitle=white,title={M.c.d.}]
Factores \textbf{comunes} con el \textbf{menor} exponente.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmB!8,colframe=fmB,coltitle=white,title={M.c.m.}]
Factores \textbf{comunes y no comunes} con el \textbf{mayor} exponente.
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplo: \ $P(x)=x^3-4x$ \ y \ $Q(x)=x^2-4x+4$]
Factorizamos: \ $P=x(x+2)(x-2)$, \quad $Q=(x-2)^2$\\[4pt]
m.c.d. $=\boxed{x-2}$ \qquad m.c.m. $=\boxed{x(x+2)(x-2)^2}$
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
¿Es $x-3$ divisor de $x^2-x-6$? \ $P(3)=0$ \be{} sí. \ ¿Y $x+3$? \ $P(-3)=6$ \mal{\text{no}}\\[4pt]
$x^2+x=x(x+1)$, \ $x^2-1=(x+1)(x-1)$: \ m.c.d. $=x+1$, \ m.c.m. $=x(x+1)(x-1)$\\[4pt]
$x^2(x-1)$, \ $(x-1)^2$, \ $2(x-1)$: \ m.c.d. $=x-1$, \ m.c.m. $=2x^2(x-1)^2$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ m.c.d. y m.c.m. de $x^3-3x+2$, \ $x^2+x-2$ \ y \ $x^2-1$]
\begin{minipage}[c]{0.55\linewidth}
Ruffini con $x=1$: \ $x^3-3x+2=(x-1)^2(x+2)$\\[3pt]
Ecuación de 2.º grado: \ $x^2+x-2=(x-1)(x+2)$\\[3pt]
Identidad notable: \ $x^2-1=(x-1)(x+1)$
\end{minipage}\hfill
\begin{minipage}[c]{0.42\linewidth}
m.c.d. $=\boxed{x-1}$ \ \nota{el único común}\\[5pt]
m.c.m. $=\boxed{(x-1)^2(x+2)(x+1)}$
\end{minipage}
\end{exemple}

\vfill
\begin{minipage}[t]{0.48\linewidth}
\begin{exemple}[Ejemplo: \ divisores de $(x+1)(x-2)$]
$1$, \ $x+1$, \ $x-2$ \ y \ $(x+1)(x-2)$\\[3pt]
\notab{y los que resultan de multiplicarlos por un número: $3(x+1)$\dots}
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
\begin{exemple}[Ejemplo: \ $k$ para que $x+2$ sea divisor de $x^3+kx+4$]
$P(-2)=0$: \ $-8-2k+4=0$\\[3pt]
$\boxed{k=-2}$
\end{exemple}
\end{minipage}

\vfill
\begin{truc}[Como con los números]
$12=2^2\cdot3$ y $18=2\cdot3^2$: \ m.c.d. $=2\cdot3=6$, \ m.c.m. $=2^2\cdot3^2=36$. \ Con polinomios es igual, pero los «primos» son los factores irreducibles. Por eso, antes de nada, \textbf{factoriza} todos los polinomios. El m.c.m. será el denominador común al sumar fracciones algebraicas (2.7).
\end{truc}

% ============================================================ 2.7
\clearpage
\seccio{sec7}{Fracciones algebraicas}{Cocientes de polinomios: simplificamos y operamos como con fracciones.}

\vfill
\begin{minipage}[t]{0.5\linewidth}
\begin{idea}[Fracción algebraica]
$\dfrac{P(x)}{Q(x)}$: solo tiene sentido si $Q(x)\neq0$.\\[3pt]
\textbf{Simplificar}: factorizamos y eliminamos los factores comunes.\\[3pt]
$\dfrac{x^3-4x}{x^3-4x^2+4x}=\dfrac{x(x+2)(x-2)}{x(x-2)^2}=\boxed{\dfrac{x+2}{x-2}}$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{compte}[Factores, no sumandos]
Solo se simplifican \textbf{factores}:\\[3pt]
\mal{\dfrac{x+3}{x}=3}\qquad \mal{\dfrac{x^2+1}{x^2}=1}
\end{compte}
\end{minipage}

\vfill
\begin{tcbraster}[raster columns=3,raster equal height,raster column skip=8pt]
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Suma y resta}]
Común denominador: el \textbf{m.c.m.} de los denominadores.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Producto},before upper=\centering]
$\dfrac PQ\cdot\dfrac RS=\dfrac{P\cdot R}{Q\cdot S}$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Cociente},before upper=\centering]
$\dfrac PQ:\dfrac RS=\dfrac{P\cdot S}{Q\cdot R}$
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplo: \ $\dfrac{x}{x-1}+\dfrac{2}{x^2-1}$]
$x^2-1=(x+1)(x-1)$: \ m.c.m. $=(x+1)(x-1)$ \qquad $=\dfrac{x(x+1)+2}{(x+1)(x-1)}=\boxed{\dfrac{x^2+x+2}{x^2-1}}$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ producto y cociente]
$\dfrac{x^2-9}{x^2+2x}\cdot\dfrac{x+2}{x-3}=\dfrac{(x+3)(x-3)(x+2)}{x(x+2)(x-3)}=\boxed{\dfrac{x+3}{x}}$ \qquad
$\dfrac{2x-4}{x^2}:\dfrac{x^2-4}{x}=\dfrac{2(x-2)\cdot x}{x^2(x+2)(x-2)}=\boxed{\dfrac{2}{x(x+2)}}$
\end{exemple}

\begin{exemple}[Ejemplo: \ $\left(\dfrac1x-\dfrac{1}{x+1}\right):\dfrac{1}{x^2+x}$]
Primero el paréntesis: \ $\dfrac{(x+1)-x}{x(x+1)}=\dfrac{1}{x(x+1)}$ \qquad Después: \ $\dfrac{1}{x(x+1)}:\dfrac{1}{x(x+1)}=\boxed{1}$ \ \nota{si $x\neq0,\,-1$}
\end{exemple}

\begin{truc}[Factoriza antes de operar]
Factoriza numeradores y denominadores \textbf{antes} de multiplicar: así simplificas enseguida.
\end{truc}
