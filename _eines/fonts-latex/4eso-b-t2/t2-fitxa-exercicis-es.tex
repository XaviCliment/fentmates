\seccio{1}{Polinomios y valor numérico}

\exf{1} Completa la tabla siguiente:\begin{center}\renewcommand{\arraystretch}{1.45}\small\begin{tabular}{|l|C{1cm}|C{1.9cm}|C{1.5cm}|C{2cm}|C{1.7cm}|}\hline\rowcolor{soft}\textbf{Polinomio} & \textbf{Grado} & \textbf{Térm. indep.} & \textbf{N.\textsuperscript{o} términos} & \textbf{¿Completo?} & \textbf{Coef. princ.}\\\hline$P(x)=5x^4-3x^3+2x+10$ &  &  &  &  & \\\hline$Q(x)=-5x^5-\frac32x^4-x^3+9x^2-6$ &  &  &  &  & \\\hline$R(x)=4x^2+x+1$ &  &  &  &  & \\\hline$S(x)=5x^2+2x^5+9x$ &  &  &  &  & \\\hline$T(x)=3x-2$ &  &  &  &  & \\\hline$U(x)=x^3-4x^2+x+1$ &  &  &  &  & \\\hline\end{tabular}\end{center}

\ex{1} Dado el polinomio $P(x)=-3x^4-2x^3-x^2-1$, calcula su valor numérico para $x=2$ y $x=-2$.

\ex{1} Calcula el valor numérico de los polinomios siguientes para $x=3$, $x=-1$ y $x=0$:\begin{tasks}(2)
  \task $P(x)=6x^3-x^2-3$
  \task $Q(x)=-3x^4+2x^3+7x^2-x+5$
  \task $R(x)=8$
  \task $S(x)=x^3-1$
\end{tasks}

\ex{2} ¿Qué valor debe tener $k$ para que el valor numérico del polinomio $P(x)=x^4-2x^2+5x+k$ para $x=2$ sea 6?

\ex{2} ¿Cuánto vale $a$ en el polinomio $P(x)=-x^4-4x^3+ax+10$, si sabemos que el valor numérico para $x=-3$ es 10?

\exf{2} Sean $P(x)=x^3+4x^2-3x+1$, $Q(x)=2x^3-x+3$ y $R(x)=4x^2-x$. Calcula:\begin{tasks}(2)
  \task $[P(x)-3\cdot Q(x)]\cdot R(x)$
  \task $[P(x)-R(x)]\cdot Q(x)$
\end{tasks}

\ex{1} Efectúa las operaciones siguientes aplicando las identidades notables:\begin{tasks}(3)
  \task $(x+3)^2$
  \task $(5x^2+7)^2$

\ex{2} Calcula:\begin{tasks}(1)

\end{tasks}

\seccio{3}{Binomio de Newton}

\exf{2} Calcula las potencias siguientes utilizando la fórmula del binomio de Newton:\begin{tasks}(3)
  \task $(x+2)^3$
  \task $(x-1)^4$
  \task $(2x+3)^3$
  \task $(2x+1)^5$
  \task $(2x-3)^3$
  \task $(-2+x)^6$
\end{tasks}\recorda{$(a+b)^n=\binom n0a^n+\binom n1a^{n-1}b+\cdots+\binom nnb^n$; los coeficientes son una fila del triángulo de Tartaglia. Si hay una resta, los signos se alternan.}

\ex{3} Calcula las potencias siguientes:\begin{tasks}(4)
  \task $(x^2+x)^3$
  \task $(2x^4-y)^5$
  \task $(4x^3+2)^4$
  \task $(x^6-3y^2)^5$
  \task $(3y^2+5x)^3$
  \task $(-2x^3+2x)^4$
  \task $(-2x^3-3x)^6$
  \task $(x-2y)^4$
\end{tasks}

  \task $(18x^6-3x^4+2x^3+x):(x^2-2)$
\end{tasks}\recorda{Deja vacío el lugar de los términos que faltan en el dividendo. Comprobación: $D(x)=d(x)\cdot c(x)+r(x)$.}

\ex{1} Haz las divisiones con la regla de Ruffini e indica el cociente y el resto:\begin{tasks}(1)

\end{tasks}

\ex{1} Determina el resto de las divisiones siguientes con la regla de Ruffini:\begin{tasks}(3)

\ex{2} Calcula el m.c.d. y el m.c.m. de:\begin{tasks}(1)
  \task $x^2-4$ y $x^2+4x+4$
  \task $x^2-x$ y $x^2-1$
  \task $x^3-x$ y $x^2-2x+1$
  \task $x^3-2x^2-x+2$ y $x^2-3x+2$
\end{tasks}\recorda{Factoriza primero. M.c.d.: factores comunes con el menor exponente. M.c.m.: todos los factores con el mayor exponente.}
