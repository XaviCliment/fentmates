\seccio{1}{Polinomis}

\exf{1} Ordena cada polinomi i indica'n el grau, el coeficient principal i el terme independent. És complet?\begin{tasks}(2)
  \task $3x^2-5+2x^4-x$
  \task $7-x^3$
  \task $x^5-2x^2+x^3-8x+1$
  \task $-4x+6x^2-\frac12$
\end{tasks}

\ex{1} Indica el coeficient, la part literal i el grau de cada monomi:\begin{tasks}(4)
  \task $-3x^4$
  \task $\frac25xy^2$
  \task $8$
  \task $-a^3b^2$
\end{tasks}

\ex{1} Redueix els termes semblants:\begin{tasks}(3)
  \task $3x^2-5x+7-x^2+2x-4$
  \task $x^3-2x+4x^3+x^2-x$
  \task $2x(x-3)-x^2$
\end{tasks}

\seccio{2}{Operacions amb polinomis}

\exf{1} Donats $P(x)=2x^3-x^2+4$, $Q(x)=x^2+3x-1$ i $R(x)=x-2$, calcula:\begin{tasks}(3)
  \task $P+Q$
  \task $P-Q$
  \task $Q\cdot R$
  \task $P\cdot R$
  \task $P-Q\cdot R$
\end{tasks}

\ex{1} Desenvolupa les identitats notables:\begin{tasks}(3)
  \task $(x+5)^2$
  \task $(2x-3)^2$
  \task $(3x+1)(3x-1)$
  \task $(x^2-2)^2$
  \task $\left(\frac x2+4\right)^2$
  \task $(5-x)(5+x)$
\end{tasks}

\ex{2} Escriu com el quadrat d'un binomi o com un producte (identitats notables a l'inrevés):\begin{tasks}(4)
  \task $x^2+6x+9$
  \task $4x^2-20x+25$
  \task $x^2-49$
  \task $9x^2-1$
\end{tasks}

\ex{2} Opera i simplifica:\begin{tasks}(1)
  \task $(x+2)^2-(x-2)^2$
  \task $3(x-1)^2-(2x+3)(2x-3)$
  \task $(2x-1)^2-2(x+1)(x-1)$
\end{tasks}

\ex{2} Fes les divisions i indica'n el quocient i el residu:\begin{tasks}(2)
  \task $(x^3-2x^2+3x-1):(x^2+1)$
  \task $(6x^4-x^2+3x):(2x^2+1)$
  \task $(2x^4+x^3-5x+2):(x^2-x+1)$
  \task $(x^4-1):(x^2+1)$
\end{tasks}\recorda{Deixa buit el lloc dels termes que falten al dividend. Comprova-ho: $D(x)=d(x)\cdot c(x)+r(x)$.}

\ex{1} Fes les divisions aplicant la regla de Ruffini:\begin{tasks}(2)
  \task $(x^3-4x^2+x+6):(x-2)$
  \task $(2x^4-3x^2+x-5):(x+1)$
  \task $(x^5-32):(x-2)$
  \task $(3x^3+2x^2-x):(x+2)$
\end{tasks}

\ex{2} Calcula $a$ i $b$ perquè $(x+a)(2x-3)=2x^2+bx-6$.

\seccio{3}{Valor numèric i teorema del residu}

\exf{1} Donat $P(x)=x^3-3x^2+2x-1$, calcula:\begin{tasks}(5)
  \task $P(0)$
  \task $P(1)$
  \task $P(-1)$
  \task $P(2)$
  \task $P\left(\frac12\right)$
\end{tasks}

\ex{2} Calcula, sense fer la divisió, el residu de:\begin{tasks}(2)
  \task $(x^4-3x^2+2):(x-1)$
  \task $(x^5+2x-3):(x+1)$
  \task $(2x^3-x+5):(x-2)$
  \task $(x^{10}-1):(x+1)$
\end{tasks}\recorda{Teorema del residu: el residu de $P(x):(x-a)$ és $P(a)$.}

\ex{2} Comprova que el residu de $(x^3-5x+3):(x-2)$ fet per Ruffini coincideix amb $P(2)$.

\ex{2} Calcula el valor de $k$ en cada cas:\begin{tasks}(1)
  \task $x^3+kx^2-5x+6$ és divisible per $x-3$.
  \task El residu de $(2x^3-kx+4):(x+2)$ és 8.
  \task $x^4+kx^2-4$ és divisible per $x+2$.
\end{tasks}

\ex{3} Troba $a$ i $b$ perquè $x^3+ax^2+bx-6$ siga divisible per $x-2$ i per $x+3$.

\seccio{4}{Arrels d'un polinomi}

\exf{1} Comprova si els nombres $1$, $2$, $-1$ i $-3$ són arrels de $P(x)=x^3-7x+6$.

\ex{2} Troba les arrels enteres dels polinomis següents:\begin{tasks}(2)
  \task $x^3-6x^2+11x-6$
  \task $x^3+2x^2-x-2$
  \task $x^4-10x^2+9$
  \task $x^3-3x^2-4x+12$
\end{tasks}\recorda{Les arrels enteres d'un polinomi amb coeficients enters són divisors del terme independent.}

\ex{2} Escriu els candidats a arrel entera de $x^3+x^2-12$ i troba quins ho són.

\ex{2} Escriu un polinomi de grau 3 que tinga per arrels $1$, $-2$ i $3$. I un de grau 2 amb una arrel doble en $x=4$.

\seccio{5}{Factorització de polinomis}

\exf{1} Factoritza traient factor comú i fent servir les identitats notables:\begin{tasks}(3)
  \task $3x^2-12x$
  \task $x^3-9x$
  \task $2x^3+8x^2+8x$
  \task $x^4-x^2$
  \task $5x^2-5$
\end{tasks}

\ex{2} Factoritza resolent l'equació de segon grau:\begin{tasks}(3)
  \task $x^2-5x+6$
  \task $x^2+x-12$
  \task $2x^2-x-1$
  \task $x^2-4x+5$
  \task $3x^2+6x-9$
\end{tasks}\recorda{Si $x_1$ i $x_2$ són les solucions de $ax^2+bx+c=0$, aleshores $ax^2+bx+c=a(x-x_1)(x-x_2)$.}

\ex{2} Factoritza amb la regla de Ruffini:\begin{tasks}(2)
  \task $x^3-2x^2-5x+6$
  \task $x^3+x^2-5x+3$
  \task $x^4-x^3-7x^2+x+6$
  \task $x^3-3x-2$
  \task $x^4-5x^2+4$
  \task $x^3-x^2-x+1$
\end{tasks}

\ex{3} Factoritza completament:\begin{tasks}(2)
  \task $x^4-2x^3-3x^2$
  \task $2x^4+x^3-5x^2+2x$
  \task $x^5-x$
  \task $x^4+x^3-x-1$
\end{tasks}

\seccio{6}{Divisibilitat. M.c.d. i m.c.m.}

\exf{1} Indica, sense fer la divisió, si el primer polinomi és divisor del segon:\begin{tasks}(2)
  \task $x-1$ i $x^3-1$
  \task $x+2$ i $x^2+4$
  \task $x+1$ i $x^4-1$
  \task $x-3$ i $x^2-9$
\end{tasks}

\ex{2} Calcula el m.c.d. i el m.c.m. de:\begin{tasks}(1)
  \task $x^2-4$ i $x^2+4x+4$
  \task $x^2-x$ i $x^2-1$
  \task $x^3-x$ i $x^2-2x+1$
  \task $2x+4$, \ $x^2+2x$ i $3x$
\end{tasks}\recorda{Factoritza primer. M.c.d.: factors comuns amb el menor exponent. M.c.m.: tots els factors amb el major exponent.}

\ex{3} Factoritza $P(x)=x^3-2x^2-x+2$ i $Q(x)=x^2-3x+2$, i calcula'n el m.c.d. i el m.c.m.

\seccio{7}{Fraccions algebraiques}

\exf{1} Indica per a quins valors de $x$ no tenen sentit les fraccions següents:\begin{tasks}(4)
  \task $\dfrac{3}{x-2}$
  \task $\dfrac{x+1}{x^2-9}$
  \task $\dfrac{x}{x^2+1}$
  \task $\dfrac{5}{x^2-4x}$
\end{tasks}

\ex{2} Simplifica les fraccions algebraiques:\begin{tasks}(3)
  \task $\dfrac{x^2-9}{x^2+3x}$
  \task $\dfrac{x^2-4x+4}{x^2-4}$
  \task $\dfrac{2x^2-2}{4x+4}$
  \task $\dfrac{x^3-x}{x^2+x}$
  \task $\dfrac{x^2-5x+6}{x^2-x-6}$
\end{tasks}\recorda{Només es poden simplificar factors: factoritza el numerador i el denominador.}

\ex{2} Comprova si són equivalents:\begin{tasks}(1)
  \task $\dfrac{x+1}{x-1}$ \ i \ $\dfrac{x^2+2x+1}{x^2-1}$
  \task $\dfrac{2}{x}$ \ i \ $\dfrac{2x+2}{x^2+1}$
\end{tasks}

\ex{2} Calcula i simplifica:\begin{tasks}(2)
  \task $\dfrac2x+\dfrac3{x^2}$
  \task $\dfrac{1}{x+1}+\dfrac{1}{x-1}$
  \task $\dfrac{x}{x-2}-\dfrac{2}{x+2}$
  \task $\dfrac{3}{x^2-1}-\dfrac{1}{x+1}$
  \task $\dfrac{1}{x^2+x}+\dfrac{2}{x}$
\end{tasks}

\ex{2} Calcula i simplifica:\begin{tasks}(2)
  \task $\dfrac{x^2-4}{3x}\cdot\dfrac{6x^2}{x+2}$
  \task $\dfrac{x+1}{x^2-1}\cdot\dfrac{x-1}{x}$
  \task $\dfrac{x^2-x}{x+3}:\dfrac{x}{x^2-9}$
  \task $\dfrac{2x}{x^2-1}:\dfrac{4x^2}{x+1}$
\end{tasks}

\ex{3} Fes les operacions combinades i simplifica:\begin{tasks}(2)
  \task $\left(1+\dfrac1x\right)\cdot\dfrac{x}{x^2-1}$
  \task $\left(\dfrac{x}{x-1}-1\right):\dfrac{1}{x^2-1}$
  \task $\left(\dfrac1x-\dfrac{1}{x+1}\right)\cdot(x^2+x)$
  \task $\dfrac{x+2}{x-2}-\dfrac{x-2}{x+2}$
\end{tasks}
