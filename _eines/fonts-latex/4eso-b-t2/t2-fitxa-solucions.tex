\seccio{1}{Polinomis}

\exf{1} \begin{tasks}(1)
  \task $\res{2x^{4}+3x^{2}-x-5}$: grau 4, coeficient principal 2, terme independent $-5$. No és complet (falta $x^3$).
  \task $\res{-x^{3}+7}$: grau 3, coeficient principal $-1$, terme independent 7. No és complet.
  \task $\res{x^{5}+x^{3}-2x^{2}-8x+1}$: grau 5, coeficient principal 1, terme independent 1. No és complet (falta $x^4$).
  \task $\res{6x^{2}-4x-\frac{1}{2}}$: grau 2, coeficient principal 6, terme independent $-\frac12$. És complet.
\end{tasks}

\seccio{6}{Divisibilitat. M.c.d. i m.c.m.}

\exf{1} \begin{tasks}(2)
  \task $1-1=0$: $\res{\text{sí}}$
  \task $4+4=8\neq0$: $\res{\text{no}}$
  \task $1-1=0$: $\res{\text{sí}}$
  \task $9-9=0$: $\res{\text{sí}}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $(x+2)(x-2)$, \ $(x+2)^2$: \ m.c.d. $=\res{x+2}$, \ m.c.m. $=\res{(x+2)^2(x-2)}$
  \task $x(x-1)$, \ $(x+1)(x-1)$: \ m.c.d. $=\res{x-1}$, \ m.c.m. $=\res{x(x+1)(x-1)}$
  \task $x(x+1)(x-1)$, \ $(x-1)^2$: \ m.c.d. $=\res{x-1}$, \ m.c.m. $=\res{x(x+1)(x-1)^2}$
  \task $2(x+2)$, \ $x(x+2)$, \ $3x$: \ m.c.d. $=\res{1}$, \ m.c.m. $=\res{6x(x+2)}$
\end{tasks}

\ex{3} $P(x)=(x-1)(x+1)(x-2)$, \ $Q(x)=(x-1)(x-2)$. \ m.c.d. $=\res{(x-1)(x-2)}$, \ m.c.m. $=\res{(x-1)(x+1)(x-2)}$.
