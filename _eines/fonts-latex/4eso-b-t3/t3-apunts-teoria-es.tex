\seccio{sec1}{Ecuaciones de segundo grado}{La fórmula, el discriminante y las incompletas.}

\vfill
\begin{minipage}[c]{0.5\linewidth}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=3mm,bottom=3mm]
{\large$ax^2+bx+c=0$}\par\medskip
{\fontsize{20}{26}\selectfont$x=\dfrac{-b\pm\sqrt{\cR{b^2-4ac}}}{2a}$}
\end{tcolorbox}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
\begin{idea}[El discriminante $\Delta=b^2-4ac$]
\chip{fmK}{$\Delta>0$} dos soluciones\\[3pt]
\chip{fmgold}{$\Delta=0$} una doble, $x=-\frac{b}{2a}$\\[3pt]
\chip{fmR}{$\Delta<0$} ninguna solución real
\end{idea}
\end{minipage}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmA!8,colframe=fmA,coltitle=white,title={Sin término independiente}]
$ax^2+bx=0\ \to\ x(ax+b)=0$\par\smallskip
$x=0$, \ $x=-\dfrac ba$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmB!8,colframe=fmB,coltitle=white,title={Sin término en $x$}]
$ax^2+c=0\ \to\ x^2=-\dfrac ca$\par\smallskip
$x=\pm\sqrt{-\frac ca}$ \ (si es $\ge0$)
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{compte}[No dividas entre $x$]
En $x^2-5x=0$, si divides entre $x$ pierdes la solución $x=0$. \textbf{Saca factor común}.
\end{compte}

\vfill
\begin{exemple}[Ejemplos]
$2x^2-x-3=0$: \ $\Delta=1+24=25$, \ $x=\dfrac{1\pm5}{4}$ \ $\to$ \ $\boxed{x=\tfrac32,\ x=-1}$\\[5pt]
$3x^2-12=0\ \to\ x^2=4\ \to\ \boxed{x=\pm2}$ \qquad $5x^2+10x=0\ \to\ 5x(x+2)=0\ \to\ \boxed{x=0,\ x=-2}$\\[5pt]
$(x-2)^2=2x-1\ \to\ x^2-6x+5=0\ \to\ \boxed{x=5,\ x=1}$ \ \nota{primero, la forma $ax^2+bx+c=0$}\\[5pt]
$4x^2-12x+9=0$: \ $\Delta=0$, \ $\boxed{x=\tfrac32}$ doble \qquad $x^2+x+5=0$: \ $\Delta=1-20<0$, \ \textbf{ninguna solución}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo con paréntesis: \ $(2x-1)^2-(x+2)(x-2)=4x+1$]
\pas{1} Quitamos los paréntesis: \ $4x^2-4x+1-(x^2-4)=4x+1$ \ \nota{el $-$ cambia todos los signos}\\[4pt]
\pas{2} Todo a un miembro: \ $3x^2-8x+4=0$ \qquad \pas{3} $x=\dfrac{8\pm\sqrt{64-48}}{6}=\dfrac{8\pm4}{6}$ \ $\to$ \ \boxed{x=2,\ x=\tfrac23}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo con denominadores: \ $\dfrac{x(x-1)}{2}-\dfrac{(x+1)^2}{3}=-2$]
\pas{1} m.c.m. $=6$: \ $3x(x-1)-2(x+1)^2=-12$\\[4pt]
\pas{2} $3x^2-3x-2x^2-4x-2=-12\ \to\ x^2-7x+10=0$ \qquad \pas{3} $x=\dfrac{7\pm3}{2}$ \ $\to$ \ \boxed{x=5,\ x=2}
\end{exemple}

\vfill

\clearpage
\seccio{sec2}{Ecuaciones bicuadradas}{Solo grados 4, 2 y 0: cambio $t=x^2$.}

\vfill
\begin{minipage}[c]{0.5\linewidth}
\passos{\pas{1} Cambio \ $t=x^2$: \ $at^2+bt+c=0$.\\[3pt]
\pas{2} Resolvemos en $t$.\\[3pt]
\pas{3} Deshacemos el cambio: \ $x=\pm\sqrt t$ \ (si $t\ge0$).}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}
\begin{truc}[¿Cuántas soluciones?]
Una bicuadrada puede tener \textbf{4, 3, 2, 1 o ninguna} solución.
\end{truc}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ $x^4-5x^2+4=0$]
$t^2-5t+4=0\ \to\ t=\dfrac{5\pm3}{2}$: \ $t=4,\ t=1$ \quad $x^2=4\to x=\pm2$; \ $x^2=1\to x=\pm1$ \quad \boxed{4\text{ soluciones}}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $x^4+3x^2-4=0$]
$t^2+3t-4=0\ \to\ t=1,\ t=-4$ \qquad $x^2=1\to\boxed{x=\pm1}$; \ $x^2=-4$ \mal{\text{no tiene solución}}
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$x^4-13x^2+36=0$: \ $t=9,\ t=4$ \ $\to$ \ $x=\pm3,\ \pm2$ \qquad $x^4-16=0$: \ $t=4$ \ $\to$ \ $x=\pm2$\\[4pt]
$x^4+5x^2+6=0$: \ $t=-2,\ -3$: \ \textbf{ninguna} \qquad $x^4-3x^2=0$: \ $x^2(x^2-3)=0$ \ $\to$ \ $x=0,\ \pm\sqrt3$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo con paréntesis: \ $(x^2-2)(x^2+1)=10$]
$x^4+x^2-2x^2-2=10\ \to\ x^4-x^2-12=0$ \quad $t^2-t-12=0\ \to\ t=4,\ -3$ \quad $x^2=4\to\boxed{x=\pm2}$; \ $x^2=-3$ \mal{\text{no}}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo con denominadores: \ $\dfrac{x^4}{4}-\dfrac{5x^2}{2}+4=0$]
Por 4: \ $x^4-10x^2+16=0$ \quad $t=\dfrac{10\pm6}{2}=8,\ 2$ \quad $x^2=8\to x=\pm2\sqrt2$; \ $x^2=2\to x=\pm\sqrt2$ \quad \boxed{4\text{ soluciones}}
\end{exemple}

\vfill
\begin{minipage}[t]{0.48\linewidth}
\begin{exemple}[Ejemplo: \ $x^2-\dfrac{36}{x^2}=5$]
Por $x^2$: \ $x^4-5x^2-36=0$\\[3pt]
$t=9,\ t=-4$ \ $\to$ \ \boxed{x=\pm3}
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
\begin{truc}[El mismo cambio, con $x^3$]
$x^6-9x^3+8=0$: \ $t=x^3$\\[3pt]
$t^2-9t+8=0\ \to\ t=8,\ t=1$ \ $\to$ \ $x=2,\ x=1$
\end{truc}
\end{minipage}

\clearpage
\seccio{sec3}{Ecuaciones de grado mayor que 2}{Factorizamos con Ruffini hasta grado 2.}

\vfill
\begin{minipage}[c]{0.5\linewidth}
\passos{\pas{1} Todo a un miembro y, si se puede, factor común.\\[3pt]
\pas{2} Ruffini con los \textbf{divisores del término independiente}, hasta grado 2.\\[3pt]
\pas{3} Resolvemos la de segundo grado. Las soluciones son todas las raíces.}
\end{minipage}\hfill
\begin{minipage}[c]{0.47\linewidth}\centering
{\small$x^3-2x^2-5x+6=0$}\\[2pt]
\ruffini{1}{1,-2,-5,6}{1,-1,-6}{1,-1,-6,0}{4}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ $x^3-2x^2-5x+6=0$]
Candidatos $\pm1,\pm2,\pm3,\pm6$. \ Con $x=1$, resto 0: \ $(x-1)(x^2-x-6)=0$\\[3pt]
$x^2-x-6=0\ \to\ x=\dfrac{1\pm5}{2}$ \qquad Soluciones: \ \boxed{x=1,\ x=3,\ x=-2}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $x^4=x^3+4x^2-4x$]
Todo a un miembro y factor común: \ $x(x^3-x^2-4x+4)=0$ \qquad Ruffini con $x=1$: \ $x(x-1)(x^2-4)=0$\\[3pt]
\boxed{x=0,\ x=1,\ x=\pm2}
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$x^3-7x+6=0$: \ $(x-1)(x-2)(x+3)=0$ \qquad $x^3+x^2-5x+3=0$: \ $(x-1)^2(x+3)=0$ \ \nota{$x=1$ doble}\\[4pt]
$2x^3-3x^2-3x+2=0$: \ $(x-2)(2x^2+x-1)=0$ \ $\to$ \ $x=2,\ \frac12,\ -1$\\[4pt]
$x^3+2x^2+2x+1=0$: \ $(x+1)(x^2+x+1)=0$ \ $\to$ \ solo $x=-1$ \ \nota{$\Delta=1-4<0$}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $x^4-2x^3-3x^2+4x+4=0$ \ (Ruffini dos veces)]
\begin{minipage}[c]{0.4\linewidth}\centering
\scalebox{0.62}{\ruffini{-1}{1,-2,-3,4,4}{-1,3,0,-4}{1,-3,0,4,0}{5}}
\end{minipage}\hfill
\begin{minipage}[c]{0.57\linewidth}
$x=-1$: \ $(x+1)(x^3-3x^2+4)=0$\\[3pt]
Ruffini con $x=2$: \ $x^3-3x^2+4=(x-2)(x^2-x-2)$\\[3pt]
$x^2-x-2=0\ \to\ x=2,\ x=-1$\\[3pt]
$(x+1)^2(x-2)^2=0$: \ \boxed{x=-1,\ x=2} \ \nota{las dos dobles}
\end{minipage}
\end{exemple}

\vfill
\begin{compte}[No dividas entre $x$]
$x^3=4x$: \ si dividimos entre $x$ perdemos el 0. \ Todo a un miembro: \ $x(x^2-4)=0$ \ $\to$ \ $x=0,\ x=\pm2$
\end{compte}

\clearpage
\seccio{sec4}{Ecuaciones factorizadas}{Si un producto vale 0, algún factor vale 0.}

\vfill
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=3mm,bottom=3mm]
{\Large$A\cdot B\cdot C=0\iff A=0\ \ \text{o}\ \ B=0\ \ \text{o}\ \ C=0$}\par\smallskip
{\small no hace falta multiplicar: igualamos cada factor a cero}
\end{tcolorbox}

\vfill
\begin{compte}[Solo si el otro miembro es 0]
En $(x-1)(x+2)=4$ \textbf{no} podemos decir $x-1=4$. Hay que multiplicar y resolver $x^2+x-6=0$.
\end{compte}

\vfill
\begin{exemple}[Ejemplo: \ $(x-2)(x+3)(2x-1)=0$]
$x-2=0\to x=2$; \quad $x+3=0\to x=-3$; \quad $2x-1=0\to x=\frac12$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $(x^2-9)(x^2+x-2)=0$]
$x^2-9=0\to x=\pm3$ \qquad $x^2+x-2=0\to x=1,\ x=-2$ \qquad Soluciones: \ \boxed{-3,\ -2,\ 1,\ 3}
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$x^2(x-4)(x^2+1)=0$: \ $x=0$ (doble), \ $x=4$; \ $x^2+1=0$ no tiene solución\\[4pt]
$(2x+1)^2(x^2+4)=0$: \ $x=-\frac12$ (doble) \qquad $(x+3)(x-1)=5$: \ $x^2+2x-8=0\to x=2,\ x=-4$\\[4pt]
$(x^2-4x+4)(x^3-x)=0$: \ $(x-2)^2\,x\,(x+1)(x-1)=0$ \ $\to$ \ $x=2$ (doble), \ $0,\ \pm1$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $(x^2-1)^2-(x^2-1)(x+1)=0$ \ (factor común con paréntesis)]
Sacamos factor común $(x^2-1)$: \ $(x^2-1)\big[(x^2-1)-(x+1)\big]=0\ \to\ (x^2-1)(x^2-x-2)=0$\\[4pt]
$x^2-1=0\to x=\pm1$ \qquad $x^2-x-2=0\to x=2,\ x=-1$ \qquad Soluciones: \ \boxed{x=-1\text{ (doble)},\ 1,\ 2}
\end{exemple}

\vfill
\begin{minipage}[t]{0.48\linewidth}
\begin{compte}[Tampoco simplifiques]
$x^2(x-3)=4x^2$: \ si dividimos entre $x^2$ perdemos $x=0$.\\[3pt]
$x^2(x-3)-4x^2=0\ \to\ x^2(x-7)=0$\\[2pt]
\boxed{x=0,\ x=7}
\end{compte}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
\begin{exemple}[Ejemplo: \ soluciones $2$, $-1$ y $\frac13$]
Un factor por solución:\\[3pt]
$(x-2)(x+1)(3x-1)=0$\\[2pt]
\notab{$x=\frac13$ da el factor $3x-1$}
\end{exemple}
\end{minipage}

\seccio{sec5}{Ecuaciones con fracciones algebraicas}{Multiplicamos por el m.c.m. y comprobamos.}

\vfill
\begin{minipage}[c]{0.55\linewidth}
\passos{\pas{1} Factorizamos los denominadores y hallamos el \textbf{m.c.m.}\\[3pt]
\pas{2} Multiplicamos los dos miembros por el m.c.m.\\[3pt]
\pas{3} Resolvemos la ecuación polinómica.\\[3pt]
\pas{4} \textbf{Comprobamos}: descartamos las que anulan un denominador.}
\end{minipage}\hfill
\begin{minipage}[c]{0.42\linewidth}
\begin{compte}[Puede no tener solución]
$\dfrac{x}{x-2}-\dfrac{2}{x-2}=3\ \to\ x=2$, pero anula el denominador: \textbf{no tiene solución}.
\end{compte}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ $\dfrac1x+\dfrac{1}{x+2}=\dfrac34$]
m.c.m. $=4x(x+2)$: \ $4(x+2)+4x=3x(x+2)\ \to\ 3x^2-2x-8=0\ \to\ x=\dfrac{2\pm10}{6}$\\[3pt]
\boxed{x=2,\ x=-\tfrac43} \ \nota{ninguna anula los denominadores}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $\dfrac{x}{x-1}+\dfrac{2}{x+1}=\dfrac{2}{x^2-1}$]
m.c.m. $=(x+1)(x-1)$: \ $x(x+1)+2(x-1)=2\ \to\ x^2+3x-4=0\ \to\ x=1,\ x=-4$\\[3pt]
$x=1$ anula $x-1$: se descarta. \ Solución: \boxed{x=-4}
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$\dfrac3x+2=5\ \to\ x=1$ \qquad $\dfrac{x}{x+1}=\dfrac2x\ \to\ x^2-2x-2=0\ \to\ x=1\pm\sqrt3$\\[5pt]
$\dfrac{6}{x^2-4}-\dfrac{1}{x-2}=\dfrac{1}{x+2}$: \ m.c.m. $(x+2)(x-2)$: \ $6-(x+2)=x-2\ \to\ x=3$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $\dfrac{x+1}{x-1}-\dfrac{x-1}{x+1}=\dfrac83$]
m.c.m. $=3(x+1)(x-1)$: \ $3(x+1)^2-3(x-1)^2=8(x^2-1)$\\[3pt]
$3(x^2+2x+1)-3(x^2-2x+1)=8x^2-8\ \to\ 12x=8x^2-8\ \to\ 2x^2-3x-2=0$ \ $\to$ \ $x=\dfrac{3\pm5}{4}$: \ \boxed{x=2,\ x=-\tfrac12}
\end{exemple}

\vfill
\newpage
\vfill
\begin{exemple}[Ejemplo: \ $\dfrac2x+\dfrac{x}{x-3}=\dfrac{9}{x^2-3x}$]
\begin{minipage}[t]{0.6\linewidth}
Factorizamos: \ $x^2-3x=x(x-3)$, \ que ya es el m.c.m.\\[3pt]
$2(x-3)+x\cdot x=9\ \to\ x^2+2x-15=0\ \to\ x=3,\ x=-5$\\[3pt]
$x=3$ anula $x-3$ \ \mal{x=3} \qquad Solución: \boxed{x=-5}
\end{minipage}\hfill
\begin{minipage}[t]{0.36\linewidth}
\notab{Comprobamos $x=-5$:}\\[2pt]
$-\frac25+\frac58=\frac{9}{40}$ \ y \ $\frac{9}{25+15}=\frac{9}{40}$ \ \be{}
\end{minipage}
\end{exemple}

\vfill
\begin{truc}[Antes del m.c.m., factoriza]
$x^2-3x=x(x-3)$, \ $x^2-1=(x+1)(x-1)$, \ $x^2-4x+4=(x-2)^2$\dots\ Así el m.c.m. es más sencillo y ves enseguida qué valores anulan los denominadores.
\end{truc}

\subsection{Problemas que llevan a fracciones}

\vfill
\begin{exemple}[Problema: \ un grupo paga 120 € a partes iguales. Si fueran 2 más, cada uno pagaría 5 € menos. ¿Cuántos son?]
$x$ personas: \ $\dfrac{120}{x}-\dfrac{120}{x+2}=5\ \to\ 120(x+2)-120x=5x(x+2)\ \to\ x^2+2x-48=0\ \to\ x=6,\ x=-8$\\[3pt]
Un número de personas no puede ser negativo: \ \boxed{6\text{ personas}} \ \nota{$20$ € cada uno; con 8, $15$ €}
\end{exemple}

\vfill
\begin{exemple}[Problema: \ un grifo llena un depósito en unas horas y otro tarda 3 h más. Juntos, lo llenan en 2 h. ¿Cuánto tarda cada uno?]
En 1 hora: \ $\dfrac1x+\dfrac{1}{x+3}=\dfrac12$ \ \nota{la parte del depósito que llenan}\\[4pt]
m.c.m. $2x(x+3)$: \ $2(x+3)+2x=x(x+3)\ \to\ x^2-x-6=0\ \to\ x=3,\ x=-2$ \qquad \boxed{3\text{ h y }6\text{ h}}
\end{exemple}

\vfill
\begin{exemple}[Problema: \ un ciclista recorre 60 km. Si fuera 5 km/h más rápido, tardaría 1 hora menos. ¿A qué velocidad va?]
Tiempo $=\dfrac{\text{distancia}}{\text{velocidad}}$: \ $\dfrac{60}{x}-\dfrac{60}{x+5}=1\ \to\ 60(x+5)-60x=x(x+5)\ \to\ x^2+5x-300=0$\\[4pt]
$x=\dfrac{-5\pm35}{2}=15,\ -20$ \qquad \boxed{15\text{ km/h}} \ \nota{tarda 4 h; a 20 km/h, 3 h}
\end{exemple}

\clearpage
\seccio{sec6}{Ecuaciones con radicales}{Aislamos la raíz, elevamos al cuadrado y comprobamos.}

\vfill
\begin{minipage}[c]{0.55\linewidth}
\passos{\pas{1} \textbf{Aislamos} el radical en un miembro.\\[3pt]
\pas{2} \textbf{Elevamos al cuadrado} los dos miembros.\\[3pt]
\pas{3} Si queda una raíz, repetimos.\\[3pt]
\pas{4} Resolvemos y \textbf{comprobamos}.}
\end{minipage}\hfill
\begin{minipage}[c]{0.42\linewidth}
\begin{compte}[Soluciones falsas]
Al elevar al cuadrado pueden aparecer soluciones que no lo son: \ $2\neq-2$, pero $2^2=(-2)^2$.
\end{compte}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ $x-\sqrt{x+1}=5$]
Aislamos: \ $x-5=\sqrt{x+1}$ \quad $\to$ \quad $x^2-10x+25=x+1\ \to\ x^2-11x+24=0\ \to\ x=8,\ x=3$\\[3pt]
Comprobamos: \ $8-3=5$ \be{} \qquad $3-2=1\neq5$ \ \mal{x=3} \qquad Solución: \boxed{x=8}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $\sqrt{2x+3}-\sqrt{x+1}=1$ \ \ (dos radicales)]
$\sqrt{2x+3}=1+\sqrt{x+1}\ \to\ 2x+3=1+2\sqrt{x+1}+x+1\ \to\ x+1=2\sqrt{x+1}$\\[3pt]
$x^2+2x+1=4x+4\ \to\ x^2-2x-3=0\ \to\ x=3,\ x=-1$ \qquad las dos valen: \ \boxed{x=3,\ x=-1}
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$\sqrt{3x+1}=4\ \to\ x=5$ \be{} \qquad $\sqrt{2x-1}=x-2\ \to\ x^2-6x+5=0$: \ $x=5$ \be{}, \ $x=1$ \mal{1\neq-1}
\end{exemple}

\vfill
\begin{truc}[El cuadrado de una suma]
$(1+\sqrt{x+1})^2=1+2\sqrt{x+1}+(x+1)$: \ no te olvides del \textbf{doble producto}.
\end{truc}

\vfill
\begin{exemple}[Ejemplo: \ $\sqrt{x+5}+\sqrt x=5$ \ (dejamos un radical en cada miembro)]
$\sqrt{x+5}=5-\sqrt x\ \to\ x+5=25-10\sqrt x+x\ \to\ 10\sqrt x=20\ \to\ \sqrt x=2\ \to\ \boxed{x=4}$ \quad $\sqrt9+\sqrt4=5$ \be{}
\end{exemple}

\vfill
\begin{minipage}[t]{0.48\linewidth}
\begin{exemple}[Problema: \ un número más su raíz cuadrada da 12]
$x+\sqrt x=12\ \to\ \sqrt x=12-x$\\[3pt]
$x=144-24x+x^2\ \to\ x^2-25x+144=0$\\[3pt]
$x=16$: \ $16+4=20$ \ \mal{\text{no}}\\[2pt] $x=9$: \ $9+3=12$ \ \be{}\\[3pt]
El número es \boxed{9}
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
\begin{compte}[Una raíz nunca es negativa]
$\sqrt{x-1}+3=1\ \to\ \sqrt{x-1}=-2$\\[3pt]
\textbf{No tiene solución}: no hace falta ni elevar al cuadrado.
\end{compte}
\end{minipage}

\clearpage
\seccio{sec7}{Ecuaciones exponenciales}{La incógnita está en el exponente.}

\vfill
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,before upper=\centering,top=2mm,bottom=2mm]
{\Large$a^M=a^N\iff M=N$}\par\smallskip
{\small $a>0$, $a\neq1$}
\end{tcolorbox}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height=rows,raster column skip=10pt,raster row skip=8pt]
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={\pas{1}\ Misma base}]
Igualamos los exponentes:\par\smallskip
$2^{x-1}=16=2^4\ \to\ x-1=4\ \to\ \boxed{x=5}$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmA!8,colframe=fmA,coltitle=white,title={\pas{2}\ Bases distintas}]
Tomamos \textbf{logaritmos}:\par\smallskip
$5^x=12\ \to\ x=\dfrac{\log12}{\log5}\approx\boxed{1{,}544}$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmB!8,colframe=fmB,coltitle=white,title={\pas{3}\ Sumas de potencias}]
Factor común $a^x$:\par\smallskip
$3^{x+1}+3^x+3^{x-1}=39$\\ $3^x\left(3+1+\tfrac13\right)=39\ \to\ 3^x=9\ \to\ \boxed{x=2}$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmV!8,colframe=fmV,coltitle=white,title={\pas{4}\ Aparecen $a^{2x}$ y $a^x$}]
Cambio $t=a^x$:\par\smallskip
$4^x-6\cdot2^x+8=0\ \to\ t^2-6t+8=0$\\ $t=4\to x=2$; \ $t=2\to x=1$
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{compte}[$a^x$ siempre es positivo]
En el cambio $t=a^x$, las soluciones $t\le0$ \textbf{no valen}: \ $9^x+5\cdot3^x-24=0$ da $t=3$ y $t=-8$: solo $x=1$.
\end{compte}

\vfill
\begin{exemple}[Ejemplos]
$2^{x^2-3}=64\ \to\ x^2-3=6\ \to\ x=\pm3$ \qquad $\left(\frac12\right)^x=8\ \to\ 2^{-x}=2^3\ \to\ x=-3$\\[4pt]
$2^{x+1}+2^x+2^{x-1}=28\ \to\ 2^x\cdot\frac72=28\ \to\ x=3$ \qquad $3^x=20\ \to\ x=\dfrac{\log20}{\log3}\approx2{,}727$
\end{exemple}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{exemple}[Ejemplo: \ $\dfrac{2^{x^2}}{2^x}=4$]
$2^{x^2-x}=2^2\ \to\ x^2-x-2=0$\\[3pt]
\boxed{x=2,\ x=-1}
\end{exemple}
\begin{exemple}[Ejemplo: \ $3^{2x}-4\cdot3^{x+1}+27=0$]
$3^{x+1}=3\cdot3^x$; \ $t=3^x$: \ $t^2-12t+27=0$\\[3pt]
$t=9\to x=2$; \ $t=3\to x=1$
\end{exemple}
\end{tcbraster}

\vfill
\begin{exemple}[Problemas]
500 bacterias que se duplican cada hora. ¿Cuándo serán 64\,000? \ $500\cdot2^t=64\,000\ \to\ 2^t=128\ \to\ \boxed{t=7\text{ h}}$\\[5pt]
1000 € al 5\,\% anual. ¿Cuándo serán 1500 €? \ $1{,}05^t=1{,}5\ \to\ t=\dfrac{\log1{,}5}{\log1{,}05}\approx8{,}31$ \ $\to$ \ \boxed{9\text{ años}}
\end{exemple}

\clearpage
\seccio{sec8}{Ecuaciones logarítmicas}{Un logaritmo en cada miembro, o la definición.}

\vfill
\begin{minipage}[c]{0.48\linewidth}
\begin{idea}[Recuerda]
$\log_aN=x\iff a^x=N$\\[3pt]
$\log(M\cdot N)=\log M+\log N$\\[2pt]
$\log\frac MN=\log M-\log N$\\[2pt]
$\log M^k=k\log M$ \qquad $1=\log10$
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.49\linewidth}
\passos{\pas{1} Agrupamos: \ $\log A=\log B\to A=B$.\\[3pt]
\pas{2} O bien $\log_aA=k\to A=a^k$.\\[3pt]
\pas{3} \textbf{Comprobamos}: argumentos \textbf{positivos}.}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ por la definición]
$\log(x+1)=2\ \to\ x+1=100\ \to\ \boxed{x=99}$ \qquad $\log_2(x-3)=3\ \to\ x-3=8\ \to\ \boxed{x=11}$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $\log x+\log(x-3)=1$]
$\log\big(x(x-3)\big)=\log10\ \to\ x^2-3x-10=0\ \to\ x=5,\ x=-2$\\[3pt]
$\log(-2)$ no existe: \ \mal{x=-2} \qquad Solución: \boxed{x=5}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $2\log x-\log(x+6)=0$]
$\log x^2=\log(x+6)\ \to\ x^2-x-6=0\ \to\ x=3,\ x=-2$ \qquad solo vale \boxed{x=3}
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$\log_3(2x+1)=2\to x=4$ \quad $\ln x+\ln2=\ln10\to x=5$ \quad $\log(15-2x)=2\log x\to x=3$ \ \nota{$-5$ no vale}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $\log(x+6)-\log(x-3)=1$]
$\log\dfrac{x+6}{x-3}=\log10\ \to\ x+6=10(x-3)\ \to\ 9x=36\ \to\ \boxed{x=4}$ \ \be{}
\end{exemple}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{exemple}[Ejemplo: \ $\log_2x+\log_2(x+2)=3$]
$\log_2\big(x(x+2)\big)=3\ \to\ x^2+2x=2^3$\\[3pt]
$x^2+2x-8=0\ \to\ x=2,\ x=-4$\\[3pt]
$x=-4$ no vale: \ \boxed{x=2}
\end{exemple}
\begin{exemple}[Ejemplo: \ $(\log x)^2-3\log x+2=0$]
Cambio $t=\log x$: \ $t^2-3t+2=0$\\[3pt]
$t=1\to x=10$; \ $t=2\to x=100$\\[3pt]
\boxed{x=10,\ x=100}
\end{exemple}
\end{tcbraster}

\vfill
\begin{minipage}[t]{0.5\linewidth}
\begin{exemple}[Ejemplo: \ $2\log x=\log(5x-6)$]
$x^2=5x-6\ \to\ x^2-5x+6=0$\\[3pt]
$x=2$ y $x=3$: las dos valen \be{}
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.47\linewidth}
\begin{compte}[El logaritmo de una suma]
$\log(x+3)\neq\log x+\log3$. \ La propiedad es para el \textbf{producto}: \ $\log(3x)=\log3+\log x$.
\end{compte}
\end{minipage}

\seccio{sec9}{Inecuaciones de primer grado}{Como una ecuación, pero con una regla especial.}

\vfill
\begin{idea}[Inecuación]
Una desigualdad ($<$, $\le$, $>$, $\ge$) con una incógnita. Las soluciones suelen ser \textbf{infinitas}: las escribimos con \textbf{intervalos}.
\end{idea}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={\faCheck\ \ Se mantiene}]
Sumar o restar, y multiplicar o dividir por un número \textbf{positivo}.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmR!6,colframe=fmR,coltitle=white,title={\faIcon{exchange-alt}\ \ Cambia de sentido}]
Multiplicar o dividir por un \textbf{negativo}:\par\smallskip
{\centering$-2x<6\ \to\ x\cR{>}-3$\par}
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{exemple}[Ejemplo: \ $3x-2\le x+4$]
$2x\le6\ \to\ x\le3$ \qquad $\boxed{\lo-\infty,\,3]}$
\end{exemple}
\begin{exemple}[Ejemplo: \ $-2x+1<7$]
$-2x<6$. \ Entre $-2$, \textbf{giramos}: \ $x>-3$ \qquad $\boxed{\lo-3,+\infty\ro}$
\end{exemple}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplo: \ $\dfrac{x-1}{2}-\dfrac{x+2}{3}\ge1$]
Por 6 (positivo): \ $3(x-1)-2(x+2)\ge6\ \to\ 3x-3-2x-4\ge6\ \to\ x\ge13$ \qquad $\boxed{[13,+\infty\ro}$
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$5x+3>2x-6\to x>-3$ \qquad $4-3x\ge10\to x\le-2$ \qquad $2(x-1)<5x+4\to x>-2$ \qquad $\frac x2-1\le\frac x3\to x\le6$\\[4pt]
Con paréntesis: \ $3(x-2)-2(1-x)>4x+1\ \to\ 5x-8>4x+1\ \to\ x>9$: \ $\lo9,+\infty\ro$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ sistema \ $2x-1>3$ \ y \ $x+4\le10$]
\begin{minipage}[c]{0.5\linewidth}
Resolvemos cada una: \ $x>2$ \ y \ $x\le6$\\[3pt]
Las soluciones son las \textbf{comunes} (intersección):\\[3pt]
$\boxed{\lo2,\,6]}$
\end{minipage}\hfill

\begin{tikzpicture}[x=0.5cm]

  \draw[fmA,line width=2.4pt,-{Latex[length=2.4mm]}] (2,0.35) -- (9,0.35); \filldraw[fill=white,draw=fmA,line width=1.2pt] (2,0.35) circle (3pt);
  \draw[fmB,line width=2.4pt,{Latex[length=2.4mm]}-] (-1,-0.35) -- (6,-0.35); \fill[fmB] (6,-0.35) circle (3.2pt);
  \fill[fmK!40,opacity=0.6] (2,-0.15) rectangle (6,0.15);
  \node[below=6pt,font=\scriptsize] at (2,-0.35) {2}; \node[below=6pt,font=\scriptsize] at (6,-0.35) {6};
\end{tikzpicture}
\end{minipage}
\end{exemple}

\vfill
\begin{minipage}[t]{0.48\linewidth}
\begin{exemple}[Ejemplo: \ $-3<2x+1\le7$]
Restamos 1 a los tres miembros: \ $-4<2x\le6$\\[3pt]
Dividimos entre 2: \ $-2<x\le3$ \ $\to$ \ $\boxed{\lo-2,\,3]}$
\end{exemple}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
\begin{exemple}[Problema: \ tarifa A: 20 € + 0,05 €/min; \ tarifa B: 0,10 €/min. ¿Cuándo es mejor A?]
$20+0{,}05x<0{,}10x\ \to\ 20<0{,}05x$\\[3pt]
$x>400$: \ \boxed{\text{más de 400 min}}
\end{exemple}
\end{minipage}

\seccio{sec10}{Inecuaciones de segundo grado o superior}{Raíces, intervalos y tabla de signos.}

\vfill
\begin{minipage}[c]{0.6\linewidth}
\passos{\pas{1} Todo a un miembro: \ $P(x)>0$ \ o \ $P(x)<0$ \ (o $\ge$, $\le$).\\[3pt]
\pas{2} Resolvemos la ecuación $P(x)=0$.\\[3pt]
\pas{3} Situamos las soluciones en la recta real y vemos el signo de cada trozo sustituyendo en $P(x)$.\\[3pt]
\pas{4} Elegimos los intervalos. Raíces, solo con $\le$ o $\ge$.}
\end{minipage}\hfill
