\seccio{1}{Ecuaciones de segundo grado}

\exf{1} Resuelve las ecuaciones incompletas:\begin{tasks}(3)
  \task $x^2-16=0$
  \task $3x^2+12x=0$
  \task $2x^2-50=0$
  \task $x^2+9=0$
  \task $5x^2=3x$
  \task $4x^2-1=0$
\end{tasks}

\ex{1} Resuelve las ecuaciones:\begin{tasks}(3)
  \task $x^2-5x+6=0$
  \task $x^2+2x-15=0$
  \task $2x^2-7x+3=0$
  \task $x^2-6x+9=0$
  \task $x^2+x+3=0$
  \task $6x^2-x-1=0$
\end{tasks}\recorda{$ax^2+bx+c=0\Rightarrow x=\dfrac{-b\pm\sqrt{b^2-4ac}}{2a}$.}

\ex{2} Opera hasta obtener la forma $ax^2+bx+c=0$ y resuelve:\begin{tasks}[after-item-skip=10pt](2)
  \task $(x+2)^2=3x+10$
  \task $x(x-3)=2(x+7)$
  \task $(2x-1)(x+3)-x^2=3$
  \task $\dfrac{x^2}{3}-\dfrac x2=\dfrac32$
\end{tasks}

\ex{3} Elimina los paréntesis y los denominadores, pasa a la forma $ax^2+bx+c=0$ y resuelve:\begin{tasks}[after-item-skip=14pt](2)
  \task $\dfrac{x(x+3)}{2}-\dfrac{(x-1)(x+1)}{3}=4$
  \task $\dfrac{(x+3)(x-3)}{4}-\dfrac{(x-1)^2}{2}=-2$
  \task $\dfrac{(2x+1)(x-3)}{5}-\dfrac{x-1}{2}=\dfrac{x^2-13}{10}$
  \task $\dfrac{(x+1)^2}{2}-\dfrac{(x-1)^2}{3}=\dfrac{x+11}{6}+x$
  \task $\dfrac{x(x-1)}{3}-\dfrac{(x+1)(x-2)}{4}=1$
  \task $\dfrac{(x-1)(x+2)}{4}-\dfrac{x+1}{2}=\dfrac{x-7}{4}$
  \task $\dfrac{(x+2)^2}{4}-\dfrac{x-1}{2}=\dfrac{x^2+11}{8}$
  \task $2(x-1)^2-3(x+1)(x-2)=x(x-5)-4$
\end{tasks}\recorda{Primero desarrolla los productos y las identidades notables; después multiplica todos los términos por el m.c.m. de los denominadores. ¡Cuidado con el signo menos delante de una fracción!}

\ex{2} Sin resolverlas, indica cuántas soluciones tiene cada ecuación:\begin{tasks}(3)
  \task $3x^2-2x+5=0$
  \task $4x^2-12x+9=0$
  \task $x^2+x-1=0$
\end{tasks}Halla después $k$ para que $x^2-6x+k=0$ tenga una solución doble.

\ex{3} Escribe una ecuación de segundo grado que tenga por soluciones $x=3$ y $x=-5$.

\seccio{2}{Ecuaciones bicuadradas}

\exf{1} Resuelve las ecuaciones bicuadradas:\begin{tasks}(2)
  \task $x^4-10x^2+9=0$
  \task $x^4-5x^2-36=0$
  \task $x^4+3x^2+2=0$
  \task $4x^4-17x^2+4=0$
\end{tasks}\recorda{Haz el cambio $t=x^2$. Cada solución $t>0$ da $x=\pm\sqrt t$; las $t<0$ no dan soluciones.}

\ex{2} Resuelve:\begin{tasks}(3)
  \task $x^4-8x^2=0$
  \task $(x^2-1)(x^2+2)=10$
  \task $x^4-16=0$
\end{tasks}

\ex{3} Opera hasta obtener una ecuación bicuadrada y resuelve:\begin{tasks}[after-item-skip=14pt](2)
  \task $(x^2-2)(x^2+2)=3x^2$
  \task $\dfrac{x^2(x^2-5)}{2}=-2$
  \task $(x^2+1)^2-3(x^2+1)=10$
  \task $\dfrac{x^4}{4}-\dfrac{5x^2}{2}+4=0$
  \task $\dfrac{x^4}{3}-\dfrac{x^2+2}{2}=x^2-\dfrac52$
  \task $(2x^2-1)^2-(x^2+2)^2=3x^4-21$
\end{tasks}

\seccio{3}{Ecuaciones de grado mayor que 2}

\exf{2} Resuelve con la regla de Ruffini:\begin{tasks}(2)
  \task $x^3-6x^2+11x-6=0$
  \task $x^3+2x^2-x-2=0$
  \task $x^3-3x+2=0$
  \task $2x^3+x^2-8x-4=0$
\end{tasks}\recorda{Prueba con los divisores del término independiente hasta llegar a una ecuación de segundo grado.}

\ex{2} Pasa todos los términos a un miembro, factoriza y resuelve:\begin{tasks}(3)
  \task $x^4-x^3-7x^2+x+6=0$
  \task $x^3=4x$
  \task $x^4+x^3=x^2+x$
\end{tasks}

\seccio{4}{Ecuaciones factorizadas}

\exf{1} Resuelve sin multiplicar los factores:\begin{tasks}(2)
  \task $(x-4)(x+1)=0$
  \task $x(2x+3)(x-5)=0$
  \task $(x-1)^2(x+7)=0$
  \task $3(x+2)(4x-1)=0$
\end{tasks}\recorda{Un producto es cero si alguno de los factores es cero.}

\ex{2} Resuelve:\begin{tasks}(2)
  \task $(x^2-16)(x+3)=0$
  \task $(x^2-3x+2)(x^2+5)=0$
  \task $x^3(x^2-2x-8)=0$
  \task $(x-2)(x+1)=4$
\end{tasks}

\seccio{5}{Ecuaciones con fracciones algebraicas}

\exf{1} Resuelve:\begin{tasks}[after-item-skip=12pt](2)
  \task $\dfrac5x-1=4$
  \task $\dfrac{x}{x-1}=\dfrac32$
  \task $\dfrac1x+\dfrac{2}{x+1}=1$
  \task $x+\dfrac4x=5$
\end{tasks}

\ex{2} Resuelve y comprueba las soluciones:\begin{tasks}[after-item-skip=14pt](2)
  \task $\dfrac{x}{x+2}-\dfrac{1}{x-2}=\dfrac{8}{x^2-4}$
  \task $\dfrac{3}{x-1}-\dfrac{2}{x+1}=\dfrac{6}{x^2-1}$
  \task $\dfrac{1}{x-1}+\dfrac{1}{x+1}=\dfrac43$
  \task $\dfrac{x}{x-2}-\dfrac{2}{x+2}=\dfrac{8}{x^2-4}$
  \task $\dfrac{x+3}{x-1}-\dfrac{x-3}{x+1}=4$
  \task $\dfrac2x-\dfrac{3}{x^2+x}=\dfrac{1}{x+1}$
\end{tasks}\recorda{Descarta las soluciones que anulan algún denominador.}

\ex{3} Resuelve $\dfrac{x+1}{x}-\dfrac{x}{x+1}=\dfrac56$.

\seccio{6}{Ecuaciones con radicales}

\exf{1} Resuelve:\begin{tasks}(4)
  \task $\sqrt{x+5}=3$
  \task $\sqrt{2x-1}=5$
  \task $\sqrt{x^2+9}=5$
  \task $\sqrt{x+3}=-2$
\end{tasks}

\ex{2} Resuelve y comprueba las soluciones:\begin{tasks}(2)
  \task $\sqrt{x+2}=x$
  \task $x+\sqrt{x-1}=7$
  \task $\sqrt{3x+4}-x=0$
  \task $2\sqrt x=x-3$
  \task $\sqrt{2x+3}=x$
  \task $x-\sqrt{x+10}=2$
  \task $\sqrt{5x+4}-2=x$
  \task $3+\sqrt{x-1}=x$
\end{tasks}\recorda{Aísla el radical, eleva al cuadrado y comprueba todas las soluciones en la ecuación inicial.}

\ex{3} Resuelve (hay dos radicales):\begin{tasks}(2)
  \task $\sqrt{x+7}-\sqrt x=1$
  \task $\sqrt{2x+1}+\sqrt{x-3}=4$
\end{tasks}

\seccio{7}{Ecuaciones exponenciales}

\exf{1} Escribe los dos miembros como potencias de la misma base y resuelve:\begin{tasks}(3)
  \task $2^x=32$
  \task $3^{x-2}=27$
  \task $5^{2x}=\frac{1}{25}$
  \task $4^x=8$
  \task $2^{x^2-1}=8$
  \task $10^{x+1}=0,001$
\end{tasks}

\ex{1} Resuelve tomando logaritmos (da el resultado con tres decimales):\begin{tasks}(4)
  \task $2^x=10$
  \task $3^x=50$
  \task $7^{x-1}=30$
  \task $1,05^x=2$
\end{tasks}\recorda{$a^x=b\Rightarrow x=\dfrac{\log b}{\log a}$.}

\ex{2} Resuelve sacando factor común:\begin{tasks}(3)
  \task $2^{x+1}+2^x=24$
  \task $3^{x+2}-3^x=72$
  \task $5^x+5^{x-1}=30$
\end{tasks}

\ex{3} Resuelve con un cambio de variable:\begin{tasks}(3)
  \task $4^x-5\cdot2^x+4=0$
  \task $9^x-10\cdot3^x+9=0$
  \task $4^x-2^x-12=0$
\end{tasks}

\seccio{8}{Ecuaciones logarítmicas}

\exf{1} Resuelve aplicando la definición de logaritmo:\begin{tasks}(4)
  \task $\log x=3$
  \task $\log_2(x+1)=4$
  \task $\ln x=0$
  \task $\log_5(3x-1)=2$
\end{tasks}

\ex{2} Resuelve y comprueba las soluciones:\begin{tasks}(2)
  \task $\log x+\log4=\log20$
  \task $\log(x+3)-\log(x-1)=\log3$
  \task $2\log x=\log(5x-6)$
  \task $\log x+\log(x+9)=1$
  \task $\log(x+1)+\log(x-2)=1$
  \task $2\log x-\log(x-2)=\log9$
  \task $\log(2x+4)-\log x=\log3$
  \task $\log_3(x+2)+\log_3(x-4)=3$
\end{tasks}\recorda{Agrupa en un solo logaritmo en cada miembro. Los argumentos de los logaritmos deben ser positivos.}

\ex{3} Resuelve:\begin{tasks}(2)
  \task $\log(x+6)=1+\log(x-3)$
  \task $\log_2x+\log_2(x-2)=3$
\end{tasks}

\seccio{9}{Inecuaciones de primer grado}

\exf{1} Resuelve y representa la solución:\begin{tasks}(4)
  \task $2x-5>3$
  \task $3-x\le5$
  \task $4x+1<2x-7$
  \task $-3x\ge12$
\end{tasks}\recorda{Si multiplicas o divides por un número negativo, la desigualdad cambia de sentido.}

\ex{2} Resuelve:\begin{tasks}[after-item-skip=12pt](2)
  \task $3(x-2)-2(x+1)\ge x-8$
  \task $\dfrac{x+1}{2}-\dfrac{x-3}{4}<2$
  \task $2x-\dfrac{3-x}{2}>\dfrac x3$
  \task $5x-1>5(x+1)$
\end{tasks}

\ex{2} Con la tarifa A, un móvil cuesta 40 € al mes. Con la tarifa B cuesta 15 € al mes más 0,50 € por cada GB. ¿A partir de cuántos GB es más cara la tarifa B?

\seccio{11}{Problemas}

\exf{2} El producto de dos números naturales consecutivos es 156. ¿Cuáles son?

\ex{2} Un rectángulo tiene 34 cm de perímetro y 13 cm de diagonal. ¿Cuánto miden los lados?

\ex{2} ¿Cuántos años deben pasar para que 5000 € a un 3\,\% de interés compuesto anual se conviertan en 6000 €?

\ex{2} Un grupo de amigos alquila un apartamento por 600 €. Si hubieran sido dos amigos más, cada uno habría pagado 25 € menos. ¿Cuántos amigos son?

\ex{3} Un grifo tarda 5 horas más que otro en llenar un depósito. Si los abrimos juntos, lo llenan en 6 horas. ¿Cuánto tarda cada uno por separado?

\ex{2} Si a un número positivo le sumamos su raíz cuadrada, obtenemos 42. ¿Cuál es el número?

\ex{2} La suma de un número y su inverso es $\frac{26}{5}$. ¿Cuál es el número?

\ex{3} ¿Para qué números su cuadrado es menor que su triple? ¿Y para cuáles su cubo es mayor que el propio número?

\ex{3} Una población de bacterias sigue la fórmula $N=500\cdot2^{t/3}$, con $t$ en horas. ¿Cuándo habrá 16\,000 bacterias?
