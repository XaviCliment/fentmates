\seccio{1}{Equacions de segon grau}

\exf{1} Resol les equacions incompletes:\begin{tasks}(3)
  \task $x^2-16=0$
  \task $3x^2+12x=0$
  \task $2x^2-50=0$
  \task $x^2+9=0$
  \task $5x^2=3x$
  \task $4x^2-1=0$
\end{tasks}

\ex{1} Resol les equacions:\begin{tasks}(3)
  \task $x^2-5x+6=0$
  \task $x^2+2x-15=0$
  \task $2x^2-7x+3=0$
  \task $x^2-6x+9=0$
  \task $x^2+x+3=0$
  \task $6x^2-x-1=0$
\end{tasks}\recorda{$ax^2+bx+c=0\Rightarrow x=\dfrac{-b\pm\sqrt{b^2-4ac}}{2a}$.}

\ex{2} Opera fins a obtindre la forma $ax^2+bx+c=0$ i resol:\begin{tasks}(2)
  \task $(x+2)^2=3x+10$
  \task $x(x-3)=2(x+7)$
  \task $(2x-1)(x+3)-x^2=3$
  \task $\dfrac{x^2}{3}-\dfrac x2=\dfrac32$
\end{tasks}

\ex{2} Sense resoldre-les, indica quantes solucions té cada equació:\begin{tasks}(3)
  \task $3x^2-2x+5=0$
  \task $4x^2-12x+9=0$
  \task $x^2+x-1=0$
\end{tasks}Troba després $k$ perquè $x^2-6x+k=0$ tinga una solució doble.

\ex{3} Escriu una equació de segon grau que tinga per solucions $x=3$ i $x=-5$.


% =====================================================
\seccio{2}{Equacions biquadrades}

\exf{1} Resol les equacions biquadrades:\begin{tasks}(2)
  \task $x^4-10x^2+9=0$
  \task $x^4-5x^2-36=0$
  \task $x^4+3x^2+2=0$
  \task $4x^4-17x^2+4=0$
\end{tasks}\recorda{Fes el canvi $t=x^2$. Cada solució $t>0$ dona $x=\pm\sqrt t$; les $t<0$ no donen solucions.}

\ex{2} Resol:\begin{tasks}(3)
  \task $x^4-8x^2=0$
  \task $(x^2-1)(x^2+2)=10$
  \task $x^4-16=0$
\end{tasks}


% =====================================================
\seccio{3}{Equacions de grau major que 2}

\exf{2} Resol amb la regla de Ruffini:\begin{tasks}(2)
  \task $x^3-6x^2+11x-6=0$
  \task $x^3+2x^2-x-2=0$
  \task $x^3-3x+2=0$
  \task $2x^3+x^2-8x-4=0$
\end{tasks}\recorda{Prova amb els divisors del terme independent fins a arribar a una equació de segon grau.}

\ex{2} Passa tots els termes a un membre, factoritza i resol:\begin{tasks}(3)
  \task $x^4-x^3-7x^2+x+6=0$
  \task $x^3=4x$
  \task $x^4+x^3=x^2+x$
\end{tasks}


% =====================================================
\seccio{4}{Equacions factoritzades}

\exf{1} Resol sense multiplicar els factors:\begin{tasks}(2)
  \task $(x-4)(x+1)=0$
  \task $x(2x+3)(x-5)=0$
  \task $(x-1)^2(x+7)=0$
  \task $3(x+2)(4x-1)=0$
\end{tasks}\recorda{Un producte és zero si algun dels factors és zero.}

\ex{2} Resol:\begin{tasks}(2)
  \task $(x^2-16)(x+3)=0$
  \task $(x^2-3x+2)(x^2+5)=0$
  \task $x^3(x^2-2x-8)=0$
  \task $(x-2)(x+1)=4$
\end{tasks}


% =====================================================
\seccio{5}{Equacions amb fraccions algebraiques}

\exf{1} Resol:\begin{tasks}(2)
  \task $\dfrac5x-1=4$
  \task $\dfrac{x}{x-1}=\dfrac32$
  \task $\dfrac1x+\dfrac{2}{x+1}=1$
  \task $x+\dfrac4x=5$
\end{tasks}

\ex{2} Resol i comprova les solucions:\begin{tasks}(1)
  \task $\dfrac{x}{x+2}-\dfrac{1}{x-2}=\dfrac{8}{x^2-4}$
  \task $\dfrac{3}{x-1}-\dfrac{2}{x+1}=\dfrac{6}{x^2-1}$
\end{tasks}\recorda{Descarta les solucions que anul·len algun denominador.}

\ex{3} Resol $\dfrac{x+1}{x}-\dfrac{x}{x+1}=\dfrac56$.


% =====================================================
\seccio{6}{Equacions amb radicals}

\exf{1} Resol:\begin{tasks}(4)
  \task $\sqrt{x+5}=3$
  \task $\sqrt{2x-1}=5$
  \task $\sqrt{x^2+9}=5$
  \task $\sqrt{x+3}=-2$
\end{tasks}

\ex{2} Resol i comprova les solucions:\begin{tasks}(2)
  \task $\sqrt{x+2}=x$
  \task $x+\sqrt{x-1}=7$
  \task $\sqrt{3x+4}-x=0$
  \task $2\sqrt x=x-3$
\end{tasks}\recorda{Aïlla el radical, eleva al quadrat i comprova totes les solucions en l'equació inicial.}

\ex{3} Resol (hi ha dos radicals):\begin{tasks}(2)
  \task $\sqrt{x+7}-\sqrt x=1$
  \task $\sqrt{2x+1}+\sqrt{x-3}=4$
\end{tasks}


% =====================================================
\seccio{7}{Equacions exponencials}

\exf{1} Escriu els dos membres com a potències de la mateixa base i resol:\begin{tasks}(3)
  \task $2^x=32$
  \task $3^{x-2}=27$
  \task $5^{2x}=\frac{1}{25}$
  \task $4^x=8$
  \task $2^{x^2-1}=8$
  \task $10^{x+1}=0,001$
\end{tasks}

\ex{1} Resol prenent logaritmes (dona el resultat amb tres decimals):\begin{tasks}(4)
  \task $2^x=10$
  \task $3^x=50$
  \task $7^{x-1}=30$
  \task $1,05^x=2$
\end{tasks}\recorda{$a^x=b\Rightarrow x=\dfrac{\log b}{\log a}$.}

\ex{2} Resol traient factor comú:\begin{tasks}(3)
  \task $2^{x+1}+2^x=24$
  \task $3^{x+2}-3^x=72$
  \task $5^x+5^{x-1}=30$
\end{tasks}

\ex{3} Resol amb un canvi de variable:\begin{tasks}(3)
  \task $4^x-5\cdot2^x+4=0$
  \task $9^x-10\cdot3^x+9=0$
  \task $4^x-2^x-12=0$
\end{tasks}


% =====================================================
\seccio{8}{Equacions logarítmiques}

\exf{1} Resol aplicant la definició de logaritme:\begin{tasks}(4)
  \task $\log x=3$
  \task $\log_2(x+1)=4$
  \task $\ln x=0$
  \task $\log_5(3x-1)=2$
\end{tasks}

\ex{2} Resol i comprova les solucions:\begin{tasks}(2)
  \task $\log x+\log4=\log20$
  \task $\log(x+3)-\log(x-1)=\log3$
  \task $2\log x=\log(5x-6)$
  \task $\log x+\log(x+9)=1$
\end{tasks}\recorda{Agrupa en un sol logaritme a cada membre. Els arguments dels logaritmes han de ser positius.}

\ex{3} Resol:\begin{tasks}(2)
  \task $\log(x+6)=1+\log(x-3)$
  \task $\log_2x+\log_2(x-2)=3$
\end{tasks}

\seccio{9}{Inequacions de primer grau}

\exf{1} Resol i representa la solució:\begin{tasks}(4)
  \task $2x-5>3$
  \task $3-x\le5$
  \task $4x+1<2x-7$
  \task $-3x\ge12$
\end{tasks}\recorda{Si multipliques o divideixes per un nombre negatiu, la desigualtat canvia de sentit.}

\ex{2} Resol:\begin{tasks}(2)
  \task $3(x-2)-2(x+1)\ge x-8$
  \task $\dfrac{x+1}{2}-\dfrac{x-3}{4}<2$
  \task $2x-\dfrac{3-x}{2}>\dfrac x3$
  \task $5x-1>5(x+1)$
\end{tasks}

\ex{2} Amb la tarifa A, un mòbil costa 40 € al mes. Amb la tarifa B costa 15 € al mes més 0,50 € per cada GB. A partir de quants GB és més cara la tarifa B?

\seccio{10}{Inequacions de segon grau o superior}

\exf{2} Resol i representa la solució:\begin{tasks}(3)
  \task $x^2-9<0$
  \task $x^2-3x-10\ge0$
  \task $x^2+4x>0$
  \task $-x^2+2x+3>0$
  \task $x^2-4x+4\le0$
  \task $x^2+x+1>0$
\end{tasks}\recorda{Calcula les arrels, divideix la recta en intervals i estudia el signe en cada un.}

\ex{3} Resol:\begin{tasks}(3)
  \task $x^3-x>0$
  \task $(x-1)(x+2)(x-4)\le0$
  \task $x^3-2x^2-3x\ge0$
\end{tasks}

\ex{2} Els costats d'un rectangle fan $x$ i $x+3$ cm. Per a quins valors de $x$ l'àrea és menor que 40 cm$^2$?
