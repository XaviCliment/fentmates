\seccio{5}{Equacions amb fraccions algebraiques}

\exf{1} \begin{tasks}(2)
  \task $\res{x=1}$
  \task $\res{x=3}$
  \task $x^2-2x-1=0$: $\res{x=1\pm\sqrt2}$
  \task $x^2-5x+4=0$: $\res{x=1,\ x=4}$
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $x^2-3x-10=0\Rightarrow x=5,\ x=-2$; $x=-2$ anul·la un denominador: $\res{x=5}$
  \task $x+5=6\Rightarrow x=1$, que anul·la $x-1$: $\res{\text{no té solució}}$
  \task $2x^2-3x-2=0$: $\res{x=2,\ x=-\frac12}$
  \task $x^2+4=8\Rightarrow x=\pm2$, i totes dues anul·len un denominador: $\res{\text{no té solució}}$
  \task $x^2-2x-1=0$: $\res{x=1\pm\sqrt2}$
  \task $2(x+1)-3=x$: $\res{x=1}$
\end{tasks}

\ex{3} m.c.m. $=6x(x+1)$: $6(x+1)^2-6x^2=5x(x+1)\Rightarrow5x^2-7x-6=0\Rightarrow\res{x=2,\ x=-\frac35}$ (totes dues valen).

\seccio{6}{Equacions amb radicals}

\exf{1} \begin{tasks}(4)
  \task $\res{x=4}$
  \task $\res{x=13}$
  \task $\res{x=\pm4}$
  \task No en té: una arrel no és negativa.
\end{tasks}

\ex{2} \begin{tasks}(1)
  \task $x^2-x-2=0\Rightarrow x=2,\ x=-1$; $x=-1$ no val ($\sqrt1\neq-1$): $\res{x=2}$
  \task $x^2-15x+50=0\Rightarrow x=5,\ x=10$; $x=10$ no val ($10+3\neq7$): $\res{x=5}$
  \task $x^2-3x-4=0\Rightarrow x=4,\ x=-1$; $x=-1$ no val ($1+1\neq0$): $\res{x=4}$
  \task $x^2-10x+9=0\Rightarrow x=1,\ x=9$; $x=1$ no val ($2\neq-2$): $\res{x=9}$
  \task $x^2-2x-3=0\Rightarrow x=3,\ x=-1$; $x=-1$ no val: $\res{x=3}$
  \task $x^2-5x-6=0\Rightarrow x=6,\ x=-1$; $x=-1$ no val ($-1-3\neq2$): $\res{x=6}$
  \task $x^2-x=0\Rightarrow x=0,\ x=1$; totes dues valen: $\res{x=0,\ x=1}$
  \task $x^2-7x+10=0\Rightarrow x=5,\ x=2$; $x=2$ no val ($3+1\neq2$): $\res{x=5}$
\end{tasks}

\ex{3} \begin{tasks}(1)
  \task $x+7=1+2\sqrt x+x\Rightarrow\sqrt x=3$: $\res{x=9}$
  \task $x-12=-8\sqrt{x-3}\Rightarrow x^2-88x+336=0\Rightarrow x=4,\ x=84$; $x=84$ no val ($13+9\neq4$): $\res{x=4}$
\end{tasks}
