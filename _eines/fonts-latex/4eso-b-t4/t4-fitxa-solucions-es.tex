\seccio{4}{Problemas con sistemas}

\exf{1} $\begin{cases}x+y=28\\[2pt] x-y=4\end{cases}$: $\res{16\ \text{chicas y}\ 12\ \text{chicos}}$.

\ex{2} $\begin{cases}3a+2c=33\\[2pt] 2a+3c=27\end{cases}$: $\res{\text{adulto}\ 9\ \text{€,\, niño}\ 3\ \text{€}}$.

\ex{2} $x$ decenas, $y$ unidades: $\begin{cases}x+y=11\\[2pt] (10x+y)-(10y+x)=27\end{cases}\Rightarrow x-y=3$: el número es $\res{74}$.

\ex{2} $\begin{cases}x+y=20\\[2pt] 8x+12y=20\cdot9{,}40=188\end{cases}$: $\res{13\ \text{kg de 8 € y}\ 7\ \text{kg de 12 €}}$.

\ex{2} $\begin{cases}x+y=14\\[2pt] x^2+y^2=100\end{cases}\Rightarrow x^2-14x+48=0$: $\res{8\ \text{m y}\ 6\ \text{m}}$.

\ex{3} $\begin{cases}x^2+y^2=169\\[9pt] \frac{xy}{2}=30\end{cases}$; como $(x+y)^2=169+120=289$, $x+y=17$ y $xy=60$: $\res{12\ \text{cm y}\ 5\ \text{cm}}$.
