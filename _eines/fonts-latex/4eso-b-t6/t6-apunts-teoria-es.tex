\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Opuesto}]
$\vect{BA}=-\vect{AB}$: mismo módulo y dirección, sentido contrario.
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmdefbg,colframe=fmdef,coltitle=white,title={Extremo u origen}]
$B=A+\vect{AB}$\\[2pt] $A=B-\vect{AB}$
\end{tcolorbox}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplo: \ $A(-1,3)$ y $B(3,0)$]
$\vect{AB}=(3-(-1),\ 0-3)=(4,-3)$ \qquad $|\vect{AB}|=\sqrt{16+9}=\boxed{5}$ \qquad $\vect{BA}=(-4,\,3)$
\end{exemple}

\vfill
\begin{exemple}[Ejemplos]
$\vec v=(2,-5)$ con origen $P(4,1)$: \ extremo $P+\vec v=\boxed{(6,-4)}$\\[4pt]
$P(2,-3)$, $Q(-4,5)$: \ $\vect{PQ}=(-6,8)$, \ $\vect{QP}=(6,-8)$, \ módulo 10 \qquad $|(5,12)|=13$, \ $|(-2,-2)|=2\sqrt2$\\[4pt]
$\vect{AB}=(-3,4)$ con $A(5,-2)$: \ $B=(2,2)$
\end{exemple}

\begin{minipage}[c]{0.55\linewidth}
\begin{idea}[Operaciones]
\textbf{Suma}: \ $\vec u+\vec v=(u_1+v_1,\ u_2+v_2)$\\[3pt]
\textbf{Resta}: \ $\vec u-\vec v=(u_1-v_1,\ u_2-v_2)$\\[3pt]

\end{idea}

{\Large$\vec u\parl\vec v\iff\dfrac{u_1}{v_1}=\dfrac{u_2}{v_2}\iff u_1v_2=u_2v_1$}\par\smallskip
{\small componentes proporcionales: \ $\vec u=k\vec v$}
\end{tcolorbox}

\vfill
\begin{exemple}[Ejemplo: \ $\vec u=(3,-1)$, \ $\vec v=(-2,4)$]
\begin{minipage}[t]{0.48\linewidth}
a) $\vec u+\vec v=\boxed{(1,\,3)}$\\[3pt]

\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
c) $2\vec u+\frac12\vec v=(6,-2)+(-1,2)=\boxed{(5,\,0)}$\\[3pt]
d) $-\vec u-\vec v=\boxed{(-1,-3)}$ \ \nota{el opuesto de a}
\end{minipage}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ ¿son paralelos?]
$\vec a=(6,-4)$, $\vec b=(-9,6)$: \ $\frac{6}{-9}=\frac{-4}{6}=-\frac23$ \ \be{} \ ($\vec b=-\frac32\vec a$, sentidos contrarios)\\[3pt]
$\vec c=(2,5)$, $\vec d=(4,8)$: \ $\frac24\neq\frac58$ \ \mal{\text{no}}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $x$ para que $(x,6)\parl(2,x)$]
$\dfrac x2=\dfrac6x\ \to\ x^2=12\ \to\ \boxed{x=\pm2\sqrt3}$ \ \nota{con $+$, mismo sentido; con $-$, contrario}
\end{exemple}

\end{exemple}

\seccio{sec5}{Vectorial, paramétricas y continua}{Un punto y un vector director.}

\vfill
\begin{minipage}[c]{0.4\linewidth}\centering
\begin{tikzpicture}[scale=0.6]
  \eixos{-1}{6}{-2}{4}
  \draw[fmnavy,line width=1.4pt] (-0.5,-2) -- (3.85,3.8);
  \draw[vec,fmK] (1,0) -- (2.5,2) node[right=2pt,font=\small\heavy] {$\vec v$};
  \fill[fmR] (1,0) circle (3.5pt) node[below right,font=\small] {$P$};
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.57\linewidth}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,top=2mm,bottom=2mm]
\textbf{Vectorial}: \ $(x,y)=(p_1,p_2)+\lambda\,(v_1,v_2)$\\[5pt]
\textbf{Paramétricas}: \ $\sis{x=p_1+\lambda v_1}{y=p_2+\lambda v_2}$\\[5pt]
\textbf{Continua}: \ $\dfrac{x-p_1}{v_1}=\dfrac{y-p_2}{v_2}$
\end{tcolorbox}
\end{minipage}

\vfill
\begin{idea}[Punto y vector director]
Si conocemos dos puntos, tomamos uno de ellos y $\vect{AB}$ (o un vector paralelo más sencillo). Cada valor del \textbf{parámetro} $\lambda$ da un punto de la recta.
\end{idea}

\vfill
\begin{exemple}[Ejemplo: \ por $A(2,-1)$ con $\vec v=(3,4)$]
$(x,y)=(2,-1)+\lambda(3,4)$ \qquad $\sis{x=2+3\lambda}{y=-1+4\lambda}$ \qquad $\dfrac{x-2}{3}=\dfrac{y+1}{4}$ \qquad $\lambda=1\to(5,3)$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ por $A(-3,2)$ y $B(1,-4)$]
$\vect{AB}=(4,-6)\parl(2,-3)$: \quad $\sis{x=-3+2\lambda}{y=2-3\lambda}$ \qquad $\dfrac{x+3}{2}=\dfrac{y-2}{-3}$
\end{exemple}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={¿Un punto es de la recta?}]
Paramétricas: el \textbf{mismo} $\lambda$ en las dos.\\
Continua: ¿se cumple la igualdad?
\end{tcolorbox}
\begin{compte}[Componente 0]
Con $\vec v=(5,0)$ no hay continua: la recta es horizontal, $y=p_2$.
\end{compte}
\end{tcbraster}

\vfill
\begin{exemple}[Ejemplo: \ $k$ para que $(k,-7)$ sea de $\sis{x=3+2\lambda}{y=1-4\lambda}$]
$-7=1-4\lambda\ \to\ \lambda=2$ \qquad $k=3+2\cdot2=\boxed{7}$
\end{exemple}

\seccio{sec6}{Pendiente. Explícita y general}{Todas las formas de escribir la misma recta.}

\vfill
\begin{minipage}[c]{0.45\linewidth}
\begin{idea}[Pendiente]
{\centering\Large$m=\dfrac{v_2}{v_1}=\dfrac{y_2-y_1}{x_2-x_1}$\par}\smallskip
$m>0$ crece, $m<0$ decrece, $m=0$ horizontal. Las verticales no tienen.
\end{idea}
\end{minipage}\hfill
\begin{minipage}[c]{0.52\linewidth}
\begin{tcolorbox}[enhanced,colback=fmyellow!18,colframe=fmgold,arc=4mm,boxrule=1.6pt,top=2mm,bottom=2mm]
\textbf{Punto-pendiente}: \ $y-p_2=m\,(x-p_1)$\\[4pt]
\textbf{Explícita}: \ $y=mx+n$\\[4pt]
\textbf{General}: \ $Ax+By+C=0$\\[2pt]
{\small director $(-B,A)$, \ normal $(A,B)$, \ $m=-\frac AB$}
\end{tcolorbox}
\end{minipage}

\vfill
\begin{center}\small
\renewcommand{\arraystretch}{1.55}
\taularod{>{\heavy}l<{\hspace{4pt}}>{\hspace{4pt}}c<{\hspace{4pt}}>{\hspace{4pt}}c<{\hspace{4pt}}>{\hspace{4pt}}c<{\hspace{4pt}}>{\hspace{4pt}}c<{\hspace{4pt}}}{
\rowcolor{fmblue}\color{white}Ecuación & \color{white}\heavy Expresión & \color{white}\heavy Un punto & \color{white}\heavy Director & \color{white}\heavy Pendiente\\
Continua & $\frac{x-p_1}{v_1}=\frac{y-p_2}{v_2}$ & $(p_1,p_2)$ & $(v_1,v_2)$ & $\frac{v_2}{v_1}$\\
\rowcolor{fmsurface}Punto-pendiente & $y-p_2=m(x-p_1)$ & $(p_1,p_2)$ & $(1,m)$ & $m$\\
Explícita & $y=mx+n$ & $(0,n)$ & $(1,m)$ & $m$\\
\rowcolor{fmsurface}General & $Ax+By+C=0$ & damos valores & $(-B,A)$ & $-\frac AB$
}
\end{center}

\vfill
\begin{compte}[Signos]
En $\dfrac{x+3}{2}=\dfrac{y-1}{5}$ el punto es $(-3,1)$; en $y+4=2(x-1)$, el punto es $(1,-4)$.
\end{compte}

\vfill
\begin{exemple}[Ejemplo: \ por $P(-2,3)$ con $m=-3$]
$y-3=-3(x+2)$ \qquad $y=-3x-3$ \qquad $3x+y+3=0$ \qquad director $(1,-3)$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ por $A(-1,4)$ y $B(3,-2)$]
$\vect{AB}=(4,-6)\parl(2,-3)$: normal $(3,2)$ \ $\to$ \ $3x+2y+C=0$; \ con $A$: $C=-5$ \qquad $\boxed{3x+2y-5=0}$, \ $y=-\frac32x+\frac52$
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $2x-5y+10=0$]
normal $(2,-5)$, \ director $(5,2)$, \ $m=\frac25$ \qquad $x=0\to(0,2)$, \ $n=2$ \qquad $y=0\to(-5,0)$
\end{exemple}

\end{tcolorbox}

\begin{idea}[Otras maneras]
\textbf{Directores}: no paralelos, secantes; paralelos, comprobamos si un punto de una es de la otra.\\[2pt]
\textbf{Pendientes}: distintas, secantes; iguales y $n$ distinta, paralelas; iguales y misma $n$, coincidentes.\\[2pt]
El \textbf{punto de corte} es la solución del sistema. Con paramétricas, sustituimos $x$ e $y$ en la otra.
\end{idea}

\vfill
\begin{exemple}[Ejemplos]
$x+2y-4=0$, $3x-y-5=0$: \ $\frac13\neq\frac2{-1}$: secantes; \ $y=3x-5\ \to\ \boxed{(2,\,1)}$\\[4pt]
$2x-3y+1=0$, $-4x+6y+5=0$: \ $\frac2{-4}=\frac{-3}{6}\neq\frac15$: \textbf{paralelas}\\[4pt] $x-y+2=0$, $3x-3y+6=0$: \ $\frac13=\frac{-1}{-3}=\frac26$: \textbf{coincidentes}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $r:\sis{x=1+2\lambda}{y=-3+\lambda}$ \ y \ $s:x+3y-2=0$]
$(1+2\lambda)+3(-3+\lambda)-2=0\ \to\ 5\lambda=10\ \to\ \lambda=2$ \qquad punto de corte: \ \boxed{(5,\,-1)}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ $a$ para que $ax-3y+2=0$ y $4x+6y-1=0$ sean paralelas]
$\dfrac a4=\dfrac{-3}{6}\ \to\ \boxed{a=-2}$ \ \nota{y $\frac{2}{-1}\neq-\frac12$: no son coincidentes}
\end{exemple}

\seccio{sec8}{Paralelas, perpendiculares y mediatriz}{Mismo director, o director perpendicular.}

\vfill
\begin{tcbraster}[raster columns=2,raster equal height,raster column skip=10pt]
\begin{tcolorbox}[fmbase,colback=fmB!8,colframe=fmB,coltitle=white,title={Paralelas $r\parl s$}]
Mismo director, misma pendiente.\\[3pt]
De $Ax+By+C=0$: \ $\boxed{Ax+By+K=0}$
\end{tcolorbox}
\begin{tcolorbox}[fmbase,colback=fmK!8,colframe=fmK,coltitle=white,title={Perpendiculares $r\perp s$}]
$\vec v_r\cdot\vec v_s=0$; \ $m_s=-\dfrac1{m_r}$.\\[3pt]
De $Ax+By+C=0$: \ $\boxed{Bx-Ay+K=0}$
\end{tcolorbox}
\end{tcbraster}
{\comic\color{fmmuted}\centering $K$ se halla haciendo que la recta pase por el punto que nos dan.\par}

\vfill
\begin{minipage}[c]{0.4\linewidth}\centering
\begin{tikzpicture}[scale=0.6]
  \eixos{-2}{6}{-3}{4}
  \draw[fmA,line width=1.3pt] (-1,2) -- (5,-2);

  \draw[fmR,line width=1.6pt] (0,-3) -- (4.67,4);
  \fill[fmR] (2,0) circle (3.5pt) node[right,font=\small\heavy] {$M$};
  \draw[fmnavy] (2.3,-0.2) -- (2.5,0.1) -- (2.2,0.3);
\end{tikzpicture}
\end{minipage}\hfill
\begin{minipage}[c]{0.57\linewidth}
\begin{idea}[Mediatriz de $AB$]
La perpendicular a $AB$ por el punto medio $M$. Todos sus puntos están a la misma distancia de $A$ y de $B$.\\[3pt]
$\vect{AB}=(a,b)$ es un vector \textbf{normal}: \ $ax+by+K=0$, y $K$ con $M$.
\end{idea}
\end{minipage}

\vfill
\begin{exemple}[Ejemplo: \ $r:2x-y+3=0$ y $P(1,-2)$]
Paralela: \ $2x-y+K=0$, \ $2+2+K=0$ \ $\to$ \ \boxed{2x-y-4=0}\\[3pt]
Perpendicular: \ $x+2y+K=0$, \ $1-4+K=0$ \ $\to$ \ \boxed{x+2y+3=0}
\end{exemple}

\vfill
\begin{exemple}[Ejemplo: \ mediatriz de $A(-1,2)$ y $B(5,-2)$]
$M=(2,0)$, \ $\vect{AB}=(6,-4)\parl(3,-2)$ \ $\to$ \ $3x-2y+K=0$; \ $6+K=0$ \ $\to$ \ \boxed{3x-2y-6=0}
